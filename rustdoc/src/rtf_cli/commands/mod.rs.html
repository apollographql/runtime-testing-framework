1//! The various user facing commands exposed through the CLI.
2//!
3//! We follow the [git CLI design][0] of splitting user facing commands into low level "plumbing"
4//! and high level "porcelain" categories.
5//!
6//! [0]: https://git-scm.com/docs
7use anyhow::bail;
8use rtf_config::context::{Context, PathKind, ResolutionContext};
9use std::{
10    collections::HashMap,
11    env::{self, current_dir},
12    path::PathBuf,
13};
14
15pub mod plumbing;
16pub mod porcelain;
17
18fn get_context() -> Context {
19    let env_vars: HashMap<String, String> = env::vars_os()
20        .map(|(k, v)| {
21            (
22                k.to_string_lossy().to_string(),
23                v.to_string_lossy().to_string(),
24            )
25        })
26        .collect();
27
28    Context::new_from_env_vars(env_vars)
29}
30
31fn get_context_and_outdir(out_dir: &str) -> anyhow::Result<(Context, PathBuf)> {
32    let ctx = get_context();
33
34    // output directories are created relative to the directory we were run from
35    let out_dir = current_dir()?.join(out_dir);
36    match ctx.path_kind(&out_dir) {
37        PathKind::File => bail!("{} is not a directory", out_dir.display()),
38        PathKind::OccupiedDir => {
39            bail!("{} already exists and is non-empty", out_dir.display())
40        }
41        _ => ctx.create_dir_all(&out_dir)?,
42    }
43
44    Ok((ctx, out_dir))
45}
