23pub(crate) fn get_context() -> Context {
24    let env_vars: HashMap<String, String> = env::vars_os()
25        .map(|(k, v)| {
26            (
27                k.to_string_lossy().to_string(),
28                v.to_string_lossy().to_string(),
29            )
30        })
31        .collect();
32
33    Context::new_from_env_vars(env_vars)
34}
35
36pub(crate) fn get_context_and_check_outdir(out_dir: &str) -> anyhow::Result<(Context, PathBuf)> {
37    let ctx = get_context();
38    let out_dir = current_dir()?.join(out_dir);
39
40    match ctx.path_kind(&out_dir) {
41        PathKind::File => bail!("{} is not a directory", out_dir.display()),
42        PathKind::OccupiedDir => {
43            bail!("{} already exists and is non-empty", out_dir.display())
44        }
45        _ => Ok((ctx, out_dir)),
46    }
47}

49/// Load a config file from a path.
50pub(crate) async fn load_config<T>(
51    path: &str,
52    type_name: &str,
53    ctx: &impl ResolutionContext,
54) -> anyhow::Result<(SourceDir, T)>
55where
56    T: for<'de> Deserialize<'de>,
57{
58    let abs_path = ctx
59        .canonicalize_path(path)
60        .with_context(|| format!("Unable to resolve path: {path}"))?;
61    let content = ctx
62        .read_path_to_string(&abs_path)
63        .with_context(|| format!("Unable to read {type_name} from {path}"))?;
64    let source = SourceDir::local(
65        abs_path
66            .parent()
67            .expect("we just read the file so it has a parent"),
68    );
69
70    let config: T = serde_yaml::from_str(&content)
71        .with_context(|| format!("Unable to parse {type_name} yaml"))?;
72
73    Ok((source, config))
74}

76/// Handles loading a local test plan and displaying user facing errors
77async fn load_and_resolve_test_plan_from_local(
78    path: &str,
79    ctx: &impl ResolutionContext,
80) -> anyhow::Result<TestPlanConfig> {
81    match TestPlanConfig::try_load_and_resolve_from_path(path, ctx).await {
82        Ok(test_plan) => Ok(test_plan),
83        Err(e) => match e {
84            formats::Error::Io(e) => bail!("Unable to load test plan from {path}: {e}"),
85            formats::Error::Yaml(e) => bail!("Unable to parse test plan yaml: {e}"),
86            _ => bail!("Unable to load and resolve test plan: {e}"),
87        },
88    }
89}

91/// Handles loading a test plan from github and displaying user facing errors
92async fn load_and_resolve_test_plan_from_github(
93    test_plan_path: &str,
94    git_ref: Option<String>,
95    ctx: &impl ResolutionContext,
96) -> anyhow::Result<TestPlanConfig> {
97    let (org, repo_and_path) = test_plan_path.split_once('/').ok_or(anyhow!(
98        "invalid GitHub uri: \"{test_plan_path}\" - GitHub uri must be in format ORG/REPO/PATH"
99    ))?;
100    let (repo, path) = repo_and_path.split_once('/').ok_or(anyhow!(
101        "invalid GitHub uri: \"{test_plan_path}\" - GitHub uri must be in format ORG/REPO/PATH"
102    ))?;
103
104    match TestPlanConfig::try_load_and_resolve_from_github(org, repo, path, git_ref, ctx).await {
105        Ok(test_plan) => Ok(test_plan),
106        Err(e) => bail!("Unable to load and resolve test plan from GitHub: {e}"),
107    }
108}
