21pub(crate) fn get_context() -> Context {
22    let env_vars: HashMap<String, String> = env::vars_os()
23        .map(|(k, v)| {
24            (
25                k.to_string_lossy().to_string(),
26                v.to_string_lossy().to_string(),
27            )
28        })
29        .collect();
30
31    Context::new_from_env_vars(env_vars)
32}
33
34pub(crate) fn get_context_and_check_outdir(out_dir: &str) -> anyhow::Result<(Context, PathBuf)> {
35    let ctx = get_context();
36
37    let out_dir = current_dir()?.join(out_dir);
38    match ctx.path_kind(&out_dir) {
39        PathKind::File => bail!("{} is not a directory", out_dir.display()),
40        PathKind::OccupiedDir => {
41            bail!("{} already exists and is non-empty", out_dir.display())
42        }
43        _ => Ok((ctx, out_dir)),
44    }
45}

47/// Handles loading the test plan and displaying user facing errors
48async fn load_and_resolve_test_plan(
49    path: &str,
50    ctx: &impl ResolutionContext,
51) -> anyhow::Result<TestPlanConfig> {
52    match TestPlanConfig::try_load_and_resolve_from_path(path, ctx).await {
53        Ok(test_plan) => Ok(test_plan),
54        Err(e) => match e {
55            formats::Error::Io(e) => bail!("Unable to load test plan from {path}: {e}"),
56            formats::Error::Yaml(e) => bail!("Unable to parse test plan yaml: {e}"),
57            _ => bail!("Unable to load and resolve test plan: {e}"),
58        },
59    }
60}
