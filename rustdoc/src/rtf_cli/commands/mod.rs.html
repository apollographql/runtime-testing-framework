23    Context::new_from_env_vars(&env::vars().collect())
24}

26pub fn get_context_and_check_outdir(
27    out_dir: &str,
28    force: bool,
29) -> anyhow::Result<(Context, PathBuf)> {
30    let ctx = get_context();
31    let out_dir = current_dir()?.join(out_dir);
32
33    match ctx.path_kind(&out_dir) {
34        PathKind::File => bail!("{} exists and is a file", out_dir.display()),
35
36        PathKind::OccupiedDir if force => {
37            ctx.remove_dir_all(&out_dir)?;
38
39            Ok((ctx, out_dir))
40        }
41
42        PathKind::OccupiedDir => {
43            bail!(
44                "{} already exists and is non-empty.\nRe-run with the '--force' flag to force removal of the existing directory",
45                out_dir.display()
46            )
47        }
48
49        _ => Ok((ctx, out_dir)),
50    }
51}

53/// Load a config file from a path.
54pub(crate) async fn load_config<T>(
55    path: &str,
56    type_name: &str,
57    ctx: &impl ResolutionContext,
58) -> anyhow::Result<(SourceDir, T)>
59where
60    T: for<'de> Deserialize<'de>,
61{
62    let abs_path = ctx
63        .canonicalize_path(path)
64        .with_context(|| format!("Unable to resolve path: {path}"))?;
65    let content = ctx
66        .read_path_to_string(&abs_path)
67        .with_context(|| format!("Unable to read {type_name} from {path}"))?;
68    let source = SourceDir::local(
69        abs_path
70            .parent()
71            .expect("we just read the file so it has a parent"),
72    );
73
74    let config: T = serde_yaml::from_str(&content)
75        .with_context(|| format!("Unable to parse {type_name} yaml"))?;
76
77    Ok((source, config))
78}

80/// Handles loading a local test plan and displaying user facing errors
81pub async fn load_and_resolve_test_plan_from_local<E: Prepare>(
82    path: &str,
83    ctx: &impl ResolutionContext,
84) -> anyhow::Result<(TestPlan<E>, Sources)> {
85    match TestPlan::<E>::try_load_and_resolve_from_path(path, ctx).await {
86        Ok(result) => Ok(result),
87        Err(e) => match e {
88            formats::Error::Io(e) => bail!("Unable to load test plan from {path}: {e}"),
89            formats::Error::Yaml(e) => bail!("Unable to parse test plan yaml: {e}"),
90            _ => bail!("Unable to load and resolve test plan: {e}"),
91        },
92    }
93}

95/// Handles loading a test plan from github and displaying user facing errors
96pub async fn load_and_resolve_test_plan_from_github<E: Prepare>(
97    test_plan_path: &str,
98    git_ref: Option<String>,
99    ctx: &impl ResolutionContext,
100) -> anyhow::Result<(TestPlan<E>, Sources)> {
101    let (org, repo_and_path) = test_plan_path.split_once('/').ok_or(anyhow!(
102        "invalid GitHub uri: \"{test_plan_path}\" - GitHub uri must be in format ORG/REPO/PATH"
103    ))?;
104    let (repo, path) = repo_and_path.split_once('/').ok_or(anyhow!(
105        "invalid GitHub uri: \"{test_plan_path}\" - GitHub uri must be in format ORG/REPO/PATH"
106    ))?;
107
108    match TestPlan::<E>::try_load_and_resolve_from_github(org, repo, path, git_ref, ctx).await {
109        Ok(result) => Ok(result),
110        Err(e) => bail!("Unable to load and resolve test plan from GitHub: {e}"),
111    }
112}
