1//! Resolve file providers for a scenario independently of executing it.
2use crate::commands::{
3    get_context_and_check_outdir, load_config,
4    plumbing::{parse_cli_variables, resolve::generate_env_file},
5};
6use anyhow::bail;
7use rtf_config::{
8    SourceDir, StableSource,
9    checks::Check,
10    context::ResolutionContext,
11    formats::{ScenarioConfig, ScenarioExecution, Sources},
12    run::{OUTPUT_PATH, PROVIDER_DIR, RunProviders},
13    templating::{Template, TemplateContext},
14};
15use rtf_core::variables::{ParsedVariables, Variables};
16use std::env::current_dir;
17use tracing::info;
18
19const SCENARIO_ENV_FILE: &str = "scenario.env";
20
21pub async fn resolve_scenario(
22    scenario_path: &str,
23    variables: Variables,
24    out_dir: &str,
25    force: bool,
26) -> anyhow::Result<()> {
27    let (mut ctx, out_dir) = get_context_and_check_outdir(out_dir, force)?;
28
29    info!("loading scenario");
30    let (source, mut scenario) =
31        load_config::<ScenarioConfig<ScenarioExecution>>(scenario_path, "scenario", &ctx).await?;
32
33    // Custom providers are not supported by resolve scenario
34    if scenario.execution.contains_custom_providers() {
35        bail!("custom_providers are not supported by resolve scenario");
36    }
37
38    let (
39        ParsedVariables {
40            variables,
41            variable_sources,
42            ..
43        },
44        vars_file_src,
45    ) = parse_cli_variables(variables, &ctx)?;
46
47    ctx.set_sources(
48        Sources::default()
49            .with_scenario(source)
50            .with_cli(SourceDir::local(current_dir()?))
51            .with_variables_file(vars_file_src),
52    );
53
54    let template_ctx = TemplateContext::new(variables, variable_sources, Default::default());
55
56    info!("templating scenario");
57    scenario.try_template(&mut Vec::new(), &StableSource::Scenario, &template_ctx)?;
58
59    info!("running static checks");
60    scenario
61        .execution
62        .try_check(&mut vec!["scenario".to_string()], &ctx)?;
63
64    info!("creating output directory");
65    ctx.create_dir_all(&out_dir)?;
66    let out_dir = ctx.canonicalize_path(&out_dir)?;
67    ctx.set_output_path(&out_dir);
68
69    info!("resolving file providers");
70    let providers_dir = out_dir.join(PROVIDER_DIR);
71    scenario
72        .execution
73        .run_providers(&providers_dir, &mut ctx)
74        .await?;
75
76    info!("writing scenario.env");
77    let output_path = out_dir.join(OUTPUT_PATH);
78    let env_vars = scenario
79        .execution
80        .all_env_vars(&out_dir, &output_path, &ctx)?;
81    let env_content = generate_env_file(env_vars);
82    ctx.write(out_dir.join(SCENARIO_ENV_FILE), env_content)?;
83
84    info!("done");
85
86    Ok(())
87}
