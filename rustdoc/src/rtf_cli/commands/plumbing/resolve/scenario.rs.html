21const SCENARIO_ENV_FILE: &str = "scenario.env";
22
23pub async fn resolve_scenario(
24    scenario_path: &str,
25    variables: Variables,
26    out_dir: &str,
27) -> anyhow::Result<()> {
28    let (mut ctx, out_dir) = get_context_and_check_outdir(out_dir)?;
29    let cwd = current_dir()?;
30    let cwd_source = SourceDir::local(cwd);
31
32    info!("loading scenario");
33    let (source, mut scenario) = load_scenario(scenario_path, &ctx).await?;
34
35    // Custom providers are not supported by resolve scenario
36    if !scenario.custom_providers.is_empty() {
37        bail!("custom_providers are not supported by resolve scenario");
38    }
39
40    let ParsedVariables {
41        variables,
42        variable_sources,
43        ..
44    } = parse_cli_variables(variables, &cwd_source, &ctx)?;
45
46    let template_ctx = TemplateContext::new(
47        variables,
48        source.clone(),
49        variable_sources,
50        Default::default(),
51    );
52
53    info!("templating scenario");
54    scenario.try_template(&mut Vec::new(), &source, &template_ctx)?;
55
56    info!("running static checks");
57    scenario
58        .command
59        .try_check(&mut vec!["scenario".to_string()], &ctx)?;
60
61    info!("creating output directory");
62    ctx.create_dir_all(&out_dir)?;
63    let out_dir = ctx.canonicalize_path(&out_dir)?;
64    ctx.set_output_path(&out_dir);
65
66    info!("resolving file providers");
67    let providers_dir = out_dir.join(PROVIDER_DIR);
68    scenario
69        .command
70        .run_providers(&providers_dir, &mut ctx)
71        .await?;
72
73    info!("writing scenario.env");
74    let output_path = out_dir.join(OUTPUT_PATH);
75    let env_vars = scenario
76        .command
77        .all_env_vars(&out_dir, &output_path, &ctx)?;
78    let env_content = generate_scenario_env(env_vars);
79    ctx.write(out_dir.join(SCENARIO_ENV_FILE), env_content)?;
80
81    info!("done");
82
83    Ok(())
84}
85
86fn generate_scenario_env(env_vars: HashMap<String, String>) -> String {
87    let mut sorted_vars: Vec<_> = env_vars.into_iter().collect();
88    sorted_vars.sort_by(|(a, _), (b, _)| a.cmp(b));
89
90    sorted_vars
91        .into_iter()
92        .fold(String::new(), |mut acc, (key, value)| {
93            acc.push_str(&format!("export {key}=\"{value}\"\n"));
94            acc
95        })
96}

98#[cfg(test)]
99mod tests {
100    use super::*;
101    use simple_test_case::test_case;
102
103    #[test_case(&[], ""; "empty")]
104    #[test_case(&[("FOO", "bar")], "export FOO=\"bar\"\n"; "single variable")]
105    #[test_case(&[("FOO", "hello world")], "export FOO=\"hello world\"\n"; "value with space")]
106    #[test_case(&[("ZZZ", "last"), ("AAA", "first")], "export AAA=\"first\"\nexport ZZZ=\"last\"\n"; "multiple variables")]
107    #[test]
108    fn generate_scenario_env_formats_correctly(vars: &[(&str, &str)], expected: &str) {
109        let env_vars = HashMap::from_iter(vars.iter().map(|(k, v)| (k.to_string(), v.to_string())));
110        let env_file_string = generate_scenario_env(env_vars);
111
112        assert_eq!(env_file_string, expected);
113    }
