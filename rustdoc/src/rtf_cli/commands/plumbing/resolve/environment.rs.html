25const SETUP_ENV_FILE: &str = "setup.env";
26const TEARDOWN_ENV_FILE: &str = "teardown.env";
27const COMPOSE_FILES_LIST: &str = "compose-files.txt";
28
29pub async fn resolve_environment(
30    environment_path: &str,
31    variables: Variables,
32    out_dir: &str,
33    force: bool,
34) -> anyhow::Result<()> {
35    let (mut ctx, out_dir) = get_context_and_check_outdir(out_dir, force)?;
36
37    info!("loading environment");
38    let (source, mut environment) = load_config::<EnvironmentConfig<EnvironmentExecution>>(
39        environment_path,
40        "environment",
41        &ctx,
42    )
43    .await?;
44
45    // Custom providers are not supported by resolve environment
46    if !environment.custom_providers.is_empty() {
47        bail!("custom_providers are not supported by resolve environment");
48    }
49
50    let (
51        ParsedVariables {
52            variables,
53            variable_sources,
54            ..
55        },
56        vars_file_src,
57    ) = parse_cli_variables(variables, &ctx)?;
58
59    ctx.set_sources(
60        Sources::default()
61            .with_environment(source)
62            .with_cli(SourceDir::local(current_dir()?))
63            .with_variables_file(vars_file_src),
64    );
65
66    let template_ctx = TemplateContext::new(variables, variable_sources, Default::default());
67
68    info!("templating environment");
69    environment.try_template(&mut Vec::new(), &StableSource::Environment, &template_ctx)?;
70
71    info!("running static checks");
72    environment.try_check(&mut vec!["environment".to_string()], &ctx)?;
73
74    info!("creating output directory");
75    ctx.create_dir_all(&out_dir)?;
76    let out_dir = ctx.canonicalize_path(&out_dir)?;
77    ctx.set_output_path(&out_dir);
78
79    match &environment.execution {
80        EnvironmentExecution::Script(script) => {
81            resolve_script_environment(script, &out_dir, &mut ctx).await?;
82        }
83        EnvironmentExecution::DockerCompose(compose) => {
84            resolve_docker_compose_environment(&environment.name, compose, &out_dir, &mut ctx)
85                .await?;
86        }
87    }
88
89    info!("done");
90
91    Ok(())
92}

94async fn resolve_script_environment(
95    env: &ScriptEnvironment,
96    out_dir: &Path,
97    ctx: &mut impl ResolutionContext,
98) -> anyhow::Result<()> {
99    info!("resolving setup providers");
100    let setup_dir = out_dir.join("setup");
101    ctx.create_dir_all(&setup_dir)?;
102    let setup_providers_dir = setup_dir.join(PROVIDER_DIR);
103    env.setup.run_providers(&setup_providers_dir, ctx).await?;
104
105    info!("writing setup.env");
106    let setup_output_path = setup_dir.join(OUTPUT_PATH);
107    let setup_vars = env
108        .setup
109        .all_env_vars(&setup_dir, &setup_output_path, ctx)?;
110    let setup_env_content = generate_env_file(setup_vars);
111    ctx.write(setup_dir.join(SETUP_ENV_FILE), setup_env_content)?;
112
113    info!("resolving teardown providers");
114    let teardown_dir = out_dir.join("teardown");
115    ctx.create_dir_all(&teardown_dir)?;
116    let teardown_providers_dir = teardown_dir.join(PROVIDER_DIR);
117    env.teardown
118        .run_providers(&teardown_providers_dir, ctx)
119        .await?;
120
121    info!("writing teardown.env");
122    let teardown_output_path = teardown_dir.join(OUTPUT_PATH);
123    let teardown_vars = env
124        .teardown
125        .all_env_vars(&teardown_dir, &teardown_output_path, ctx)?;
126    let teardown_env_content = generate_env_file(teardown_vars);
127    ctx.write(teardown_dir.join(TEARDOWN_ENV_FILE), teardown_env_content)?;
128
129    Ok(())
130}

132async fn resolve_docker_compose_environment(
133    name: &str,
134    env: &DockerComposeEnvironment,
135    out_dir: &Path,
136    ctx: &mut impl ResolutionContext,
137) -> anyhow::Result<()> {
138    info!("resolving setup providers");
139    let setup_dir = out_dir.join("setup");
140    ctx.create_dir_all(&setup_dir)?;
141    let setup_providers_dir = setup_dir.join(PROVIDER_DIR);
142    env.run_providers(&setup_providers_dir, ctx).await?;
143
144    // Write compose file paths to a file (one path per line)
145    info!("writing compose-files.txt");
146    let compose_paths: Vec<String> = env
147        .compose_file_paths(ctx)?
148        .iter()
149        .map(|p| p.to_string_lossy().to_string())
150        .collect();
151    let compose_files_path = setup_dir.join(COMPOSE_FILES_LIST);
152    let compose_files_content = compose_paths.join("\n");
153    ctx.write(&compose_files_path, compose_files_content)?;
154
155    info!("writing setup.env");
156    let setup_output_path = setup_dir.join(OUTPUT_PATH);
157    let mut setup_vars = env.all_env_vars(&setup_dir, &setup_output_path, ctx)?;
158    setup_vars.insert(
159        "COMPOSE_FILES".to_string(),
160        compose_files_path.to_string_lossy().to_string(),
161    );
162
163    let setup_env_content = generate_env_file(setup_vars);
164    ctx.write(setup_dir.join(SETUP_ENV_FILE), setup_env_content)?;
165
166    info!("writing teardown.env");
167    let teardown_dir = out_dir.join("teardown");
168    ctx.create_dir_all(&teardown_dir)?;
169    // Create empty providers dir for consistency
170    ctx.create_dir_all(teardown_dir.join(PROVIDER_DIR))?;
171
172    let project_name = env.project_name.as_deref().unwrap_or(name);
173    let teardown_vars =
174        HashMap::from([("COMPOSE_PROJECT_NAME".to_string(), project_name.to_string())]);
175    let teardown_env_content = generate_env_file(teardown_vars);
176    ctx.write(teardown_dir.join(TEARDOWN_ENV_FILE), teardown_env_content)?;
177
178    Ok(())
179}
