18fn parse_cli_variables(
19    variables: Variables,
20    cwd_source: &SourceDir,
21    ctx: &impl ResolutionContext,
22) -> anyhow::Result<ParsedVariables> {
23    let parsed = variables.parse(cwd_source, ctx)?;
24
25    if !parsed.matrix_dimensions.is_empty() {
26        let mut keys: Vec<_> = parsed
27            .matrix_dimensions
28            .keys()
29            .map(|s| s.as_str())
30            .collect();
31        keys.sort_unstable();
32        return Err(anyhow!(
33            "Expected only scalar variables but found matrix dimensions for: {}",
34            keys.join(", ")
35        ));
36    }
37
38    Ok(parsed)
39}
