11const N_PARALLEL_FETCH: usize = 20;

13pub async fn pull_run_output(run_id: Uuid, out_dir: &str, force: bool) -> anyhow::Result<()> {
14    let (ctx, out_dir) = get_context_and_check_outdir(out_dir, force)?;
15    let client = OrchestratorClient::new_from_env().await?;
16
17    let tr: TestRunSummary = client
18        .get_json(&format!("test-run/{run_id}/status"))
19        .await?;
20
21    info!("creating output directory");
22    ctx.create_dir_all(&out_dir)?;
23    ctx.write(
24        out_dir.join("run-summary.json"),
25        serde_json::to_string_pretty(&tr)?,
26    )?;
27
28    let out_dir = ctx.canonicalize_path(&out_dir)?;
29
30    for chunk in tr.executions.chunks(N_PARALLEL_FETCH) {
31        let sub_dirs = chunk
32            .iter()
33            .map(|ex| {
34                let sub_dir = out_dir.join(&ex.name);
35                ctx.create_dir_all(&sub_dir)?;
36                Ok(sub_dir)
37            })
38            .collect::<anyhow::Result<Vec<_>>>()?;
39
40        try_join_all(
41            chunk
42                .iter()
43                .zip(sub_dirs.iter())
44                .map(|(ex, sub_dir)| write_output_for_execution(ex, sub_dir, &client, &ctx)),
45        )
46        .await?;
47    }
48
49    Ok(())
50}

52pub async fn pull_execution_output(ex_id: Uuid, out_dir: &str, force: bool) -> anyhow::Result<()> {
53    let (ctx, out_dir) = get_context_and_check_outdir(out_dir, force)?;
54    let client = OrchestratorClient::new_from_env().await?;
55
56    let ex: TestExecutionSummary = client
57        .get_json(&format!("test-execution/{ex_id}/status"))
58        .await?;
59
60    info!("creating output directory");
61    ctx.create_dir_all(&out_dir)?;
62    ctx.write(
63        out_dir.join("execution-summary.json"),
64        serde_json::to_string_pretty(&ex)?,
65    )?;
66
67    let out_dir = ctx.canonicalize_path(&out_dir)?;
68
69    write_output_for_execution(&ex, &out_dir, &client, &ctx).await
70}

72async fn write_output_for_execution(
73    ex: &TestExecutionSummary,
74    out_dir: &Path,
75    client: &OrchestratorClient,
76    ctx: &impl ResolutionContext,
77) -> anyhow::Result<()> {
78    info!("pulling execution log for {}", ex.id);
79    fetch_execution_file(client, ex.id, "log.txt", out_dir, ctx).await?;
80
81    info!("pulling execution output for {}", ex.id);
82    fetch_execution_file(client, ex.id, "output.zip", out_dir, ctx).await?;
83
84    Ok(())
85}

87/// Fetch a single execution file from the Orchestrator and write it to `out_dir/file_name`.
88///
89/// If the file cannot be found (404), the command shouldn't abort the whole pull -- instead
90/// write the error to `out_dir/<file_name>.txt` (appending `.txt` only if `file_name` doesn't
91/// already have it) so the rest of the executions still get pulled.
92async fn fetch_execution_file(
93    client: &OrchestratorClient,
94    ex_id: Uuid,
95    file_name: &str,
96    out_dir: &Path,
97    ctx: &impl ResolutionContext,
98) -> anyhow::Result<()> {
99    let endpoint = format!("test-execution/{ex_id}/{file_name}");
100    let resp = client.request(Method::GET, &endpoint).await?.send().await?;
101
102    if resp.status() == StatusCode::NOT_FOUND {
103        let body = resp.text().await?;
104        return write_not_found_placeholder(ex_id, file_name, &body, out_dir, ctx);
105    }
106
107    let bytes = resp.error_for_status()?.bytes().await?;
108    ctx.write(out_dir.join(file_name), bytes)?;
109
110    Ok(())
111}
112
113/// Record that `file_name` could not be found for execution `ex_id`, instead of failing the
114/// whole pull.
115///
116/// The error is written to `out_dir/<file_name>.txt` (appending `.txt` only if `file_name`
117/// doesn't already have it) so the rest of the executions still get pulled.
118fn write_not_found_placeholder(
119    ex_id: Uuid,
120    file_name: &str,
121    body: &str,
122    out_dir: &Path,
123    ctx: &impl ResolutionContext,
124) -> anyhow::Result<()> {
125    let error_file_name = if file_name.ends_with(".txt") {
126        file_name.to_owned()
127    } else {
128        format!("{file_name}.txt")
129    };
130
131    warn!("{file_name} not found for execution {ex_id}, writing error to {error_file_name}");
132    ctx.write(
133        out_dir.join(error_file_name),
134        format!("failed to fetch test-execution/{ex_id}/{file_name}: 404 Not Found\n\n{body}"),
135    )?;
136
137    Ok(())
138}

140#[cfg(test)]
141mod tests {
142    use super::*;
143    use assert_fs::{TempDir, prelude::*};
144    use predicates::{prelude::*, str::contains};
145    use rtf_config::context::Context;
146    use simple_test_case::test_case;
147
148    const EX_ID: Uuid = Uuid::nil();
149
150    #[test_case("log.txt", "log.txt"; "already_txt")]
151    #[test_case("output.zip", "output.zip.txt"; "non_txt_gets_txt_appended")]
152    #[test]
153    fn not_found_placeholder_only_appends_txt_when_needed(
154        file_name: &str,
155        expected_error_file: &str,
156    ) {
157        let out_dir = TempDir::new().unwrap();
158        let ctx = Context::new();
159
160        write_not_found_placeholder(
161            EX_ID,
162            file_name,
163            "execution output has expired",
164            out_dir.path(),
165            &ctx,
166        )
167        .unwrap();
168
169        out_dir
170            .child(expected_error_file)
171            .assert(contains("execution output has expired").and(contains(EX_ID.to_string())));
172
173        // The real (non-placeholder) file should never exist unless it's the same
174        // path as the error placeholder (i.e. file_name already ended in .txt).
175        if file_name != expected_error_file {
176            out_dir.child(file_name).assert(predicate::path::missing());
177        }
178    }
