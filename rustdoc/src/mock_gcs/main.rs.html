1//! Minimal in-memory GCS mock server for local integration testing of the rep-orchestrator crate.
2use axum::{
3    Router,
4    body::Bytes,
5    extract::{Path, State},
6    http::StatusCode,
7    routing::put,
8};
9use std::{
10    collections::HashMap,
11    net::SocketAddr,
12    sync::{Arc, Mutex},
13};
14use tokio::net::TcpListener;
15use tracing::info;
16
17type Store = Arc<Mutex<HashMap<String, Bytes>>>;
18
19#[tokio::main]
20async fn main() {
21    tracing_subscriber::fmt().init();
22
23    let app = Router::new()
24        .route("/{*path}", put(put_handler).get(get_handler))
25        .with_state(Store::default());
26    let addr: SocketAddr = "0.0.0.0:8080".parse().unwrap();
27    let listener = TcpListener::bind(addr).await.unwrap();
28
29    info!(%addr, "listening on");
30    axum::serve(listener, app).await.unwrap();
31}
32
33async fn put_handler(
34    Path(path): Path<String>,
35    State(store): State<Store>,
36    body: Bytes,
37) -> StatusCode {
38    info!(%path, bytes = body.len(), "PUT");
39    store.lock().unwrap().insert(path, body);
40
41    StatusCode::OK
42}
43
44async fn get_handler(Path(path): Path<String>, State(store): State<Store>) -> (StatusCode, Bytes) {
45    info!(%path, "GET");
46    match store.lock().unwrap().get(&path).cloned() {
47        Some(bytes) => (StatusCode::OK, bytes),
48        None => (StatusCode::NOT_FOUND, Bytes::from("not found")),
49    }
50}
