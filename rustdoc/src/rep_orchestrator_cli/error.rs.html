7/// All errors that prevent test execution from proceeding.

25#[cfg(test)]
26mod tests {
27    use super::*;
28    use crate::context::FsErrorKind;
29    use std::{io, path::PathBuf};
30
31    fn filesystem_error() -> Error {
32        FsError {
33            path: PathBuf::from("/some/file"),
34            kind: FsErrorKind::Read,
35            source: io::Error::new(io::ErrorKind::NotFound, "not found"),
36        }
37        .into()
38    }
39
40    fn shell_spawn_error() -> Error {
41        ShellError::Spawn {
42            cmd: "my-cmd".to_owned(),
43            source: io::Error::new(io::ErrorKind::NotFound, "not found"),
44        }
45        .into()
46    }
47
48    fn shell_failed_error(code: i32) -> Error {
49        use std::{os::unix::process::ExitStatusExt, process::ExitStatus};
50        ShellError::Failed {
51            cmd: "my-cmd".to_owned(),
52            exit_status: ExitStatus::from_raw(code << 8),
53        }
54        .into()
55    }
56
57    #[test]
58    fn filesystem_error_message_includes_path() {
59        let msg = filesystem_error().to_string();
60        assert!(msg.contains("/some/file"), "expected path in error: {msg}");
61    }
62
63    #[test]
64    fn shell_failed_error_message_includes_exit_code() {
65        let msg = shell_failed_error(2).to_string();
66        assert!(msg.contains("(2)"), "expected exit code in error: {msg}");
67        assert!(msg.contains("my-cmd"), "expected cmd in error: {msg}");
68    }
69
70    #[test]
71    fn shell_spawn_error_message_includes_cmd() {
72        let msg = shell_spawn_error().to_string();
73        assert!(msg.contains("my-cmd"), "expected cmd in spawn error: {msg}");
74    }
