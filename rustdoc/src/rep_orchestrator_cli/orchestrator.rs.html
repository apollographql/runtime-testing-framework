13/// Errors produced when communicating with the REP orchestrator HTTP API.
14#[derive(Debug, Error)]
15pub enum Error {
16    #[error("failed to fetch {kind} config: {message}")]
17    FetchConfig { kind: &'static str, message: String },
18
19    #[error("failed to request artifact upload URLs: {message}")]
20    GenerateUploadUrls { message: String },
21
22    #[error("failed to upload artifact: {message}")]
23    Upload { message: String },
24
25    #[error("failed to update execution status: {message}")]
26    UpdateStatus { message: String },
27}

29const KUSTOMIZE_PATCH: &str = include_str!("../resources/kustomization.yaml");

31pub trait Client: Send + Sync {
32    fn kustomize_patch_for_execution(
33        &self,
34        outdir: &Path,
35        toolbox_image_pull_policy: &str,
36    ) -> String;
37
38    /// Update the [Status] of the current test execution, with an optional `message` to write to the database.
39    ///
40    /// If there was a subprocess associated with this status update, its [ExitStatus] may be included for additional context.
41    fn update_status(
42        &self,
43        status: Status,
44        exit_status: Option<ExitStatus>,
45        message: Option<String>,
46    ) -> impl Future<Output = Result<(), Error>> + Send;
47
48    /// Request signed URLs from the orchestrator for uploading the log file and output zip
49    /// associated with the current test execution.
50    fn generate_upload_urls(&self) -> impl Future<Output = Result<UploadUrls, Error>> + Send;
51
52    /// PUT `body` to a previously-issued signed upload URL.
53    fn upload_to_signed_url(
54        &self,
55        url: &str,
56        body: Vec<u8>,
57    ) -> impl Future<Output = Result<(), Error>> + Send;
58
59    /// Fetch the resolved environment YAML for the current test execution.
60    fn fetch_environment_config(&self) -> impl Future<Output = Result<Vec<u8>, Error>> + Send;
61
62    /// Fetch the resolved scenario YAML for the current test execution.
63    fn fetch_scenario_config(&self) -> impl Future<Output = Result<Vec<u8>, Error>> + Send;
64}

68    execution_id: Uuid,
69    execution_token: Uuid,

72impl HttpClient {
73    /// Creates a new [HttpClient] that will report updates to `orchestrator_url` for the test execution associated
74    /// with `execution_id`, authenticating requests with `execution_token` as a Bearer token.
75    pub fn try_new(
76        orchestrator_url: String,
77        execution_id: Uuid,
78        execution_token: Uuid,
79    ) -> anyhow::Result<Self> {
80        Ok(Self {
81            orchestrator_url: Url::parse(&orchestrator_url)?,
82            execution_id,
83            execution_token,
84        })
85    }
86
87    async fn fetch_config(&self, kind: &'static str) -> Result<Vec<u8>, Error> {
88        let url = self
89            .orchestrator_url
90            .join(&format!(
91                "test-execution/{}/{kind}-config",
92                self.execution_id
93            ))
94            .map_err(|e| Error::FetchConfig {
95                kind,
96                message: e.to_string(),
97            })?;
98
99        info!(id=%self.execution_id, "fetching {kind} config");
100        let resp = reqwest::Client::new()
101            .get(url)
102            .bearer_auth(self.execution_token)
103            .send()
104            .await
105            .map_err(|e| Error::FetchConfig {
106                kind,
107                message: e.to_string(),
108            })?;
109
110        let status_code = resp.status();
111        if !status_code.is_success() {
112            let body = resp.text().await.unwrap_or_default();
113            return Err(Error::FetchConfig {
114                kind,
115                message: format!(
116                    "{kind} fetch failed for {}; ({status_code}): {body}",
117                    self.execution_id,
118                ),
119            });
120        }
121
122        resp.bytes()
123            .await
124            .map(|b| b.to_vec())
125            .map_err(|e| Error::FetchConfig {
126                kind,
127                message: e.to_string(),
128            })
129    }
130}

132impl Client for HttpClient {
133    fn kustomize_patch_for_execution(
134        &self,
135        outdir: &Path,
136        toolbox_image_pull_policy: &str,
137    ) -> String {
138        KUSTOMIZE_PATCH
139            .replace("__ORCHESTRATOR_URL__", self.orchestrator_url.as_str())
140            .replace("__EXECUTION_ID__", &self.execution_id.to_string())
141            .replace("__EXECUTION_TOKEN__", &self.execution_token.to_string())
142            .replace("__OUTDIR__", &outdir.to_string_lossy())
143            .replace("__IMAGE_PULL_POLICY__", toolbox_image_pull_policy)
144            .replace("__FILE_PROVIDERS_LABEL__", FILE_PROVIDERS_LABEL)
145            .replace("__LOG_COLLECTION_LABEL__", LOG_COLLECTION_LABEL)
146    }
147
148    async fn update_status(
149        &self,
150        status: Status,
151        exit_status: Option<ExitStatus>,
152        message: Option<String>,
153    ) -> Result<(), Error> {
154        let url = self
155            .orchestrator_url
156            .join(&format!("test-execution/{}/status", self.execution_id))
157            .map_err(|e| Error::UpdateStatus {
158                message: e.to_string(),
159            })?;
160
161        let payload = SetStatusPayload {
162            status,
163            exit_code: exit_status
164                .and_then(|status| status.code())
165                .map(|code| code as u8),
166            message,
167        };
168
169        info!(id=%self.execution_id, %status, "updating execution status");
170        let resp = reqwest::Client::new()
171            .post(url)
172            .bearer_auth(self.execution_token)
173            .json(&payload)
174            .send()
175            .await
176            .map_err(|e| Error::UpdateStatus {
177                message: e.to_string(),
178            })?;
179
180        let status_code = resp.status();
181        if status_code.is_success() {
182            info!(id=%self.execution_id, %status, "update successful");
183            Ok(())
184        } else {
185            let body = match resp.text().await {
186                Ok(body) => body,
187                Err(err) => format!("[body could not be read: {err}]"),
188            };
189            Err(Error::UpdateStatus {
190                message: format!(
191                    "status update to {status} failed for {}; ({status_code}): {body}",
192                    self.execution_id
193                ),
194            })
195        }
196    }
197
198    async fn generate_upload_urls(&self) -> Result<UploadUrls, Error> {
199        let url = self
200            .orchestrator_url
201            .join(&format!(
202                "test-execution/{}/generate-upload-urls",
203                self.execution_id
204            ))
205            .map_err(|e| Error::GenerateUploadUrls {
206                message: e.to_string(),
207            })?;
208
209        info!(id=%self.execution_id, "requesting upload URLs");
210        let resp = reqwest::Client::new()
211            .post(url)
212            .bearer_auth(self.execution_token)
213            .json(&GenerateUploadUrlsPayload {})
214            .send()
215            .await
216            .map_err(|e| Error::GenerateUploadUrls {
217                message: e.to_string(),
218            })?;
219
220        let status_code = resp.status();
221        if !status_code.is_success() {
222            let body = resp.text().await.unwrap_or_default();
223            return Err(Error::GenerateUploadUrls {
224                message: format!(
225                    "generate-upload-urls failed for {}; ({status_code}): {body}",
226                    self.execution_id,
227                ),
228            });
229        }
230
231        resp.json().await.map_err(|e| Error::GenerateUploadUrls {
232            message: e.to_string(),
233        })
234    }
235
236    async fn upload_to_signed_url(&self, url: &str, body: Vec<u8>) -> Result<(), Error> {
237        info!(id=%self.execution_id, "uploading artifact");
238        let resp = reqwest::Client::new()
239            .put(url)
240            .body(body)
241            .send()
242            .await
243            .map_err(|e| Error::Upload {
244                message: e.to_string(),
245            })?;
246
247        let status_code = resp.status();
248        if !status_code.is_success() {
249            let body = resp.text().await.unwrap_or_default();
250            return Err(Error::Upload {
251                message: format!("signed upload PUT failed ({status_code}): {body}"),
252            });
253        }
254
255        Ok(())
256    }
257
258    async fn fetch_environment_config(&self) -> Result<Vec<u8>, Error> {
259        self.fetch_config("environment").await
260    }
261
262    async fn fetch_scenario_config(&self) -> Result<Vec<u8>, Error> {
263        self.fetch_config("scenario").await
264    }
