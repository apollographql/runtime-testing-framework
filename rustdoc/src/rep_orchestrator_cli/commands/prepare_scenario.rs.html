10// On exit, the wrapper touches a sentinel file so the output-collector container (which runs
11// in parallel with the scenario-runner) can detect completion via polling. The real scenario
12// exit code is written to __EXIT_STATUS_FILE__ for the output-collector to read.
13const RUN_SCRIPT_TEMPLATE: &str = r#"#!/bin/sh
14set -ex
15
16trap 'touch __EXIT_SENTINEL__' EXIT
17
18mkdir -p __OUTPUT_DIR__
19
20{
21  (
22    set -ex
23__SCENARIO_ENV__
24    export OUTDIR=__OUTPUT_DIR__
25    export RTF_OUTPUT="$OUTDIR/RTF_OUTPUT"
26
27__SCENARIO_COMMAND__
28  ) 2>&1
29  echo $? > __EXIT_STATUS_FILE__
30} | tee __OUTPUT_DIR__/output.log
31"#;

33/// Resolve the scenario config into `<shared_dir>/providers` and write a `run.sh` script into
34/// `<shared_dir>` that inlines the resolved env vars and invokes the user's scenario command.
35pub async fn prepare_scenario(
36    shared_dir: &Path,
37    scenario_command: &str,
38    ctx: &impl CliContext,
39) -> crate::Result<()> {
40    let paths = SharedPaths::new(shared_dir);
41
42    info_status!(
43        ctx,
44        Status::Provisioning,
45        "resolving scenario configuration"
46    )?;
47
48    let cfg_bytes = ctx.orchestrator_client().fetch_scenario_config().await?;
49    let cfg_path = &temp_dir().join("scenario.yaml");
50    ctx.write_file(cfg_path, &cfg_bytes)?;
51
52    info!("resolving scenario");
53    ctx.run_shell(
54        Command::new("rtf")
55            .args(["resolve", "scenario"])
56            .arg(cfg_path)
57            .args(["--outdir", &paths.providers_dir.to_string_lossy()]),
58    )
59    .await?;
60
61    let scenario_env = ctx.read_file_to_string(&paths.providers_dir.join("scenario.env"))?;
62
63    let run_sh = render_run_script(&paths, &scenario_env, scenario_command);
64    ctx.write_file(&paths.run_script, run_sh.as_bytes())?;
65    ctx.set_permissions_mode(&paths.run_script, 0o777)?;
66
67    info_status!(ctx, Status::Running, "beginning scenario execution")?;
68
69    Ok(())
70}

72struct SharedPaths {
73    providers_dir: PathBuf,
74    output_dir: PathBuf,
75    run_script: PathBuf,
76    exit_sentinel: PathBuf,
77    exit_status_file: PathBuf,
78}

80impl SharedPaths {
81    fn new(shared_dir: &Path) -> Self {
82        Self {
83            providers_dir: shared_dir.join("providers"),
84            output_dir: shared_dir.join("output"),
85            run_script: shared_dir.join("run.sh"),
86            exit_sentinel: shared_dir.join("scenario-exited"),
87            exit_status_file: shared_dir.join("scenario_exit_status"),
88        }
89    }
90}

92fn render_run_script(paths: &SharedPaths, scenario_env: &str, command: &str) -> String {
93    let indented_env = scenario_env
94        .lines()
95        .map(|l| format!("    {l}"))
96        .collect::<Vec<_>>()
97        .join("\n");
98
99    let indented_command = command
100        .lines()
101        .map(|l| format!("    {l}"))
102        .collect::<Vec<_>>()
103        .join("\n");
104
105    RUN_SCRIPT_TEMPLATE
106        .replace("__EXIT_SENTINEL__", &paths.exit_sentinel.to_string_lossy())
107        .replace("__OUTPUT_DIR__", &paths.output_dir.to_string_lossy())
108        .replace("__SCENARIO_ENV__", &indented_env)
109        .replace("__SCENARIO_COMMAND__", &indented_command)
110        .replace(
111            "__EXIT_STATUS_FILE__",
112            &paths.exit_status_file.to_string_lossy(),
113        )
114}

116#[cfg(test)]
117mod tests {
118    use super::*;
119
120    fn paths() -> SharedPaths {
121        SharedPaths::new(&PathBuf::from("/shared"))
122    }
123
124    #[test]
125    fn run_script_substitutes_all_placeholders() {
126        let rendered = render_run_script(
127            &paths(),
128            "export FOO=bar\nexport BAZ=qux",
129            "echo hi > $RTF_OUTPUT",
130        );
131
132        assert!(
133            !rendered.contains("__"),
134            "unsubstituted placeholder in:\n{rendered}"
135        );
136        assert!(rendered.contains("trap 'touch /shared/scenario-exited' EXIT"));
137        assert!(rendered.contains("mkdir -p /shared/output"));
138        assert!(rendered.contains("echo $? > /shared/scenario_exit_status"));
139        assert!(rendered.contains("tee /shared/output/output.log"));
140    }
141
142    #[test]
143    fn run_script_inlines_scenario_env_and_command_with_indent() {
144        let rendered = render_run_script(
145            &paths(),
146            "export FOO=bar\nexport BAZ=qux",
147            "first_line\nsecond_line",
148        );
149
150        // Both env and command should land inside the subshell with 4-space indent
151        assert!(rendered.contains("    export FOO=bar\n    export BAZ=qux"));
152        assert!(rendered.contains("    first_line\n    second_line"));
153    }
