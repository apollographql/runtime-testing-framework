21/// Build a kube client from a kubeconfig file path.
22// TODO: abstract this into Context. We may need to newtype Client to make it actually useful for testing.
23async fn client_from_kubeconfig(kubeconfig_path: Option<&Path>) -> CliResult<Client> {
24    match kubeconfig_path {
25        Some(path) => {
26            let kfg = Kubeconfig::read_from(path)
27                .with_context(|| format!("failed to read kubeconfig from {}", path.display()))
28                .map_err(CliError::unrunnable)?;
29
30            let config = Config::from_custom_kubeconfig(kfg, &KubeConfigOptions::default())
31                .await
32                .context("failed to build kube config")
33                .map_err(CliError::unrunnable)?;
34
35            Client::try_from(config)
36        }
37        None => Client::try_default().await,
38    }
39    .context("failed to create kube client from in-cluster config")
40    .map_err(CliError::unrunnable)
41}
