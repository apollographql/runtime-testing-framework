1use crate::{
2    commands::{MANAGER_NAME, client_from_kubeconfig},
3    error::{CliError, CliResult},
4};
5use anyhow::Context;
6use k8s_openapi::api::core::v1::{Secret, ServiceAccount};
7use kube::{
8    Api,
9    api::{ObjectMeta, Patch, PatchParams},
10};
11use serde_json::json;
12use std::collections::BTreeMap;
13use std::path::Path;
14use tracing::info;
15
16pub async fn create_pull_secret(
17    namespace: &str,
18    kubeconfig_path: &Path,
19    docker_config_path: &Path,
20) -> CliResult<()> {
21    info!("Creating image pull secret in namespace '{namespace}'...");
22
23    let client = client_from_kubeconfig(Some(kubeconfig_path)).await?;
24
25    let docker_config_json = std::fs::read(docker_config_path)
26        .with_context(|| {
27            format!(
28                "Failed to read docker config from {}",
29                docker_config_path.display()
30            )
31        })
32        .map_err(CliError::unrunnable)?;
33
34    let secret = Secret {
35        metadata: ObjectMeta {
36            name: Some("gcr-secret".to_owned()),
37            namespace: Some(namespace.to_owned()),
38            ..Default::default()
39        },
40        type_: Some("kubernetes.io/dockerconfigjson".to_owned()),
41        data: Some(BTreeMap::from([(
42            ".dockerconfigjson".to_owned(),
43            k8s_openapi::ByteString(docker_config_json),
44        )])),
45        ..Default::default()
46    };
47
48    let secrets: Api<Secret> = Api::namespaced(client.clone(), namespace);
49    secrets
50        .patch(
51            "gcr-secret",
52            &PatchParams::apply(MANAGER_NAME),
53            &Patch::Apply(&secret),
54        )
55        .await
56        .context("Failed to create pull secret")
57        .map_err(CliError::unrunnable)?;
58
59    info!("Patching default service account...");
60    let sa_api: Api<ServiceAccount> = Api::namespaced(client, namespace);
61    let patch = json!({
62        "imagePullSecrets": [{"name": "gcr-secret"}]
63    });
64    sa_api
65        .patch("default", &PatchParams::default(), &Patch::Strategic(patch))
66        .await
67        .context("Failed to patch default service account")
68        .map_err(CliError::unrunnable)?;
69    info!("Pull secret created successfully.");
70
71    Ok(())
72}
