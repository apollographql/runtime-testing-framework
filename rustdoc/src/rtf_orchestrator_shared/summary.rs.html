7/// A page of historic [TestRunSummary]s matching a set of query filters, along with the total
8/// number of runs matching those filters (ignoring pagination) so that callers can page through
9/// the full result set.
10#[derive(Default, Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
11pub struct TestRunListResponse {
12    pub runs: Vec<TestRunSummary>,
13    pub total: i64,

31pub struct TestExecutionSummary {

33    /// The parent [TestRunSummary]'s id. Only populated when this summary is fetched directly
