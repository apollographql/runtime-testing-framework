1//! Types describing the workload clusters available to the orchestrator
2use chrono::{DateTime, Utc};
3use serde::{Deserialize, Serialize};
4use std::collections::BTreeMap;
5
6#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
7pub struct ClusterSummaryResponse {
8    pub max_queued_executions: usize,
9    pub pools: Vec<WorkloadPoolSummary>,
10}
11
12#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
13pub struct WorkloadPoolSummary {
14    pub name: String,
15    pub clusters: Vec<WorkloadClusterSummary>,
16}
17
18#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
19pub struct WorkloadClusterSummary {
20    pub name: String,
21    pub execution: ClusterExecutionSummary,
22    pub hourly_executions: Vec<HourlyCount>,
23    /// `None` when this cluster's node data could not be fetched (e.g. it is unreachable).
24    pub nodes: Option<NodesSummary>,
25}
26
27#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
28pub struct NodesSummary {
29    pub by_instance_type: BTreeMap<String, usize>,
30}
31
32#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
33pub struct ClusterExecutionSummary {
34    pub max_concurrent: usize,
35    pub failed_execution_ttl_secs: u64,
36    pub retry_window_secs: u64,
37    pub poll_interval_secs: u64,
38    pub exclusive_nodes: bool,
39    pub scenario_node_selector: BTreeMap<String, String>,
40    pub namespace_cleanup_timeout_secs: u64,
41    pub per_user: PerUserExecutionSummary,
42}
43
44#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
45pub struct PerUserExecutionSummary {
46    pub max_concurrent_runs: usize,
47    pub max_queued_runs: usize,
48    pub max_queued_executions: usize,
49    pub max_runs_per_hour: usize,
50}
51
52#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
53pub struct HourlyCount {
54    pub hour: DateTime<Utc>,
55    pub count: u64,
56}
