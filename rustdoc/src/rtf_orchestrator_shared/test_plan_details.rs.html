52impl TestPlanDetailsParams {
53    pub fn history_window(&self, now: DateTime<Utc>) -> HistoryWindow {
54        let days_back = self.days_back.min(MAX_DAYS_BACK);
55        let days = self.days.clamp(1, MAX_DAYS);
56        let today = now.date_naive();
57        let midnight = |date: NaiveDate| date.and_time(NaiveTime::MIN).and_utc();
58
59        let end_of_today = midnight(
60            today
61                .checked_add_days(Days::new(1))
62                .expect("a date one day after now is representable"),
63        );
64        let from = midnight(
65            today
66                .checked_sub_days(Days::new(days_back.into()))
67                .unwrap_or(today),
68        );
69        let raw_to = from
70            .checked_add_days(Days::new(days.into()))
71            .unwrap_or(end_of_today);
72
73        let to = if raw_to >= midnight(today) {
74            end_of_today
75        } else {
76            raw_to
77        };
78
79        HistoryWindow { from, to }
80    }
81}

83#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
84pub struct HistoryWindow {
85    pub from: DateTime<Utc>,
86    pub to: DateTime<Utc>,
87}
88
89impl HistoryWindow {
90    pub fn days(&self) -> Vec<String> {
91        let mut days = Vec::new();
92        let mut date = self.from.date_naive();
93        let last = self.to.date_naive();
94
95        while date < last {
96            days.push(date.format(DAY_FORMAT).to_string());
97            match date.checked_add_days(Days::new(1)) {
98                Some(next) => date = next,
99                None => break,
100            }
101        }
102
103        days
104    }
105}

147impl TestPlanVariable {
148    pub fn from_test_plan(test_plan: &OrchestratorTestPlan) -> Vec<Self> {
149        let include_keys: HashSet<&String> = test_plan
150            .matrix
151            .compound
152            .values()
153            .flat_map(|entries| entries.iter().take(1).flat_map(|group| group.keys()))
154            .collect();
155
156        let sections = [
157            (
158                ConfigSection::Scenario,
159                &test_plan.scenario.variable_definitions,
160            ),
161            (
162                ConfigSection::Environment,
163                &test_plan.environment.variable_definitions,
164            ),
165        ];
166
167        let mut by_name: BTreeMap<&String, Vec<VariableDeclaration>> = BTreeMap::new();
168        for (section, definitions) in sections {
169            for def in definitions
170                .iter()
171                .filter(|def| !include_keys.contains(&def.name))
172            {
173                by_name
174                    .entry(&def.name)
175                    .or_default()
176                    .push(VariableDeclaration::new(section, def));
177            }
178        }
179
180        by_name
181            .into_iter()
182            .map(|(name, declarations)| {
183                let current_value = VariableValue::try_new(test_plan, name);
184
185                TestPlanVariable {
186                    name: name.clone(),
187                    required: current_value.is_none()
188                        && declarations.iter().any(|d| d.default.is_none()),
189                    declarations,
190                    current_value,
191                }
192            })
193            .collect()
194    }
195}

343impl EnvironmentSummary {
344    /// A summary of what the environment will deploy, resolved for the first matrix variant.
345    /// `None` if the environment is skipped entirely (there's nothing to summarise).
346    pub async fn try_from_test_plan(
347        test_plan: &OrchestratorTestPlan,
348        config_file_url: &str,
349        ctx: &impl ResolutionContext,
350    ) -> Result<Option<Self>, EnvironmentSummaryError> {
351        let (variant_name, mut variant) = test_plan
352            .try_expand_variant(0)
353            .map_err(EnvironmentSummaryError::Formats)?
354            .expect("a test plan always expands to at least one variant");
355
356        let variables = take(&mut variant.variables);
357        let template_ctx =
358            TemplateContext::new(variables, HashMap::new(), ctx.custom_provider_definitions());
359        variant
360            .try_template(&mut Vec::new(), &StableSource::TestPlan, &template_ctx)
361            .map_err(EnvironmentSummaryError::Templating)?;
362
363        let resolved_for_variant = (!test_plan.matrix.is_empty()).then_some(variant_name);
364
365        match variant.environment.execution {
366            OrchestratorEnvironment::K8s(k8s_env) => {
367                let mut manifests: Vec<_> = k8s_env
368                    .manifest_urls(ctx)
369                    .into_iter()
370                    .map(|(name, url)| match url {
371                        Some(url) => ManifestLocation::gh(name, url),
372                        None => ManifestLocation::inline(name, config_file_url),
373                    })
374                    .collect();
375
376                manifests.sort_unstable();
377
378                Ok(Some(Self::K8s(K8sEnvironmentSummary {
379                    manifests,
380                    resolved_for_variant,
381                })))
382            }
383
384            OrchestratorEnvironment::Null(_) => Ok(None),
385
386            OrchestratorEnvironment::DockerCompose(mut dce) => {
387                dce.inline_manifests(ctx, &mut HashMap::new())
388                    .await
389                    .map_err(EnvironmentSummaryError::Inlining)?;
390
391                let services = dce
392                    .services()
393                    .map_err(|e| EnvironmentSummaryError::Formats(e.into()))?;
394
395                Ok(Some(Self::DockerCompose(ComposeEnvironmentSummary::new(
396                    services,
397                    services_vary_by_matrix(test_plan),
398                    resolved_for_variant,
399                ))))
400            }
401        }
402    }
403}

405/// Whether any compose file provider is templated on a matrix variable, which is what would make
406/// different executions deploy a different set of services.
407fn services_vary_by_matrix(test_plan: &OrchestratorTestPlan) -> bool {
408    let matrix_keys: HashSet<&String> = test_plan.matrix.keys().collect();
409
410    test_plan
411        .environment
412        .execution
413        .manifest_template_variables()
414        .iter()
415        .any(|name| matrix_keys.contains(name))
416}
