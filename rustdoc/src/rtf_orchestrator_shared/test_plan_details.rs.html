53impl TestPlanDetailsParams {
54    pub fn history_window(&self, now: DateTime<Utc>) -> HistoryWindow {
55        let days_back = self.days_back.min(MAX_DAYS_BACK);
56        let days = self.days.clamp(1, MAX_DAYS);
57        let today = now.date_naive();
58        let midnight = |date: NaiveDate| date.and_time(NaiveTime::MIN).and_utc();
59
60        let end_of_today = midnight(
61            today
62                .checked_add_days(Days::new(1))
63                .expect("a date one day after now is representable"),
64        );
65        let from = midnight(
66            today
67                .checked_sub_days(Days::new(days_back.into()))
68                .unwrap_or(today),
69        );
70        let to = from
71            .checked_add_days(Days::new(days.into()))
72            .unwrap_or(end_of_today)
73            .min(end_of_today);
74
75        HistoryWindow { from, to }
76    }
77}

79#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
80pub struct HistoryWindow {
81    pub from: DateTime<Utc>,
82    pub to: DateTime<Utc>,
83}
84
85impl HistoryWindow {
86    pub fn days(&self) -> Vec<String> {
87        let mut days = Vec::new();
88        let mut date = self.from.date_naive();
89        let last = self.to.date_naive();
90
91        while date < last {
92            days.push(date.format(DAY_FORMAT).to_string());
93            match date.checked_add_days(Days::new(1)) {
94                Some(next) => date = next,
95                None => break,
96            }
97        }
98
99        days
100    }
101}

133#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
134pub struct TestPlanVariable {
135    pub name: String,
136    pub declarations: Vec<VariableDeclaration>,
137    #[serde(skip_serializing_if = "Option::is_none")]
138    pub current_value: Option<VariableValue>,
139    pub required: bool,
140}
141
142impl TestPlanVariable {
143    pub fn from_test_plan(test_plan: &OrchestratorTestPlan) -> Vec<Self> {
144        let include_keys: HashSet<&String> = test_plan
145            .matrix
146            .include
147            .iter()
148            .flat_map(|group| group.keys())
149            .collect();
150
151        let sections = [
152            (
153                ConfigSection::Scenario,
154                &test_plan.scenario.variable_definitions,
155            ),
156            (
157                ConfigSection::Environment,
158                &test_plan.environment.variable_definitions,
159            ),
160        ];
161
162        let mut by_name: BTreeMap<&String, Vec<VariableDeclaration>> = BTreeMap::new();
163        for (section, definitions) in sections {
164            for def in definitions
165                .iter()
166                .filter(|def| !include_keys.contains(&def.name))
167            {
168                by_name
169                    .entry(&def.name)
170                    .or_default()
171                    .push(VariableDeclaration::new(section, def));
172            }
173        }
174
175        by_name
176            .into_iter()
177            .map(|(name, declarations)| {
178                let current_value = VariableValue::try_new(test_plan, name);
179
180                TestPlanVariable {
181                    name: name.clone(),
182                    required: current_value.is_none()
183                        && declarations.iter().any(|d| d.default.is_none()),
184                    declarations,
185                    current_value,
186                }
187            })
188            .collect()
189    }
190}

213#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
214#[serde(rename_all = "snake_case")]
215pub enum VariableValue {
216    Scalar(Scalar),
217    Dimension(Vec<Scalar>),
218}
219
220impl VariableValue {
221    fn try_new(test_plan: &OrchestratorTestPlan, name: &String) -> Option<Self> {
222        if let Some(scalar) = test_plan.variables.get(name) {
223            return Some(Self::Scalar(scalar.clone()));
224        }
225
226        test_plan
227            .matrix
228            .dimensions
229            .get(name)
230            .map(|values| Self::Dimension(values.clone()))
231    }

266#[derive(Debug, Default, Clone, PartialEq, Eq, Serialize, Deserialize)]
267pub struct EnvironmentSummary {
268    pub services: Vec<EnvironmentService>,
269    pub has_variable_replicas: bool,
270    pub services_vary_by_matrix: bool,
271    #[serde(skip_serializing_if = "Option::is_none")]
272    pub resolved_for_variant: Option<String>,
273}
274
275impl EnvironmentSummary {
276    /// The services deployed for each execution, resolved for the first matrix variant.
277    pub async fn try_from_test_plan(
278        test_plan: &OrchestratorTestPlan,
279        ctx: &impl ResolutionContext,
280    ) -> Result<Self, EnvironmentSummaryError> {
281        let (variant_name, mut variant) = test_plan
282            .try_expand_variant(0)
283            .map_err(EnvironmentSummaryError::Formats)?
284            .expect("a test plan always expands to at least one variant");
285
286        let variables = take(&mut variant.variables);
287        let template_ctx =
288            TemplateContext::new(variables, HashMap::new(), ctx.custom_provider_definitions());
289        variant
290            .try_template(&mut Vec::new(), &StableSource::TestPlan, &template_ctx)
291            .map_err(EnvironmentSummaryError::Templating)?;
292
293        let mut env = variant.environment.execution;
294        env.inline_compose_files(ctx, &mut HashMap::new())
295            .await
296            .map_err(EnvironmentSummaryError::Inlining)?;
297
298        let services = env
299            .services()
300            .map_err(|e| EnvironmentSummaryError::Formats(e.into()))?;
301
302        Ok(Self::new(
303            services,
304            services_vary_by_matrix(test_plan),
305            (!test_plan.matrix.is_empty()).then_some(variant_name),
306        ))
307    }
308
309    fn new(
310        services: Vec<EnvironmentService>,
311        services_vary_by_matrix: bool,
312        resolved_for_variant: Option<String>,
313    ) -> Self {
314        Self {
315            has_variable_replicas: services
316                .iter()
317                .any(|s| !matches!(s.replicas, ServiceReplicas::Fixed(_))),
318            services,
319            services_vary_by_matrix,
320            resolved_for_variant,
321        }
322    }
323}

325/// Whether any compose file provider is templated on a matrix variable, which is what would make
326/// different executions deploy a different set of services.
327fn services_vary_by_matrix(test_plan: &OrchestratorTestPlan) -> bool {
328    let matrix_keys: HashSet<&String> = test_plan.matrix.keys().collect();
329
330    test_plan
331        .environment
332        .execution
333        .compose_template_variables()
334        .iter()
335        .any(|name| matrix_keys.contains(name))
336}
