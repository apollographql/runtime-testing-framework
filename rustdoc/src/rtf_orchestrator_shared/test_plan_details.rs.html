53impl TestPlanDetailsParams {
54    pub fn history_window(&self, now: DateTime<Utc>) -> HistoryWindow {
55        let days_back = self.days_back.min(MAX_DAYS_BACK);
56        let days = self.days.clamp(1, MAX_DAYS);
57        let today = now.date_naive();
58        let midnight = |date: NaiveDate| date.and_time(NaiveTime::MIN).and_utc();
59
60        let end_of_today = midnight(
61            today
62                .checked_add_days(Days::new(1))
63                .expect("a date one day after now is representable"),
64        );
65        let from = midnight(
66            today
67                .checked_sub_days(Days::new(days_back.into()))
68                .unwrap_or(today),
69        );
70        let raw_to = from
71            .checked_add_days(Days::new(days.into()))
72            .unwrap_or(end_of_today);
73
74        let to = if raw_to >= midnight(today) {
75            end_of_today
76        } else {
77            raw_to
78        };
79
80        HistoryWindow { from, to }
81    }
82}

84#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
85pub struct HistoryWindow {
86    pub from: DateTime<Utc>,
87    pub to: DateTime<Utc>,
88}
89
90impl HistoryWindow {
91    pub fn days(&self) -> Vec<String> {
92        let mut days = Vec::new();
93        let mut date = self.from.date_naive();
94        let last = self.to.date_naive();
95
96        while date < last {
97            days.push(date.format(DAY_FORMAT).to_string());
98            match date.checked_add_days(Days::new(1)) {
99                Some(next) => date = next,
100                None => break,
101            }
102        }
103
104        days
105    }
106}

139#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
140pub struct TestPlanVariable {
141    pub name: String,
142    pub declarations: Vec<VariableDeclaration>,
143    #[serde(skip_serializing_if = "Option::is_none")]
144    pub current_value: Option<VariableValue>,
145    pub required: bool,
146}
147
148impl TestPlanVariable {
149    pub fn from_test_plan(test_plan: &OrchestratorTestPlan) -> Vec<Self> {
150        let include_keys: HashSet<&String> = test_plan
151            .matrix
152            .compound
153            .values()
154            .flat_map(|entries| entries.iter().take(1).flat_map(|group| group.keys()))
155            .collect();
156
157        let sections = [
158            (
159                ConfigSection::Scenario,
160                &test_plan.scenario.variable_definitions,
161            ),
162            (
163                ConfigSection::Environment,
164                &test_plan.environment.variable_definitions,
165            ),
166        ];
167
168        let mut by_name: BTreeMap<&String, Vec<VariableDeclaration>> = BTreeMap::new();
169        for (section, definitions) in sections {
170            for def in definitions
171                .iter()
172                .filter(|def| !include_keys.contains(&def.name))
173            {
174                by_name
175                    .entry(&def.name)
176                    .or_default()
177                    .push(VariableDeclaration::new(section, def));
178            }
179        }
180
181        by_name
182            .into_iter()
183            .map(|(name, declarations)| {
184                let current_value = VariableValue::try_new(test_plan, name);
185
186                TestPlanVariable {
187                    name: name.clone(),
188                    required: current_value.is_none()
189                        && declarations.iter().any(|d| d.default.is_none()),
190                    declarations,
191                    current_value,
192                }
193            })
194            .collect()
195    }
196}

219#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
220#[serde(rename_all = "snake_case")]
221pub enum VariableValue {
222    Scalar(Scalar),
223    Dimension(Vec<Scalar>),
224}
225
226impl VariableValue {
227    fn try_new(test_plan: &OrchestratorTestPlan, name: &String) -> Option<Self> {
228        if let Some(scalar) = test_plan.variables.get(name) {
229            return Some(Self::Scalar(scalar.clone()));
230        }
231
232        test_plan
233            .matrix
234            .dimensions
235            .get(name)
236            .map(|values| Self::Dimension(values.clone()))
237    }

247impl MatrixSummary {
248    pub fn new(matrix: &Matrix) -> Self {
249        Self {
250            n_executions: matrix.n_variants(),
251            dimensions: matrix
252                .dimensions
253                .iter()
254                .map(|(k, v)| (k.clone(), v.clone()))
255                .collect(),
256            compound_groups: matrix
257                .compound
258                .iter()
259                .map(|(name, entries)| {
260                    let entries = entries
261                        .iter()
262                        .map(|g| g.iter().map(|(k, v)| (k.clone(), v.clone())).collect())
263                        .collect();
264
265                    (name.clone(), entries)
266                })
267                .collect(),
268        }
269    }

288impl EnvironmentSummary {
289    /// The services deployed for each execution, resolved for the first matrix variant.
290    pub async fn try_from_test_plan(
291        test_plan: &OrchestratorTestPlan,
292        ctx: &impl ResolutionContext,
293    ) -> Result<Self, EnvironmentSummaryError> {
294        let (variant_name, mut variant) = test_plan
295            .try_expand_variant(0)
296            .map_err(EnvironmentSummaryError::Formats)?
297            .expect("a test plan always expands to at least one variant");
298
299        let variables = take(&mut variant.variables);
300        let template_ctx =
301            TemplateContext::new(variables, HashMap::new(), ctx.custom_provider_definitions());
302        variant
303            .try_template(&mut Vec::new(), &StableSource::TestPlan, &template_ctx)
304            .map_err(EnvironmentSummaryError::Templating)?;
305
306        let mut env = variant.environment.execution;
307        env.inline_compose_files(ctx, &mut HashMap::new())
308            .await
309            .map_err(EnvironmentSummaryError::Inlining)?;
310
311        let services = env
312            .services()
313            .map_err(|e| EnvironmentSummaryError::Formats(e.into()))?;
314
315        Ok(Self::new(
316            services,
317            services_vary_by_matrix(test_plan),
318            (!test_plan.matrix.is_empty()).then_some(variant_name),
319        ))
320    }
321
322    fn new(
323        services: Vec<EnvironmentService>,
324        services_vary_by_matrix: bool,
325        resolved_for_variant: Option<String>,
326    ) -> Self {
327        Self {
328            has_variable_replicas: services
329                .iter()
330                .any(|s| !matches!(s.replicas, ServiceReplicas::Fixed(_))),
331            services,
332            services_vary_by_matrix,
333            resolved_for_variant,
334        }
335    }
336}

338/// Whether any compose file provider is templated on a matrix variable, which is what would make
339/// different executions deploy a different set of services.
340fn services_vary_by_matrix(test_plan: &OrchestratorTestPlan) -> bool {
341    let matrix_keys: HashSet<&String> = test_plan.matrix.keys().collect();
342
343    test_plan
344        .environment
345        .execution
346        .compose_template_variables()
347        .iter()
348        .any(|name| matrix_keys.contains(name))
349}
