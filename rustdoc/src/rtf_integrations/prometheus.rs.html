1//! A lightweight Prometheus API client
2use chrono::{DateTime, Utc};
3use reqwest::StatusCode;
4use serde_json::Value;
5
6/// Errors that can occur when building or using a [`PrometheusClient`].
7#[derive(Debug, thiserror::Error)]
8pub enum Error {
9    /// An underlying error from the reqwest crate
10    #[error(transparent)]
11    Reqwest(#[from] reqwest::Error),
12
13    /// Prometheus returned an API-level error response.
14    ///
15    /// See the [Prometheus HTTP API docs](https://prometheus.io/docs/prometheus/latest/querying/api/#format-overview).
16    #[error("prometheus query failed ({status}): {error_type}: {message}")]
17    Api {
18        /// 400 (bad_data), 422 (execution error), or 503 (timeout/abort)
19        status: StatusCode,
20        /// Prometheus's `errorType` field, e.g. `bad_data`, `execution`, `timeout`
21        error_type: String,
22        /// Prometheus's `error` field — human-readable message
23        message: String,
24    },
25}
26
27/// Alias for a [Result][std::result::Result] where the error variant is an [Error].
28pub type Result<T> = std::result::Result<T, Error>;
29
30/// An API client that can make requests to the Prometheus REST API.
31pub trait Client: Send + Sync {
32    /// Executes a Prometheus range query.
33    ///
34    /// See the [Prometheus HTTP API docs](https://prometheus.io/docs/prometheus/latest/querying/api/#range-queries).
35    fn query_range(
36        &self,
37        query: &str,
38        step: &str,
39        start: DateTime<Utc>,
40        end: DateTime<Utc>,
41    ) -> impl Future<Output = Result<Value>> + Send;
42}
43
44/// A lightweight Prometheus API HTTP client
45#[derive(Debug)]
46pub struct PrometheusClient {
47    url: String,
48    http_client: reqwest::Client,
49}
50
51impl PrometheusClient {
52    /// Build a new client using the supplied prometheus url
53    pub fn new(url: impl Into<String>) -> Self {
54        Self {
55            url: url.into(),
56            http_client: reqwest::Client::new(),
57        }
58    }
59}
60
61impl Client for PrometheusClient {
62    async fn query_range(
63        &self,
64        query: &str,
65        step: &str,
66        start: DateTime<Utc>,
67        end: DateTime<Utc>,
68    ) -> Result<Value> {
69        let resp = self
70            .http_client
71            .post(format!("{}/api/v1/query_range", self.url))
72            .form(&[
73                ("query", query),
74                ("start", &start.timestamp().to_string()),
75                ("end", &end.timestamp().to_string()),
76                ("step", step),
77            ])
78            .send()
79            .await?;
80
81        let status = resp.status();
82        let body: Value = resp.json().await?;
83
84        handle_response(status, body)
85    }
86}
87
88/// Turns a Prometheus HTTP response's status and JSON body into a result, surfacing the
89/// `errorType`/`error` fields Prometheus returns on non-2xx responses instead of silently
90/// treating them as an empty result.
91fn handle_response(status: StatusCode, body: Value) -> Result<Value> {
92    if !status.is_success() {
93        return Err(Error::Api {
94            status,
95            error_type: body["errorType"].as_str().unwrap_or("unknown").to_owned(),
96            message: body["error"]
97                .as_str()
98                .unwrap_or("no error message returned")
99                .to_owned(),
100        });
101    }
102
103    Ok(body["data"]["result"].clone())
104}

106#[cfg(test)]
107mod tests {
108    use super::*;
109    use reqwest::StatusCode;
110    use serde_json::json;
111    use simple_test_case::test_case;
112
113    #[test_case(
114        StatusCode::OK,
115        json!({"status": "success", "data": {"resultType": "matrix", "result": ["ok"]}}),
116        Ok(json!(["ok"]));
117        "200 success returns data.result"
118    )]
119    #[test_case(
120        StatusCode::BAD_REQUEST,
121        json!({"status": "error", "errorType": "bad_data", "error": "parse error at char 1"}),
122        Err(Error::Api {
123            status: StatusCode::BAD_REQUEST,
124            error_type: "bad_data".to_owned(),
125            message: "parse error at char 1".to_owned(),
126        });
127        "400 bad_data surfaces errorType and error"
128    )]
129    #[test_case(
130        StatusCode::UNPROCESSABLE_ENTITY,
131        json!({"status": "error", "errorType": "execution", "error": "query timed out in expression evaluation"}),
132        Err(Error::Api {
133            status: StatusCode::UNPROCESSABLE_ENTITY,
134            error_type: "execution".to_owned(),
135            message: "query timed out in expression evaluation".to_owned(),
136        });
137        "422 execution error surfaces errorType and error"
138    )]
139    #[test_case(
140        StatusCode::SERVICE_UNAVAILABLE,
141        json!({"status": "error", "errorType": "timeout", "error": "query aborted"}),
142        Err(Error::Api {
143            status: StatusCode::SERVICE_UNAVAILABLE,
144            error_type: "timeout".to_owned(),
145            message: "query aborted".to_owned(),
146        });
147        "503 timeout surfaces errorType and error"
148    )]
149    #[test_case(
150        StatusCode::INTERNAL_SERVER_ERROR,
151        json!({}),
152        Err(Error::Api {
153            status: StatusCode::INTERNAL_SERVER_ERROR,
154            error_type: "unknown".to_owned(),
155            message: "no error message returned".to_owned(),
156        });
157        "non-2xx with no errorType/error falls back to defaults"
158    )]
159    #[test]
160    fn handle_response_cases(status: StatusCode, body: Value, expected: Result<Value>) {
161        let actual = handle_response(status, body);
162
163        match (actual, expected) {
164            (Ok(a), Ok(e)) => assert_eq!(a, e),
165            (Err(a), Err(e)) => assert_eq!(a.to_string(), e.to_string()),
166            (actual, expected) => panic!("expected {expected:?}, got {actual:?}"),
167        }
168    }
