1//! A lightweight GitHub API client
2use bytes::Bytes;
3use std::{future::Future, string::FromUtf8Error, sync::Arc};
4
5const PKG_VERSION: &str = env!("CARGO_PKG_VERSION");
6pub(crate) const GITHUB_API_URL: &str = "https://api.github.com";
7
8/// Error variants that we can encounter when making requests to the GitHub REST API
9#[derive(Debug, thiserror::Error)]
10pub enum Error {
11    /// No GitHub API client available
12    #[error("no GitHub client available")]
13    NoClient,
14
15    // Wrapped errors
16    /// Invalid utf-8 found while trying to decode file content from GitHub
17    #[error(transparent)]
18    InvalidUtf8(#[from] FromUtf8Error),
19
20    /// An underlying error from the reqwest crate
21    #[error(transparent)]
22    Reqwest(#[from] reqwest::Error),
23}
24
25/// An API client that can make requests to the GitHub REST API.
26pub trait Client: Send + Sync {
27    /// Attempt to pull the raw file content of a given file as [Bytes] from the specified GitHub
28    /// repo.
29    ///
30    /// The API token used to create this client must have access to the repo in question.
31    fn raw_file_content<G: AsRef<str> + Send>(
32        &self,
33        org: &str,
34        repo: &str,
35        path: &str,
36        git_ref: Option<G>,
37    ) -> impl Future<Output = Result<Bytes, Error>> + Send;
38
39    /// Attempt to pull the raw file content of a given file as a utf-8 [String] from the specified
40    /// GitHub repo.
41    ///
42    /// The API token used to create this client must have access to the repo in question.
43    fn string_file_content<G: AsRef<str> + Send>(
44        &self,
45        org: &str,
46        repo: &str,
47        path: &str,
48        git_ref: Option<G>,
49    ) -> impl Future<Output = Result<String, Error>> + Send {
50        async move {
51            let bytes = self.raw_file_content(org, repo, path, git_ref).await?;
52            Ok(String::from_utf8(bytes.to_vec())?)
53        }
54    }
55}

57/// A lightweight GitHub API client for the subset of REST endpoints we need to work with.
58///
59/// This client makes use of bearer auth as documented for the GitHub API here:
60///   <https://docs.github.com/en/rest/authentication/authenticating-to-the-rest-api?apiVersion=2022-11-28>
61///
62/// Creation of access tokens is documented here:
63///   <https://docs.github.com/en/authentication/keeping-your-account-and-data-secure/managing-your-personal-access-tokens#types-of-personal-access-tokens>
64#[derive(Clone, Debug)]
65pub struct GithubClient {
66    pub(crate) inner: reqwest::Client,
67    pub(crate) base_url: Arc<str>,
68    pub(crate) api_token: Arc<str>,
69}
70
71impl GithubClient {
72    /// Construct a new client with a dedicated underlying [reqwest::Client].
73    pub fn new(api_token: impl Into<String>) -> Self {
74        Self {
75            inner: reqwest::Client::new(),
76            base_url: GITHUB_API_URL.into(),
77            api_token: api_token.into().into(),
78        }
79    }
80
81    /// Construct a new client with a dedicated underlying [reqwest::Client] for the provided base
82    /// URL.
83    pub fn new_with_base_url(base_url: impl Into<String>, api_token: impl Into<String>) -> Self {
84        Self {
85            inner: reqwest::Client::new(),
86            base_url: base_url.into().into(),
87            api_token: api_token.into().into(),
88        }
89    }
90}
91
92impl Client for GithubClient {
93    async fn raw_file_content<G: AsRef<str> + Send>(
94        &self,
95        org: &str,
96        repo: &str,
97        path: &str,
98        git_ref: Option<G>,
99    ) -> Result<Bytes, Error> {
100        let mut url = format!("{}/repos/{org}/{repo}/contents/{path}", self.base_url);
101        if let Some(r) = git_ref {
102            url = format!("{url}?ref={}", r.as_ref())
103        }
104
105        let res = self
106            .inner
107            .get(url)
108            .bearer_auth(&self.api_token)
109            .header("User-Agent", format!("apollo-rtf-{PKG_VERSION}"))
110            .header("X-GitHub-Api-Version", "2022-11-28")
111            .header("accept", "application/vnd.github.v3.raw")
112            .send()
113            .await?
114            .error_for_status()?;
115
116        Ok(res.bytes().await?)
117    }
118}
