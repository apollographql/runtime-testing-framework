1//! IAP-authenticated HTTP client for the REP orchestrator.
2//!
3//! This module provides everything needed to make authenticated HTTP requests
4//! to the REP orchestrator service, which is protected by Google Cloud IAP.
5//! Authentication uses an interactive user OAuth loopback flow against the
6//! same Web OAuth client that IAP itself is configured with: the CLI opens a
7//! browser, the user consents, and the resulting refresh token is cached on
8//! disk. The Web client's id and secret are fetched at runtime from Secret
9//! Manager (using the user's ADC access token to bootstrap).
10use reqwest::StatusCode;
11
12mod auth;
13mod client;
14
15pub use client::OrchestratorClient;

23/// The Secret Manager secret name storing the IAP OAuth client ID.
24///
25/// This is the Web OAuth client that IAP is configured with — its ID is both
26/// the audience IAP validates against and the `client_id` the CLI uses to
27/// drive the user-consent loopback flow. `http://localhost` must be in the
28/// client's authorized redirect URIs.
29const IAP_OAUTH_CLIENT_ID_SECRET_NAME: &str = "iap-orchestrator-client-id";

31/// The Secret Manager secret name storing the IAP OAuth client secret.

34/// Errors that can occur when building or using an [`OrchestratorClient`].
35#[derive(Debug, thiserror::Error)]
36pub enum Error {
37    /// A request to the orchestrator failed
38    #[error("got {status} response from orchestrator: {body:?}")]
39    FailedRequest {
40        /// The status code returned by the orchestrator
41        status: StatusCode,
42        /// The response body returned by the orchestrator
43        body: String,
44    },
45
46    /// The secret payload returned by Secret Manager could not be decoded.
47    #[error("could not decode secret payload: {0}")]
48    InvalidSecret(String),
49
50    /// The provided endpoint produced an invalid URL when combined with the client base URL.
51    #[error("invalid URL: {0}")]
52    InvalidUrl(String),
53
54    /// The interactive user OAuth flow failed.
55    #[error(transparent)]
56    OauthFlow(#[from] OauthError),
57
58    /// Application Default Credentials error.
59    #[error(transparent)]
60    Adc(#[from] AdcError),
61
62    /// The on-disk token cache could not be read or written.
63    #[error("token cache error: {0}")]
64    TokenCache(#[from] std::io::Error),
65
66    /// An error from the Google Cloud client builder.
67    #[error(transparent)]
68    GoogleClientError(#[from] google_cloud_gax::client_builder::Error),
69
70    /// An error from the Google Secret Manager API.
71    #[error(transparent)]
72    GoogleSecretManager(#[from] google_cloud_secretmanager_v1::Error),
73
74    /// The caller's GCP credentials lack IAM permission to read a Secret Manager secret.
75    #[error(
76        "permission denied reading secret `{secret_name}` from project `{project}` — \
77         ensure your account has the `roles/secretmanager.secretAccessor` role on \
78         the `{project}` GCP project"
79    )]
80    SecretManagerPermissionDenied {
81        /// The short secret name that was requested.
82        secret_name: String,
83        /// The GCP project that hosts the secret.
84        project: &'static str,
85    },
86
87    /// An HTTP transport error.
88    #[error(transparent)]
89    Http(#[from] reqwest::Error),
90
91    /// A serde json error.
92    #[error(transparent)]
93    Json(#[from] serde_json::Error),
94}

96/// Alias for a [Result][std::result::Result] where the error variant is an [Error].
97pub type Result<T> = std::result::Result<T, Error>;

99/// Errors from Application Default Credentials.
100#[derive(Debug, thiserror::Error)]
101pub enum AdcError {
102    /// No GCP credentials found — the user needs to run `gcloud auth application-default login`.
103    #[error("not authenticated to GCP — run `gcloud auth application-default login` first")]
104    NotAuthenticated,
105
106    /// The home directory cannot be determined, so the ADC file path is unknown.
107    #[error("unable to find home directory")]
108    NoHomeDir,
109
110    /// The ADC credentials file could not be read from disk.
111    #[error("could not read ADC credentials file: {0}")]
112    ReadFailed(#[source] std::io::Error),
113
114    /// The ADC credentials file could not be parsed as JSON.
115    #[error("could not parse ADC credentials file: {0}")]
116    ParseFailed(#[source] serde_json::Error),
117
118    /// The service-account ID token credentials object could not be built.
119    #[error("could not build service-account ID token credentials: {0}")]
120    CredentialsBuild(String),
121
122    /// The service-account ID token could not be minted.
123    #[error("could not mint service-account ID token: {0}")]
124    TokenMint(String),
125
126    /// external_account credentials are missing `service_account_impersonation_url`,
127    /// which is required to extract the service account email for signJwt.
128    #[error(
129        "external_account credentials missing service_account_impersonation_url; \
130         ensure the WIF provider is configured with service account impersonation"
131    )]
132    NoImpersonationUrl,
133
134    /// The `service_account_impersonation_url` field could not be parsed to
135    /// extract a service account email address.
136    #[error("could not extract service account email from impersonation URL: {0}")]
137    InvalidImpersonationUrl(String),
138
139    /// The IAM Credentials API rejected the `generateIdToken` call with a 403.
140    ///
141    /// The WIF principal needs `iam.serviceAccounts.generateIdToken` on the
142    /// target service account, granted by `roles/iam.serviceAccountTokenCreator`.
143    #[error(
144        "permission denied generating ID token for service account `{sa_email}` — \
145         ensure the WIF principal has `roles/iam.serviceAccountTokenCreator` on that \
146         service account"
147    )]
148    GenerateIdTokenDenied {
149        /// The service account email the token was requested for.
150        sa_email: String,
151    },
152}
153/// Errors from the interactive user OAuth loopback flow.
154#[derive(Debug, thiserror::Error)]
155pub enum OauthError {
156    /// The token cache path cannot be determined because both `HOME` and `XDG_CONFIG_HOME` are
157    /// unset.
158    #[error("cannot determine token cache location (HOME and XDG_CONFIG_HOME both unset)")]
159    NoCacheLocation,
160
161    /// A Google OAuth/token endpoint URL constant failed to parse.
162    #[error("invalid {endpoint} URL: {message}")]
163    InvalidUrl {
164        /// Which endpoint (`"auth"` or `"token"`).
165        endpoint: &'static str,
166        /// The underlying parse error message.
167        message: String,
168    },
169
170    /// The reqwest client used for token-endpoint calls could not be built.
171    #[error("could not build OAuth HTTP client: {0}")]
172    HttpClientBuild(String),
173
174    /// The loopback TCP listener could not be bound.
175    #[error("could not bind loopback listener: {0}")]
176    LoopbackBind(#[source] std::io::Error),
177
178    /// The loopback port could not be read from the bound listener.
179    #[error("could not read loopback port: {0}")]
180    LoopbackPort(#[source] std::io::Error),
181
182    /// The computed redirect URI failed to parse.
183    #[error("invalid redirect URI: {0}")]
184    InvalidRedirectUri(String),
185
186    /// The browser did not complete the redirect within the timeout window.
187    #[error("OAuth flow timed out after {0}s without a redirect")]
188    FlowTimeout(u64),
189
190    /// The `state` returned by Google did not match the issued CSRF token.
191    #[error("OAuth state mismatch — aborting to prevent CSRF")]
192    CsrfMismatch,
193
194    /// Google's token endpoint rejected the code exchange or refresh grant.
195    ///
196    /// Stringified because `oauth2`'s `RequestTokenError` carries unwieldy generic parameters.
197    #[error("Google token endpoint error: {0}")]
198    TokenEndpoint(String),
199
200    /// Google completed the flow but did not return a `refresh_token`.
201    #[error(
202        "Google did not return a refresh_token; check that the OAuth consent screen grants \
203         offline access"
204    )]
205    NoRefreshToken,
206
207    /// The loopback TCP accept call failed.
208    #[error("loopback accept failed: {0}")]
209    LoopbackAccept(#[source] std::io::Error),
210
211    /// Reading from the loopback TCP stream failed.
212    #[error("loopback read failed: {0}")]
213    LoopbackRead(#[source] std::io::Error),
214
215    /// The HTTP request received on the loopback port was malformed.
216    #[error("malformed OAuth callback: {0}")]
217    MalformedCallback(&'static str),
218
219    /// The authorization server returned an `error` parameter in the callback.
220    #[error("authorization denied: {0}")]
221    AuthorizationDenied(String),
222
223    /// The OAuth callback did not include a `code` parameter.
224    #[error("missing code in OAuth callback")]
225    MissingCode,
226
227    /// The OAuth callback did not include a `state` parameter.
228    #[error("missing state in OAuth callback")]
229    MissingState,
