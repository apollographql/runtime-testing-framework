1//! IAP-authenticated HTTP client for the REP orchestrator.
2//!
3//! This module provides everything needed to make authenticated HTTP requests
4//! to the REP orchestrator service, which is protected by Google Cloud IAP.
5//! Authentication uses an interactive user OAuth loopback flow against the
6//! same Web OAuth client that IAP itself is configured with: the CLI opens a
7//! browser, the user consents, and the resulting refresh token is cached on
8//! disk. The Web client's id and secret are fetched at runtime from Secret
9//! Manager (using the user's ADC access token to bootstrap).
10
11mod auth;
12mod client;
13
14pub use client::OrchestratorClient;

22/// The Secret Manager secret name storing the IAP OAuth client ID.
23///
24/// This is the Web OAuth client that IAP is configured with — its ID is both
25/// the audience IAP validates against and the `client_id` the CLI uses to

33/// Errors that can occur when building or using an [`OrchestratorClient`].
34#[derive(Debug, thiserror::Error)]
35pub enum Error {
36    /// The secret payload returned by Secret Manager could not be decoded.
37    #[error("could not decode secret payload: {0}")]
38    InvalidSecret(String),
39
40    /// The interactive user OAuth flow failed.
41    #[error(transparent)]
42    OauthFlow(#[from] OauthError),
43
44    /// Application Default Credentials error.
45    #[error(transparent)]
46    Adc(#[from] AdcError),
47
48    /// The on-disk token cache could not be read or written.
49    #[error("token cache error: {0}")]
50    TokenCache(#[from] std::io::Error),
51
52    /// The ID token could not be formatted as an HTTP Authorization header value.
53    #[error("invalid bearer token: {0}")]
54    InvalidBearerToken(#[source] reqwest::header::InvalidHeaderValue),
55
56    /// An error from the Google Cloud client builder.
57    #[error(transparent)]
58    GoogleClientError(#[from] google_cloud_gax::client_builder::Error),
59
60    /// An error from the Google Secret Manager API.
61    #[error(transparent)]
62    GoogleSecretManager(#[from] google_cloud_secretmanager_v1::Error),
63
64    /// An HTTP transport error.
65    #[error(transparent)]
66    Http(#[from] reqwest::Error),
67
68    /// A serde json error.
69    #[error(transparent)]
70    Json(#[from] serde_json::Error),
71}
72
73/// Alias for a [Result][std::result::Result] where the error variant is an [Error].
74pub type Result<T> = std::result::Result<T, Error>;

76/// Errors from Application Default Credentials.
77#[derive(Debug, thiserror::Error)]
78pub enum AdcError {
79    /// No GCP credentials found — the user needs to run `gcloud auth application-default login`.
80    #[error("not authenticated to GCP — run `gcloud auth application-default login` first")]
81    NotAuthenticated,
82
83    /// The home directory cannot be determined, so the ADC file path is unknown.
84    #[error("unable to find home directory")]
85    NoHomeDir,
86
87    /// The ADC credentials file could not be read from disk.
88    #[error("could not read ADC credentials file: {0}")]
89    ReadFailed(#[source] std::io::Error),
90
91    /// The ADC credentials file could not be parsed as JSON.
92    #[error("could not parse ADC credentials file: {0}")]
93    ParseFailed(#[source] serde_json::Error),
94
95    /// The service-account ID token credentials object could not be built.
96    #[error("could not build service-account ID token credentials: {0}")]
97    CredentialsBuild(String),
98
99    /// The service-account ID token could not be minted.
100    #[error("could not mint service-account ID token: {0}")]
101    TokenMint(String),
102}
103/// Errors from the interactive user OAuth loopback flow.
104#[derive(Debug, thiserror::Error)]
105pub enum OauthError {
106    /// The token cache path cannot be determined because both `HOME` and `XDG_CONFIG_HOME` are
107    /// unset.
108    #[error("cannot determine token cache location (HOME and XDG_CONFIG_HOME both unset)")]
109    NoCacheLocation,
110
111    /// A Google OAuth/token endpoint URL constant failed to parse.
112    #[error("invalid {endpoint} URL: {message}")]
113    InvalidUrl {
114        /// Which endpoint (`"auth"` or `"token"`).
115        endpoint: &'static str,
116        /// The underlying parse error message.
117        message: String,
118    },
119
120    /// The reqwest client used for token-endpoint calls could not be built.
121    #[error("could not build OAuth HTTP client: {0}")]
122    HttpClientBuild(String),
123
124    /// The loopback TCP listener could not be bound.
125    #[error("could not bind loopback listener: {0}")]
126    LoopbackBind(#[source] std::io::Error),
127
128    /// The loopback port could not be read from the bound listener.
129    #[error("could not read loopback port: {0}")]
130    LoopbackPort(#[source] std::io::Error),
131
132    /// The computed redirect URI failed to parse.
133    #[error("invalid redirect URI: {0}")]
134    InvalidRedirectUri(String),
135
136    /// The browser did not complete the redirect within the timeout window.
137    #[error("OAuth flow timed out after {0}s without a redirect")]
138    FlowTimeout(u64),
139
140    /// The `state` returned by Google did not match the issued CSRF token.
141    #[error("OAuth state mismatch — aborting to prevent CSRF")]
142    CsrfMismatch,
143
144    /// Google's token endpoint rejected the code exchange or refresh grant.
145    ///
146    /// Stringified because `oauth2`'s `RequestTokenError` carries unwieldy generic parameters.
147    #[error("Google token endpoint error: {0}")]
148    TokenEndpoint(String),
149
150    /// Google completed the flow but did not return a `refresh_token`.
151    #[error(
152        "Google did not return a refresh_token; check that the OAuth consent screen grants \
153         offline access"
154    )]
155    NoRefreshToken,
156
157    /// The loopback TCP accept call failed.
158    #[error("loopback accept failed: {0}")]
159    LoopbackAccept(#[source] std::io::Error),
160
161    /// Reading from the loopback TCP stream failed.
162    #[error("loopback read failed: {0}")]
163    LoopbackRead(#[source] std::io::Error),
164
165    /// The HTTP request received on the loopback port was malformed.
166    #[error("malformed OAuth callback: {0}")]
167    MalformedCallback(&'static str),
168
169    /// The authorization server returned an `error` parameter in the callback.
170    #[error("authorization denied: {0}")]
171    AuthorizationDenied(String),
172
173    /// The OAuth callback did not include a `code` parameter.
174    #[error("missing code in OAuth callback")]
175    MissingCode,
176
177    /// The OAuth callback did not include a `state` parameter.
178    #[error("missing state in OAuth callback")]
179    MissingState,
