1//! IAP-authenticated HTTP client for the REP orchestrator.
2
3use crate::orchestrator::{
4    Error, GCP_PROJECT, IAP_OAUTH_CLIENT_ID_SECRET_NAME, IAP_OAUTH_CLIENT_SECRET_SECRET_NAME,
5    Result,
6    auth::{AdcCredentials, id_token},
7};
8use google_cloud_secretmanager_v1::client::SecretManagerService;
9use oauth2::http::HeaderValue;
10use reqwest::{Client, Request, Response, header::AUTHORIZATION};
11
12/// Authenticated HTTP client for the REP orchestrator, protected by Google Cloud IAP.
13#[derive(Debug)]
14pub struct OrchestratorClient {
15    adc: AdcCredentials,
16    client_id: String,
17    client_secret: String,
18    http_client: Client,
19}
20
21impl OrchestratorClient {
22    /// Build a new client.
23    ///
24    /// Loads Application Default Credentials from disk and fetches the IAP
25    /// OAuth client credentials from Secret Manager.
26    pub async fn new() -> Result<Self> {
27        let adc = AdcCredentials::load()?;
28
29        let (client_id, client_secret) = fetch_secrets().await?;
30
31        Ok(Self {
32            adc,
33            client_id,
34            client_secret,
35            http_client: Client::new(),
36        })
37    }
38
39    /// Send an authenticated request to the orchestrator and return the full response.
40    pub async fn send(&self, mut request: Request) -> Result<Response> {
41        let id_token = id_token(&self.adc, &self.client_id, &self.client_secret).await?;
42
43        let mut auth_value = HeaderValue::from_str(&format!("Bearer {id_token}"))
44            .map_err(Error::InvalidBearerToken)?;
45        auth_value.set_sensitive(true);
46
47        request.headers_mut().insert(AUTHORIZATION, auth_value);
48
49        let response = self.http_client.execute(request).await?;
50
51        Ok(response)
52    }
53}
54
55async fn fetch_secrets() -> Result<(String, String)> {
56    let client = SecretManagerService::builder().build().await?;
57
58    let client_id = fetch_secret(&client, IAP_OAUTH_CLIENT_ID_SECRET_NAME).await?;
59    let client_secret = fetch_secret(&client, IAP_OAUTH_CLIENT_SECRET_SECRET_NAME).await?;
60
61    Ok((client_id, client_secret))
62}
63
64async fn fetch_secret(client: &SecretManagerService, secret_name: &str) -> Result<String> {
65    let response = client
66        .access_secret_version()
67        .set_name(format!(
68            "projects/{GCP_PROJECT}/secrets/{secret_name}/versions/latest"
69        ))
70        .send()
71        .await?;
72
73    let payload = response
74        .payload
75        .ok_or_else(|| Error::InvalidSecret("response missing payload".to_owned()))?;
76
77    String::from_utf8(payload.data.to_vec()).map_err(|e| Error::InvalidSecret(e.to_string()))
78}
