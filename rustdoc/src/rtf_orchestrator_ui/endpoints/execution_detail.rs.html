18/// `GET /ui/execution/{eid}` — one execution's detail: its status-history timeline (newest-first),
19/// exit code, and timestamps.
20///
21/// Fetches the execution status directly from the orchestrator rather than pulling the whole
22/// parent run.
23pub async fn handler<C: Client>(
24    State(orchestrator_client): State<C>,
25    Extension(links_cfg): Extension<LinksConfig>,
26    Path(execution_id): Path<Uuid>,
27) -> Response {
28    let result = orchestrator_client.execution_summary(execution_id).await;
29
30    to_response(execution_detail_body(execution_id, result, &links_cfg))
31}
32
33/// Maps the result of fetching an execution summary to a rendered `(status, body)` pair: the detail
34/// page on success, a "not found" page for an unknown id, or an error page for a fetch failure. Kept
35/// free of the orchestrator [`Client`] so it can be tested directly against hand-built results,
36/// without a mock client or handler plumbing.
37fn execution_detail_body(
38    execution_id: Uuid,
39    result: Result<Option<TestExecutionSummary>, orchestrator::Error>,
40    links_cfg: &LinksConfig,
41) -> (StatusCode, String) {
42    match result {
43        Ok(Some(execution)) => render_body(
44            StatusCode::OK,
45            ExecutionTemplate {
46                execution: ExecutionDetailView::new(execution, links_cfg),
47            },
48        ),
49        Ok(None) => render_body(
50            StatusCode::NOT_FOUND,
51            ExecutionNotFoundTemplate {
52                execution_id: execution_id.to_string(),
53            },
54        ),
55        Err(error) => {
56            error!(%error, %execution_id, "failed to fetch execution summary from orchestrator");
57            render_body(
58                StatusCode::BAD_GATEWAY,
59                ErrorTemplate {
60                    message: "Could not load this execution from the orchestrator.".to_owned(),
61                },
62            )
63        }
64    }
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70    use crate::{
71        endpoints::body_text,
72        links::sample_config,
73        orchestrator::mocks::{MockClient, sample_execution},
74    };
75    use rtf_orchestrator_shared::status::Status;
76
77    #[test]
78    fn execution_detail_body_renders_a_known_execution() {
79        let run_id = Uuid::from_u128(1);
80        let ex_id = Uuid::from_u128(2);
81        let (status, body) = execution_detail_body(
82            ex_id,
83            Ok(Some(sample_execution(run_id, ex_id))),
84            &sample_config(),
85        );
86
87        assert_eq!(status, StatusCode::OK);
88        assert!(body.contains("exec-alpha"), "execution name should render");
89        assert!(
90            body.contains(&format!("/ui/run/{run_id}")),
91            "execution detail page should link back to its parent run"
92        );
93    }
94
95    #[test]
96    fn execution_detail_body_renders_not_found_for_an_unknown_execution() {
97        let ex_id = Uuid::from_u128(2);
98        let (status, body) = execution_detail_body(ex_id, Ok(None), &sample_config());
99
100        assert_eq!(status, StatusCode::NOT_FOUND);
101        assert!(body.contains("Execution not found"));
102    }
103
104    #[test]
105    fn execution_detail_body_renders_error_when_the_fetch_fails() {
106        let ex_id = Uuid::from_u128(2);
107        let (status, body) = execution_detail_body(
108            ex_id,
109            Err(orchestrator::Error::TestExecutionStatus {
110                status: StatusCode::BAD_GATEWAY,
111                id: ex_id,
112            }),
113            &sample_config(),
114        );
115
116        assert_eq!(status, StatusCode::BAD_GATEWAY);
117        assert!(body.contains("went wrong"));
118    }
119
120    #[tokio::test]
121    async fn execution_detail_calls_the_client_and_renders_its_result() {
122        let run_id = Uuid::from_u128(1);
123        let ex_id = Uuid::from_u128(2);
124        let resp = handler(

126            Extension(sample_config()),
127            Path(ex_id),

129        .await;
130
131        assert_eq!(resp.status(), StatusCode::OK);
132        assert!(
133            body_text(resp).await.contains("exec-alpha"),

135        );
136    }
