1use crate::{endpoints::download_response, orchestrator::Client};
2use axum::{
3    extract::{Path, State},
4    response::Response,
5};
6use uuid::Uuid;

8/// `GET /ui/execution/{eid}/output.zip`
9///
10/// The orchestrator answers this with a 307 to a signed GCS URL, which this handler relays
11/// verbatim rather than following - the UI never downloads the artifact's bytes itself.
12pub async fn handler<C: Client>(
13    State(orchestrator_client): State<C>,
14    Path(execution_id): Path<Uuid>,
15) -> Response {
16    download_response(
17        orchestrator_client.execution_output_zip(execution_id).await,
18        "application/zip",
19        format!("{execution_id}-output.zip"),
20    )
21}
