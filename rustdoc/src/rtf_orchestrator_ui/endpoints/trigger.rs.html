21/// `GET /ui/trigger` - the "trigger a run from a GitHub-hosted test plan" form.
22pub async fn get_handler() -> Response {
23    to_response(render_body(StatusCode::OK, TriggerTemplate::default()))
24}
25
26/// `POST /ui/trigger` - builds a [`GitHubPayload`] from the submitted form and POSTs it to the
27/// orchestrator's `test-run/trigger` endpoint, forwarding the caller's IAP identity so the run's
28/// `initiated_by` reflects who submitted the form rather than `"unknown"`.
29pub async fn post_handler<C: Client>(
30    State(orchestrator_client): State<C>,
31    headers: HeaderMap,
32    Form(form): Form<TriggerForm>,
33) -> Response {
34    let payload = match form.try_into_payload() {
35        Ok(payload) => payload,
36        Err(message) => {
37            return to_response(render_body(
38                StatusCode::BAD_REQUEST,
39                TriggerTemplate {
40                    error: Some(message),
41                    ..form.into()
42                },
43            ));
44        }
45    };
46
47    let initiated_by = headers
48        .get(IAP_USER_EMAIL_HEADER)
49        .and_then(|value| value.to_str().ok());
50
51    let result = orchestrator_client
52        .trigger(&TriggerPayload::GitHub(payload), initiated_by)
53        .await;
54
55    trigger_result_response(form, result)
56}
57
58/// Form fields submitted by the "trigger a run from GitHub" form.
59#[derive(Debug, Default, Deserialize)]
60pub struct TriggerForm {
61    org: String,
62    repo: String,
63    path: String,
64    #[serde(rename = "ref")]
65    git_ref: String,
66    /// A JSON object of `HashMap<String, VariableOverride>` - a plain value templates a single
67    /// variable, an array value defines a matrix dimension. Blank means no overrides.
68    variables: String,
69}
70
71impl TriggerForm {
72    /// Builds the orchestrator payload, or the message to show inline if `variables` isn't valid
73    /// JSON. Trims every field first, so stray leading/trailing whitespace doesn't turn into a
74    /// bogus org/repo/path or a spuriously "non-blank" variables box.
75    fn try_into_payload(&self) -> Result<GitHubPayload, String> {
76        let (git_ref, variables) = parse_trigger_ref_and_variables(&self.git_ref, &self.variables)?;
77
78        Ok(GitHubPayload {
79            org: self.org.trim().to_owned(),
80            repo: self.repo.trim().to_owned(),
81            path: self.path.trim().to_owned(),
82            git_ref,
83            variables,
84        })
85    }
86}
87
88impl From<TriggerForm> for TriggerTemplate {
89    fn from(form: TriggerForm) -> Self {
90        Self {
91            org: form.org,
92            repo: form.repo,
93            path: form.path,
94            git_ref: form.git_ref,
95            variables: form.variables,
96            error: None,
97        }
98    }
99}

101/// Maps the result of triggering a run to a response: a redirect to the new run's status page on
102/// success, or the form re-rendered with its fields intact and an inline error otherwise. Kept
103/// free of the orchestrator [`Client`] so it's testable directly against hand-built results.
104fn trigger_result_response(
105    form: TriggerForm,
106    result: Result<TestRunSummary, orchestrator::Error>,
107) -> Response {
108    match result {
109        Ok(summary) => Redirect::to(&format!("/ui/run/{}", summary.id)).into_response(),
110
111        Err(error) => {
112            error!(%error, "failed to trigger a run from the orchestrator");
113            let (status, message) = match error {
114                orchestrator::Error::Trigger { status, message } => (status, message),
115                e => (
116                    StatusCode::BAD_GATEWAY,
117                    format!("Unable to trigger run: {e}"),
118                ),
119            };
120
121            to_response(render_body(
122                status,
123                TriggerTemplate {
124                    error: Some(message),
125                    ..form.into()
126                },
127            ))
128        }
129    }
130}

143    fn sample_form() -> TriggerForm {
144        TriggerForm {
145            org: "apollographql".to_owned(),
146            repo: "runtime-testing-framework".to_owned(),
147            path: "test-plans/smoke/test-plan.yaml".to_owned(),

153    #[test]
154    fn try_into_payload_trims_fields_and_wires_the_parsed_ref_and_variables_into_the_payload() {
155        let form = TriggerForm {
156            org: "  apollographql  ".to_owned(),
157            repo: "  runtime-testing-framework  ".to_owned(),
158            path: "  test-plans/smoke/test-plan.yaml  ".to_owned(),
159            git_ref: "  main  ".to_owned(),
160            variables: r#"{"message": "hello"}"#.to_owned(),
161        };
162        let payload = form.try_into_payload().expect("should parse");
163
164        assert_eq!(payload.org, "apollographql");
165        assert_eq!(payload.repo, "runtime-testing-framework");
166        assert_eq!(payload.path, "test-plans/smoke/test-plan.yaml");
167        assert_eq!(payload.git_ref, Some("main".to_owned()));
168        assert!(payload.variables.is_some());
169    }

171    #[tokio::test]
172    async fn get_handler_renders_the_empty_form() {
173        let resp = get_handler().await;
174
175        assert_eq!(resp.status(), StatusCode::OK);
176
177        let body = body_text(resp).await;
178
179        assert!(
180            body.contains("<form"),
181            "expected a form on the trigger page"
182        );
183        assert!(body.contains(r#"name="org""#));
184        assert!(body.contains(r#"name="variables""#));
185    }
186
187    #[test]
188    fn trigger_result_response_redirects_to_the_new_run_on_success() {
189        let run_id = Uuid::from_u128(1);
190        let ex_id = Uuid::from_u128(2);
191        let resp = trigger_result_response(
192            sample_form(),
193            Ok(sample_summary(run_id, ex_id, Status::Initialising)),
194        );
195
196        assert_eq!(resp.status(), StatusCode::SEE_OTHER);
197        assert_eq!(
198            resp.headers().get(LOCATION).unwrap(),
199            &format!("/ui/run/{run_id}")
200        );
201    }
