25/// Query params accepted by the known test plan detail page. `trigger_ref`/`trigger_variables`/
26/// `trigger_error` are only ever set by [`redirect_with_trigger_error`] after a failed trigger
27/// submission, to re-populate the form and show the failure inline.
28#[derive(Debug, Default, serde::Deserialize)]
29pub struct TestPlanDetailParams {
30    offset: Option<i64>,
31    #[serde(default, rename = "trigger_ref")]
32    trigger_git_ref: Option<String>,
33    #[serde(default)]
34    trigger_variables: Option<String>,
35    #[serde(default)]
36    trigger_error: Option<String>,
37}
38
39/// `GET /ui/test-plan/{uuid}` — the plan's metadata (with a GitHub link), a form to trigger a new
40/// run of it, and a paginated table of its recent runs. The plan and its runs are fetched
41/// concurrently, since the runs fetch only needs `uuid` from the path, not anything from the plan
42/// lookup.
43pub async fn handler<C: Client>(
44    State(orchestrator_client): State<C>,
45    Path(uuid): Path<Uuid>,
46    Query(params): Query<TestPlanDetailParams>,
47) -> Response {
48    let offset = params.offset.unwrap_or(0).max(0);
49    let runs_params = KnownTestPlanRunsParams {
50        limit: Some(DEFAULT_LIMIT),
51        offset: Some(offset),
52        ..Default::default()
53    };
54
55    let (plan_result, runs_result) = tokio::join!(
56        orchestrator_client.known_test_plan_summary(uuid),
57        orchestrator_client.list_known_test_plan_runs(uuid, &runs_params)
58    );
59
60    to_response(test_plan_detail_body(
61        uuid,
62        plan_result,
63        runs_result,
64        DEFAULT_LIMIT,
65        offset,
66        params.trigger_git_ref.unwrap_or_default(),
67        params.trigger_variables.unwrap_or_default(),
68        params.trigger_error,
69    ))
70}

72/// Maps the plan lookup and runs-list results to a rendered `(status, body)` pair: the plan header,
73/// trigger form, and runs table on success, a "not found" page for an unknown uuid, or an error
74/// page for a plan-fetch failure. A runs-list failure still renders the page - with the plan header
75/// and GitHub link intact and an inline error in place of the runs table - mirroring the home
76/// page's `index_body`. `runs_result` is only rendered when `plan_result` is `Ok(Some(_))`; kept as
77/// a plain parameter (rather than only fetched conditionally) so this whole mapping stays a single
78/// pure, directly-testable function, matching `run_status_body`/`execution_detail_body`.
79#[allow(clippy::too_many_arguments)]
80fn test_plan_detail_body(
81    uuid: Uuid,
82    plan_result: Result<Option<KnownTestPlanSummary>, orchestrator::Error>,
83    runs_result: Result<TestRunListResponse, orchestrator::Error>,
84    limit: i64,
85    offset: i64,
86    trigger_git_ref: String,
87    trigger_variables: String,
88    trigger_error: Option<String>,
89) -> (StatusCode, String) {
90    let plan = match plan_result {
91        Ok(Some(plan)) => plan,
92        Ok(None) => {
93            return render_body(
94                StatusCode::NOT_FOUND,
95                TestPlanNotFoundTemplate {
96                    uuid: uuid.to_string(),
97                },
98            );
99        }
100        Err(error) => {
101            error!(%error, %uuid, "failed to fetch known test plan from orchestrator");
102            return render_body(
103                StatusCode::BAD_GATEWAY,
104                ErrorTemplate {
105                    message: "Could not load this known test plan from the orchestrator."
106                        .to_owned(),
107                },
108            );
109        }
110    };
111
112    let plan_view = KnownTestPlanRowView::from(plan);
113
114    match runs_result {
115        Ok(response) => render_body(
116            StatusCode::OK,
117            TestPlanDetailTemplate {
118                plan: plan_view,
119                runs: Some(RunListView::for_known_test_plan(
120                    response, limit, offset, uuid,
121                )),
122                runs_error: None,
123                trigger_git_ref,
124                trigger_variables,
125                trigger_error,
126            },
127        ),
128        Err(error) => {
129            error!(%error, %uuid, "failed to list runs for known test plan from orchestrator");
130            render_body(
131                StatusCode::OK,
132                TestPlanDetailTemplate {
133                    plan: plan_view,
134                    runs: None,
135                    runs_error: Some("Could not load recent runs for this test plan.".to_owned()),
136                    trigger_git_ref,
137                    trigger_variables,
138                    trigger_error,
139                },
140            )
141        }
142    }
143}

145/// Form fields submitted by the trigger form embedded on the known test plan detail page.
146#[derive(Debug, Default, serde::Deserialize)]
147pub struct KnownTestPlanTriggerForm {
148    #[serde(rename = "ref")]
149    git_ref: String,
150    /// A JSON object of `HashMap<String, ScalarOrArray>` - a plain value templates a single
151    /// variable, an array value defines a matrix dimension. Blank means no overrides.
152    variables: String,
153}
154
155impl KnownTestPlanTriggerForm {
156    /// Builds the orchestrator payload for `uuid`, or the message to show inline if `variables`
157    /// isn't valid JSON. Trims both fields first, mirroring `trigger::TriggerForm::try_into_payload`.
158    fn try_into_payload(&self, uuid: Uuid) -> Result<KnownTestPlanUuidPayload, String> {
159        let variables_json = self.variables.trim();
160        let variables = if variables_json.is_empty() {
161            None
162        } else {
163            serde_json::from_str(variables_json)
164                .map_err(|error| format!("Variables must be a JSON object: {error}"))?
165        };
166        let git_ref = self.git_ref.trim();
167
168        Ok(KnownTestPlanUuidPayload {
169            test_plan_uuid: uuid,
170            git_ref: (!git_ref.is_empty()).then(|| git_ref.to_owned()),
171            variables,
172        })
173    }
174}

176/// `POST /ui/test-plan/{uuid}/trigger` — builds a [`KnownTestPlanUuidPayload`] from the submitted
177/// form and POSTs it to the orchestrator's `test-run/trigger` endpoint, forwarding the caller's IAP
178/// identity as with the GitHub trigger form. On success, redirects to the new run's status page. On
179/// failure, redirects back to this plan's detail page with the submitted fields and the failure
180/// message carried as query params - simpler than re-rendering the whole detail page directly here,
181/// which would mean duplicating the `GET` handler's plan/runs fetching in the `POST` handler too.
182pub async fn post_trigger<C: Client>(
183    State(orchestrator_client): State<C>,
184    headers: HeaderMap,
185    Path(uuid): Path<Uuid>,
186    Form(form): Form<KnownTestPlanTriggerForm>,
187) -> Response {
188    let payload = match form.try_into_payload(uuid) {
189        Ok(payload) => payload,
190        Err(message) => return redirect_with_trigger_error(uuid, &form, &message),
191    };
192
193    let initiated_by = headers
194        .get(IAP_USER_EMAIL_HEADER)
195        .and_then(|value| value.to_str().ok());
196
197    match orchestrator_client
198        .trigger(&TriggerPayload::KnownTestPlanUuid(payload), initiated_by)
199        .await
200    {
201        Ok(summary) => Redirect::to(&format!("/ui/run/{}", summary.id)).into_response(),
202        Err(error) => {
203            error!(%error, %uuid, "failed to trigger a run of a known test plan from the orchestrator");
204            let message = match &error {
205                orchestrator::Error::Trigger { message, .. } => message.clone(),
206                e => format!("Unable to trigger run: {e}"),
207            };
208            redirect_with_trigger_error(uuid, &form, &message)
209        }
210    }
211}

213/// Redirects back to the plan's detail page carrying the submitted form fields and the failure
214/// message as query params, so the subsequent `GET` re-populates the form and shows the error
215/// inline.
216fn redirect_with_trigger_error(
217    uuid: Uuid,
218    form: &KnownTestPlanTriggerForm,
219    message: &str,
220) -> Response {
221    let mut qs = form_urlencoded::Serializer::new(String::new());
222    if !form.git_ref.is_empty() {
223        qs.append_pair("trigger_ref", &form.git_ref);
224    }
225    if !form.variables.is_empty() {
226        qs.append_pair("trigger_variables", &form.variables);
227    }
228    qs.append_pair("trigger_error", message);
229
230    Redirect::to(&format!("/ui/test-plan/{uuid}?{}", qs.finish())).into_response()
231}
