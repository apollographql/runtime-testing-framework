19/// Render an askama template to a `(status, body)` pair. Template render errors are mapped to a 500
20/// with an empty body — rendering only fails on genuine bugs (e.g. a missing field), not on bad
21/// input, so there's no more specific message to show.
22fn render_body<T: Template>(status: StatusCode, template: T) -> (StatusCode, String) {
23    match template.render() {
24        Ok(body) => (status, body),
25        Err(error) => {
26            error!(%error, "template render failed");
27            (StatusCode::INTERNAL_SERVER_ERROR, String::new())
28        }
29    }
30}

52/// Maps a [`Download`] outcome to an HTTP response: the file's bytes with a `Content-Disposition`
53/// download header on success, the orchestrator's redirect relayed verbatim, or a plain-text
54/// response carrying the corresponding status otherwise.
55fn download_response(
56    result: Result<Download, orchestrator::Error>,
57    content_type: &'static str,
58    filename: String,
59) -> Response {
60    match result {
61        Ok(Download::Ready(bytes)) => (
62            [
63                (CONTENT_TYPE, content_type.to_owned()),
64                (
65                    CONTENT_DISPOSITION,
66                    format!("attachment; filename=\"{filename}\""),
67                ),
68            ],
69            bytes,
70        )
71            .into_response(),
72        Ok(Download::Redirect { status, location }) => {
73            (status, [(LOCATION, location)]).into_response()
74        }
75        Ok(Download::NotFound) => (StatusCode::NOT_FOUND, "not found").into_response(),
76        Ok(Download::NotReady) => (StatusCode::CONFLICT, "output not ready yet").into_response(),
77        Err(error) => {
78            error!(%error, "failed to fetch download from orchestrator");
79            (
80                StatusCode::BAD_GATEWAY,
81                "could not fetch download from orchestrator",
82            )
83                .into_response()
84        }
85    }
86}

98    #[tokio::test]
99    async fn download_response_returns_bytes_with_a_download_header_on_success() {
100        let resp = download_response(
101            Ok(Download::Ready(b"hello".to_vec())),
102            "text/plain; charset=utf-8",
103            "example.txt".to_owned(),
104        );
105
106        assert_eq!(resp.status(), StatusCode::OK);
107        assert_eq!(
108            resp.headers().get(CONTENT_DISPOSITION).unwrap(),
109            "attachment; filename=\"example.txt\""
110        );
111        assert_eq!(body_text(resp).await, "hello");
112    }
113
114    #[tokio::test]
115    async fn download_response_relays_a_redirect_verbatim() {
116        let resp = download_response(
117            Ok(Download::Redirect {
118                status: StatusCode::TEMPORARY_REDIRECT,
119                location: "https://storage.googleapis.com/signed-url".to_owned(),
120            }),
121            "application/zip",
122            "f.zip".to_owned(),
123        );
124
125        assert_eq!(resp.status(), StatusCode::TEMPORARY_REDIRECT);
126        assert_eq!(
127            resp.headers().get(LOCATION).unwrap(),
128            "https://storage.googleapis.com/signed-url"
129        );
130        assert_eq!(
131            body_text(resp).await,
132            "",
133            "a redirect should carry no body - the browser follows Location itself"
134        );
135    }
