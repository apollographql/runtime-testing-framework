10/// Upper bound on how long a non-terminal run is auto-refreshed. A run that never reaches a
11/// terminal state (e.g. a stuck orchestrator) would otherwise be polled forever; past this age the
12/// UI stops polling and the user can refresh manually.
13const MAX_POLL_AGE_SECS: i64 = 60 * 60;
14
15/// The run status page content: the overall-status banner and the executions table. This whole
16/// region is re-rendered on every htmx poll (via `hx-select` against the same page route), so the
17/// banner, elapsed time, and "last updated" indicator all refresh together and stop together on
18/// terminal.
19pub struct RunView {
20    pub id: Uuid,
21    pub name: String,
22    pub status_label: String,
23    pub status_class: &'static str,
24    pub started_at: String,
25    pub updated_at: String,
26    pub completed_at: Option<String>,
27    /// Wall-clock time from start to completion (or to `now` while still running).
28    pub elapsed: String,
29    /// When this render happened, shown as the "last updated" indicator.
30    pub last_updated: String,
31    /// Whether the rendered region should keep polling for updates (drives the htmx trigger).
32    pub should_poll: bool,
33    pub executions: Vec<ExecutionView>,
34}
35
36impl RunView {
37    /// Build the run view. `now` decides whether the region keeps polling and anchors the elapsed
38    /// time and "last updated" indicator. `links_cfg` supplies the GCP/Grafana deep-link
39    /// configuration for each execution row.
40    pub fn new(run: TestRunSummary, now: DateTime<Utc>, links_cfg: &LinksConfig) -> Self {
41        // Trust `current_status`, not just the presence of `completed_at`, to decide whether the run
42        // is actually done — a run that is still in progress should never show a completion time,
43        // even if the wire data is momentarily inconsistent.
44        let completed_at = run
45            .completed_at
46            .filter(|_| run.current_status.is_terminal());
47        let end = completed_at.unwrap_or(now);
48
49        Self {
50            id: run.id,
51            name: run.name,
52            status_label: run.current_status.to_string(),
53            status_class: status::css_class(run.current_status),
54            started_at: run.started_at.to_rfc3339(),
55            updated_at: run.updated_at.to_rfc3339(),
56            completed_at: completed_at.map(|ts| ts.to_rfc3339()),
57            elapsed: format_duration(Duration::from_secs(
58                (end.timestamp() - run.started_at.timestamp()).max(0) as u64,
59            ))
60            .to_string(),
61            last_updated: now.format("%H:%M:%S UTC").to_string(),
62            should_poll: should_poll(run.current_status, run.started_at, now),
63            executions: run
64                .executions
65                .into_iter()
66                .map(|execution| ExecutionView::new(execution, links_cfg))
67                .collect(),
68        }
69    }
70}

72/// The execution view within a test run summary
73pub struct ExecutionView {
74    pub id: Uuid,
75    pub name: String,
76    pub status_label: String,
77    pub status_class: &'static str,
78    pub exit_code: Option<i32>,

87impl ExecutionView {
88    fn new(execution: TestExecutionSummary, links_cfg: &LinksConfig) -> Self {
89        let namespace = execution.id.to_string();
90        let window_end = execution.completed_at.unwrap_or_else(Utc::now);
91        let logs_url = gcp_logs(links_cfg, &namespace, execution.started_at, window_end);
92        let grafana_url = grafana(links_cfg, &namespace, execution.started_at, window_end);
93
94        Self {
95            id: execution.id,
96            name: execution.name,
97            status_label: execution.current_status.to_string(),
98            status_class: status::css_class(execution.current_status),
99            exit_code: execution.exit_code,
100            started_at: execution.started_at.to_rfc3339(),
101            updated_at: execution.updated_at.to_rfc3339(),
102            logs_url,
103            grafana_url,
104        }
105    }

108/// The execution detail page: the execution's status-history timeline plus its metadata.

111    pub run_id: Option<Uuid>,
112    pub name: String,
113    pub status_label: String,
114    pub status_class: &'static str,
115    pub exit_code: Option<i32>,
116    pub started_at: String,
117    pub updated_at: String,
118    pub completed_at: Option<String>,
119    /// Status updates, newest-first (as returned by the orchestrator).
120    pub history: Vec<StatusEntryView>,
121    /// Cloud Logging deep link scoped to this execution's workload namespace and time window.
122    pub logs_url: String,
123    /// Grafana deep link scoped to this execution's workload namespace and time window.
124    pub grafana_url: String,
125}
126
127impl ExecutionDetailView {
128    pub fn new(execution: TestExecutionSummary, links_cfg: &LinksConfig) -> Self {
129        let namespace = execution.id.to_string();
130        let end = execution.completed_at.unwrap_or_else(Utc::now);
131        let logs_url = gcp_logs(links_cfg, &namespace, execution.started_at, end);
132        let grafana_url = grafana(links_cfg, &namespace, execution.started_at, end);
133
134        Self {
135            run_id: execution.test_run_id,
136            name: execution.name,
137            status_label: execution.current_status.to_string(),
138            status_class: status::css_class(execution.current_status),
139            exit_code: execution.exit_code,
140            started_at: execution.started_at.to_rfc3339(),
141            updated_at: execution.updated_at.to_rfc3339(),
142            completed_at: execution.completed_at.map(|ts| ts.to_rfc3339()),
143            history: execution
144                .status_history
145                .into_iter()
146                .map(StatusEntryView::from)
147                .collect(),
148            logs_url,
149            grafana_url,
150        }
151    }

173/// Whether a run should keep being polled: only while it is non-terminal and has not been running
174/// longer than [`MAX_POLL_AGE_SECS`] (the stuck-run guard).
175fn should_poll(status: Status, started_at: DateTime<Utc>, now: DateTime<Utc>) -> bool {
176    !status.is_terminal() && (now.timestamp() - started_at.timestamp()) < MAX_POLL_AGE_SECS
177}
