12/// Upper bound on how long a non-terminal run is auto-refreshed. A run that never reaches a
13/// terminal state (e.g. a stuck orchestrator) would otherwise be polled forever; past this age the
14/// UI stops polling and the user can refresh manually.
15const MAX_POLL_AGE_SECS: i64 = 60 * 60;
16
17/// Renders a timestamp as RFC 3339 at second precision (e.g. `2026-07-27T14:23:01Z`), the format
18/// every timestamp on the UI is shown in.
19fn format_rfc3339(dt: DateTime<Utc>) -> String {
20    let seconds_since_epoch = dt.timestamp().max(0) as u64;
21    let system_time = SystemTime::UNIX_EPOCH + Duration::from_secs(seconds_since_epoch);
22
23    format_rfc3339_seconds(system_time).to_string()
24}

26/// The run status page content: the overall-status banner and the executions table. This whole
27/// region is re-rendered on every htmx poll (via `hx-select` against the same page route), so the
28/// banner, elapsed time, and "last updated" indicator all refresh together and stop together on
29/// terminal.
30pub struct RunView {
31    pub id: Uuid,
32    pub name: String,
33    pub status_label: String,
34    pub status_class: &'static str,
35    pub initiated_by: String,
36    pub started_at: String,
37    pub updated_at: String,
38    pub completed_at: Option<String>,
39    /// Wall-clock time from start to completion (or to `now` while still running).
40    pub elapsed: String,
41    /// When this render happened, shown as the "last updated" indicator.
42    pub last_updated: String,
43    /// Whether the rendered region should keep polling for updates (drives the htmx trigger).
44    pub should_poll: bool,
45    /// How many executions this run has in total, unaffected by `execution_status_filter` — shown
46    /// next to the table heading so filtering down to a status doesn't read as "the run only has
47    /// this many executions."
48    pub total_executions: usize,
49    /// The `current_status` value (its `Display` string, e.g. `"RUNNING"`) executions are filtered
50    /// down to, or empty for no filter. Kept as a string, not a parsed [`Status`], since its only
51    /// uses are re-populating the `<select>` and round-tripping through URLs.
52    pub execution_status_filter: String,
53    /// How many executions are in each status, in lifecycle order, excluding statuses no execution
54    /// currently has. Always reflects every execution, regardless of `execution_status_filter`.
55    pub status_breakdown: Vec<StatusCountView>,
56    /// Executions matching `execution_status_filter` (all of them, if empty).
57    pub executions: Vec<ExecutionView>,
58}

60impl RunView {
61    /// Build the run view. `now` decides whether the region keeps polling and anchors the elapsed
62    /// time and "last updated" indicator. `links_cfg` supplies the GCP/Grafana deep-link
63    /// configuration for each execution row. `execution_status_filter` restricts the executions
64    /// table to rows whose status matches exactly (empty means no filter).
65    pub fn new(
66        run: TestRunSummary,
67        now: DateTime<Utc>,
68        links_cfg: &LinksConfig,
69        execution_status_filter: String,
70    ) -> Self {
71        // Trust `current_status`, not just the presence of `completed_at`, to decide whether the run
72        // is actually done — a run that is still in progress should never show a completion time,
73        // even if the wire data is momentarily inconsistent.
74        let completed_at = run
75            .completed_at
76            .filter(|_| run.current_status.is_terminal());
77        let end = completed_at.unwrap_or(now);
78        let total_executions = run.executions.len();
79        let status_breakdown = status_breakdown(&run.executions);
80
81        Self {
82            id: run.id,
83            name: run.name,
84            status_label: run.current_status.to_string(),
85            status_class: status::css_class(run.current_status),
86            initiated_by: run.initiated_by,
87            started_at: format_rfc3339(run.started_at),
88            updated_at: format_rfc3339(run.updated_at),
89            completed_at: completed_at.map(format_rfc3339),
90            elapsed: format_duration(Duration::from_secs(
91                (end.timestamp() - run.started_at.timestamp()).max(0) as u64,
92            ))
93            .to_string(),
94            last_updated: now.format("%H:%M:%S UTC").to_string(),
95            should_poll: should_poll(run.current_status, run.started_at, now),
96            total_executions,
97            status_breakdown,
98            executions: run
99                .executions
100                .into_iter()
101                .filter(|execution| {
102                    execution_status_filter.is_empty()
103                        || execution.current_status.to_string() == execution_status_filter
104                })
105                .map(|execution| ExecutionView::new(execution, links_cfg))
106                .collect(),
107            execution_status_filter,
108        }
109    }
110
111    /// The URL the executions table's htmx auto-poll and manual "Refresh now" button fetch from —
112    /// this run's status page, carrying `execution_status_filter` forward so a refresh doesn't
113    /// silently drop the active filter.
114    pub fn poll_url(&self) -> String {
115        if self.execution_status_filter.is_empty() {
116            format!("/ui/run/{}", self.id)
117        } else {
118            let mut qs = form_urlencoded::Serializer::new(String::new());
119            qs.append_pair("execution_status", &self.execution_status_filter);
120            format!("/ui/run/{}?{}", self.id, qs.finish())
121        }
122    }
123
124    /// The message shown in the executions table in place of rows when there's nothing to show:
125    /// either the run genuinely has no executions yet, or a status filter matched none of them.
126    pub fn executions_empty_message(&self) -> &'static str {
127        if self.execution_status_filter.is_empty() {
128            "No executions yet."
129        } else {
130            "No executions match this filter."
131        }
132    }
133}

135/// Every [`Status`] variant, in lifecycle order — the order the run's status breakdown and the
136/// execution-status filter's `<select>` present statuses in.
137const STATUS_LIFECYCLE_ORDER: [Status; 8] = [
138    Status::Initialising,
139    Status::Resolving,
140    Status::Provisioning,
141    Status::EnvironmentReady,
142    Status::Running,
143    Status::Successful,
144    Status::Failed,
145    Status::Unrunnable,
146];

148/// One row in the run's status-breakdown summary.
149pub struct StatusCountView {
150    pub status_label: String,
151    pub status_class: &'static str,
152    pub count: usize,
153}
154
155/// How many `executions` are in each status, in lifecycle order, omitting statuses none of them are
156/// in — a run with no `Unrunnable` executions shouldn't show an "Unrunnable: 0" row.
157fn status_breakdown(executions: &[TestExecutionSummary]) -> Vec<StatusCountView> {
158    STATUS_LIFECYCLE_ORDER
159        .into_iter()
160        .filter_map(|candidate| {
161            let count = executions
162                .iter()
163                .filter(|execution| execution.current_status == candidate)
164                .count();
165
166            (count > 0).then(|| StatusCountView {
167                status_label: candidate.to_string(),
168                status_class: status::css_class(candidate),
169                count,
170            })
171        })
172        .collect()
173}

315    fn prev_offset(&self) -> i64 {
316        if self.rows.is_empty() && self.total > 0 {
317            ((self.total - 1) / self.limit) * self.limit
318        } else {
319            (self.offset - self.limit).max(0)

323    fn href_with_offset(&self, offset: i64) -> String {
324        match &self.scope {
325            RunListScope::KnownTestPlan(plan_uuid) => {
326                format!("/ui/test-plan/{plan_uuid}?offset={offset}")
327            }
328            RunListScope::Filtered {
329                initiated_by,
330                started_within,
331            } => {
332                let mut qs = form_urlencoded::Serializer::new(String::new());
333                if !initiated_by.is_empty() {
334                    qs.append_pair("initiated_by", initiated_by);
335                }
336                if !started_within.is_empty() {
337                    qs.append_pair("started_within", started_within);
338                }
339                qs.append_pair("offset", &offset.to_string());
340                format!("/ui?{}", qs.finish())
341            }
342        }
343    }

353        Some(format!("{}-{} of {}", self.offset + 1, last, self.total))
354    }

356    pub fn empty_message(&self) -> Option<&'static str> {

358            None
359        } else if self.total == 0 {

361                RunListScope::KnownTestPlan(_) => "This test plan has no runs yet.",
362                RunListScope::Filtered { .. } => "No runs match these filters.",
363            })
364        } else {
365            Some("No runs on this page.")
366        }
367    }

548/// Whether a run should keep being polled: only while it is non-terminal and has not been running
549/// longer than [`MAX_POLL_AGE_SECS`] (the stuck-run guard).
550fn should_poll(status: Status, started_at: DateTime<Utc>, now: DateTime<Utc>) -> bool {
551    !status.is_terminal() && (now.timestamp() - started_at.timestamp()) < MAX_POLL_AGE_SECS
552}
