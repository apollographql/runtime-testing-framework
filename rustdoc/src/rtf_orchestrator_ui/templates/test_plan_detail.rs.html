2use askama::Template;

4#[derive(Debug, Template)]
5#[template(
6    path = "test_plan_detail.html",
7    blocks = ["plan_header", "overview", "trigger_form", "runs"]
8)]
9pub struct TestPlanDetailTemplate {
10    pub plan: KnownTestPlanRowView,
11    pub details: Option<TestPlanDetailsView>,
12    pub details_error: Option<String>,
13    pub runs: Option<RunListView>,
14    pub runs_error: Option<String>,
15    pub trigger_git_ref: String,
16    pub days_back: u32,
17    pub days: u32,
18    pub trigger_variables: String,
19    pub trigger_error: Option<String>,
20}
21
22impl TestPlanDetailTemplate {
23    /// Falls back to `HEAD` when details failed to load.
24    pub fn github_url(&self) -> &str {
25        self.details
26            .as_ref()
27            .map_or(&self.plan.github_url, |d| &d.github_url)
28    }
29
30    pub fn write_access_label(&self) -> &'static str {
31        if self.plan.allow_k8s_write {
32            "Enabled"
33        } else {
34            "Disabled"
35        }
36    }
37
38    pub fn customize_expanded(&self) -> bool {
39        self.trigger_error.is_some() || !self.trigger_variables.is_empty()
40    }
41
42    pub fn focus_variables(&self) -> bool {
43        self.trigger_error.is_some()
44    }
45}
