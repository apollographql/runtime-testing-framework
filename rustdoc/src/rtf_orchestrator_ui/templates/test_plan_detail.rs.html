10#[derive(Debug, Template)]
11#[template(
12    path = "test_plan_detail.html",

15pub struct TestPlanDetailTemplate {
16    pub plan: KnownTestPlanRowView,

19    pub runs: Option<RunListView>,

31impl TestPlanDetailTemplate {
32    /// Falls back to `HEAD` when details failed to load.
33    pub fn github_url(&self) -> &str {
34        self.details
35            .as_ref()
36            .map_or(&self.plan.github_url, |d| &d.github_url)
37    }
38
39    pub fn write_access_label(&self) -> &'static str {
40        match self.plan.allow_k8s_write {
41            Some(true) => "Enabled",
42            Some(false) => "Disabled",
43            None => "Pool default",
44        }
45    }
46}

106    /// The JS unit tests (tests/js/) load this fixture as the page data, so they exercise exactly

113        );
114        let t = preview::test_plan_detail();

116        let actual = format!(
117            "{}\n",
118            serde_json::to_string_pretty(&details.js_data()).unwrap()
119        );
120
121        if env::var_os("UPDATE_JS_FIXTURES").is_some() {
122            fs::write(path, &actual).unwrap();
123        }
124
125        let expected = fs::read_to_string(path).unwrap_or_default();
126
127        assert_eq!(
128            actual, expected,
129            "page data no longer matches {path}: rerun with UPDATE_JS_FIXTURES=1 and check the JS tests still pass"
