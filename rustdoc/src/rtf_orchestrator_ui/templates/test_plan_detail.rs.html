7#[derive(Debug, Template)]
8#[template(
9    path = "test_plan_detail.html",
10    blocks = ["plan_header", "overview", "trigger_form", "runs"]
11)]
12pub struct TestPlanDetailTemplate {
13    pub plan: KnownTestPlanRowView,
14    pub details: Option<TestPlanDetailsView>,
15    pub details_error: Option<String>,
16    pub runs: Option<RunListView>,
17    pub runs_error: Option<String>,
18    pub trigger_git_ref: String,
19    pub days_back: u32,
20    pub days: u32,
21    pub trigger_variables: String,
22    pub trigger_error: Option<String>,
23    pub badge: Option<ViewerBadge>,
24    pub can_trigger: bool,
25    pub edit: Option<TestPlanEditView>,
26}
27
28impl TestPlanDetailTemplate {
29    /// Falls back to `HEAD` when details failed to load.
30    pub fn github_url(&self) -> &str {
31        self.details
32            .as_ref()
33            .map_or(&self.plan.github_url, |d| &d.github_url)
34    }
35
36    pub fn write_access_label(&self) -> &'static str {
37        match self.plan.allow_k8s_write {
38            Some(true) => "Enabled",
39            Some(false) => "Disabled",
40            None => "Pool default",
41        }
42    }
43
44    pub fn customize_expanded(&self) -> bool {
45        self.trigger_error.is_some() || !self.trigger_variables.is_empty()
46    }
47
48    pub fn focus_variables(&self) -> bool {
49        self.trigger_error.is_some()
50    }
51}
