1use rep_orchestrator_shared::status::{Status, StatusCategory};

3/// CSS modifier class for a status, used to colour-code the UI.
4pub fn css_class(status: Status) -> &'static str {
5    match status.category() {
6        StatusCategory::Pending => "status--pending",
7        StatusCategory::Active => "status--active",
8        StatusCategory::Success => "status--success",
9        StatusCategory::Failed => "status--failed",
10    }
11}
