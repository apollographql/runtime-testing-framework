5/// Configuration needed to build GCP Cloud Logging and Grafana deep links. Threaded through
6/// explicitly (rather than read from a global [Config]) so link-building stays testable without
7/// needing a populated environment.
8#[derive(Debug, Clone)]
9pub struct LinksConfig {
10    pub gcp_project: String,
11    pub grafana_base_url: String,
12    pub grafana_dashboard_uid: String,
13    pub grafana_dashboard_slug: String,
14    pub grafana_datasource_uid: String,
15}
16
17impl From<&Config> for LinksConfig {
18    fn from(cfg: &Config) -> Self {
19        Self {
20            gcp_project: cfg.gcp_project.clone(),
21            grafana_base_url: cfg.grafana_base_url.clone(),
22            grafana_dashboard_uid: cfg.grafana_dashboard_uid.clone(),
23            grafana_dashboard_slug: cfg.grafana_dashboard_slug.clone(),
24            grafana_datasource_uid: cfg.grafana_datasource_uid.clone(),
25        }
26    }
27}

29/// Build a Cloud Logging deep link scoped to one execution's workload namespace (namespace name
30/// == execution id) within `cluster`, with the time range narrowed to the execution's own window.
31pub fn gcp_logs(
32    cfg: &LinksConfig,
33    cluster: &str,
34    namespace: &str,
35    start: DateTime<Utc>,
36    end: DateTime<Utc>,
37) -> String {
38    let query = format!(
39        "resource.labels.cluster_name=\"{cluster}\"\nresource.labels.namespace_name=\"{namespace}\"",
40    );
41    let start = start.to_rfc3339_opts(SecondsFormat::Millis, true);
42    let end = end.to_rfc3339_opts(SecondsFormat::Millis, true);
43
44    format!(
45        "https://console.cloud.google.com/logs/query;query={};cursorTimestamp={start};startTime={start};endTime={end}?referrer=search&project={}&supportedpurview=project",
46        percent_encode(&query),
47        cfg.gcp_project,
48    )
49}

51/// Build a Grafana deep link to the dashboard configured in [LinksConfig], scoped to one
52/// execution's workload namespace (namespace name == execution id), with the time range narrowed
53/// to the execution's own window.
54pub fn grafana(
55    cfg: &LinksConfig,
56    namespace: &str,
57    start: DateTime<Utc>,
58    end: DateTime<Utc>,
59) -> String {
60    let start = start.to_rfc3339_opts(SecondsFormat::Millis, true);
61    let end = end.to_rfc3339_opts(SecondsFormat::Millis, true);
62
63    format!(
64        "{}/d/{}/{}?orgId=1&from={start}&to={end}&timezone=utc&var-datasource={}&var-cluster=&var-namespace={}&refresh=10s",
65        cfg.grafana_base_url,
66        cfg.grafana_dashboard_uid,
67        cfg.grafana_dashboard_slug,
68        cfg.grafana_datasource_uid,
69        percent_encode(namespace),
70    )
71}
72
73/// Percent-encode `input` for embedding in a URL.
74fn percent_encode(input: &str) -> String {
75    byte_serialize(input.as_bytes()).collect()
76}
