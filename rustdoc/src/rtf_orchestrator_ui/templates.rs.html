7/// The landing page: the run-id lookup form plus the recent-runs table.
8#[derive(Template)]
9#[template(path = "index.html")]
10pub struct IndexTemplate {
11    /// Re-populates the "Initiated by" field after a search.
12    pub initiated_by: String,
13    /// Which `started_within` preset is selected ("", "hour", "day", "week", "month").
14    pub started_within: String,
15    pub list: Option<RunListView>,
16    pub list_error: Option<String>,
17}

86/// The "trigger a run from GitHub" form. Fields are re-populated from the submission on a failed
