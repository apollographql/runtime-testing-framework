16fn current_value_list(v: &TestPlanVariable) -> Vec<String> {
17    match &v.current_value {
18        Some(VariableValue::Dimension(values)) => values.iter().map(ToString::to_string).collect(),
19        Some(VariableValue::Scalar(s)) => vec![s.to_string()],
20        None => v
21            .declarations
22            .iter()
23            .find_map(|d| d.default.as_ref())
24            .map(|d| vec![d.to_string()])
25            .unwrap_or_default(),
26    }
27}

29#[derive(Debug, Clone, Serialize)]
30pub struct TriggerVariableView {
31    pub name: String,
32    pub sections: Vec<&'static str>,
33    pub description: String,
34    pub ghost_text: String,
35    pub full_values: Vec<String>,
36    pub full_values_multiline: String,
37    pub allowed_values: Option<Vec<String>>,
38}
39
40impl TriggerVariableView {
41    fn new(v: &TestPlanVariable) -> Self {
42        let mut sections: Vec<&'static str> = v
43            .declarations
44            .iter()
45            .map(|d| match d.section {
46                ConfigSection::Scenario => "scenario",
47                ConfigSection::Environment => "environment",
48            })
49            .collect();
50        sections.dedup();
51
52        let full_values = current_value_list(v);
53        let allowed_values = v
54            .declarations
55            .iter()
56            .find_map(|d| d.allowed_values.as_ref())
57            .map(|values| values.iter().map(ToString::to_string).collect());
58
59        Self {
60            name: v.name.clone(),
61            sections,
62            description: v
63                .declarations
64                .first()
65                .map(|d| d.description.clone())
66                .unwrap_or_default(),
67            ghost_text: full_values.join(", "),
68            full_values_multiline: full_values.join("\n"),
69            full_values,
70            allowed_values,
71        }
72    }
73
74    pub fn is_selected(&self, value: &str) -> bool {
75        self.full_values.iter().any(|v| v == value)
76    }
77}

112pub struct TestPlanDetailsView {
113    pub uuid: Uuid,
114    pub github_url: String,
115    pub source_sha: String,
116    pub variables: Vec<TriggerVariableView>,
117    pub n_executions: usize,
118    pub n_services: usize,
119    pub matrix_formula: String,
120    pub compound_groups: Vec<CompoundGroupView>,
121    pub services: Vec<ServiceRowView>,
122    pub services_vary_by_matrix: bool,
123    pub n_containers_per_execution: Option<usize>,
124    pub history_from_label: String,
125    pub history_to_label: String,
126    pub sampled_runs: u64,
127    days_back: u32,
128    days: u32,
129    history: TestPlanHistory,
130}
131
132impl TestPlanDetailsView {
133    pub fn new(details: TestPlanDetails, days_back: u32, days: u32) -> Self {
134        let source = details.source;
135        let matrix = details.matrix;
136        let environment = details.environment;
137        let history = details.history;
138
139        let github_url = format!(
140            "https://github.com/{}/{}/blob/{}/{}",
141            source.org, source.repo, source.sha, source.path
142        );
143
144        let n_containers_per_execution = (!environment.has_variable_replicas).then(|| {
145            environment
146                .services
147                .iter()
148                .map(|s| match s.replicas {
149                    ServiceReplicas::Fixed(n) => n as usize,
150                    ServiceReplicas::Variable(_) => 0,
151                })
152                .sum()
153        });
154
155        let mut matrix_formula_parts: Vec<String> = matrix
156            .dimensions
157            .iter()
158            .map(|(name, values)| format!("{} {name}", values.len()))
159            .collect();
160        matrix_formula_parts.extend(
161            matrix
162                .compound_groups
163                .iter()
164                .map(|(name, entries)| format!("{} {name}", entries.len())),
165        );
166
167        Self {
168            uuid: details.uuid,
169            github_url,
170            source_sha: source.sha,
171            n_executions: matrix.n_executions,
172            n_services: environment.services.len(),
173            matrix_formula: matrix_formula_parts.join(" × "),
174            compound_groups: matrix
175                .compound_groups
176                .iter()
177                .map(|(name, entries)| CompoundGroupView {
178                    name: name.clone(),
179                    entries: entries
180                        .iter()
181                        .map(|group| {
182                            group
183                                .iter()
184                                .map(|(k, v)| (k.clone(), v.to_string()))
185                                .collect()
186                        })
187                        .collect(),
188                })
189                .collect(),
190            services: environment
191                .services
192                .iter()
193                .map(ServiceRowView::new)
194                .collect(),
195            services_vary_by_matrix: environment.services_vary_by_matrix,
196            n_containers_per_execution,
197            history_from_label: format_day(history.window.from),
198            history_to_label: format_day(history.window.to - Duration::days(1)),
199            sampled_runs: history.execution_durations.total,
200            variables: details
201                .variables
202                .iter()
203                .map(TriggerVariableView::new)
204                .collect(),
205            days_back,
206            days,
207            history,
208        }
209    }
210
211    pub fn js_data(&self) -> TestPlanDetailsJsData<'_> {
212        TestPlanDetailsJsData {
213            variables: &self.variables,
214            compound_groups: self
215                .compound_groups
216                .iter()
217                .map(|g| (g.name.clone(), g.entries.len()))
218                .collect(),
219            history: &self.history,
220        }
221    }
222
223    pub fn prev_history_href(&self, current_ref: &str) -> String {
224        self.history_href_with_days_back(current_ref, self.days_back + self.days)
225    }
226
227    pub fn next_history_href(&self, current_ref: &str) -> String {
228        self.history_href_with_days_back(current_ref, self.days_back.saturating_sub(self.days))
229    }
230
231    pub fn has_next_history(&self) -> bool {
232        self.days_back > 0
233    }
234
235    fn history_href_with_days_back(&self, current_ref: &str, days_back: u32) -> String {
236        let mut qs = form_urlencoded::Serializer::new(String::new());
237        if !current_ref.is_empty() {
238            qs.append_pair("trigger_ref", current_ref);
239        }
240        qs.append_pair("days_back", &days_back.to_string());
241        format!("/ui/test-plan/{}?{}", self.uuid, qs.finish())
242    }
243}
