16/// Upper bound on how long a non-terminal run is auto-refreshed. A run that never reaches a
17/// terminal state (e.g. a stuck orchestrator) would otherwise be polled forever; past this age the
18/// UI stops polling and the user can refresh manually.
19const MAX_POLL_AGE_SECS: i64 = 60 * 60;
20
21/// The run status page content: the overall-status banner and the executions table. This whole
22/// region is re-rendered on every htmx poll (via `hx-select` against the same page route), so the
23/// banner, elapsed time, and "last updated" indicator all refresh together and stop together on
24/// terminal.
25pub struct RunView {
26    pub id: Uuid,
27    pub name: String,
28    pub status_label: String,
29    pub status_class: &'static str,
30    pub initiated_by: String,
31    pub started_at: String,
32    pub updated_at: String,
33    pub completed_at: Option<String>,
34    /// Wall-clock time from start to completion (or to `now` while still running).
35    pub elapsed: String,
36    /// When this render happened, shown as the "last updated" indicator.
37    pub last_updated: String,
38    /// Whether the rendered region should keep polling for updates (drives the htmx trigger).
39    pub should_poll: bool,
40    /// How many executions this run has in total, unaffected by `execution_status_filter` — shown
41    /// next to the table heading so filtering down to a status doesn't read as "the run only has
42    /// this many executions."
43    pub total_executions: usize,
44    /// The `current_status` value (its `Display` string, e.g. `"RUNNING"`) executions are filtered
45    /// down to, or empty for no filter. Kept as a string, not a parsed [`Status`], since its only
46    /// uses are re-populating the `<select>` and round-tripping through URLs.
47    pub execution_status_filter: String,
48    /// How many executions are in each status, in lifecycle order, excluding statuses no execution
49    /// currently has. Always reflects every execution, regardless of `execution_status_filter`.
50    pub status_breakdown: Vec<StatusCountView>,
51    /// Executions matching `execution_status_filter` (all of them, if empty).
52    pub executions: Vec<ExecutionView>,
53}

55impl RunView {
56    /// Build the run view. `now` decides whether the region keeps polling and anchors the elapsed
57    /// time and "last updated" indicator. `links_cfg` supplies the GCP/Grafana deep-link
58    /// configuration for each execution row. `execution_status_filter` restricts the executions
59    /// table to rows whose status matches exactly (empty means no filter).
60    pub fn new(
61        run: TestRunSummary,
62        now: DateTime<Utc>,
63        links_cfg: &LinksConfig,
64        execution_status_filter: String,
65    ) -> Self {
66        // Trust `current_status`, not just the presence of `completed_at`, to decide whether the run
67        // is actually done — a run that is still in progress should never show a completion time,
68        // even if the wire data is momentarily inconsistent.
69        let completed_at = run
70            .completed_at
71            .filter(|_| run.current_status.is_terminal());
72        let end = completed_at.unwrap_or(now);
73        let total_executions = run.executions.len();
74        let status_breakdown = status_breakdown(&run.executions);
75
76        Self {
77            id: run.id,
78            name: run.name,
79            status_label: run.current_status.to_string(),
80            status_class: status::css_class(run.current_status),
81            initiated_by: run.initiated_by,
82            started_at: format_rfc3339(run.started_at),
83            updated_at: format_rfc3339(run.updated_at),
84            completed_at: completed_at.map(format_rfc3339),
85            elapsed: format_duration(Duration::from_secs(
86                (end.timestamp() - run.started_at.timestamp()).max(0) as u64,
87            ))
88            .to_string(),
89            last_updated: now.format("%H:%M:%S UTC").to_string(),
90            should_poll: should_poll(run.current_status, run.started_at, now),
91            total_executions,
92            status_breakdown,
93            executions: run
94                .executions
95                .into_iter()
96                .filter(|execution| {
97                    execution_status_filter.is_empty()
98                        || execution.current_status.to_string() == execution_status_filter
99                })
100                .map(|execution| ExecutionView::new(execution, links_cfg))
101                .collect(),
102            execution_status_filter,
103        }
104    }
105
106    /// The URL the executions table's htmx auto-poll and manual "Refresh now" button fetch from —
107    /// this run's status page, carrying `execution_status_filter` forward so a refresh doesn't
108    /// silently drop the active filter.
109    pub fn poll_url(&self) -> String {
110        if self.execution_status_filter.is_empty() {
111            format!("/ui/run/{}", self.id)
112        } else {
113            let mut qs = form_urlencoded::Serializer::new(String::new());
114            qs.append_pair("execution_status", &self.execution_status_filter);
115            format!("/ui/run/{}?{}", self.id, qs.finish())
116        }
117    }
118
119    /// The message shown in the executions table in place of rows when there's nothing to show:
120    /// either the run genuinely has no executions yet, or a status filter matched none of them.
121    pub fn executions_empty_message(&self) -> &'static str {
122        if self.execution_status_filter.is_empty() {
123            "No executions yet."
124        } else {
125            "No executions match this filter."
126        }
127    }
128}

130/// Every [`Status`] variant, in lifecycle order — the order the run's status breakdown and the
131/// execution-status filter's `<select>` present statuses in.
132const STATUS_LIFECYCLE_ORDER: [Status; 8] = [
133    Status::Initialising,
134    Status::Resolving,
135    Status::Provisioning,
136    Status::EnvironmentReady,
137    Status::Running,
138    Status::Successful,
139    Status::Failed,
140    Status::Unrunnable,
141];
142
143/// One row in the run's status-breakdown summary.
144pub struct StatusCountView {
145    pub status_label: String,
146    pub status_class: &'static str,
147    pub count: usize,
148}
149
150/// How many `executions` are in each status, in lifecycle order, omitting statuses none of them are
151/// in — a run with no `Unrunnable` executions shouldn't show an "Unrunnable: 0" row.
152fn status_breakdown(executions: &[TestExecutionSummary]) -> Vec<StatusCountView> {
153    STATUS_LIFECYCLE_ORDER
154        .into_iter()
155        .filter_map(|candidate| {
156            let count = executions
157                .iter()
158                .filter(|execution| execution.current_status == candidate)
159                .count();
160
161            (count > 0).then(|| StatusCountView {
162                status_label: candidate.to_string(),
163                status_class: status::css_class(candidate),
164                count,
165            })
166        })
167        .collect()
168}

170/// The execution view within a test run summary
171pub struct ExecutionView {
172    pub id: Uuid,
173    pub name: String,
174    pub status_label: String,
175    pub status_class: &'static str,
176    pub exit_code: Option<i32>,
177    pub started_at: String,
178    pub updated_at: String,
179    /// Cloud Logging deep link scoped to this execution's workload namespace and time window.
180    pub logs_url: String,
181    /// Grafana deep link scoped to this execution's workload namespace and time window.
182    pub grafana_url: String,
183}
184
185impl ExecutionView {
186    fn new(execution: TestExecutionSummary, links_cfg: &LinksConfig) -> Self {
187        let namespace = execution.id.to_string();
188        let window_end = execution.completed_at.unwrap_or_else(Utc::now);
189        let logs_url = gcp_logs(links_cfg, &namespace, execution.started_at, window_end);
190        let grafana_url = grafana(links_cfg, &namespace, execution.started_at, window_end);
191
192        Self {
193            id: execution.id,
194            name: execution.name,
195            status_label: execution.current_status.to_string(),
196            status_class: status::css_class(execution.current_status),
197            exit_code: status::effective_exit_code(execution.current_status, execution.exit_code),
198            started_at: format_rfc3339(execution.started_at),
199            updated_at: format_rfc3339(execution.updated_at),
200            logs_url,
201            grafana_url,
202        }
203    }
204}

206/// Whether a run should keep being polled: only while it is non-terminal and has not been running
207/// longer than [`MAX_POLL_AGE_SECS`] (the stuck-run guard).
208fn should_poll(status: Status, started_at: DateTime<Utc>, now: DateTime<Utc>) -> bool {
209    !status.is_terminal() && (now.timestamp() - started_at.timestamp()) < MAX_POLL_AGE_SECS
210}
