6/// Errors produced when communicating with the orchestrator HTTP API.
7#[derive(Debug, Error)]
8pub enum Error {
9    #[error("orchestrator returned {status} fetching run id {id}")]
10    TestRunStatus { status: StatusCode, id: Uuid },
11
12    #[error("orchestrator returned {status} fetching execution id {id}")]
13    TestExecutionStatus { status: StatusCode, id: Uuid },
14
15    #[error(transparent)]
16    Reqwest(#[from] reqwest::Error),
17}
18
19pub trait Client: Send + Sync + Clone + 'static {
20    fn run_summary(
21        &self,
22        id: Uuid,
23    ) -> impl Future<Output = Result<Option<TestRunSummary>, Error>> + Send;
24
25    fn execution_summary(
26        &self,
27        id: Uuid,
28    ) -> impl Future<Output = Result<Option<TestExecutionSummary>, Error>> + Send;
29}

31#[derive(Debug, Clone)]
32pub struct HttpClient {
33    client: reqwest::Client,
34    orchestrator_url: Url,
35}
36
37impl HttpClient {
38    /// Creates a new [HttpClient] that will report updates to `orchestrator_url` for the test execution associated
39    /// with `execution_id`, authenticating requests with `execution_token` as a Bearer token.
40    pub fn try_new(orchestrator_url: String) -> anyhow::Result<Self> {
41        Ok(Self {
42            client: reqwest::Client::new(),
43            orchestrator_url: Url::parse(&orchestrator_url)?,
44        })
45    }
46}
47
48impl Client for HttpClient {
49    async fn run_summary(&self, id: Uuid) -> Result<Option<TestRunSummary>, Error> {
50        let url = test_run_status_url(&self.orchestrator_url, id);
51        let response = self.client.get(url).send().await?;
52
53        match response.status() {
54            StatusCode::OK => Ok(Some(response.json().await?)),
55            StatusCode::NOT_FOUND => Ok(None),
56            status => Err(Error::TestRunStatus { status, id }),
57        }
58    }
59
60    async fn execution_summary(&self, id: Uuid) -> Result<Option<TestExecutionSummary>, Error> {
61        let url = test_execution_status_url(&self.orchestrator_url, id);
62        let response = self.client.get(url).send().await?;
63
64        match response.status() {
65            StatusCode::OK => Ok(Some(response.json().await?)),
66            StatusCode::NOT_FOUND => Ok(None),
67            status => Err(Error::TestExecutionStatus { status, id }),
68        }
69    }
70}
71
72fn test_run_status_url(base_url: &Url, id: Uuid) -> Url {
73    base_url
74        .join(&format!("/test-run/{id}/status"))
75        .expect("base url should be valid")
76}
77
78fn test_execution_status_url(base_url: &Url, id: Uuid) -> Url {
79    base_url
80        .join(&format!("/test-execution/{id}/status"))
81        .expect("base url should be valid")
82}
