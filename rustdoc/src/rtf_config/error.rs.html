1//! Generic helpers for working with operations that can return multiple errors, such as templating
2//! and checks.
3use std::{fmt, slice, vec};
4
5pub trait ErrorKind: fmt::Debug + fmt::Display + Copy {
6    const HEADER: &str;
7}
8
9/// One or more [Error]s.
10///
11/// If you know that you only have a single error to report then [Errors::new] can be used to
12/// quickly construct an new [Errors]. Otherwise, you should programmatically build up the set of
13/// errors using an [ErrorBuilder]. See the documentation there for usage.
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub struct Errors<K>
16where
17    K: ErrorKind,
18{
19    inner: Vec<Error<K>>,
20}
21
22impl<K> Errors<K>
23where
24    K: ErrorKind,
25{
26    /// Construct a new [Errors] containing a single [Error].
27    ///
28    /// Prefer using [ErrorBuilder] when there is the possibility of multiple errors arising from a
29    /// single operation.
30    pub fn new(kind: K, message: impl Into<String>, path: &[String]) -> Self {
31        Self {
32            inner: vec![Error {
33                kind,
34                message: message.into(),
35                path: path.join("."),
36            }],
37        }
38    }
39
40    /// Assert that this [Errors] only contains a single [Error] and extract it.
41    ///
42    /// # Panics
43    /// This method will panic if there is more than one error.
44    pub fn unwrap_single(mut self) -> Error<K> {
45        if self.inner.len() > 1 {
46            panic!(
47                "expected a single error but there were {}",
48                self.inner.len()
49            )
50        }
51
52        self.inner.remove(0)
53    }
54
55    /// Extract the underlying [Vec] of [Error]s.
56    pub fn into_vec(self) -> Vec<Error<K>> {
57        self.inner
58    }
59
60    /// Iterate over the underlying errors in the order they were encountered.
61    pub fn iter(&self) -> slice::Iter<'_, Error<K>> {
62        self.inner.iter()
63    }
64
65    /// Append another [Errors] onto this one.
66    pub fn append(&mut self, other: Errors<K>) {
67        self.inner.extend(other.inner);
68    }
69}
70
71impl<K> IntoIterator for Errors<K>
72where
73    K: ErrorKind,
74{
75    type Item = Error<K>;
76    type IntoIter = vec::IntoIter<Self::Item>;
77
78    fn into_iter(self) -> Self::IntoIter {
79        self.inner.into_iter()
80    }
81}
82
83impl<K> fmt::Display for Errors<K>
84where
85    K: ErrorKind,
86{
87    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
88        let msgs: Vec<String> = self.inner.iter().map(|e| e.to_string()).collect();
89
90        write!(f, "{}\n{}", K::HEADER, msgs.join("\n\n"))
91    }
92}

94impl<K> std::error::Error for Errors<K> where K: ErrorKind {}

96/// Programmatically build up an ordered list of errors encountered by a single operation.
97#[derive(Default, Debug)]
98pub struct ErrorBuilder<K>
99where
100    K: ErrorKind,
101{
102    inner: Vec<Error<K>>,
103}
104
105impl<K> From<Result<(), Errors<K>>> for ErrorBuilder<K>
106where
107    K: ErrorKind,
108{
109    fn from(res: Result<(), Errors<K>>) -> Self {
110        match res {
111            Ok(_) => Self::new(),
112            Err(e) => Self { inner: e.inner },
113        }
114    }
115}
116
117impl<K> ErrorBuilder<K>
118where
119    K: ErrorKind,
120{
121    /// Construct a new empty [ErrorBuilder].
122    pub fn new() -> Self {
123        Self { inner: Vec::new() }
124    }
125
126    /// Construct and append a new [Error] to the builder.
127    pub fn push(&mut self, kind: K, message: impl Into<String>, path: &[String]) {
128        self.inner.push(Error {
129            kind,
130            message: message.into(),
131            path: path.join("."),
132        });
133    }
134
135    /// Append a new [Error] to the builder.
136    pub fn push_err(&mut self, err: Error<K>) {
137        self.inner.push(err);
138    }
139
140    /// Add all errors from `res` to the end of this builder.
141    ///
142    /// Typically used to combine errors from nested sources when working with a larger structure
143    /// such as a config file.
144    pub fn append(&mut self, res: Result<(), Errors<K>>) {
145        if let Err(e) = res {
146            self.inner.extend(e.inner);
147        }
148    }
149
150    /// Construct a result from this builder, returning `Ok(t)` if the builder is empty or
151    /// `Err(Errors)` if at least one [Error] is present.
152    ///
153    /// Other than calling [Errors::new] for a single error, this is the only way to construct a
154    /// new [Errors] instance.
155    pub fn into_result<T>(self, t: T) -> Result<T, Errors<K>> {
156        if self.inner.is_empty() {
157            Ok(t)
158        } else {
159            Err(Errors { inner: self.inner })
160        }
161    }
162}
163
164impl<K> Extend<Error<K>> for ErrorBuilder<K>
165where
166    K: ErrorKind,
167{
168    fn extend<T>(&mut self, iter: T)
169    where
170        T: IntoIterator<Item = Error<K>>,
171    {
172        self.inner.extend(iter);
173    }
174}

176/// An individual error pairing a given error kind with a string message.
177///
178/// See [Errors::new] and [ErrorBuilder::push] for details on how to construct a new error.
179#[derive(Debug, Clone, PartialEq, Eq)]
180pub struct Error<K>
181where
182    K: ErrorKind,
183{
184    /// The kind associated with this error.
185    pub kind: K,
186    /// The meta-data message attached to this error.
187    pub message: String,
188    /// The dotted path to the source of this error.
189    pub path: String,
190}
191
192impl<K> fmt::Display for Error<K>
193where
194    K: ErrorKind,
195{
196    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
197        write!(f, "({}) {}\n{}", self.path, self.kind, self.message)
198    }
199}
