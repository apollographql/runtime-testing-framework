20/// # Text File Provider
21///
22/// A subset of file providers that can produce arbitrary utf-8 text as their output.
23#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
24#[serde(rename_all = "snake_case", tag = "kind")]
25pub enum TextFileProvider {
26    GithubFile(GithubFile),
27    GraphosSubgraphRouterUrlOverrides(GraphosSubgraphRouterUrlOverrides),
28    Inline(InlineFile),
29    RelativePath(RelativeFile),
30    Required(RequiredFile),
31}
32
33// Each time we add a new variant to the TextFileProvider enum above we need to remember to add it
34// to the macro invocation below in order to update the trait implementations for the enum. (You
35// can't really forget to do this as the compiler will complain about missing match arms if you
36// do!)
37macro_rules! enum_impl_text_file_provider {
38    ($($variant:ident),+) => {
39        enum_impl_check!(TextFileProvider => $($variant),+);
40        enum_impl_template!(TextFileProvider => $($variant),+);
41        enum_impl_as_utf8_file_content!(TextFileProvider => $($variant),+);
42    };
43}

47    GraphosSubgraphRouterUrlOverrides,

53/// # Merge YAML
54///
55/// Merge the YAML output of two text based file providers into a single YAML file.
56///
57/// Matching keys in the overrides file will replace scalar values, concatenate arrays
58/// and merge keys for maps.
59///
60/// ```yaml
61/// - name: router-config.yaml
62///   env_var: ROUTER_CONFIG
63///   kind: merge_yaml
64///   base:
65///     kind: relative_path
66///     path: "data/base-router-config.yaml"
67///   overrides:
68///     kind: graphos_subgraph_router_url_overrides
69///     graph_ref: "foo@bar"
70///     url_format: "docker"
71/// ```
72#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
73pub struct MergeYaml {
74    /// A base YAML file to start with.
75    pub(crate) base: TextFileProvider,
76    /// An second YAML file to merge on top of the base file.
77    pub(crate) overrides: TextFileProvider,
78}
79
80impl AsUtf8FileContent for MergeYaml {
81    async fn try_get_file_content(
82        &self,
83        src: &Source,
84        ctx: &impl ResolutionContext,
85    ) -> Result<String> {
86        let base_str = self.base.try_get_file_content(src, ctx).await?;
87        let overrides_str = self.overrides.try_get_file_content(src, ctx).await?;
88
89        let mut base: serde_yaml::Value = match serde_yaml::from_str(&base_str) {
90            Ok(v) => v,
91            Err(e) => {
92                error!("found invalid YAML in base file for merge_yaml file provider");
93                return Err(e.into());
94            }
95        };
96        let overrides: serde_yaml::Value = match serde_yaml::from_str(&overrides_str) {
97            Ok(v) => v,
98            Err(e) => {
99                error!("found invalid YAML in overrides file for merge_yaml file provider");
100                return Err(e.into());
101            }
102        };
103
104        merge_yaml(overrides, &mut base);
105
106        Ok(serde_yaml::to_string(&base)?.trim().to_string())
107    }
108}
109
110impl_template!(MergeYaml => [base, overrides]);

112impl Check for MergeYaml {
113    fn try_check(
114        &self,
115        path: &mut Vec<String>,
116        src: &Source,
117        ctx: &impl ResolutionContext,
118    ) -> checks::Result<()> {
119        let mut errs = checks::ErrorBuilder::new();
120        errs.append(self.base.try_check(path, src, ctx));
121        errs.append(self.overrides.try_check(path, src, ctx));
122
123        errs.into_result(())
124    }
