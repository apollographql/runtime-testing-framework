15/// The source directory of where a particular config file was obtained.
16///
17/// In its simplest form this is a local file path to the directory containing the config file, but
18/// this may also include things like pulling the file over the network.
19#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
20#[serde(rename_all = "snake_case", tag = "kind")]
21pub enum SourceDir {
22    /// The config file was read from disk
23    Local {
24        /// The absolute path to the config file
25        abs_path: PathBuf,
26    },
27
28    /// The config file was downloaded from GitHub
29    Github {
30        /// The GitHub org
31        org: String,
32        /// The GitHub repository
33        repo: String,
34        /// The path to the file within the GitHub repository
35        path: PathBuf,
36        /// An optional ref of the repo to use (the default branch is used when None)
37        git_ref: Option<String>,
38    },
39}
40
41impl SourceDir {
42    pub fn local(abs_path: impl Into<PathBuf>) -> Self {
43        Self::Local {
44            abs_path: abs_path.into(),
45        }
46    }
47
48    pub fn github(
49        org: impl Into<String>,
50        repo: impl Into<String>,
51        path: impl Into<PathBuf>,
52        git_ref: Option<impl Into<String>>,
53    ) -> Self {
54        Self::Github {
55            org: org.into(),
56            repo: repo.into(),
57            path: path.into(),
58            git_ref: git_ref.map(Into::into),
59        }
60    }
61
62    pub(crate) fn github_permalink_for(
63        &self,
64        child_path: impl AsRef<Path>,
65        is_file: bool,
66    ) -> Option<String> {
67        match self {
68            SourceDir::Local { .. } => None,
69            SourceDir::Github {
70                org,
71                repo,
72                path,
73                git_ref,
74            } => Some(github_permalink(
75                org,
76                repo,
77                git_ref.as_deref(),
78                path.join(child_path).display(),
79                is_file,
80            )),
81        }
82    }
83
84    pub(crate) fn to_uri_for(&self, child_path: impl AsRef<Path>) -> String {
85        self.to_uri(Some(child_path))
86    }
87
88    fn to_uri(&self, child_path: Option<impl AsRef<Path>>) -> String {
89        let full_path = |base: &Path| match child_path {
90            Some(tail) => base.join(tail),
91            None => base.to_path_buf(),
92        };
93
94        match self {
95            Self::Local { abs_path } => {
96                format!("file://{}", full_path(abs_path).display())
97            }
98
99            Self::Github {
100                org,
101                repo,
102                path,
103                git_ref: Some(git_ref),
104            } => format!(
105                "https://github.com/{org}/{repo}/{}?ref={git_ref}",
106                full_path(path).display()
107            ),
108
109            Self::Github {
110                org,
111                repo,
112                path,
113                git_ref: None,
114            } => format!(
115                "https://github.com/{org}/{repo}/{}",
116                full_path(path).display()
117            ),
118        }
119    }
120
121    pub async fn try_get_file_content(
122        &self,
123        fname: impl AsRef<Path>,
124        ctx: &impl ResolutionContext,
125    ) -> Result<String> {
126        match self {
127            Self::Local { abs_path } => Ok(ctx.read_path_to_string(abs_path.join(fname))?),
128            Self::Github {
129                org,
130                repo,
131                path,
132                git_ref,
133            } => {
134                let client = ctx.github_client().ok_or(providers::Error::Github(
135                    rtf_integrations::github::Error::NoClient,
136                ))?;
137
138                Ok(client
139                    .string_file_content(
140                        org,
141                        repo,
142                        &path.join(fname).display().to_string(),
143                        git_ref.as_ref(),
144                    )
145                    .await?)
146            }
147        }
148    }
149}

191/// A stable, human-readable logical name for the source of a configuration value.
192///
193/// Unlike [SourceDir] which carries absolute disk paths or GitHub refs, `StableSource` carries
194/// only a logical name that is safe to serialize and share without leaking local path information.
195#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
196#[serde(rename_all = "snake_case", tag = "kind")]
197pub enum StableSource {
198    /// The value came from the test plan config file
199    TestPlan,
200    /// The value came from the environment config file
201    Environment,
202    /// The value came from the scenario config file
203    Scenario,
204    /// The value came from a custom provider definition. The first element is the provider name
205    /// and the second is the section in which it was declared.
206    CustomProvider {
207        section: CustomProviderSection,
208        ident: String,
209    },
210    /// The value was provided on the command line
211    Cli,
212    /// The value was provided from a variables file
213    VariablesFile,
214}

216/// # Config Source
217#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
218#[serde(rename_all = "snake_case", tag = "kind")]
219pub enum RawSource {
220    Local {
221        /// A relative path to the target config file
222        relative_path: PathBuf,
223    },
224    Github {
225        /// The GitHub org to pull the config file from
226        org: String,
227        /// The GitHub repository to pull the config file from
228        repo: String,
229        /// The absolute path to the config file within the target GitHub repository
230        path: PathBuf,
231        /// An optional git ref to pull the config file from (defaults to mainline)
232        #[serde(default)]
233        git_ref: Option<String>,
234    },
235}
236
237impl RawSource {
238    pub fn try_into_source_and_filename(
239        self,
240        file_source: &SourceDir,
241        ctx: &impl ResolutionContext,
242    ) -> io::Result<(SourceDir, String)> {
243        let (mut src, file_name) = self.try_into_source_without_canonical_path(file_source)?;
244        if let SourceDir::Local { abs_path } = &mut src {
245            *abs_path = ctx.canonicalize_path(&*abs_path)?;
246        }
247
248        Ok((src, file_name))
249    }
250
251    pub(crate) fn with_child_path(&self, child_path: impl AsRef<Path>) -> Self {
252        let mut new = self.clone();
253        match &mut new {
254            Self::Local { relative_path } => *relative_path = relative_path.join(child_path),
255            Self::Github { path, .. } => *path = path.join(child_path),
256        }
257
258        new
259    }
260
261    fn try_into_source_without_canonical_path(
262        self,
263        file_source: &SourceDir,
264    ) -> io::Result<(SourceDir, String)> {
265        let split = |p: PathBuf| {
266            let file_name = p
267                .file_name()
268                .and_then(|os_str| os_str.to_str())
269                .ok_or_else(|| io::Error::other("expected a filename"))?
270                .to_string();
271            let p = p.parent().unwrap().to_owned();
272
273            io::Result::Ok((p, file_name))
274        };
275
276        match self {
277            Self::Local { relative_path } => {
278                let (relative_dir, file_name) = split(relative_path)?;
279
280                let src = match file_source {
281                    SourceDir::Local {
282                        abs_path: containing_dir,
283                    } => SourceDir::Local {
284                        abs_path: containing_dir.join(relative_dir),
285                    },
286
287                    SourceDir::Github {
288                        org,
289                        repo,
290                        path: containing_dir,
291                        git_ref,
292                    } => SourceDir::Github {
293                        org: org.clone(),
294                        repo: repo.clone(),
295                        path: containing_dir.join(relative_dir),
296                        git_ref: git_ref.clone(),
297                    },
298                };
299
300                Ok((src, file_name))
301            }
302
303            Self::Github {
304                org,
305                repo,
306                path,
307                git_ref,
308            } => {
309                let (path, file_name) = split(path)?;
310
311                let src = SourceDir::Github {
312                    org,
313                    repo,
314                    path,
315                    git_ref,
316                };
317
318                Ok((src, file_name))
319            }
320        }
321    }
322}
