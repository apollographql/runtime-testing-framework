7use rtf_core::github::Client;
8use schemars::JsonSchema;
9use serde::{Deserialize, Serialize};
10use std::{io, path::PathBuf};

12/// The source of how a particular config file was obtained.
13///
14/// In its simplest form this is a local file path to the directory containing the config file, but
15/// this may also include things like pulling the file over the network.
16#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
17#[serde(rename_all = "snake_case", tag = "kind")]
18pub enum Source {
19    /// The config file was read from disk
20    Local {
21        /// The absolute path to the config file
22        abs_path: PathBuf,
23    },
24
25    /// The config file was downloaded from GitHub
26    Github {
27        /// The GitHub org
28        org: String,
29        /// The GitHub repository
30        repo: String,
31        /// The path to the file within the GitHub repository
32        path: PathBuf,
33        /// An optional ref of the repo to use (the default branch is used when None)
34        git_ref: Option<String>,
35    },
36}
37
38impl Source {
39    pub fn local(abs_path: impl Into<PathBuf>) -> Self {
40        Self::Local {
41            abs_path: abs_path.into(),
42        }
43    }
44
45    pub fn github(
46        org: impl Into<String>,
47        repo: impl Into<String>,
48        path: impl Into<PathBuf>,
49        git_ref: Option<impl Into<String>>,
50    ) -> Self {
51        Self::Github {
52            org: org.into(),
53            repo: repo.into(),
54            path: path.into(),
55            git_ref: git_ref.map(Into::into),
56        }
57    }
58
59    pub async fn try_get_file_content(&self, ctx: &impl ResolutionContext) -> Result<String> {
60        match self {
61            Self::Local { abs_path } => Ok(ctx.read_path_to_string(abs_path)?),
62            Self::Github {
63                org,
64                repo,
65                path,
66                git_ref,
67            } => {
68                let client = ctx
69                    .github_client()
70                    .ok_or(providers::Error::Github(rtf_core::github::Error::NoClient))?;
71
72                Ok(client
73                    .string_file_content(org, repo, &path.display().to_string(), git_ref.as_ref())
74                    .await?)
75            }
76        }
77    }
78}
79
80impl Default for Source {
81    fn default() -> Self {
82        Self::Local {
83            abs_path: PathBuf::new(),
84        }
85    }
86}

88/// # Config Source
89#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize, JsonSchema)]
90#[serde(rename_all = "snake_case", tag = "kind")]
91pub enum RawSource {
92    Local {
93        /// A relative path to the target config file
94        relative_path: PathBuf,
95    },
96    Github {
97        /// The GitHub org to pull the config file from
98        org: String,
99        /// The GitHub repository to pull the config file from
100        repo: String,
101        /// The absolute path to the config file within the target GitHub repository
102        path: String,
103        /// An optional git ref to pull the config file from (defaults to mainline)
104        #[serde(default)]
105        git_ref: Option<String>,
106    },
107}
108
109impl RawSource {
110    pub fn try_into_source(
111        self,
112        tp_source: &Source,
113        ctx: &impl ResolutionContext,
114    ) -> io::Result<Source> {
115        match self.try_into_source_without_canonical_path(tp_source) {
116            Source::Local { abs_path } => Ok(Source::Local {
117                abs_path: ctx.canonicalize_path(abs_path)?,
118            }),
119
120            gh => Ok(gh),
121        }
122    }
123
124    fn try_into_source_without_canonical_path(self, tp_source: &Source) -> Source {
125        match self {
126            Self::Local { relative_path } => match tp_source {
127                Source::Local {
128                    abs_path: test_plan_path,
129                } => {
130                    let abs_path = match test_plan_path.parent() {
131                        Some(parent) => parent.join(relative_path),
132                        None => relative_path,
133                    };
134
135                    Source::Local { abs_path }
136                }
137
138                Source::Github {
139                    org,
140                    repo,
141                    path: test_plan_path,
142                    git_ref,
143                } => {
144                    let path = match test_plan_path.parent() {
145                        Some(parent) => parent.join(relative_path),
146                        None => relative_path,
147                    };
148
149                    Source::Github {
150                        org: org.clone(),
151                        repo: repo.clone(),
152                        path,
153                        git_ref: git_ref.clone(),
154                    }
155                }
156            },
157
158            Self::Github {
159                org,
160                repo,
161                path,
162                git_ref,
163            } => Source::Github {
164                org,
165                repo,
166                path: PathBuf::from(path),
167                git_ref,
168            },
169        }
170    }
171}

178    fn raw_local(path: &str) -> RawSource {
179        RawSource::Local {

193    // Option<impl Into<String>> requires explicit typing for "None"

196    // Github RawSource is independent of the test plan source so these should all just map
197    // directly from their raw to "cooked" counterpart

214        "github test plan github raw"
215    )]

218        Source::github("org", "repo", "foo/test-plan.yaml", Some("branch")),
