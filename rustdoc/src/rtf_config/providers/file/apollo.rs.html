33/// # GraphOS Supergraph SDL
34///
35/// The user specifies the ref that should be used to fetch a supergraph SDL
36/// file from the GraphOS API.
37///
38/// ```yaml
39/// - name: "supergraph.graphql"
40///   env_var: SUPERGRAPH
41///   kind: graphos_supergraph
42///   graph_ref: graph@variant
43///   with_subgraph_overrides: docker
44/// ```
45#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
46pub struct GraphosSupergraph {
47    /// The Apollo graph ref to pull supergraph SDL for.
48    pub graph_ref: Field<String>,
49    /// Replace the supergraph's subgraph urls with overridden values for testing.
50    ///
51    /// Defaults to null if unset.
52    #[serde(default)]
53    #[template(skip)]
54    pub with_subgraph_overrides: Option<UrlFormat>,
55}
56
57impl AsUtf8FileContent for GraphosSupergraph {
58    async fn try_get_file_content(
59        &self,
60        _src: &Source,
61        ctx: &impl ResolutionContext,
62    ) -> providers::Result<String> {
63        let (graph_id, variant) = self
64            .graph_ref
65            .as_resolved()
66            .split_once('@')
67            .expect("validated graph_ref");
68
69        match self.with_subgraph_overrides.as_ref() {
70            Some(url_format) => {
71                let mut arc_details = ctx
72                    .with_supergraph_details(graph_id, variant, |details| Ok(details.clone()))
73                    .await?;
74
75                let details = Arc::make_mut(&mut arc_details);
76                let subgraph_urls = url_format.urls_for_subgraphs(&details.subgraphs);
77                details
78                    .rewrite_subgraph_urls(&subgraph_urls)
79                    .expect("unable to rewrite subgraph URLs");
80
81                Ok(details.supergraph_sdl.clone())
82            }
83
84            None => {
85                ctx.with_supergraph_details(graph_id, variant, |details| {
86                    Ok(details.supergraph_sdl.clone())
87                })
88                .await
89            }
90        }
91    }
92}

125impl ResolveFileContent for GraphosSubgraphs {
126    async fn try_get_all_file_contents(
127        &self,
128        target: impl AsRef<Path>,
129        _src: &Source,
130        ctx: &mut impl ResolutionContext,
131    ) -> providers::Result<Vec<(PathBuf, String)>> {
132        let (graph_id, variant) = self
133            .graph_ref

135            .split_once('@')
136            .expect("validated graph_ref");

146            .collect();
147
148        Ok(contents)
149    }

163/// # GraphOS Subgraph Docker Compose
164///
165/// The user specifies the graph ref that should be used to fetch the supergraph

169///
170/// ```yaml
171/// - name: "subgraph-compose.yaml"
172///   env_var: SUBGRAPH_COMPOSE
173///   kind: graphos_subgraph_docker_compose
174///   graph_ref: graph@variant
175///   image: ghcr.io/apollographql/runtime-testing-framework/router-scale-subgraph:main
176///   command:
177///   - -schema
178///   - /app/supergraph.graphql
179///   replicas: 5
180///   resource_limits:
181///     cpus: '0.5'
182///     memory: 1G
183///   resource_reservations:
184///    cpus: '0.1'
185///     memory: 512M
186///   mem_swappiness: 0
187///   loadbalancer:
188///     resource_limits:
189///       cpus: '0.5'

191///     resource_reservations:
192///       cpus: '0.1'

195/// ```

365        loadbalancer_resources.insert(
366            "limits".to_string(),

368                cpus: self

551/// # GraphOS Supergraph Router URL Overrides
552///
553/// The user specifies the graph ref that should be used to fetch subgraph
554/// SDL files from the GraphOS API and generates a the override_subgraph_urls
555/// YAML snippet that can be merged into a router config file
556///
557/// This should be used when generating the subgraph docker compose using
558/// [GraphosSubgraphDockerCompose]. This will ensure the router subgraph urls
559/// map to the loadbalancer url in that compose file.
560///
561/// ```yaml
562/// - name: subgraph-url-overrides.yaml
563///   env_var: SUBGRAPH_URL_OVERRIDES
564///   kind: graphos_subgraph_router_url_overrides
565///   graph_ref: graph@variant
566///   url_format: localhost
567/// ```
568#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
569pub struct GraphosSubgraphRouterUrlOverrides {
570    /// The Apollo graph ref to pull the subgraphs for.
571    pub graph_ref: Field<String>,
572    /// The format of the overrides url.
573    #[serde(default = "default_url_format")]
574    #[template(skip)]
575    pub url_format: UrlFormat,
576}
577
578fn default_url_format() -> UrlFormat {
579    UrlFormat::Localhost
580}
581
582/// The allowed url formats for the [GraphosSubgraphRouterUrlOverrides]
583#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
584#[serde(rename_all = "snake_case")]
585pub enum UrlFormat {
586    Localhost,
587    Docker,
588}
589
590impl UrlFormat {
591    fn urls_for_subgraphs(&self, subgraphs: &[Subgraph]) -> HashMap<String, String> {
592        let mut subgraph_urls: HashMap<String, String> = HashMap::new();
593        let base_port = 4001;
594
595        for (offset, sg) in subgraphs.iter().enumerate() {
596            let port = base_port + offset;
597
598            let url = self.subgraph_url(&port);
599            subgraph_urls.insert(sg.name.clone(), url);
600        }
601
602        subgraph_urls
603    }
604
605    fn subgraph_url(&self, port: &usize) -> String {
606        match self {
607            UrlFormat::Localhost => format!("http://localhost:{port}"),
608            // The docker compose generated by rtf puts all subgraph containers behind a loadbalancer
609            UrlFormat::Docker => "http://loadbalancer:8080".to_string(),
610        }
611    }
612}

738/// ```
739#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
740pub struct GraphosCannedOpsById {
741    /// The Apollo graph ref to pull operations for.
742    pub graph_ref: Field<String>,
743    /// Operation IDs from the Apollo studio API for the operations you want to
744    /// work with as queried from an `OperationInsightsListItem` in the Studio
745    /// graphQL API.
746    pub operation_ids: Vec<Field<String>>,
747}
748
749impl AsUtf8FileContent for GraphosCannedOpsById {
750    async fn try_get_file_content(
751        &self,
752        _src: &Source,
753        ctx: &impl ResolutionContext,
754    ) -> providers::Result<String> {
755        let (graph_id, variant) = self
756            .graph_ref
757            .as_resolved()
758            .split_once('@')
759            .expect("validated graph_ref");
760
761        let details: Arc<SupergraphDetails> = ctx
762            .with_supergraph_details(graph_id, variant, |details| Ok(details.clone()))
763            .await?;
764
765        let client = ctx.platform_client().expect("to have a platform client");
766        let ids: Vec<String> = self
767            .operation_ids
768            .iter()
769            .map(|id| id.as_resolved().clone())
770            .collect();
771        let canned_ops = canned_ops_for_ids(&details, ids, client).await?;
772
773        canned_ops_json_lines(canned_ops)
774    }
775}
776
777impl Check for GraphosCannedOpsById {
778    fn try_check(
779        &self,
780        path: &mut Vec<String>,
781        _src: &Source,
782        ctx: &impl ResolutionContext,
783    ) -> checks::Result<()> {
784        validate_graph_ref_and_client(self.graph_ref.as_resolved(), path, ctx)
785    }
786}
787
788fn canned_ops_json_lines(canned_ops: Vec<CannedOperation>) -> providers::Result<String> {
789    // Create a json line file for each of the canned operations
790    let mut json_file = canned_ops
791        .iter()
792        .map(|v| v.to_json_string())
793        .collect::<Result<Vec<_>, _>>()?
794        .join("\n");
795
796    // This appends a new line to the json file
797    // Without this, when shell scripts iterate over the operations they count the lines to iterate over as N-1
798    json_file.push('\n');
799
800    Ok(json_file)
801}

833impl Check for OfflineGraphosLicense {
834    fn try_check(
835        &self,
836        path: &mut Vec<String>,
837        _src: &Source,
838        ctx: &impl ResolutionContext,
839    ) -> checks::Result<()> {
840        validate_client(path, ctx)
841    }
842}
843
844fn validate_graph_ref_and_client(
845    graph_ref: &str,
846    path: &[String],
847    ctx: &impl ResolutionContext,
848) -> checks::Result<()> {
849    let mut errs = checks::ErrorBuilder::new();
850    if !graph_ref.contains('@') {
851        errs.push(
852            checks::ErrorKind::InvalidGraphRef,
853            format!("expected a string of the form 'graph_id@variant', got {graph_ref}"),
854            path,
855        );
856    }
857    errs.append(validate_client(path, ctx));
858
859    errs.into_result(())
860}
861
862fn validate_client(path: &[String], ctx: &impl ResolutionContext) -> checks::Result<()> {
863    if ctx.platform_client().is_none() {
864        return Err(checks::Errors::new(
865            checks::ErrorKind::MissingGraphOsApiKey,
866            "expected os env key APOLLO_KEY",
867            path,
868        ));
869    }
870
871    Ok(())
872}

874/// # Router Download Script
875///
876/// Produces a POSIX shell script that can be run in order to download a target version of the
877/// Apollo Router.
878///
879/// ```yaml
880/// - name: "router-download.sh"
881///   env_var: ROUTER_DOWNLOAD
882///   kind: router_download_script
883///   version: "v2.6.0"
884/// ```
885#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
886pub struct RouterDownloadScript {
887    /// The version of the Apollo Router to download.
888    pub(crate) version: Field<String>,
889}
890
891impl AsUtf8FileContent for RouterDownloadScript {
892    async fn try_get_file_content(
893        &self,
894        _src: &Source,
895        ctx: &impl ResolutionContext,
896    ) -> providers::Result<String> {
897        let version = self.version.as_resolved();
898        let url = format!("https://router.apollo.dev/download/nix/{version}");
899        let client = ctx.http_client().expect("to have an http client");
900        let response = client.get(&url).await?;
901        if response.status == StatusCode::NOT_FOUND {
902            return Err(providers::Error::UnknownRouterVersion(version.to_string()));
903        }
904        let script = std::str::from_utf8(&response.body)
905            .map_err(|_| providers::Error::Utf8DecodingError)?
906            .to_string();
907
908        Ok(script)
909    }
910}
911
912impl Check for RouterDownloadScript {
913    fn try_check(
914        &self,
915        path: &mut Vec<String>,
916        _src: &Source,
917        ctx: &impl ResolutionContext,
918    ) -> checks::Result<()> {
919        if ctx.http_client().is_none() {
920            return Err(checks::Errors::new(
921                checks::ErrorKind::HttpClientNotFound,
922                "",
923                path,
924            ));
925        }
926
927        Ok(())
928    }
929}

931/// # Build Router From Source
932///
933/// A file provider used for building the Router from source at a specific git commit
934/// or reference.
935///
936/// ```yaml
937/// - name: "router-build.sh"
938///   env_var: ROUTER_BUILD_SCRIPT
939///   kind: build_router_from_source
940///   git_ref: "some-ref"
941///   rust_version: "1.89.0"
942/// ```
943#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
944pub struct BuildRouterFromSource {
945    /// A git reference that can be passed to `git checkout`. This may be
946    /// a full or partial commit hash, branch name, or tag.
947    pub(crate) git_ref: Field<String>,
948
949    /// A Rust version string that can be passed to `rustup run {rust_version}`,
950    /// such as `"1.78.0"`, `"beta"`, or `"nightly"`.
951    ///
952    /// Defaults to `"stable"` if unset.
953    #[serde(default = "default_rust_version")]
954    pub(crate) rust_version: Field<String>,
955}
956
957fn default_rust_version() -> Field<String> {
958    Field::Resolved("stable".to_string())
959}
960
961impl AsUtf8FileContent for BuildRouterFromSource {
962    async fn try_get_file_content(
963        &self,
964        _src: &Source,
965        _ctx: &impl ResolutionContext,
966    ) -> providers::Result<String> {
967        let commit_ref = self.git_ref.as_resolved();
968        let rust_version = self.rust_version.as_resolved();
969
970        let install_script = format!(
971            indoc!(
972                r#"mkdir router-source && \
973                cd router-source && \
974                git clone https://github.com/apollographql/router.git && \
975                cd router && \
976                git checkout {} && \
977                rustup toolchain install {} && \
978                rustup run {} cargo build --release && \
979                cp ${{CARGO_TARGET_DIR}}/release/router ~/.cargo/bin/"#
980            ),
981            commit_ref, rust_version, rust_version
982        );
983
984        Ok(install_script)
985    }
986}

990        &self,
991        _path: &mut Vec<String>,
992        _src: &Source,
993        _ctx: &impl ResolutionContext,
994    ) -> checks::Result<()> {
995        Ok(())
996    }
