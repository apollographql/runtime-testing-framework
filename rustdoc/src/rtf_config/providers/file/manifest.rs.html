28/// # Named Manifest File Provider
29///
30/// Shared metadata that wraps every manifest file provider. This is specifically for the
31/// ManifestEnvironment and is implemented to restrict how users can specify manifest files in that
32/// config.
33#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
34pub struct NamedManifestFileProvider {
35    /// The name to use for the output produced by this provider
36    ///
37    /// For single-file providers, this is the output filename.
38    /// For directory providers, this is the directory name.
39    pub name: String,
40    #[serde(flatten)]
41    pub provider: ManifestFileProvider,
42}

58impl Template for NamedManifestFileProvider {
59    fn required_variables(&self) -> Vec<String> {
60        self.provider.required_variables()
61    }
62
63    fn validate_context(
64        &self,
65        path: &mut Vec<String>,
66        allowed_variables: &HashSet<&String>,
67        file_source: &StableSource,
68        ctx: &TemplateContext,
69    ) -> templating::Result<()> {
70        let tail = self.name.clone();
71        self.provider
72            .validate_context_nested(path, &tail, allowed_variables, file_source, ctx)
73    }
74
75    fn try_template(
76        &mut self,
77        path: &mut Vec<String>,
78        file_source: &StableSource,
79        ctx: &TemplateContext,
80    ) -> templating::Result<()> {
81        // Normalize the name: replace '.' and '/' with '_'
82        let tail = self.name.replace(['.', '/'], "_");
83
84        self.provider
85            .try_template_nested(path, &tail, file_source, ctx)
86    }
87}

89impl Check for NamedManifestFileProvider {
90    fn try_check(
91        &self,
92        path: &mut Vec<String>,
93        ctx: &impl ResolutionContext,
94    ) -> checks::Result<()> {
95        // Normalize the name: replace '.' and '/' with '_'
96        let tail = self.name.clone().replace(['.', '/'], "_");
97
98        let mut errs = checks::ErrorBuilder::new();
99        let mut err_path = path.clone();
100        err_path.push(tail.clone());
101
102        let name_relative_path = match PathBuf::from_str(&self.name) {
103            Ok(p) => p,
104            Err(_) => {
105                errs.push(
106                    checks::ErrorKind::InvalidRelativePath,
107                    "file provider name is not a valid path",
108                    &err_path,
109                );
110                PathBuf::new()
111            }
112        };
113        errs.append(check_relative_path_specifiers(
114            &name_relative_path,
115            &mut err_path,
116        ));
117
118        errs.append(self.provider.try_check_nested(path, tail, ctx));
119
120        errs.into_result(())
121    }
122}

124/// Compose file provider
125#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
126#[serde(tag = "kind", rename_all = "snake_case")]
127pub enum ManifestFileProvider {
128    GithubFile(GithubFile),
129    Inline(InlineFile),
130    InlineDir(InlineDir),
131    RelativeDir(RelativeDir),
132    RelativePath(RelativeFile),
133    Required(RequiredFile),
134    Templated(TemplatedFile),
135}
136
137impl ManifestFileProvider {
138    pub fn github_permalink(&self, ctx: &impl ResolutionContext) -> Option<String> {
139        match self {
140            Self::GithubFile(gh) => Some(gh.permalink()),
141
142            Self::RelativePath(rf) => rf.src.as_ref().and_then(|src| {
143                ctx.source_dir_for(src)
144                    .github_permalink_for(rf.path.as_resolved(), true)
145            }),
146
147            Self::RelativeDir(rd) => rd.src.as_ref().and_then(|src| {
148                ctx.source_dir_for(src)
149                    .github_permalink_for(rd.path.as_resolved(), false)
150            }),
151
152            Self::Inline(_) | Self::InlineDir(_) | Self::Required(_) | Self::Templated(_) => None,
153        }
154    }
155}

157impl Inline for ManifestFileProvider {
158    fn try_inline<'a>(
159        &'a mut self,
160        mode: InlineMode,
161        ctx: &'a impl ResolutionContext,
162        cache: &'a mut HashMap<u64, InlinedProvider>,
163    ) -> Pin<Box<dyn Future<Output = inlining::Result<()>> + Send + 'a>> {
164        Box::pin(async move {
165            if matches!(self, Self::Inline(_) | Self::InlineDir(_)) {
166                return Ok(());
167            }
168
169            let key = provider_cache_key(&*self);
170            if let Some(cached) = cache.get(&key) {
171                match cached.clone() {
172                    InlinedProvider::File(f) => *self = Self::Inline(f),
173                    InlinedProvider::Dir(d) => *self = Self::InlineDir(d),
174                }
175
176                return Ok(());
177            }
178
179            match (&mut *self, mode) {
180                (ManifestFileProvider::RelativeDir(inner), _) => {
181                    *self =
182                        ManifestFileProvider::InlineDir(inner.try_into_inline_files(ctx).await?);
183                }
184                (ManifestFileProvider::RelativePath(inner), _) => {
185                    *self = ManifestFileProvider::Inline(inner.try_into_inline_file(ctx).await?);
186                }
187                (_, InlineMode::RelativeFiles) => return Ok(()),
188                (ManifestFileProvider::GithubFile(inner), InlineMode::All) => {
189                    *self = ManifestFileProvider::Inline(inner.try_into_inline_file(ctx).await?);
190                }
191                (
192                    ManifestFileProvider::Inline(_)
193                    | ManifestFileProvider::InlineDir(_)
194                    | ManifestFileProvider::Templated(_),
195                    InlineMode::All,
196                ) => return Ok(()),
197                (ManifestFileProvider::Required(inner), InlineMode::All) => {
198                    *self = ManifestFileProvider::Inline(inner.try_into_inline_file(ctx).await?);
199                }
200            }
201
202            match self {
203                Self::Inline(f) => {
204                    cache.insert(key, InlinedProvider::File(f.clone()));
205                }
206                Self::InlineDir(d) => {
207                    cache.insert(key, InlinedProvider::Dir(d.clone()));
208                }
209                _ => {}
210            }
211
212            Ok(())
213        })
214    }
215}

234impl ExtractRelativeFiles for ManifestFileProvider {
235    async fn try_extract_relative_files(
236        &self,
237        files: &mut HashMap<(StableSource, String), String>,
238        ctx: &impl ResolutionContext,
239    ) -> providers::Result<()> {
240        match self {
241            Self::RelativePath(rf) => try_read_relative_file(rf, files, ctx).await,
242            Self::RelativeDir(rd) => try_read_relative_dir(rd, files, ctx).await,
243
244            Self::GithubFile(_)
245            | Self::Inline(_)
246            | Self::InlineDir(_)
247            | Self::Required(_)
248            | Self::Templated(_) => Ok(()),
249        }
250    }
251}
