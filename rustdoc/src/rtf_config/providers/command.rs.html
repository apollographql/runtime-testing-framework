30/// # Command Section
31///
32/// Defines an executable command along with environment variables that should be set prior to
33/// execution and file providers that should be made available.
34#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema, Template)]
35pub struct CommandSection {
36    /// The command to be run
37    pub command: CommandSpec,
38    /// Environment variables to set
39    #[serde(default)]
40    pub env_vars: HashMap<String, Field<Scalar>>,
41    /// File providers to run and make available prior to execution
42    #[serde(default)]
43    pub file_providers: Vec<NamedFileProvider>,
44}

46impl CommandSection {
47    /// Run all of the [FileProviders][0] associated with this command and write out their file
48    /// contents to the specified directory before executing the command with the specified
49    /// environment, returning the the output path passed to the command. If no `output_path` is
50    /// specified it will be defaulted to `out_dir`/[OUTPUT_PATH].
51    ///
52    /// [0]: crate::providers::file::FileProvider
53    pub async fn run_providers_and_execute(
54        &self,
55        out_dir: &Path,
56        output_path: Option<PathBuf>,
57        src: &Source,
58        ctx: &mut impl ResolutionContext,
59    ) -> providers::Result<PathBuf> {
60        let output_path = output_path.unwrap_or_else(|| out_dir.join(OUTPUT_PATH));
61        self.run_providers(out_dir, src, ctx).await?;
62        self.execute(out_dir, &output_path, ctx)?;
63
64        Ok(output_path)
65    }
66
67    /// Run all of the [FileProviders][0] associated with this command and write out their file
68    /// contents to the specified directory before executing the command with the specified
69    /// environment, returning the the content written to [OUTPUT_PATH] and removing the output
70    /// file if it was created.
71    ///
72    /// This method assumes that the command being executed is writing a single file out at the
73    /// provided output path rather than a directory of files. To run a command and leave the
74    /// resources available on disk after execution, use [run_providers_and_execute][1] instead.
75    ///
76    /// [0]: crate::providers::file::FileProvider
77    /// [1]: CommandSection::run_providers_and_execute
78    pub async fn run_providers_and_execute_for_output(
79        &self,
80        out_dir: &Path,
81        src: &Source,
82        ctx: &mut impl ResolutionContext,
83    ) -> providers::Result<String> {
84        let output_path = self
85            .run_providers_and_execute(out_dir, None, src, ctx)
86            .await?;
87
88        try_read_output_and_remove(&output_path, ctx)
89    }
90
91    /// Execute this command with the specified environment, returning the output path used.
92    ///
93    /// [CommandSection::run_providers] must have been run successfully before calling this method
94    /// in order to ensure that all file providers have written out their file content to the
95    /// expected location.
96    fn execute(
97        &self,
98        out_dir: &Path,
99        output_path: &Path,
100        ctx: &impl ResolutionContext,
101    ) -> providers::Result<()> {
102        let env_vars = self.all_env_vars(out_dir, output_path, ctx)?;
103
104        let file_path = ctx
105            .known_provider_output_path(Provider::Command {
106                name: &self.command.name,
107                cmd: &self.command.command_provider,
108            })
109            .ok_or(providers::Error::MissingProviderOutput {
110                name: self.command.name.clone(),
111            })?;
112
113        let prog = match file_path.to_str() {
114            Some(v) => v,
115            None => {
116                return Err(io::Error::new(
117                    io::ErrorKind::InvalidData,
118                    "unable to convert command path to valid string",
119                )
120                .into());
121            }
122        };
123        let it = self
124            .command
125            .args
126            .iter()
127            .map(|arg| arg.as_resolved().as_str());
128        ctx.run_command_blocking(prog, it, &env_vars)?;
129
130        Ok(())
131    }

133    /// Combine the base environment variables we have with the ones coming from the file providers
134    /// we need to run. The `out_dir` argument here needs to match the one used when running
135    /// and outputting the content of the file providers.
136    pub fn all_env_vars(
137        &self,
138        out_dir: &Path,
139        output_path: &Path,
140        ctx: &impl ResolutionContext,
141    ) -> providers::Result<HashMap<String, String>> {
142        let mut vars: HashMap<String, String> = self
143            .env_vars
144            .iter()
145            .map(|(k, v)| (k.to_string(), v.as_resolved().to_string()))
146            .collect();
147
148        for nfp in self.file_providers.iter() {
149            let path = ctx
150                .known_provider_output_path(Provider::File { fp: &nfp.provider })
151                .ok_or(providers::Error::MissingProviderOutput {
152                    name: nfp.name.clone(),
153                })?;
154            vars.insert(nfp.env_var.clone(), path.to_string_lossy().to_string());
155        }
156
157        vars.insert(OUTDIR.to_string(), out_dir.display().to_string());
158        vars.insert(OUTPUT_PATH.to_string(), output_path.display().to_string());
159
160        Ok(vars)
161    }

163    /// Run all of the [FileProviders][0] associated with this command and write out their file
164    /// contents to the specified directory.
165    ///
166    /// [0]: crate::providers::file::FileProvider
167    pub async fn run_providers(
168        &self,
169        out_dir: &Path,
170        src: &Source,
171        ctx: &mut impl ResolutionContext,
172    ) -> providers::Result<()> {
173        let provider_dir = out_dir.join(PROVIDER_DIR);
174
175        if ctx
176            .known_provider_output_path(Provider::Command {
177                name: &self.command.name,
178                cmd: &self.command.command_provider,
179            })
180            .is_none()
181        {
182            trace!(name=%self.command.name, "running command provider");
183            let file_path = provider_dir.join(&self.command.name);
184            self.command
185                .command_provider
186                .resolve_and_write(&file_path, src, ctx)
187                .await?;
188            ctx.make_executable(&file_path)?;
189            ctx.store_provider_output_path(
190                Provider::Command {
191                    name: &self.command.name,
192                    cmd: &self.command.command_provider,
193                },
194                file_path,
195            );
196        }
197
198        for nfp in self.file_providers.iter() {
199            if ctx
200                .known_provider_output_path(Provider::File { fp: &nfp.provider })
201                .is_some()
202            {
203                continue;
204            }
205
206            trace!(name=%nfp.name, "running command provider");
207            let file_path = provider_dir.join(&nfp.name);
208
209            // We need to box the future here in order to prevent us ending up with a recursive
210            // type definition for the Future we are building with this method. We end up being
211            // recursively defined because of the FromCommand file provider which is just a wrapper
212            // around this struct, meaning that the call to resolve_and_write below ends up calling
213            // back into run_providers_and_execute which then calls this method (run_providers).
214            Box::pin(nfp.resolve_and_write(&file_path, src, ctx)).await?;
215
216            ctx.store_provider_output_path(Provider::File { fp: &nfp.provider }, file_path);
217        }
218
219        Ok(())
220    }

239impl Check for CommandSection {
240    fn try_check(
241        &self,
242        path: &mut Vec<String>,
243        src: &Source,
244        ctx: &impl ResolutionContext,
245    ) -> checks::Result<()> {
246        let mut errs = checks::ErrorBuilder::new();
247
248        errs.append(self.command.try_check_nested(path, "command", src, ctx));
249
250        for nfp in self.file_providers.iter() {
251            let tail = nfp.name.clone();
252            errs.append(nfp.provider.try_check_nested(path, tail, src, ctx));
253        }
254
255        // We are checking whether the env vars in the command are duplicates of any env vars
256        // defined in the file providers. Each individual list has been checks for duplicates
257        // by this point.
258        let env_var_names = self
259            .env_vars
260            .keys()
261            .map(|k| k.as_str())
262            .chain(self.file_providers.iter().map(|f| f.env_var.as_str()));
263
264        let duplicates = duplicate_keys(env_var_names, |name| name);
265
266        if !duplicates.is_empty() {
267            errs.push(
268                checks::ErrorKind::DuplicateEnvironmentVariables,
269                duplicates.join("\n"),
270                path,
271            );
272        }
273
274        errs.into_result(())
275    }

278/// It is possible that no output is set in the environment setup script. If the output is
279/// blank then default to an empty json object. If there is an error reading the user defined
280/// output to the file then we still pass that error to the user. This is a quality of life
281/// improvement so if the user does not define any output the rtf execution will continue.
282fn try_read_output_and_remove(
283    output_path: &Path,
284    ctx: &impl ResolutionContext,
285) -> providers::Result<String> {
286    let output = match ctx.read_path_to_string(output_path) {
287        Ok(s) => {
288            ctx.remove_file(output_path)?;
289            s
290        }
291        Err(e) if e.kind() == io::ErrorKind::NotFound => "{}".to_string(),
292        Err(e) => return Err(e.into()),
293    };
294
295    Ok(output)
296}

644    /// files have been written to disk.
645    #[derive(Debug, Default)]
646    struct MockCommandContext {
647        written_files: Mutex<HashMap<String, String>>,
648        writes: Mutex<HashMap<String, usize>>,
649        fp_output_paths: HashMap<String, PathBuf>,
650    }
651
652    impl ResolutionContext for MockCommandContext {
653        type PlatformClient = NullClient;
654        type GithubClient = NullClient;
655        type HttpClient = NullClient;
656
657        fn http_client(&self) -> &Self::HttpClient {
658            &NullClient
659        }
660
661        fn store_provider_output_path(&mut self, provider: Provider<'_>, path: PathBuf) {
662            let key = serde_yaml::to_string(&provider).unwrap();
663
664            self.fp_output_paths.insert(key, path);
665        }
666
667        fn known_provider_output_path(&self, provider: Provider<'_>) -> Option<PathBuf> {
668            let key = serde_yaml::to_string(&provider).ok()?;
669
670            self.fp_output_paths.get(&key).cloned()
671        }
672
673        fn run_command_blocking<'a>(
674            &self,
675            _prog: &str,
676            args: impl IntoIterator<Item = &'a str>,
677            env_vars: &HashMap<String, String>,
678        ) -> io::Result<()> {
679            let path = env_vars
680                .get(OUTPUT_PATH)
681                .expect("RTF_OUTPUT env var not set");
682            let mut iter = args.into_iter();
683            if let Some(content) = iter.next() {
684                self.written_files
685                    .lock()
686                    .unwrap()
687                    .insert(path.to_string(), content.to_string());
688            }
689
690            Ok(())
691        }
692
693        fn write(&self, path: impl AsRef<Path>, content: impl AsRef<[u8]>) -> io::Result<()> {
694            let s = String::from_utf8(content.as_ref().to_vec()).expect("valid utf8");
695            self.written_files
696                .lock()
697                .unwrap()
698                .insert(path.as_ref().display().to_string(), s);
699
700            // Record each time we write to support checking that we correctly use the provider
701            // cache
702            *self
703                .writes
704                .lock()
705                .unwrap()
706                .entry(path.as_ref().display().to_string())
707                .or_default() += 1;
708
709            Ok(())
710        }
711
712        fn path_kind(&self, _path: impl AsRef<Path>) -> crate::context::PathKind {
713            PathKind::File
714        }

850    #[tokio::test]
851    async fn the_provider_cache_is_used_to_avoid_rerunning_providers() {
852        let c = test_cmd_section();
853        let mut ctx = MockCommandContext::default();
854        let dir = PathBuf::from("/example-dir");
855        let src = Source::Local {
856            abs_path: dir.join("example.yaml"),
857        };
858
859        let writes = ctx.writes.lock().unwrap().clone();
860        assert!(
861            writes.is_empty(),
862            "should start without any recorded writes"
863        );
864
865        // Running the providers should result in each provider writing output once
866        let expected: HashMap<String, usize> = [
867            ("/example-dir/providers/example.sh".to_string(), 1),
868            ("/example-dir/providers/fp1.txt".to_string(), 1),
869            ("/example-dir/providers/fp2.txt".to_string(), 1),
870        ]
871        .into_iter()
872        .collect();
873
874        let res = c.run_providers(&dir, &src, &mut ctx).await;
875        assert!(res.is_ok(), "unexpected error: {res:?}");
876
877        let writes = ctx.writes.lock().unwrap().clone();
878        assert_eq!(writes, expected);
879
880        // Running the providers a second time should still succeed and should not result in any
881        // further calls to ctx.write
882        let res = c.run_providers(&dir, &src, &mut ctx).await;
883        assert!(res.is_ok(), "unexpected error: {res:?}");
884
885        let writes = ctx.writes.into_inner().unwrap();
886        assert_eq!(writes, expected);
887    }
