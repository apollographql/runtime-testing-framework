19/// # Custom Provider Definition
20///
21/// The definition for a custom provider to be executed as part of a test plan.
22#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
23pub struct CustomProviderDefinition {
24    /// The name of this custom provider
25    pub name: String,
26    /// A brief description of the purpose / behaviour of this custom provider
27    pub description: String,
28    /// The variables that are required for this custom provider
29    #[serde(default)]
30    pub variable_definitions: Vec<VariableDefinition>,
31    /// The command to execute as this custom provider
32    #[serde(flatten)]
33    pub command: CommandSection,
34}
35
36impl CustomProviderDefinition {
37    /// Validate variable definitions and provided values against allowed_values constraints.
38    pub fn validate_variables(
39        &self,
40        variables: &HashMap<String, Scalar>,
41        cli_overrides: Option<&HashMap<String, StableSource>>,
42    ) -> templating::Result<()> {
43        let mut errs = templating::ErrorBuilder::new();
44
45        for vd in &self.variable_definitions {
46            vd.validate(
47                &["variable_definitions".to_string(), vd.name.to_string()],
48                &mut errs,
49            );
50
51            if let Some(value) = variables.get(&vd.name) {
52                let source_desc = match cli_overrides {
53                    None => "test plan variable",
54                    Some(overrides) => match overrides.get(&vd.name) {
55                        Some(StableSource::Cli | StableSource::VariablesFile) => "CLI variable",
56                        _ => "test plan variable",
57                    },
58                };
59                vd.validate_value(
60                    value,
61                    source_desc,
62                    &["variables".to_string(), vd.name.to_string()],
63                    &mut errs,
64                );
65            }
66        }
67
68        errs.into_result(())
69    }
70
71    /// Create an empty [CustomProviderDefinition] for tests
72    #[cfg(test)]
73    pub(crate) fn empty() -> CustomProviderDefinition {
74        CustomProviderDefinition {
75            name: Default::default(),
76            description: Default::default(),
77            variable_definitions: Default::default(),
78            command: CommandSection::empty(),
79        }
80    }
81}

83// The Template implementation for CustomProviderDefinition is used to check that the custom provider definition is self-consistent
84// i.e. the template variables it uses are defined in the variable_definitions field. It does not check that the user has provided
85// all the required variables for the custom provider in the NamedFileProviders for the config file it is being used in.
86// Check is intentionally not implemented for CustomProviderDefinition as there is no way to guarantee it will be in its fully templated
87// form when a check is performed. It is only possible to check a custom provider definition after it has been transformed into a
88// FromCommand Provider.
89impl Template for CustomProviderDefinition {
90    fn required_variables(&self) -> Vec<String> {
91        self.command.required_variables()
92    }
93
94    fn validate_context(
95        &self,
96        path: &mut Vec<String>,
97        allowed_variables: &HashSet<&String>,
98        file_source: &StableSource,
99        ctx: &TemplateContext,
100    ) -> templating::Result<()> {
101        let file_ctx = ctx.for_config_file(file_source, None, self.variable_definitions.iter());
102
103        self.command
104            .validate_context(path, allowed_variables, file_source, &file_ctx)
105    }
106
107    fn try_template(
108        &mut self,
109        path: &mut Vec<String>,
110        source: &StableSource,
111        ctx: &TemplateContext,
112    ) -> templating::Result<()> {
113        let file_ctx = ctx.for_config_file(source, None, self.variable_definitions.iter());
114
115        self.command
116            .try_template_nested(path, "command_section", source, &file_ctx)
117    }
118}

131/// # Custom Provider Declaration
132///
133/// Allows for declaring one or more custom providers that should be loaded from the given
134/// directory. The directory can be specified as a relative path from the file containing
135/// this declaration or as coming from a GitHub repository.
136///
137/// ## Example using a relative directory
138///
139/// ```yaml
140/// kind: local
141/// relative_path: ../providers
142/// using:
143///   my_custom_provider: my_custom_provider.yaml
144///   my_other_provider: my_other_provider.yaml
145/// ```
146///
147/// ## Example using a directory from a GitHub repository
148///
149/// ```yaml
150/// kind: github
151/// org: my-org
152/// repo: my-repo
153/// path: path/to/providers
154/// git_ref: main
155/// using:
156///   my_custom_provider: my_custom_provider.yaml
157///   my_other_provider: my_other_provider.yaml
158/// ```
159#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
160pub struct CustomProviderDeclaration {
161    /// The source directory containing the custom provider definition files
162    #[serde(flatten)]
163    pub source: RawSource,
164
165    /// A map of the name to use for a given custom provider to the child path within this
166    /// directory containing the definition for that provider.
167    pub using: HashMap<String, String>,
168}
169
170impl CustomProviderDeclaration {
171    pub async fn try_load_all(
172        &self,
173        file_source: &SourceDir,
174        ctx: &impl ResolutionContext,
175    ) -> Result<
176        HashMap<String, (SourceDir, CustomProviderDefinition)>,
177        Vec<(String, providers::Error)>,
178    > {
179        let mut providers = HashMap::with_capacity(self.using.len());
180        let mut errs = Vec::new();
181
182        for (provider_name, filename) in self.using.iter() {
183            match load_one(self.source.with_child_path(filename), file_source, ctx).await {
184                Ok(entry) => {
185                    providers.insert(provider_name.clone(), entry);
186                }
187                Err(e) => errs.push((provider_name.clone(), e)),
188            }
189        }
190
191        if errs.is_empty() {
192            Ok(providers)
193        } else {
194            errs.sort_by(|l, r| l.0.cmp(&r.0));
195            Err(errs)
196        }
197    }
198}
199
200async fn load_one(
201    source: RawSource,
202    file_source: &SourceDir,
203    ctx: &impl ResolutionContext,
204) -> providers::Result<(SourceDir, CustomProviderDefinition)> {
205    let (definition_source, file_name) = source.try_into_source_and_filename(file_source, ctx)?;
206    let content = definition_source
207        .try_get_file_content(file_name, ctx)
208        .await?;
209    let raw: serde_yaml::Value = serde_yaml::from_str(&content)?;
210
211    if let Some(mapping) = raw.as_mapping()
212        && mapping.contains_key("custom_providers")
213    {
214        return Err(providers::Error::NestedCustomProvider);
215    }
216
217    Ok((definition_source, serde_yaml::from_value(raw)?))
218}
