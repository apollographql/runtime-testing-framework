27/// # Environment Config
28///
29/// Configuration for preparing and cleaning up the test environment as part of a test plan.
30#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
31pub struct EnvironmentConfig<R: RunEnvironment> {
32    /// The name of this environment configuration
33    pub name: String,
34    /// A brief description of how this environment setup works
35    pub description: String,
36    /// Definitions for the required variables for templating this environment
37    #[serde(default, alias = "values")]
38    // This alias is for backwards compatibility with the original name
39    pub variable_definitions: Vec<VariableDefinition>,
40    /// Custom provider declarations to load for this environment
41    #[serde(default)]
42    pub custom_providers: Vec<CustomProviderDeclaration>,
43    #[serde(flatten)]
44    pub execution: R,
45}

52    }

101impl<R: RunEnvironment> Template for EnvironmentConfig<R> {
102    fn required_variables(&self) -> Vec<String> {
103        self.execution.required_variables()
104    }
105
106    fn validate_context(
107        &self,
108        path: &mut Vec<String>,
109        allowed_variables: &HashSet<&String>,
110        file_source: &StableSource,
111        ctx: &TemplateContext,
112    ) -> templating::Result<()> {
113        let mut allowed_variables = allowed_variables.clone();
114        allowed_variables.extend(self.variable_definitions.iter().map(|vd| &vd.name));
115
116        let ctx = ctx.for_config_file(
117            file_source,
118            Some(FileType::Environment),
119            self.variable_definitions.iter(),
120        );
121
122        self.execution
123            .validate_context(path, &allowed_variables, file_source, &ctx)
124    }
125
126    fn try_template(
127        &mut self,
128        path: &mut Vec<String>,
129        source: &StableSource,
130        ctx: &TemplateContext,
131    ) -> templating::Result<()> {
132        let ctx = ctx.for_config_file(
133            source,
134            Some(FileType::Environment),
135            self.variable_definitions.iter(),
136        );
137
138        // We call try_template here instead of try_template_nested to avoid appending
139        // an unnecessary entry to the path
140        self.execution.try_template(path, source, &ctx)
141    }
142}

144impl<R: RunEnvironment> Check for EnvironmentConfig<R> {
145    fn try_check(
146        &self,
147        path: &mut Vec<String>,
148        ctx: &impl ResolutionContext,
149    ) -> checks::Result<()> {
150        let mut errs = checks::ErrorBuilder::new();
151
152        // Check that hard coded variables are unique
153        let all_variables = self.variable_definitions.iter();
154        let duplicates = duplicate_keys(all_variables, |v| &v.name);
155        if !duplicates.is_empty() {
156            errs.push(
157                checks::ErrorKind::DuplicateVariableNames,
158                duplicates.join("\n"),
159                path,
160            );
161        }
162
163        // Check that each command is valid in isolation
164        // We call try_check here instead of try_check_nested to avoid appending
165        // an unnecessary entry to the path
166        errs.append(self.execution.try_check(path, ctx));
167
168        errs.into_result(())
169    }
170}

291impl Check for ScriptEnvironment {

293        &self,
294        path: &mut Vec<String>,

326            errs.into_result(())

440    /// Create temp files and store their paths in context, returning the paths for use in assertions.
441    /// The returned TempDir must be kept alive for the duration of the test to prevent cleanup.
442    pub(crate) fn register_compose_paths(
443        ctx: &mut Context,
444        env: &DockerComposeEnvironment,
445    ) -> (TempDir, Vec<String>) {
446        let temp_dir = TempDir::new().unwrap();
447        let paths = env
448            .compose_files
449            .iter()
450            .map(|ncfp| {
451                let file = temp_dir.child(&ncfp.name);
452                file.write_str(&format!("# {}\nservices: {{}}", ncfp.name))
453                    .unwrap();
454                let path = file.path().to_string_lossy().to_string();
455                ctx.store_provider_output_path(
456                    Provider::ComposeFile { fp: &ncfp.provider },
457                    PathBuf::from(&path),
458                );
459                path
460            })
461            .collect();
462        (temp_dir, paths)
463    }

1023                    command: CommandSpec {

1026                        args: Vec::new(),
1027                    },
1028                    ..CommandSection::empty()
1029                },
1030                teardown: CommandSection {
1031                    command: CommandSpec {
1032                        name: "teardown.sh".to_string(),
1033                        command_provider: relative_command_provider,
1034                        args: Vec::new(),
1035                    },
1036                    ..CommandSection::empty()
1037                },
1038            }),
1039            ..EnvironmentConfig::empty()
1040        };
1041
1042        let result = environment
1043            .inline(&InlineMode::All, &ctx, &mut HashMap::new())
1044            .await;
