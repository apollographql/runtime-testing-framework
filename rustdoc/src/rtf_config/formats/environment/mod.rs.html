31/// # Environment Config
32///
33/// Configuration for preparing and cleaning up the test environment as part of a test plan.
34#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
35pub struct EnvironmentConfig<R: RunEnvironment> {
36    /// The name of this environment configuration
37    pub name: String,
38    /// A brief description of how this environment setup works
39    pub description: String,
40    /// Definitions for the required variables for templating this environment
41    #[serde(default, alias = "values")]
42    // This alias is for backwards compatibility with the original name
43    pub variable_definitions: Vec<VariableDefinition>,
44    /// Custom provider declarations to load for this environment
45    #[serde(default)]
46    pub custom_providers: Vec<CustomProviderDeclaration>,
47    #[serde(flatten)]
48    pub execution: R,
49}
50
51impl EnvironmentConfig<EnvironmentExecution> {
52    pub fn try_load_from_path(p: impl AsRef<Path>) -> Result<Self> {
53        let content = fs::read_to_string(p)?;
54
55        Ok(serde_yaml::from_str(&content)?)
56    }
57
58    pub fn output_collection(&self) -> Option<&OutputCollection> {
59        match &self.execution {
60            EnvironmentExecution::DockerCompose(ex) => Some(&ex.output_collection),
61            EnvironmentExecution::Script(_) => None,
62            EnvironmentExecution::Null(_) => None,
63        }
64    }
65
66    /// Returns `true` if this environment actually has output configured to collect
67    pub fn output_collection_defined(&self) -> bool {
68        self.output_collection()
69            .is_some_and(OutputCollection::is_defined)
70    }
71
72    /// Create an empty [EnvironmentConfig] for tests
73    #[cfg(test)]
74    pub(crate) fn empty() -> EnvironmentConfig<EnvironmentExecution> {
75        use crate::providers::command::CommandSection;
76
77        EnvironmentConfig {
78            name: Default::default(),
79            description: Default::default(),
80            variable_definitions: Vec::new(),
81            custom_providers: Default::default(),
82            execution: EnvironmentExecution::Script(ScriptEnvironment {
83                setup: CommandSection::empty(),
84                teardown: CommandSection::empty(),
85            }),
86        }
87    }
88}

121impl<R: RunEnvironment> Template for EnvironmentConfig<R> {
122    fn required_variables(&self) -> Vec<String> {
123        self.execution.required_variables()
124    }
125
126    fn validate_context(
127        &self,
128        path: &mut Vec<String>,
129        allowed_variables: &HashSet<&String>,
130        file_source: &StableSource,
131        ctx: &TemplateContext,
132    ) -> templating::Result<()> {
133        let mut allowed_variables = allowed_variables.clone();
134        allowed_variables.extend(self.variable_definitions.iter().map(|vd| &vd.name));
135
136        let ctx = ctx.for_config_file(
137            file_source,
138            Some(FileType::Environment),
139            self.variable_definitions.iter(),
140        );
141
142        self.execution
143            .validate_context(path, &allowed_variables, file_source, &ctx)
144    }
145
146    fn try_template(
147        &mut self,
148        path: &mut Vec<String>,
149        source: &StableSource,
150        ctx: &TemplateContext,
151    ) -> templating::Result<()> {
152        let ctx = ctx.for_config_file(
153            source,
154            Some(FileType::Environment),
155            self.variable_definitions.iter(),
156        );
157
158        // We call try_template here instead of try_template_nested to avoid appending
159        // an unnecessary entry to the path
160        self.execution.try_template(path, source, &ctx)
161    }
162}

164impl<R: RunEnvironment> Check for EnvironmentConfig<R> {
165    fn try_check(
166        &self,
167        path: &mut Vec<String>,
168        ctx: &impl ResolutionContext,
169    ) -> checks::Result<()> {
170        let mut errs = checks::ErrorBuilder::new();
171
172        // Check that hard coded variables are unique
173        let all_variables = self.variable_definitions.iter();
174        let duplicates = duplicate_keys(all_variables, |v| &v.name);
175        if !duplicates.is_empty() {
176            errs.push(
177                checks::ErrorKind::DuplicateVariableNames,
178                duplicates.join("\n"),
179                path,
180            );
181        }
182
183        // Check that each command is valid in isolation
184        // We call try_check here instead of try_check_nested to avoid appending
185        // an unnecessary entry to the path
186        errs.append(self.execution.try_check(path, ctx));
187
188        errs.into_result(())
189    }
190}

211#[allow(clippy::large_enum_variant)] // We only ever allocate one of these, not multiples, so the difference in variant size should not be an issue
212pub enum EnvironmentExecution {
213    // Null needs to be the first variant in this enum to ensure that any time `skip: true` is set,
214    // we resolve to a NullEnvironment.
215    Null(NullEnvironment),
216    DockerCompose(DockerComposeEnvironment),
217    Script(ScriptEnvironment),
218}

220enum_impl_check!(EnvironmentExecution => Null, DockerCompose, Script);

222impl RunEnvironment for EnvironmentExecution {
223    async fn execute_setup(
224        &self,
225        name: &str,
226        out_dir: &Path,
227        ctx: &mut impl ResolutionContext,
228    ) -> providers::Result<String> {
229        match self {
230            EnvironmentExecution::Null(inner) => inner.execute_setup(name, out_dir, ctx).await,
231            EnvironmentExecution::DockerCompose(inner) => {
232                inner.execute_setup(name, out_dir, ctx).await
233            }
234            EnvironmentExecution::Script(inner) => inner.execute_setup(name, out_dir, ctx).await,
235        }

359    /// Create a named compose file with inline content unique to the name.
360    /// Using the name in the content ensures each compose file has a unique provider

390    /// Create temp files and store their paths in context, returning the paths for use in assertions.
391    /// The returned TempDir must be kept alive for the duration of the test to prevent cleanup.
392    pub(crate) fn register_compose_paths(
393        ctx: &mut Context,
394        env: &DockerComposeEnvironment,
395    ) -> (TempDir, Vec<String>) {
396        let temp_dir = TempDir::new().unwrap();
397        let paths = env
398            .compose_files
399            .iter()
400            .map(|ncfp| {
401                let file = temp_dir.child(&ncfp.name);
402                file.write_str(&format!("# {}\nservices: {{}}", ncfp.name))
403                    .unwrap();
404                let path = file.path().to_string_lossy().to_string();
405                ctx.store_provider_output_path(
406                    Provider::ComposeFile { fp: &ncfp.provider },
407                    PathBuf::from(&path),
408                );
409                path
410            })
411            .collect();
412        (temp_dir, paths)
413    }
