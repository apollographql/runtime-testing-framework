16/// Errors that can be encountered resolving config files
17#[derive(Debug, thiserror::Error)]
18pub enum Error {
19    #[error("One or more file providers failed to run:\n{}", .errs.join("\n"))]
20    FailedFileProviders { errs: Vec<String> },
21
22    #[error("Environment setup output not valid json: {output:?}")]
23    MalformedSetupOutputFormat { output: String },
24
25    #[error("Missing required output fields from environment setup: {missing:?}")]
26    MissingSetupOutputFields { missing: Vec<String> },
27
28    #[error(
29        "The provided matrix.variant_names template references unknown matrix variables: {variables:?}"
30    )]
31    UnknownMatrixVariantTemplateVariables { variables: Vec<String> },
32
33    #[error("The provided variant_names template produced duplicate names: {duplicates:?}")]
34    NonUniqueMatrixVariantNames { duplicates: Vec<String> },
35
36    #[error("The config file being parsed was invalid:\n{0}")]
37    Validation(#[from] checks::Errors),
38
39    // wrapped errors
40    #[error(transparent)]
41    GitHub(#[from] github::Error),
42
43    #[error(transparent)]
44    Io(#[from] io::Error),
45
46    #[error(transparent)]
47    Json(#[from] serde_json::Error),
48
49    #[error(transparent)]
50    Provider(#[from] providers::Error),
51
52    #[error(transparent)]
53    Yaml(#[from] serde_yaml::Error),
54}
55
56pub type Result<T> = std::result::Result<T, Error>;

72    /// Return a pending field
73    pub(crate) fn p(name: &str) -> Field<String> {
74        Field::Pending(name.to_string())
75    }
76
77    /// Return a resolved field
78    pub(crate) fn r(name: &str) -> Field<String> {
79        Field::Resolved(name.to_string())
80    }
81
82    /// Return a NamedFileProvider with a field
83    pub(crate) fn named_file_provider_with_field(
84        name: &str,
85        f: Field<String>,
86    ) -> NamedFileProvider {
87        NamedFileProvider {
88            name: name.to_string(),
89            env_var: name.to_ascii_uppercase(),
90            provider: FileProvider::RelativePath(RelativeFile { path: f, src: None }),
91        }
92    }
93
94    /// Return named file providers with fields
95    pub(crate) fn named_file_providers_with_fields(
96        fields: &[Field<String>],
97    ) -> Vec<NamedFileProvider> {
98        fields
99            .iter()
100            .enumerate()
101            .map(|(i, f)| named_file_provider_with_field(&format!("file{}", i), f.clone()))
102            .collect()
103    }
104
105    /// Create NamedFileProviders with pending fields from string names
106    pub(crate) fn templatable_file_providers(field_names: &[&str]) -> Vec<NamedFileProvider> {
107        field_names
108            .iter()
109            .map(|name| named_file_provider_with_field(name, p(name)))
110            .collect()
111    }

113    /// Create a HashMap of variables from string names (each name maps to itself as a Scalar::String)
114    pub(crate) fn template_variables(variable_names: &[&str]) -> TemplateVariables {
115        TemplateVariables::new_stubbed(
116            variable_names
117                .iter()
118                .map(|&name| (name.to_string(), Scalar::String(name.to_string())))
119                .collect(),
120        )
121    }

123    /// Create VariableDefinitions from string names with default description
124    pub(crate) fn variable_definitions(variable_names: &[&str]) -> Vec<VariableDefinition> {
125        variable_names
126            .iter()
127            .map(|&name| VariableDefinition {

135    /// Generate expected error details for fields
136    pub(crate) fn expected_error_details(
137        field_names: &[&str],
138        path_prefix: &str,
139    ) -> (Vec<String>, Vec<String>) {
140        let mut expected_messages: Vec<String> =
141            field_names.iter().map(|name| name.to_string()).collect();
142        expected_messages.sort();
143
144        let mut expected_paths: Vec<String> = field_names
145            .iter()
146            .map(|name| {
147                format!(
148                    "{}.file_providers.{}.path",
149                    path_prefix,
150                    name.to_ascii_uppercase()
151                )
152            })
153            .collect();
154        expected_paths.sort();
155
156        (expected_messages, expected_paths)
157    }

159    /// Assert template errors
160    pub(crate) fn assert_template_errors(
161        t: &mut impl Template,
162        variables: TemplateVariables,
163        expected_err_messages: Vec<String>,
164        expected_err_paths: Vec<String>,
165    ) {
166        let res = t.try_template(&mut Vec::new(), &Source::local("/"), &variables);
167        assert!(res.is_err(), "expected templating to fail, got {res:?}");
168
169        let errors = res.unwrap_err();
170        assert!(
171            errors
172                .iter()
173                .all(|e| matches!(e.kind, ErrorKind::UnknownVariable)),
174            "expected all errors to be UnknownVariable, got {:?}",
175            errors
176        );
177
178        let mut messages: Vec<String> = errors.iter().map(|e| e.message.clone()).collect();
179        messages.sort();
180        assert_eq!(
181            messages, expected_err_messages,
182            "we are testing we get all expected error messages"
183        );
184
185        let mut paths: Vec<String> = errors.iter().map(|e| e.path.clone()).collect();
186        paths.sort();
187        assert_eq!(
188            paths, expected_err_paths,
189            "we are testing we get all expected error paths"
190        );
191    }

193    /// Assert check errors
194    pub(crate) fn assert_check_errors(
195        c: impl Check,
196        ctx: &Context,
197        expected_err_kinds: &[checks::ErrorKind],
198    ) {
199        let res = c.try_check(&mut Vec::new(), ctx);
200        assert!(res.is_err(), "expected check to fail, got {res:?}");
201
202        let err = res.unwrap_err();
203        let err_kinds: Vec<checks::ErrorKind> = err.iter().map(|e| e.kind).collect();
204        assert_eq!(
205            err_kinds, expected_err_kinds,
206            "check the error kind is correct"
207        );
