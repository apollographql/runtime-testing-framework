18/// Errors that can be encountered resolving config files
19#[derive(Debug, thiserror::Error)]
20pub enum Error {
21    #[error("One or more file providers failed to run:\n{}", .errs.join("\n"))]
22    FailedFileProviders { errs: Vec<String> },
23
24    #[error("One or more custom provider definitions failed to load:\n{}", .errs.join("\n"))]
25    FailedCustomProviderDefinitions { errs: Vec<String> },
26
27    #[error("Custom provider declarations can not be specified as part of overrides.")]
28    InvalidCustomProviderOverride,
29
30    #[error("The provided variant_names template produced duplicate names: {duplicates:?}")]
31    NonUniqueMatrixVariantNames { duplicates: Vec<String> },
32
33    #[error(
34        "The provided matrix.variant_names template references unknown matrix variables: {variables:?}"
35    )]
36    UnknownMatrixVariantTemplateVariables { variables: Vec<String> },
37
38    #[error("The config file being parsed was invalid:\n{0}")]
39    Validation(#[from] checks::Errors),
40
41    // wrapped errors
42    #[error(transparent)]
43    GitHub(#[from] github::Error),
44
45    #[error(transparent)]
46    Io(#[from] io::Error),
47
48    #[error(transparent)]
49    Json(#[from] serde_json::Error),
50
51    #[error(transparent)]
52    Provider(#[from] providers::Error),
53
54    #[error(transparent)]
55    Yaml(#[from] serde_yaml::Error),
56}
57
58pub type Result<T> = std::result::Result<T, Error>;

74    /// Return a pending field
75    pub(crate) fn p(name: &str) -> Field<String> {
76        Field::Pending(name.to_string())
77    }
78
79    /// Return a resolved field
80    pub(crate) fn r(name: &str) -> Field<String> {
81        Field::Resolved(name.to_string())
82    }
83
84    /// Return a NamedFileProvider with a field
85    pub(crate) fn named_file_provider_with_field(
86        name: &str,
87        f: Field<String>,
88    ) -> NamedFileProvider {
89        NamedFileProvider {
90            name: name.to_string(),
91            env_var: name.to_ascii_uppercase(),
92            provider: FileProvider::RelativePath(RelativeFile { path: f, src: None }),
93        }
94    }
95
96    /// Return named file providers with fields
97    pub(crate) fn named_file_providers_with_fields(
98        fields: &[Field<String>],
99    ) -> Vec<NamedFileProvider> {
100        fields
101            .iter()
102            .enumerate()
103            .map(|(i, f)| named_file_provider_with_field(&format!("file{}", i), f.clone()))
104            .collect()
105    }
106
107    /// Create NamedFileProviders with pending fields from string names
108    pub(crate) fn templatable_file_providers(field_names: &[&str]) -> Vec<NamedFileProvider> {
109        field_names
110            .iter()
111            .map(|name| named_file_provider_with_field(name, p(name)))
112            .collect()
113    }

115    /// Create a HashMap of variables from string names (each name maps to itself as a Scalar::String)

138    /// Generate expected error details for fields
139    pub(crate) fn expected_error_details(
140        field_names: &[&str],
141        path_prefix: &str,
142    ) -> (Vec<String>, Vec<String>) {
143        let mut expected_messages: Vec<String> =
144            field_names.iter().map(|name| name.to_string()).collect();
145        expected_messages.sort();
146
147        let mut expected_paths: Vec<String> = field_names
148            .iter()
149            .map(|name| {
150                format!(
151                    "{}.file_providers.{}.path",
152                    path_prefix,
153                    name.to_ascii_uppercase()
154                )
155            })
156            .collect();
157        expected_paths.sort();
158
159        (expected_messages, expected_paths)
160    }

162    /// Assert template errors
163    pub(crate) fn assert_template_errors(
164        t: &mut impl Template,
165        ctx: TemplateContext,
166        expected_err_messages: Vec<String>,
167        expected_err_paths: Vec<String>,
168    ) {
169        let res = t.try_template(&mut Vec::new(), &SourceDir::local("/"), &ctx);
170        assert!(res.is_err(), "expected templating to fail, got {res:?}");
171
172        let errors = res.unwrap_err();
173        assert!(
174            errors
175                .iter()
176                .all(|e| matches!(e.kind, ErrorKind::UnknownVariable)),
177            "expected all errors to be UnknownVariable, got {:?}",
178            errors
179        );
180
181        let mut messages: Vec<String> = errors.iter().map(|e| e.message.clone()).collect();
182        messages.sort();
183        assert_eq!(
184            messages, expected_err_messages,
185            "we are testing we get all expected error messages"
186        );
187
188        let mut paths: Vec<String> = errors.iter().map(|e| e.path.clone()).collect();
189        paths.sort();
190        assert_eq!(
191            paths, expected_err_paths,
192            "we are testing we get all expected error paths"
193        );
194    }
