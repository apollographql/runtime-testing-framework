14/// # Environment Config
15///
16/// Configuration for preparing and cleaning up the test environment as part of a test plan.
17#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
18pub struct EnvironmentConfig {
19    /// The name of this environment configuration
20    pub name: String,
21    /// A brief description of how this environment setup works
22    pub description: String,
23    /// Definitions for the required variables for templating this environment
24    #[serde(default, alias = "values")]
25    // This alias is for backwards compatibility with the original name
26    pub variable_definitions: Vec<VariableDefinition>,
27    /// Custom provider declarations to load for this environment
28    #[serde(default)]
29    pub custom_providers: Vec<CustomProviderDeclaration>,
30    /// The command to execute to prepare the environment for executing the test scenario
31    pub setup: SetupSection,
32    /// The command to execute to clean up the environment after executing the test scenario
33    pub teardown: CommandSection,
34}

36impl EnvironmentConfig {
37    pub fn try_load_from_path(p: impl AsRef<Path>) -> Result<Self> {
38        let content = fs::read_to_string(p)?;
39
40        Ok(serde_yaml::from_str(&content)?)
41    }
42
43    fn ctx_for_setup(&self, source: &SourceDir, ctx: &TemplateContext) -> TemplateContext {
44        ctx.for_config_file(
45            source,
46            Some(FileType::Environment),
47            self.variable_definitions.iter(),
48        )
49    }
50
51    fn ctx_for_teardown(&self, source: &SourceDir, ctx: &TemplateContext) -> TemplateContext {
52        ctx.for_config_file(
53            source,
54            Some(FileType::Environment),
55            self.variable_definitions
56                .iter()
57                .chain(self.setup.provides.iter()),
58        )
59    }
60
61    /// Try to template the setup [CommandSection].
62    ///
63    /// Setup is only allowed to reference variables that are declared in the variables section of this
64    /// config file.
65    pub fn try_template_setup(
66        &mut self,
67        path: &mut Vec<String>,
68        source: &SourceDir,
69        ctx: &TemplateContext,
70    ) -> templating::Result<()> {
71        self.setup.command.try_template_nested(
72            path,
73            "setup",
74            source,
75            &self.ctx_for_setup(source, ctx),
76        )
77    }
78
79    /// Try to template the teardown [CommandSection].
80    ///
81    /// Teardown is allowed to reference variables that come from the output of setup in addition to
82    /// the variables decalered in the variables section of this config file.
83    pub fn try_template_teardown(
84        &mut self,
85        path: &mut Vec<String>,
86        source: &SourceDir,
87        ctx: &TemplateContext,
88    ) -> templating::Result<()> {
89        self.teardown.try_template_nested(
90            path,
91            "teardown",
92            source,
93            &self.ctx_for_teardown(source, ctx),
94        )
95    }

114impl Template for EnvironmentConfig {
115    fn has_pending_fields(&self) -> bool {
116        self.setup.command.has_pending_fields() || self.teardown.has_pending_fields()
117    }
118
119    fn required_variables(&self) -> Vec<String> {
120        let mut vals = self.setup.command.required_variables();
121        vals.extend(self.teardown.required_variables());
122
123        vals
124    }
125
126    fn validate_context(
127        &self,
128        path: &mut Vec<String>,
129        allowed_variables: &HashSet<&String>,
130        source: &SourceDir,
131        ctx: &TemplateContext,
132    ) -> templating::Result<()> {
133        let mut allowed_variables = allowed_variables.clone();
134        allowed_variables.extend(self.variable_definitions.iter().map(|vd| &vd.name));
135
136        let mut errs = templating::ErrorBuilder::from(self.setup.command.validate_context_nested(
137            path,
138            "setup",
139            &allowed_variables,
140            source,
141            &self.ctx_for_setup(source, ctx),
142        ));
143
144        errs.append(self.teardown.validate_context_nested(
145            path,
146            "teardown",
147            &allowed_variables,
148            source,
149            &self.ctx_for_teardown(source, ctx),
150        ));
151
152        errs.into_result(())
153    }
154
155    fn try_template(
156        &mut self,
157        path: &mut Vec<String>,
158        source: &SourceDir,
159        ctx: &TemplateContext,
160    ) -> templating::Result<()> {
161        let mut errs = templating::ErrorBuilder::from(self.try_template_setup(path, source, ctx));
162        errs.append(self.try_template_teardown(path, source, ctx));
163
164        errs.into_result(())
165    }
166}

168impl Check for EnvironmentConfig {
169    fn try_check(
170        &self,
171        path: &mut Vec<String>,
172        ctx: &impl ResolutionContext,
173    ) -> checks::Result<()> {
174        let mut errs = checks::ErrorBuilder::new();
175
176        // Check that hard coded variables and the ones coming from setup.provides are unique
177        let all_variables = self
178            .variable_definitions
179            .iter()
180            .chain(self.setup.provides.iter());
181        let duplicates = duplicate_keys(all_variables, |v| &v.name);
182        if !duplicates.is_empty() {
183            errs.push(
184                checks::ErrorKind::DuplicateVariableNames,
185                duplicates.join("\n"),
186                path,
187            );
188        }
189
190        // Check that each command is valid in isolation
191        errs.append(self.setup.command.try_check_nested(path, "setup", ctx));
192        errs.append(self.teardown.try_check_nested(path, "teardown", ctx));
193
194        errs.into_result(())
195    }
196}

890        let (temp, _) = create_temp_dir_with_file("config.yaml", "");
891
892        let providers_dir = temp.child("providers");
893        providers_dir.create_dir_all().unwrap();
894
895        let simple_provider = indoc!(
896            r#"
897        name: simple provider
898        description: A simple custom provider for integration testing

900          - name: test_var
901            description: a test variable
902        command:
903          name: provider.sh
904          kind: relative_path
905          path: ./provider.sh
906        "#
907        );
908
909        providers_dir
910            .child("provider1.yaml")
911            .write_str(simple_provider)
912            .unwrap();
913        providers_dir
914            .child("provider2.yaml")
915            .write_str(simple_provider)
916            .unwrap();
917
918        let env_config_yaml = indoc!(
919            r#"
920            name: test environment
921            description: Environment with custom providers
922            variable_definitions: []
923            custom_providers:
924              - kind: local
925                relative_path: providers
926                using:
927                  provider1: provider1.yaml
928                  provider2: provider2.yaml
929            setup:
930              command:
931                name: setup.sh
932                kind: relative_path
933                path: ./setup.sh
934            teardown:
935              command:
936                name: teardown.sh
937                kind: relative_path
938                path: ./teardown.sh
939            "#
940        );
941
942        let env_config: EnvironmentConfig =
943            serde_yaml::from_str(env_config_yaml).expect("environment config to parse");
944
945        assert_eq!(env_config.custom_providers.len(), 1);
946
947        let declaration = &env_config.custom_providers[0];
948        let loaded_providers = declaration
949            .try_load_all(&SourceDir::local(temp.path()), &Context::new())
950            .await
951            .expect("custom providers should load successfully");
952
953        assert_eq!(loaded_providers.len(), 2);
954
955        let (provider1_source, provider1_def) = loaded_providers
956            .get("provider1")
957            .expect("provider1 should exist");
958        assert_eq!(
959            provider1_source,
960            &SourceDir::local(providers_dir.canonicalize().unwrap())
961        );
962        assert_eq!(provider1_def.name, "simple provider");
963        assert_eq!(
964            provider1_def.description,
965            "A simple custom provider for integration testing"
966        );
967        assert_eq!(provider1_def.variable_definitions.len(), 1);
968
969        let (provider2_source, provider2_def) = loaded_providers
970            .get("provider2")
971            .expect("provider2 should exist");
972        assert_eq!(
973            provider2_source,
974            &SourceDir::local(providers_dir.canonicalize().unwrap())
975        );
976        assert_eq!(provider2_def.name, "simple provider");
977        assert_eq!(
978            provider2_def.description,
979            "A simple custom provider for integration testing"
980        );
981        assert_eq!(provider2_def.variable_definitions.len(), 1);
