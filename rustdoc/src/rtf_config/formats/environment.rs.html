14/// # Environment Config
15///
16/// Configuration for preparing and cleaning up the test environment as part of a test plan.
17#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
18pub struct EnvironmentConfig {
19    /// The name of this environment configuration
20    pub name: String,
21    /// A brief description of how this environment setup works
22    pub description: String,
23    /// Definitions for the required variables for templating this environment
24    #[serde(default, alias = "values")]
25    // This alias is for backwards compatibility with the original name
26    pub variable_definitions: Vec<VariableDefinition>,
27    /// The command to execute to prepare the environment for executing the test scenario
28    pub setup: SetupSection,
29    /// The command to execute to clean up the environment after executing the test scenario
30    pub teardown: CommandSection,
31}
32
33impl EnvironmentConfig {
34    pub fn try_load_from_path(p: impl AsRef<Path>) -> Result<Self> {
35        let content = fs::read_to_string(p)?;
36
37        Ok(serde_yaml::from_str(&content)?)
38    }
39
40    /// Try to template the setup [CommandSection].
41    ///
42    /// Setup is only allowed to reference variables that are declared in the variables section of this
43    /// config file.
44    pub fn try_template_setup(
45        &mut self,
46        path: &mut Vec<String>,
47        source: &Source,
48        ctx: &TemplateContext,
49    ) -> templating::Result<()> {
50        let ctx_for_file = ctx.for_config_file(source, self.variable_definitions.iter());
51
52        self.setup
53            .command
54            .try_template_nested(path, "setup", source, &ctx_for_file)
55    }
56
57    /// Try to template the teardown [CommandSection].
58    ///
59    /// Teardown is allowed to reference variables that come from the output of setup in addition to
60    /// the variables decalered in the variables section of this config file.
61    pub fn try_template_teardown(
62        &mut self,
63        path: &mut Vec<String>,
64        source: &Source,
65        ctx: &TemplateContext,
66    ) -> templating::Result<()> {
67        let ctx_for_file = ctx.for_config_file(
68            source,
69            self.variable_definitions
70                .iter()
71                .chain(self.setup.provides.iter()),
72        );
73
74        self.teardown
75            .try_template_nested(path, "teardown", source, &ctx_for_file)
76    }
77
78    /// Create an empty [EnvironmentConfig] for tests
79    #[cfg(test)]
80    pub(crate) fn empty() -> EnvironmentConfig {
81        EnvironmentConfig {
82            name: Default::default(),
83            description: Default::default(),
84            variable_definitions: Vec::new(),
85            setup: SetupSection {
86                command: CommandSection::empty(),
87                provides: Vec::new(),
88            },
89            teardown: CommandSection::empty(),
90        }
91    }
92}
93
94impl Template for EnvironmentConfig {
95    fn has_pending_fields(&self) -> bool {
96        self.setup.command.has_pending_fields() || self.teardown.has_pending_fields()
97    }
98
99    fn required_variables(&self) -> Vec<String> {
100        let mut vals = self.setup.command.required_variables();
101        vals.extend(self.teardown.required_variables());
102
103        vals
104    }
105
106    fn try_template(
107        &mut self,
108        path: &mut Vec<String>,
109        source: &Source,
110        ctx: &TemplateContext,
111    ) -> templating::Result<()> {
112        let mut errs = templating::ErrorBuilder::from(self.try_template_setup(path, source, ctx));
113        errs.append(self.try_template_teardown(path, source, ctx));
114
115        errs.into_result(())
116    }
117}

119impl Check for EnvironmentConfig {
120    fn try_check(
121        &self,
122        path: &mut Vec<String>,
123        ctx: &impl ResolutionContext,
124    ) -> checks::Result<()> {
125        let mut errs = checks::ErrorBuilder::new();
126
127        // Check that hard coded variables and the ones coming from setup.provides are unique
128        let all_variables = self
129            .variable_definitions
130            .iter()
131            .chain(self.setup.provides.iter());
132        let duplicates = duplicate_keys(all_variables, |v| &v.name);
133        if !duplicates.is_empty() {
134            errs.push(
135                checks::ErrorKind::DuplicateVariableNames,
136                duplicates.join("\n"),
137                path,
138            );
139        }
140
141        // Check that each command is valid in isolation
142        errs.append(self.setup.command.try_check_nested(path, "setup", ctx));
143        errs.append(self.teardown.try_check_nested(path, "teardown", ctx));
144
145        errs.into_result(())
146    }
147}

169            ),
170        ]
171    }
