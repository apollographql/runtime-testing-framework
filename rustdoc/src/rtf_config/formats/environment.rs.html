14/// # Environment Config
15///
16/// Configuration for preparing and cleaning up the test environment as part of a test plan.
17#[derive(Debug, Clone, PartialEq, Deserialize, Serialize, JsonSchema)]
18pub struct EnvironmentConfig {
19    /// The name of this environment configuration
20    pub name: String,
21    /// A brief description of how this environment setup works
22    pub description: String,
23    /// Definitions for the required values for templating this environment
24    #[serde(default)]
25    pub values: Vec<ValueDefinition>,
26    /// The command to execute to prepare the environment for executing the test scenario
27    pub setup: SetupSection,
28    /// The command to execute to clean up the environment after executing the test scenario
29    pub teardown: CommandSection,
30}
31
32impl EnvironmentConfig {
33    pub fn try_load_from_path(p: impl AsRef<Path>) -> Result<Self> {
34        let content = fs::read_to_string(p)?;
35
36        Ok(serde_yaml::from_str(&content)?)
37    }
38
39    /// Try to template the setup [CommandSection].
40    ///
41    /// Setup is only allowed to reference values that are declared in the values section of this
42    /// config file.
43    pub fn try_template_setup(
44        &mut self,
45        path: &mut Vec<String>,
46        values: &HashMap<String, Scalar>,
47    ) -> templating::Result<()> {
48        let definitions = self.values.iter();
49        let allowed_values = values_for_config_file(values, definitions);
50
51        self.setup
52            .command
53            .try_template_nested(path, "setup", &allowed_values)
54    }
55
56    /// Try to template the teardown [CommandSection].
57    ///
58    /// Teardown is allowed to reference values that come from the output of setup in addition to
59    /// the values decalered in the values section of this config file.
60    pub fn try_template_teardown(
61        &mut self,
62        path: &mut Vec<String>,
63        values: &HashMap<String, Scalar>,
64    ) -> templating::Result<()> {
65        let definitions = self.values.iter().chain(self.setup.provides.iter());
66        let allowed_values = values_for_config_file(values, definitions);
67
68        self.teardown
69            .try_template_nested(path, "teardown", &allowed_values)
70    }
71
72    /// Create an empty [EnvironmentConfig] for tests
73    #[cfg(test)]
74    pub(crate) fn empty() -> EnvironmentConfig {
75        EnvironmentConfig {
76            name: Default::default(),
77            description: Default::default(),
78            values: Vec::new(),
79            setup: SetupSection {
80                command: CommandSection::empty(),
81                provides: Vec::new(),
82            },
83            teardown: CommandSection::empty(),
84        }
85    }
86}

88impl Template for EnvironmentConfig {
89    fn has_pending_fields(&self) -> bool {
90        self.setup.command.has_pending_fields() || self.teardown.has_pending_fields()
91    }
92
93    fn required_values(&self) -> Vec<String> {
94        let mut vals = self.setup.command.required_values();
95        vals.extend(self.teardown.required_values());
96
97        vals
98    }
99
100    fn try_template(
101        &mut self,
102        path: &mut Vec<String>,
103        values: &HashMap<String, Scalar>,
104    ) -> templating::Result<()> {
105        let mut errs = templating::ErrorBuilder::from(self.try_template_setup(path, values));
106        errs.append(self.try_template_teardown(path, values));
107
108        errs.into_result(())
109    }
110}

112impl Check for EnvironmentConfig {
113    fn try_check(
114        &self,
115        path: &mut Vec<String>,
116        src: &Source,
117        ctx: &impl ResolutionContext,
118    ) -> checks::Result<()> {
119        let mut errs = checks::ErrorBuilder::new();
120
121        // Check that hard coded values and the ones coming from setup.provides are unique
122        let all_values = self.values.iter().chain(self.setup.provides.iter());
123        let duplicates = duplicate_keys(all_values, |v| &v.name);
124        if !duplicates.is_empty() {
125            errs.push(
126                checks::ErrorKind::DuplicateValueNames,
127                duplicates.join("\n"),
128                path,
129            );
130        }
131
132        // Check that each command is valid in isolation
133        errs.append(self.setup.command.try_check_nested(path, "setup", src, ctx));
134        errs.append(self.teardown.try_check_nested(path, "teardown", src, ctx));
135
136        errs.into_result(())
137    }
138}
