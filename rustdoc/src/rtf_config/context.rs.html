31/// Types that implement ResolutionContext may be used to perform IO while checking and templating
32/// providers.
33///
34/// This trait is used to allow us to inject mock IO implementations in tests so we do not need to
35/// make use of temp directories or mock servers. For the most part the methods provided by this
36/// trait are wrappers around [std::fs] and API clients for making requests to third party APIs.
37///
38/// For the canonical real implementations that should be mocked see [Context].
39#[allow(async_fn_in_trait)]
40pub trait ResolutionContext {
41    type PlatformClient: platform_query::Client;
42    type GithubClient: github::Client;
43    type HttpClient: HttpClient;
44
45    /// Provide a [Client][platform_query::Client] for making requests to the Apollo platform API.
46    ///
47    /// If it is not possible for this current context to make requests to the platform API then
48    /// this method should return [None].
49    fn platform_client(&self) -> Option<&Self::PlatformClient> {
50        None
51    }
52
53    /// Provide a [Client][github::Client] for making requests to the GitHub REST API.
54    ///
55    /// If it is not possible for this current context to make requests to the GitHub API then
56    /// this method should return [None].
57    fn github_client(&self) -> Option<&Self::GithubClient> {
58        None
59    }
60
61    fn http_client(&self) -> Option<&Self::HttpClient> {
62        None
63    }
64
65    /// Record the path that the given provider's output was written to.
66    fn store_provider_output_path(&mut self, provider: Provider<'_>, path: PathBuf);
67
68    /// Query the output path of a given provider.
69    #[allow(unused_variables)]
70    fn known_provider_output_path(&self, provider: Provider<'_>) -> Option<PathBuf> {
71        None
72    }
73
74    #[allow(unused_variables)]
75    fn set_values(&mut self, values: &HashMap<String, Scalar>) {}
76
77    fn values(&self) -> Option<&HashMap<String, Scalar>> {
78        None
79    }
80
81    /// Make use of potentially cached supergraph details to derive related data.
82    /// The default implementation of this method will simply pull the required [SupergraphDetails]
83    /// and pass them to the mapping function provided. Custom implementations can be used to
84    /// implement caching and other custom behaviour.
85    ///
86    /// # Panics
87    /// The default implementation will panic if [ResolutionContext::platform_client] is [None].
88    async fn with_supergraph_details<T>(
89        &self,
90        graph_id: impl Into<String>,
91        variant: impl Into<String>,
92        f: impl FnOnce(&Arc<SupergraphDetails>) -> providers::Result<T>,
93    ) -> providers::Result<T> {
94        let client = self.platform_client().expect("no platform client");
95        let details = SupergraphDetails::fetch(graph_id, variant, client).await?;
96
97        f(&Arc::new(details))
98    }
99
100    /// Returns the canonical, absolute form of the path with all intermediate
101    /// components normalized and symbolic links resolved.
102    fn canonicalize_path(&self, relative_path: impl AsRef<Path>) -> io::Result<PathBuf>;
103
104    fn dir_containing(&self, path: impl AsRef<Path>) -> PathBuf {
105        match path.as_ref().parent() {
106            Some(p) => p.to_path_buf(),
107            None => PathBuf::new(),
108        }
109    }
110
111    /// Categorise the provided path.
112    ///
113    /// If you cannot access the metadata of the file, e.g. because of a permission error or broken
114    /// symbolic links, this will return [PathKind::Missing].
115    fn path_kind(&self, path: impl AsRef<Path>) -> PathKind;
116
117    /// Reads the entire contents of a file into a string.
118    ///
119    /// # Errors
120    ///
121    /// This function will return an error if `path` does not already exist. Other errors may also
122    /// be returned according to [OpenOptions::open][0].
123    ///
124    /// If the contents of the file are not valid UTF-8, then an error will also be returned.
125    ///
126    /// While reading from the file, this function handles [io::ErrorKind::Interrupted] with
127    /// automatic retries. See [io::Read] documentation for details.
128    ///
129    /// [0]: std::fs::OpenOptions::open
130    fn read_path_to_string(&self, path: impl AsRef<Path>) -> io::Result<String>;
131
132    /// Writes a slice as the entire contents of a file.
133    ///
134    /// This function will create a file if it does not exist, and will entirely replace its
135    /// contents if it does.
136    ///
137    /// Depending on the platform, this function may fail if the full directory path does not
138    /// exist.
139    fn write(&self, path: impl AsRef<Path>, content: impl AsRef<[u8]>) -> io::Result<()>;
140
141    /// Spawn the specified program as a subprocess with the provided arguments.
142    /// This method will block until the process completes and return the stdout of the process as
143    /// a utf-8 string.
144    fn run_command_blocking<'a>(
145        &self,
146        prog: &str,
147        args: impl IntoIterator<Item = &'a str>,
148        env_vars: &HashMap<String, String>,
149    ) -> io::Result<()>;
150
151    /// Changes the current working directory to the specified path.
152    fn set_current_dir(&mut self, path: impl AsRef<Path>) -> io::Result<()>;
153
154    /// Changes the permissions of the specified file
155    fn make_executable(&self, path: impl AsRef<Path>) -> io::Result<()> {
156        let file = File::open(path.as_ref())?;
157        let mut permissions = file.metadata()?.permissions();
158        permissions.set_mode(0o777);
159
160        file.set_permissions(permissions)
161    }
162
163    fn remove_file(&self, path: impl AsRef<Path>) -> io::Result<()>;
164
165    /// Recursively create a directory and all of its parent components if they
166    /// are missing.
167    ///
168    /// If this function returns an error, some of the parent components might have
169    /// been created already.
170    ///
171    /// If the empty path is passed to this function, it always succeeds without
172    /// creating any directories.
173    fn create_dir_all(&self, path: impl AsRef<Path>) -> io::Result<()>;
174}

176/// A [ResolutionContext] that will perform real IO.
177#[derive(Default, Debug)]
178pub struct Context {
179    client: ReqwestClient,
180    supergraph_details: Mutex<HashMap<String, Arc<SupergraphDetails>>>,
181    values: HashMap<String, Scalar>,
182    fp_output_paths: HashMap<String, PathBuf>,
183}
184
185impl Context {
186    /// Construct a new `Context` with default configuration.
187    pub fn new() -> Self {
188        Self::default()
189    }
190
191    /// Construct a new `Context` with environment variables.
192    pub fn new_from_env_vars(mut env_vars: HashMap<String, String>) -> Self {
193        let mut ctx = Self::new();
194
195        if let Some(api_key) = env_vars.remove(APOLLO_KEY_ENV_VAR) {
196            let staging = matches!(
197                env_vars.remove(GRAPH_OS_STAGING_ENV_VAR).as_deref(),
198                Some("true" | "1")
199            );
200            let sudo = matches!(
201                env_vars.remove(APOLLO_SUDO_ENV_VAR).as_deref(),
202                Some("true" | "1")
203            );
204
205            ctx.with_platform_config(api_key, staging, sudo);
206        }
207
208        if let Some(api_token) = env_vars.remove(GITHUB_TOKEN_ENV_VAR) {
209            ctx.with_github_config(api_token);
210        }
211
212        ctx
213    }
214
215    /// Provide configuration for making requests to the Apollo platform API.
216    pub fn with_platform_config(
217        &mut self,
218        api_key: impl Into<String>,
219        staging: bool,
220        sudo: bool,
221    ) -> &mut Self {
222        self.client.with_platform_config(api_key, staging, sudo);
223        self
224    }
225
226    /// Provide configuration for making requests to the GitHub REST API.
227    pub fn with_github_config(&mut self, api_token: impl Into<String>) -> &mut Self {
228        self.client.with_github_config(api_token);
229        self
230    }
231}

233impl ResolutionContext for Context {
234    type PlatformClient = rtf_core::graphos::PlatformClient;
235    type GithubClient = github::GithubClient;
236    type HttpClient = ReqwestClient;

238    fn platform_client(&self) -> Option<&Self::PlatformClient> {
239        self.client.platform_client()
240    }
241
242    fn github_client(&self) -> Option<&Self::GithubClient> {
243        self.client.github_client()
244    }
245
246    fn http_client(&self) -> Option<&Self::HttpClient> {
247        Some(&self.client)
248    }

250    fn store_provider_output_path(&mut self, provider: Provider<'_>, path: PathBuf) {
251        let key = match serde_yaml::to_string(&provider) {
252            Ok(s) => s,
253            Err(error) => {
254                error!(?path, %error, "unable to generate file provider cache key");
255                return;
256            }
257        };
258
259        self.fp_output_paths.insert(key, path);
260    }
261
262    fn known_provider_output_path(&self, provider: Provider<'_>) -> Option<PathBuf> {
263        let key = serde_yaml::to_string(&provider).ok()?;
264
265        self.fp_output_paths.get(&key).cloned()
266    }
267
268    fn set_values(&mut self, values: &HashMap<String, Scalar>) {
269        self.values = values.clone();
270    }
271
272    fn values(&self) -> Option<&HashMap<String, Scalar>> {
273        Some(&self.values)
274    }
275
276    async fn with_supergraph_details<T>(
277        &self,
278        graph_id: impl Into<String>,
279        variant: impl Into<String>,
280        f: impl FnOnce(&Arc<SupergraphDetails>) -> providers::Result<T>,
281    ) -> providers::Result<T> {
282        let graph_id = graph_id.into();
283        let variant = variant.into();
284
285        let mut guard = self.supergraph_details.lock().await;
286        let client = self.client.platform_client().expect("no platform client");
287        let graph_ref = format!("{graph_id}@{variant}");
288
289        if let Entry::Vacant(e) = guard.entry(graph_ref.clone()) {
290            let details = Arc::new(SupergraphDetails::fetch(graph_id, variant, client).await?);
291            let res = f(&details);
292            e.insert(details);
293
294            return res;
295        }
296
297        let details = guard.get(&graph_ref).expect("details should be cached");
298
299        f(details)
300    }
301
302    fn canonicalize_path(&self, relative_path: impl AsRef<Path>) -> io::Result<PathBuf> {
303        relative_path.as_ref().canonicalize()
304    }
305
306    fn path_kind(&self, path: impl AsRef<Path>) -> PathKind {
307        let p = path.as_ref();
308
309        if !p.exists() {
310            return PathKind::Missing;
311        } else if p.is_file() {
312            return PathKind::File;
313        }
314
315        match p.read_dir() {
316            Ok(mut rd) => {
317                if rd.next().is_some() {
318                    PathKind::OccupiedDir
319                } else {
320                    PathKind::EmptyDir
321                }
322            }
323            Err(_) => PathKind::Missing,
324        }
325    }
326
327    fn read_path_to_string(&self, path: impl AsRef<Path>) -> io::Result<String> {
328        fs::read_to_string(path)
329    }
330
331    fn write(&self, path: impl AsRef<Path>, contents: impl AsRef<[u8]>) -> io::Result<()> {
332        fs::write(path, contents)
333    }
334
335    fn run_command_blocking<'a>(
336        &self,
337        prog: &str,
338        args: impl IntoIterator<Item = &'a str>,
339        env_vars: &HashMap<String, String>,
340    ) -> io::Result<()> {
341        let status = Command::new(prog)
342            .args(args)
343            .envs(env_vars)
344            .spawn()?
345            .wait()?;
346
347        if !status.success() {
348            return Err(io::Error::other(format!(
349                "{prog:?} failed to terminate successfully"
350            )));
351        }
352
353        Ok(())
354    }
355
356    fn remove_file(&self, path: impl AsRef<Path>) -> io::Result<()> {
357        fs::remove_file(path)
358    }
359
360    fn set_current_dir(&mut self, path: impl AsRef<Path>) -> io::Result<()> {
361        set_current_dir(path.as_ref())?;
362
363        Ok(())
364    }
365
366    fn create_dir_all(&self, path: impl AsRef<Path>) -> io::Result<()> {
367        fs::create_dir_all(path)
368    }
