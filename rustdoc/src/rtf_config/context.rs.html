37/// Types that implement ResolutionContext may be used to perform IO while checking and templating
38/// providers.
39///
40/// This trait is used to allow us to inject mock IO implementations in tests so we do not need to
41/// make use of temp directories or mock servers. For the most part the methods provided by this
42/// trait are wrappers around [std::fs] and API clients for making requests to third party APIs.
43///
44/// For the canonical real implementations that should be mocked see [Context].
45pub trait ResolutionContext: Send + Sync {
46    type PlatformClient: platform_query::Client;
47    type GithubClient: github::Client;
48    type HttpClient: HttpClient;

82    /// Make use of potentially cached supergraph details to derive related data.
83    /// The default implementation of this method will simply pull the required [SupergraphDetails]
84    /// and pass them to the mapping function provided. Custom implementations can be used to
85    /// implement caching and other custom behaviour.
86    ///
87    /// # Panics
88    /// The default implementation will panic if [ResolutionContext::platform_client] is [None].
89    fn with_supergraph_details<T: Send>(
90        &self,
91        graph_id: impl Into<String> + Send,
92        variant: impl Into<String> + Send,
93        f: impl FnOnce(&Arc<SupergraphDetails>) -> providers::Result<T> + Send,
94    ) -> impl Future<Output = providers::Result<T>> + Send {
95        async move {
96            let client = self.platform_client().expect("no platform client");
97            let details = SupergraphDetails::fetch(graph_id, variant, client).await?;
98
99            f(&Arc::new(details))
100        }
101    }

107    fn dir_containing(&self, path: impl AsRef<Path>) -> PathBuf {
108        match path.as_ref().parent() {
109            Some(p) => p.to_path_buf(),
110            None => PathBuf::new(),
111        }

114    fn check_path_kind(
115        &self,
116        stable_src: &StableSource,
117        str_path: &str,
118        err_path: &[String],
119    ) -> checks::Result<Option<PathKind>> {
120        let src = self.source_dir_for(stable_src);
121        let res = match src {
122            SourceDir::Local { abs_path } => self
123                .canonicalize_path(abs_path.join(str_path))
124                .map(|p| Some(self.path_kind(p))),
125
126            SourceDir::Github { .. } => {
127                return if self.github_client().is_none() {
128                    Err(checks::Errors::new(
129                        checks::ErrorKind::MissingGithubApiKey,
130                        "expected os env key GITHUB_TOKEN",
131                        err_path,
132                    ))
133                } else {
134                    Ok(None)
135                };
136            }
137        };
138
139        res.map_err(|e| {
140            let kind = if e.kind() == io::ErrorKind::NotFound {
141                checks::ErrorKind::FileNotFound
142            } else {
143                checks::ErrorKind::InvalidRelativePath
144            };
145
146            checks::Errors::new(
147                kind,
148                format!("provided path was {}", src.to_uri_for(str_path)),
149                err_path,
150            )
151        })
152    }

164    ///
165    /// # Errors
166    ///
167    /// This function will return an error if `path` does not already exist. Other errors may also
168    /// be returned according to [OpenOptions::open][0].
169    ///
170    /// If the contents of the file are not valid UTF-8, then an error will also be returned.
171    ///
172    /// While reading from the file, this function handles [io::ErrorKind::Interrupted] with
173    /// automatic retries. See [io::Read] documentation for details.
174    ///
175    /// [0]: std::fs::OpenOptions::open
176    fn read_path_to_string(&self, path: impl AsRef<Path>) -> io::Result<String>;
177

185    /// Writes a slice as the entire contents of a file.
186    ///
187    /// This function will create a file if it does not exist, and will entirely replace its
188    /// contents if it does.
189    ///
190    /// Depending on the platform, this function may fail if the full directory path does not
191    /// exist.
192    fn write(&self, path: impl AsRef<Path>, content: impl AsRef<[u8]>) -> io::Result<()>;
193
194    /// Spawn the specified program as a subprocess with the provided arguments.

202    ) -> io::Result<()>;

204    /// Changes the permissions of the specified file
205    fn make_executable(&self, path: impl AsRef<Path>) -> io::Result<()> {
206        let file = File::open(path.as_ref())?;
207        let mut permissions = file.metadata()?.permissions();
208        permissions.set_mode(0o777);
209
210        file.set_permissions(permissions)
211    }

213    /// Store a key-value pair of run metadata for cross-phase communication.
214    fn store_run_metadata(&mut self, _key: &'static str, _value: impl Into<String>) {}
215
216    /// Retrieve a previously stored run metadata value by key.
217    fn run_metadata(&self, _key: &str) -> Option<&str> {
218        None
219    }

221    fn remove_file(&self, path: impl AsRef<Path>) -> io::Result<()>;

224
225    /// Recursively create a directory and all of its parent components if they
226    /// are missing.
227    ///
228    /// If this function returns an error, some of the parent components might have

240}

256impl Context {
257    /// Construct a new `Context` with default configuration.
258    pub fn new() -> Self {
259        Self::default()
260    }
261
262    /// Construct a new `Context` with environment variables.
263    pub fn new_from_env_vars(env_vars: &HashMap<String, String>) -> Self {
264        let mut ctx = Self::new();
265
266        if let Some(api_key) = env_vars.get(APOLLO_KEY_ENV_VAR) {
267            let staging = matches!(
268                env_vars.get(GRAPH_OS_STAGING_ENV_VAR).map(|s| s.as_str()),
269                Some("true" | "1")
270            );
271            let sudo = matches!(
272                env_vars.get(APOLLO_SUDO_ENV_VAR).map(|s| s.as_str()),
273                Some("true" | "1")
274            );
275
276            ctx.with_platform_config(api_key, staging, sudo);
277        }
278
279        if let Some(api_token) = env_vars.get(GITHUB_TOKEN_ENV_VAR) {
280            ctx.with_github_config(api_token);
281        }
282
283        ctx
284    }

378    async fn with_supergraph_details<T: Send>(
379        &self,
380        graph_id: impl Into<String> + Send,
381        variant: impl Into<String> + Send,
382        f: impl FnOnce(&Arc<SupergraphDetails>) -> providers::Result<T> + Send,
383    ) -> providers::Result<T> {
384        let graph_id = graph_id.into();
385        let variant = variant.into();
386
387        let mut guard = self.supergraph_details.lock().await;
388        let client = self.client.platform_client().expect("no platform client");
389        let graph_ref = format!("{graph_id}@{variant}");
390
391        if let Entry::Vacant(e) = guard.entry(graph_ref.clone()) {
392            let details = Arc::new(SupergraphDetails::fetch(graph_id, variant, client).await?);
393            let res = f(&details);
394            e.insert(details);
395
396            return res;
397        }
398
399        let details = guard.get(&graph_ref).expect("details should be cached");
400
401        f(details)
402    }

404    fn canonicalize_path(&self, path: impl AsRef<Path>) -> io::Result<PathBuf> {

406    }

408    fn path_kind(&self, path: impl AsRef<Path>) -> PathKind {
409        let p = path.as_ref();
410
411        if !p.exists() {
412            return PathKind::Missing;
413        } else if p.is_file() {
414            return PathKind::File;
415        }
416
417        match p.read_dir() {
418            Ok(mut rd) => {
419                if rd.next().is_some() {
420                    PathKind::OccupiedDir
421                } else {
422                    PathKind::EmptyDir
423                }
424            }
425            Err(_) => PathKind::Missing,
426        }
427    }

451    fn run_command_blocking<'a>(
452        &self,
453        prog: &str,
454        args: impl IntoIterator<Item = &'a str>,
455        env_vars: &HashMap<String, String>,
456    ) -> io::Result<()> {
457        if self.capture_output {
458            let output = Command::new(prog).args(args).envs(env_vars).output()?;
459
460            self.captured_stdout.write().unwrap().extend(&output.stdout);
461            self.captured_stderr.write().unwrap().extend(&output.stderr);
462
463            if !output.status.success() {
464                return Err(io::Error::other(format!(
465                    "{prog:?} failed to terminate successfully"
466                )));
467            }
468        } else {
469            let status = Command::new(prog)
470                .args(args)
471                .envs(env_vars)
472                .spawn()?
473                .wait()?;
474
475            if !status.success() {
476                return Err(io::Error::other(format!(
477                    "{prog:?} failed to terminate successfully"
478                )));
479            }
480        }
481
482        Ok(())
483    }

524    #[test_case(None, None, true; "no sources")]
525    #[test_case(Some(StableSource::TestPlan), Some(StableSource::TestPlan), true; "same source")]
526    #[test_case(Some(StableSource::TestPlan), None, false; "cache with source new without")]
527    #[test_case(None, Some(StableSource::TestPlan), false; "cache without source new with")]
528    #[test_case(Some(StableSource::TestPlan), Some(StableSource::Environment), false; "different sources")]
529    #[test]
530    fn context_provider_caching_respects_relative_path_sources(
531        source_1: Option<StableSource>,
532        source_2: Option<StableSource>,
533        path_is_known: bool,
534    ) {
535        let provider_1 = FileProvider::RelativePath(RelativeFile {
536            path: Field::Resolved("scripts/run.sh".to_string()),
537            src: source_1,
538        });
539
540        let provider_2 = FileProvider::RelativePath(RelativeFile {
541            path: Field::Resolved("scripts/run.sh".to_string()),
542            src: source_2,
543        });
544
545        let mut ctx = Context::new();
546
547        ctx.store_provider_output_path(
548            Provider::File { fp: &provider_1 },
549            PathBuf::from("/some/path"),
550        );
551        let maybe_path = ctx.known_provider_output_path(Provider::File { fp: &provider_2 });
552
553        assert_eq!(maybe_path.is_some(), path_is_known);
554    }
