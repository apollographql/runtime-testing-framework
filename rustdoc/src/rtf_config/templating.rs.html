93    /// Helper for filtering allowed templating variables based on [VariableDefinition]s present in a
94    /// config file. This is also where defaults defined in variable definitions are applied, being
95    /// overwritten by any explicitly provided variables coming from `all_variables`.
96    ///
97    /// # Constructing the definitions argument
98    ///
99    /// The trait bound here is to support both direct calls to `Vec<VariableDefinition>.iter()` and
100    /// calls to [Iterator::chain] to joining together multiple vecs of VariableDefinitions:
101    ///
102    /// ```ignore
103    /// // from EnvironmentConfig: both of these will work
104    /// let definitions = self.variable_definitions.iter();
105    /// let definitions = self.variable_definitions.iter().chain(self.setup.provides.iter());
106    /// ```
107    pub(crate) fn for_config_file<'a>(
108        &self,
109        file_source: &Source,
110        definitions: impl Iterator<Item = &'a VariableDefinition> + Clone,
111    ) -> Self {
112        let mut new = self.clone();
113
114        new.variables
115            .retain(|k, _| definitions.clone().any(|val| &val.name == k));
116
117        for vd in definitions {
118            if let Some(default) = vd.default.as_ref() {
119                new.variables.entry(vd.name.clone()).or_insert_with(|| {
120                    new.override_sources
121                        .insert(vd.name.clone(), file_source.clone());
122
123                    default.clone()
124                });
125            }
126        }
127
128        new
129    }
130
131    pub fn extend(&mut self, source: Source, variables: HashMap<String, Scalar>) {
132        for k in variables.keys() {
133            self.override_sources.insert(k.to_owned(), source.clone());
134        }
135
136        self.variables.extend(variables);
137    }
138
139    pub fn get<Q>(&self, key: &Q) -> Option<&Scalar>
140    where
141        String: Borrow<Q>,
142        Q: Hash + Eq + ?Sized,
143    {
144        self.variables.get(key)
145    }
146
147    pub fn get_with_source<Q>(&self, key: &Q) -> Option<(&Source, &Scalar)>
148    where
149        String: Borrow<Q>,
150        Q: Hash + Eq + ?Sized,
151    {
152        let s = self.variables.get(key)?;
153        let source = self
154            .override_sources
155            .get(key)
156            .unwrap_or(&self.test_plan_source);
157
158        Some((source, s))
159    }

162/// In order to support controlled templating of config files with [Scalar] variables we make use of a
163/// wrapper [Field] type to identify where variables need to be injected. A type that implements
164/// [Template] supports walking its contents to locate and template fields using a provided map
165/// of scalar variables.
166pub trait Template {
167    /// Whether or not there are any pending [Field]s contained within this variable.
168    fn has_pending_fields(&self) -> bool;
169
170    /// The list of template variables that are required to template this type fully.
171    fn required_variables(&self) -> Vec<String>;
172
173    /// Attempt to resolve all pending [Field]s, appending encountered errors to the `errs` vec
174    /// provided.
175    fn try_template(
176        &mut self,
177        path: &mut Vec<String>,
178        file_source: &Source,
179        ctx: &TemplateContext,
180    ) -> Result<()>;
181
182    /// Attempt to resolve all pending [Field]s when this type is a child of some parent
183    /// [Template], appending encountered errors to the `errs` vec provided. The provided `tail`
184    /// will be appended to `path` before calling through to [Template::try_template].
185    fn try_template_nested(
186        &mut self,
187        path: &mut Vec<String>,
188        tail: &str,
189        file_source: &Source,
190        ctx: &TemplateContext,
191    ) -> Result<()> {
192        let mut path = path.clone();
193        path.push(tail.to_string());
194        self.try_template(&mut path, file_source, ctx)
195    }
196
197    /// Attempt to resolve all known [Field]s, reporting required variables that are not present in
198    /// the provided map. If there are any deserialization errors then then this method as an
199    /// aggregate operation will fail.
200    fn try_template_known(
201        &mut self,
202        path: &mut Vec<String>,
203        file_source: &Source,
204        ctx: &TemplateContext,
205    ) -> Result<Vec<String>> {
206        let all_errs = match self.try_template(path, file_source, ctx) {
207            Ok(_) => return Ok(Vec::new()),
208            Err(errs) => errs,
209        };
210
211        let mut errs = ErrorBuilder::new();
212        let mut missing = Vec::new();
213
214        for err in all_errs.into_iter() {
215            match err.kind {
216                ErrorKind::UnknownVariable => missing.push(err.message),
217                _ => errs.push_err(err),
218            }
219        }
220
221        errs.into_result(missing)
222    }
223}

314/// A [Field] wraps some scalar type that implements [Template] in order to mark it as
315/// requriring a templated variable coming from user provided variables as part of resolving the config
316/// file.
317///
318/// Fields must be resolved in order to be usable during a test run.
319#[derive(Debug, Clone, PartialEq)]
320pub enum Field<T>
321where
322    T: ValidField,
323{

325    Pending(String),
326    /// A field containing the final data needed for resolving the config file.
327    Resolved(T),
328}

401impl<T> Template for Field<T>
402where
403    T: ValidField,
404{
405    fn has_pending_fields(&self) -> bool {
406        matches!(self, Self::Pending(_))
407    }
408
409    fn required_variables(&self) -> Vec<String> {
410        match self {
411            Self::Pending(field_name) => vec![field_name.clone()],
412            _ => Vec::new(),
413        }
414    }
415
416    fn try_template(
417        &mut self,
418        path: &mut Vec<String>,
419        _file_source: &Source,
420        ctx: &TemplateContext,
421    ) -> Result<()> {
422        if let Self::Pending(variable) = self {
423            match ctx.get(variable) {
424                Some(raw) => match T::try_from_scalar(raw.clone()) {
425                    Ok(t) => *self = Self::Resolved(t),
426                    Err(reason) => {
427                        return Err(Errors::new(ErrorKind::InvalidData, reason, path));
428                    }
429                },
430                None => {
431                    return Err(Errors::new(
432                        ErrorKind::UnknownVariable,
433                        variable.clone(),
434                        path,
435                    ));
436                }
437            }
438        }
439
440        Ok(())
441    }
442}

465/// A custom serde [Visitor] for locating template strings of the form `"{{ some_variable }}"` and
466/// marking them as pending fields. Malformed templates are reported as deserialization errors and
467/// fields that do not contain template strings are deserialized as normal by deferring to the
468/// default deserializer implementation for the type found in the input. (If the type in the input
469/// is incorrect for the target then this will result in a deserialization error as normal).
470struct FieldVisitor<T>(PhantomData<T>);
471
472impl<T> Visitor<'_> for FieldVisitor<T>
473where
474    T: ValidField,
475{
476    type Value = Field<T>;
477
478    fn expecting(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
479        formatter.write_str("a bool, integer, float or string")
480    }
481
482    /// When visiting a string we need to check to see if we have a valid template pattern or not.
483    /// If we do then we extract the variable name from it an return a Pending, otherwise we defer to
484    /// the default handling for strings as with our other supported scalar types.
485    /// If we detect a malformed template then we error _here_ rather treating it as a string and
486    /// potentially leading to confusing runtime behaviour.
487    fn visit_str<E: de::Error>(self, value: &str) -> std::result::Result<Self::Value, E> {
488        let ident = match value.strip_prefix("{{ ") {
489            Some(s) => s
490                .strip_suffix(" }}")
491                .ok_or(E::custom("unclosed field template"))?,
492
493            None if value.starts_with("{{") => {
494                return Err(E::custom(
495                    "malformed template string: expected a single space after '{{'",
496                ));
497            }
498
499            None => {
500                return Deserialize::deserialize(de::value::StrDeserializer::new(value))
501                    .map(|t| Field::Resolved(t));
502            }
503        };
504
505        let is_valid_identifier = !ident.is_empty()
506            && ident.starts_with(char::is_alphabetic)
507            && ident.chars().all(|ch| ch == '_' || ch.is_alphanumeric());
508
509        if is_valid_identifier {
510            Ok(Field::Pending(ident.to_string()))
511        } else {
512            Err(E::custom(
513                "expected variable identifier with a single space either side",
514            ))
515        }
516    }
517
518    fn visit_bool<E: de::Error>(self, v: bool) -> std::result::Result<Self::Value, E> {
519        Deserialize::deserialize(de::value::BoolDeserializer::new(v)).map(|t| Field::Resolved(t))
520    }
521
522    // The default impls for i8, i16 and i32 will forward to this
523    fn visit_i64<E: de::Error>(self, v: i64) -> std::result::Result<Self::Value, E> {
524        Deserialize::deserialize(de::value::I64Deserializer::new(v)).map(|t| Field::Resolved(t))
525    }
526
527    // The default impls for u8, u16 and u32 will forward to this
528    fn visit_u64<E: de::Error>(self, v: u64) -> std::result::Result<Self::Value, E> {
529        Deserialize::deserialize(de::value::U64Deserializer::new(v)).map(|t| Field::Resolved(t))
530    }
531
532    // The default impl for f32 will forward to this
533    fn visit_f64<E: de::Error>(self, v: f64) -> std::result::Result<Self::Value, E> {
534        Deserialize::deserialize(de::value::F64Deserializer::new(v)).map(|t| Field::Resolved(t))
535    }
536}

606/// A [Scalar] type which may appear inside of a templated [Field] within a config file.
607pub trait ValidField: fmt::Debug + Clone + Serialize + DeserializeOwned {
608    /// We need this try_from operation to always return a String error on failure so we can report
609    /// invalid types being used in Test Plans to the user. This doesn't work for a `Field<Scalar>`
610    /// as Scalar::try_from(Scalar) has an error type of Infallible which can't be constructed (and
611    /// therefore can't be turned into a string).
612    fn try_from_scalar(s: Scalar) -> std::result::Result<Self, String>;
613}
614
615impl ValidField for Scalar {
616    fn try_from_scalar(s: Scalar) -> std::result::Result<Self, String> {
617        Ok(s)
618    }
619}
620
621impl ValidField for bool {
622    fn try_from_scalar(s: Scalar) -> std::result::Result<Self, String> {
623        Self::try_from(s)
624    }
625}

674// We want to handle all signed and unsigned integers in a similar way so we're stamping out the
675// impls we need to satisfy ValidField using a macro.
676macro_rules! impl_integer_scalars {
677    ( $([$($ty:ty),+] => $as_method:ident;)+ ) => {
678        $($(
679            impl ValidField for $ty {
680                fn try_from_scalar(s: Scalar) -> std::result::Result<Self, String> {
681                    Self::try_from(s)
682                }
683            }
684
685            impl From<$ty> for Scalar {
686                fn from(variable: $ty) -> Self {
687                    Scalar::Number(Number(serde_json::Number::from(variable)))
688                }
689            }
690
691            impl TryFrom<Scalar> for $ty {
692                type Error = String;
693
694                fn try_from(variable: Scalar) -> std::result::Result<Self, Self::Error> {
695                    let maybe_t = match &variable {
696                        Scalar::Number(Number(v)) => v.$as_method().map(|n| n as $ty),
697                        _ => None
698                    };
699
700                    maybe_t.ok_or_else(|| format!("invalid variable `{variable}`, expected {}", stringify!($ty)))
701                }
702            }
703        )+)+
704    };
705}

881    macro_rules! field_map {
882        ($slice:expr) => {{
883            let mut m = ::std::collections::HashMap::new();
884            for field in $slice {
885                match field {
886                    Field::Pending(variable) => {
887                        m.insert(variable.clone(), field.clone());
888                    }
889                    Field::Resolved(value) => {
890                        m.insert(value.clone(), field.clone());
891                    }
892                }
893            }
894            m
895        }};
896    }
