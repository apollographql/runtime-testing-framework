50pub trait Check {
51    /// Run any initial static check available to error early if this provider contains
52    /// invalid data.
53    fn try_check(
54        &self,
55        path: &mut Vec<String>,
56        src: &Source,
57        ctx: &impl ResolutionContext,
58    ) -> Result<()>;
59
60    fn try_check_nested(
61        &self,
62        path: &mut Vec<String>,
63        tail: impl Into<String>,
64        src: &Source,
65        ctx: &impl ResolutionContext,
66    ) -> Result<()> {
67        let mut path = path.clone();
68        path.push(tail.into());
69
70        self.try_check(&mut path, src, ctx)
71    }
72}

74/// Helper macro for stamping out implementations of the [Check] trait on an enum where each
75/// variant is a wrapper around a type that already implements the trait.
76#[macro_export]
77macro_rules! enum_impl_check {
78    ($enum:ident => $($variant:ident),+) => {
79        impl Check for $enum {
80            fn try_check(
81                &self,
82                path: &mut Vec<String>,
83                src: &Source,
84                ctx: &impl ResolutionContext,
85            ) -> $crate::checks::Result<()> {
86                match self {
87                    $(Self::$variant(inner) => inner.try_check(path, src, ctx),)+
88                }
89            }
90            fn try_check_nested(
91                &self,
92                path: &mut Vec<String>,
93                tail: impl Into<String>,
94                src: &Source,
95                ctx: &impl ResolutionContext,
96            ) -> $crate::checks::Result<()> {
97                match self {
98                    $(Self::$variant(inner) => inner.try_check_nested(path, tail, src, ctx),)+
99                }
100            }
101        }
102    };
103}

105/// Determine if there are any duplicates within a given slices of elements using a given key
106/// function.
107///
108/// See the tests in the validation module for example usage.
109pub(crate) fn duplicate_keys<'a, T: 'a>(
110    elems: impl Iterator<Item = T>,
111    key_fn: impl Fn(T) -> &'a str,
112) -> Vec<&'a str> {
113    let mut seen = HashSet::new();
114    let mut duplicates = Vec::new();
115
116    for elem in elems.into_iter() {
117        let k = key_fn(elem);
118        if seen.contains(k) {
119            duplicates.push(k);
120        } else {
121            seen.insert(k);
122        }
123    }
124
125    // ensure that our reported duplicates are in alphabetical order
126    duplicates.sort_unstable();
127    duplicates.dedup();
128
129    duplicates
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    #[derive(Debug)]
137    struct Mapping {
138        k: &'static str,
139        _v: usize,
140    }
141
142    #[test]
143    fn duplicate_keys_returns_empty_vec_for_no_duplicates() {
144        let vals = [
145            Mapping { k: "a", _v: 1 },
146            Mapping { k: "c", _v: 2 },
147            Mapping { k: "b", _v: 3 },
148            Mapping { k: "d", _v: 4 },
149            Mapping { k: "x", _v: 5 },
150        ];
151        let duplicates = duplicate_keys(vals.iter(), |s| s.k);
152
153        assert!(
154            duplicates.is_empty(),
155            "expected no duplicates, got {duplicates:?}"
156        );
157    }
158
159    #[test]
160    fn duplicate_keys_returns_sorted_results() {
161        let vals = [
162            Mapping { k: "c", _v: 1 },
163            Mapping { k: "c", _v: 2 },
164            Mapping { k: "a", _v: 3 },
165            Mapping { k: "a", _v: 4 },
166            Mapping { k: "b", _v: 5 },
167        ];
168        let duplicates = duplicate_keys(vals.iter(), |s| s.k);
169
170        assert_eq!(duplicates, vec!["a", "c"]);
171    }
172
173    #[test]
174    fn duplicate_keys_returns_unique_results() {
175        let vals = [
176            Mapping { k: "a", _v: 1 },
177            Mapping { k: "c", _v: 2 },
178            Mapping { k: "c", _v: 3 },
179            Mapping { k: "c", _v: 4 },
180            Mapping { k: "c", _v: 5 },
181        ];
182        let duplicates = duplicate_keys(vals.iter(), |s| s.k);
183
184        assert_eq!(duplicates, vec!["c"]);
185    }
