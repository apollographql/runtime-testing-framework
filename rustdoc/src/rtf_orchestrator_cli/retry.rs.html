1//! Retry for HTTP requests with exponential backoff and jitter.
2use rand::RngExt;
3use reqwest::StatusCode;
4use std::{fmt::Display, future::Future, time::Duration};
5use tokio::time::sleep;
6use tracing::warn;
7
8const INITIAL_INTERVAL: Duration = Duration::from_secs(2);
9const MAX_RETRIES: u32 = 5;
10const RATIO: f64 = 1.5;
11const JITTER: f64 = 0.25;
12
13#[derive(Debug)]
14pub(crate) struct Failure<E> {
15    error: E,
16    retryable: bool,
17}
18
19impl<E> Failure<E> {
20    pub(crate) fn from_reqwest(error: E, cause: &reqwest::Error) -> Self {
21        let is_transient = !(cause.is_builder() || cause.is_redirect() || cause.is_decode());
22
23        Self {
24            error,
25            retryable: is_transient,
26        }
27    }
28
29    pub(crate) fn from_status(error: E, status: StatusCode) -> Self {
30        let retryable = status.is_server_error()
31            || status == StatusCode::REQUEST_TIMEOUT
32            || status == StatusCode::TOO_MANY_REQUESTS;
33
34        Self { error, retryable }
35    }
36}
37
38/// How many times, and how slowly, a failed request is retried.
39#[derive(Debug, Clone, Copy, PartialEq)]
40pub struct RetryConfig {
41    max_retries: u32,
42    initial_interval: Duration,
43    ratio: f64,
44}
45
46impl Default for RetryConfig {
47    fn default() -> Self {
48        Self {
49            max_retries: MAX_RETRIES,
50            initial_interval: INITIAL_INTERVAL,
51            ratio: RATIO,
52        }
53    }
54}
55
56impl RetryConfig {
57    fn jittered_delay(&self, retry: u32) -> Duration {
58        let factor = rand::rng().random_range(1.0 - JITTER..=1.0 + JITTER);
59
60        self.scaled_delay(retry, factor)
61    }
62
63    fn scaled_delay(&self, retry: u32, factor: f64) -> Duration {
64        let secs = self.initial_interval.as_secs_f64() * self.ratio.powi(retry as i32) * factor;
65
66        Duration::try_from_secs_f64(secs).unwrap_or(Duration::MAX)
67    }
68
69    /// Run `op` according to this [RetryConfig]'s parameters.
70    ///
71    /// Non-retryable errors are returned eagerly and maxing out the number of retries will return
72    /// the final error encountered.
73    pub(crate) async fn run<T, E, F, Fut>(&self, mut op: F) -> Result<T, E>
74    where
75        E: Display,
76        F: FnMut() -> Fut,
77        Fut: Future<Output = Result<T, Failure<E>>>,
78    {
79        let mut retries = 0;
80
81        loop {
82            match op().await {
83                Ok(value) => return Ok(value),
84
85                Err(Failure { error, retryable }) if retryable && retries < self.max_retries => {
86                    let delay = self.jittered_delay(retries);
87                    retries += 1;
88                    warn!(
89                        retry = retries,
90                        max_retries = self.max_retries,
91                        ?delay,
92                        "request failed, retrying: {error}"
93                    );
94
95                    sleep(delay).await;
96                }
97
98                Err(Failure { error, .. }) => return Err(error),
99            }
100        }
101    }
102}

104#[cfg(test)]
105mod tests {
106    use super::*;
107    use std::sync::atomic::{AtomicU32, Ordering};
108
109    impl<E> Failure<E> {
110        fn new(error: E, retryable: bool) -> Self {
111            Self { error, retryable }
112        }
113    }
114
115    fn no_delay_cfg(max_retries: u32) -> RetryConfig {
116        RetryConfig {
117            max_retries,
118            initial_interval: Duration::ZERO,
119            ratio: 1.5,
120        }
121    }
122
123    #[test]
124    fn scaled_delay_grows_by_ratio_from_initial_interval() {
125        let cfg = RetryConfig::default();
126
127        assert_eq!(cfg.scaled_delay(0, 1.0), Duration::from_millis(2000));
128        assert_eq!(cfg.scaled_delay(1, 1.0), Duration::from_millis(3000));
129        assert_eq!(cfg.scaled_delay(2, 1.0), Duration::from_millis(4500));
130    }
131
132    #[tokio::test]
133    async fn run_returns_first_success_without_retrying() {
134        let calls = AtomicU32::new(0);
135
136        let res: Result<_, String> = no_delay_cfg(3)
137            .run(|| async {
138                calls.fetch_add(1, Ordering::SeqCst);
139                Ok("done")
140            })
141            .await;
142
143        assert_eq!(res.unwrap(), "done");
144        assert_eq!(calls.load(Ordering::SeqCst), 1);
145    }
146
147    #[tokio::test]
148    async fn run_retries_retryable_failures_until_success() {
149        let calls = AtomicU32::new(0);
150
151        let res: Result<_, String> = no_delay_cfg(5)
152            .run(|| async {
153                match calls.fetch_add(1, Ordering::SeqCst) {
154                    0..=2 => Err(Failure::new("transient".to_string(), true)),
155                    _ => Ok("done"),
156                }
157            })
158            .await;
159
160        assert_eq!(res.unwrap(), "done");
161        assert_eq!(calls.load(Ordering::SeqCst), 4);
162    }
163
164    #[tokio::test]
165    async fn run_returns_last_error_once_retries_are_exhausted() {
166        let calls = AtomicU32::new(0);
167
168        let res: Result<(), String> = no_delay_cfg(3)
169            .run(|| async {
170                let n = calls.fetch_add(1, Ordering::SeqCst);
171                Err(Failure::new(format!("attempt {n}"), true))
172            })
173            .await;
174
175        // The initial attempt plus 3 retries
176        assert_eq!(res.unwrap_err(), "attempt 3");
177        assert_eq!(calls.load(Ordering::SeqCst), 4);
178    }
