25/// Error variants that we can encounter when making graphQL requests to the platform API.
26#[derive(Debug, thiserror::Error)]
27pub enum Error {
28    /// The operation being run returned graphQL errors
29    #[error(
30        "graphql errors returned when running operation: {:?}",
31        .errors.iter().map(|e| &e.message).collect::<Vec<_>>()
32    )]
33    GraphqlErrors {
34        /// The raw graphQL errors that were returned
35        errors: Vec<GqlError>,
36    },
37
38    /// The operation being run returned unexpected graphQL extensions
39    #[error(
40        "unexpected graphql extensions returned running operation: {:?}",
41        .extensions.iter().map(|(k, v)| format!("{k}: {v:?}")).collect::<Vec<_>>()
42    )]
43    GraphqlExtensions {
44        /// The raw graphQL extensions returned with the response for the operation
45        extensions: serde_json::Map<String, serde_json::Value>,
46    },
47
48    /// The value for a header was invalid
49    #[error("the value provided for the header {key:?} is not a valid header value")]
50    InvalidHeaderValue {
51        /// The header key that was being set
52        key: String,
53    },
54
55    /// No data or errors were returned from the platform API in response to running an operation.
56    #[error("no data returned from graphql operation")]
57    NoData,
58
59    // Wrapped errors
60    /// An underlying error from the reqwest crate
61    #[error(transparent)]
62    Reqwest(#[from] reqwest::Error),
63
64    /// An underlying error from the serde-json crate
65    #[error(transparent)]
66    SerdeJson(#[from] serde_json::Error),
67}

69/// Serialization format for graphQL errors: <https://spec.graphql.org/October2021/#sec-Errors.Error-result-format>
70#[derive(Debug, Deserialize)]
71pub struct GqlError {
72    /// The string error message
73    pub message: String,
74    /// The locations within the operation being run where the errors occurred
75    #[serde(default)]
76    pub locations: Option<Vec<BTreeMap<String, u32>>>,
77    /// The path through the provided operation to the location of the error
78    #[serde(default)]
79    pub path: Option<Vec<serde_json::Value>>,
80}

82/// Helper trait for packaging up making a graphql request to the Apollo platform API and parsing the
83/// response into a more ergonomic type.
84///
85/// # Deriving GraphQLQuery
86/// Types that implement this trait need to derive the [GraphQLQuery] trait from [graphql_client]
87/// first. The schema file for the platform API (`engine@prod`) along with the query files used to
88/// define new queries are located in the top level `resources` directory of this crate. The details
89/// on how to work with the graphql_client crate are covered in [their docs][0] but there are couple
90/// of things we need to specify as boiler plate in order for the [PlatformQuery] trait to work:
91///   - variables need to derive [Clone].
92///   - responses need to derive [Deserialize].
93///   - The name of the query in your graphql file needs to exactly match the name of the Rust
94///     struct
95///
96/// The macro will then generate a module with a name that is the snake_case transform of your
97/// struct (so `MyQuery` becomes `my_query`) that contains all of the generated types needed to
98/// make your request and parse the response.
99///
100/// # Example
101/// ```graphql
102/// # contents of queries/my-query.graphql
103/// query MyQuery(
104///   $my_arg: String!,
105///   $my_other_arg: Int,
106///   $my_enums: [MyEnum]!
107/// ) {
108///   foo(arg: $my_arg) {
109///     bar(arg: $my_other_arg, enums: $my_enums) {
110///       myData # a nullable string
111///      }
112///   }
113/// }
114/// ```
115///
116/// ```ignore
117/// #[derive(GraphQLQuery)]
118/// #[graphql(
119///     schema_path = "resources/engine-prod-schema.graphql",
120///     query_path = "queries/my-query.graphql",
121///     response_derives = "Deserialize",
122///     variables_derives = "Clone"
123/// )]
124/// pub struct MyQuery;
125///
126/// impl PlatformQuery for MyQuery {
127///     type Output = String;
128///     type Error = &'static str;
129///
130///     fn try_parse(
131///         data: Self::ResponseData,
132///         _vars: my_query::Variables
133///     ) -> Result<String, &'static str> {
134///         // The nested structure of Self::ResponseData matches the your graphQL query
135///         let maybe_my_data = data.foo.bar.my_data;
136///
137///         match maybe_my_data {
138///             Some(s) => Ok(s),
139///             None => Err("myData was null"),
140///         }
141///     }
142/// }
143///
144/// // Making a request
145/// let vars = my_query::Variables {
146///     my_arg: "some arg".to_string(),      // non-nullable types in graphql map to Rust types
147///     my_other_arg: Some(42),              // nullable types map to options
148///     my_enums: vec![my_query::MyEnum::A], // enums are available in the generated module
149/// };
150/// let api_key = "...";
151/// let client = ReqwestClient::new_prod(api_key);
152///
153/// let parsed_response = MyQuery::fetch(vars, &client).await?;
154/// ```
155///
156///   [0]: https://github.com/graphql-rust/graphql-client?tab=readme-ov-file#getting-started
157#[allow(async_fn_in_trait)]
158pub trait PlatformQuery: GraphQLQuery + Sized
159where
160    Self::Variables: Clone,
161{
162    /// The output type returned from `try_parse`
163    type Output;
164    /// The error type returned from `try_parse`
165    type Error: From<Error>;
166
167    /// Attempt to parse the raw data returned from a graphQL query into a usable Rust type.
168    fn try_parse(
169        data: Self::ResponseData,
170        variables: Self::Variables,
171    ) -> Result<Self::Output, Self::Error>;
172
173    /// Execute this query and parse the returned data
174    async fn fetch(
175        variables: Self::Variables,
176        client: &impl Client,
177    ) -> Result<Self::Output, Self::Error> {
178        let raw = client.execute_operation::<Self>(variables.clone()).await?;
179
180        Self::try_parse(raw, variables)
181    }
182}

184/// An API client that can make requests to the Apollo platform API.
185#[allow(async_fn_in_trait)]
186pub trait Client {
187    /// POST a GraphQL operation to the studio API with appropriate headers, returning the raw JSON
188    /// response.
189    ///
190    /// [Client::execute_operation] is used to handle creating the POST body and parsing the
191    /// response.
192    async fn post_operation(&self, body: &impl Serialize) -> Result<serde_json::Value, Error>;
193
194    /// Helper for making an API request to studio and handling any serialization or graphQL errors so
195    /// that implementations of [PlatformQuery::try_parse] only need to care about mapping valid data.
196    async fn execute_operation<T>(&self, variables: T::Variables) -> Result<T::ResponseData, Error>
197    where
198        T: GraphQLQuery,
199    {
200        let body = T::build_query(variables);
201        let raw = self.post_operation(&body).await?;
202
203        let resp: GqlResponse<T::ResponseData> = serde_json::from_value(raw)?;
204
205        if !resp.errors.is_empty() {
206            error!("errors returned when running graphql operation");
207            return Err(Error::GraphqlErrors {
208                errors: resp.errors,
209            });
210        } else if !resp.extensions.is_empty() {
211            error!("unexpected extensions returned when running graphql operation");
212            return Err(Error::GraphqlExtensions {
213                extensions: resp.extensions,
214            });
215        }
216
217        return resp.data.ok_or(Error::NoData);
218
219        // Serde type for parsing the graphql response from the server
220
221        #[derive(Debug, Deserialize)]
222        struct GqlResponse<T> {
223            data: Option<T>,
224            #[serde(default)]
225            errors: Vec<GqlError>,
226            #[serde(default)]
227            extensions: serde_json::Map<String, serde_json::Value>,
228        }
229    }
230}

232impl Client for PlatformClient {
233    async fn post_operation(&self, body: &impl Serialize) -> Result<serde_json::Value, Error> {
234        let mut api_key = match HeaderValue::from_str(&self.api_key) {
235            Ok(val) => val,
236            Err(_) => {
237                return Err(Error::InvalidHeaderValue {
238                    key: "x-api-key".to_string(),
239                });
240            }
241        };
242
243        api_key.set_sensitive(true);
244
245        let mut req_builder = self
246            .inner
247            .post(&self.url)
248            .json(body)
249            .header("x-api-key", api_key)
250            .header("apollographql-client-name", "runtime-testing-framework")
251            .header("apollographql-client-version", "0.1.0");
252
253        if self.sudo {
254            req_builder = req_builder.header("apollo-sudo", "true");
255        }
256
257        let raw = req_builder
258            .send()
259            .await?
260            .json::<serde_json::Value>()
261            .await?;
262
263        Ok(raw)
264    }
