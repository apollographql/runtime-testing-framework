8/// Error variants that we can encounter when interacting with Apollo graphOS.
9#[derive(Debug, thiserror::Error)]
10pub enum Error {
11    /// Error variants that we can encounter when making graphQL requests to the platform API.
12    #[error(transparent)]
13    Fetch(#[from] supergraph::FetchError),
14
15    /// A graphQL error was encountered while attempting to pull supergraph details
16    #[error(transparent)]
17    Graphql(#[from] platform_query::Error),
18
19    /// IO errors
20    #[error(transparent)]
21    Io(#[from] io::Error),
22
23    /// GraphQL parsing errors
24    #[error("error parsing GraphQL document")]
25    InvalidDocument {
26        /// Errors encountered while parsing or validating a GraphQL document
27        errors: DiagnosticList,
28        /// Context for the document that the errors occurred in
29        context: String,
30    },
31
32    /// JSON parsing errors
33    #[error(transparent)]
34    Json(#[from] serde_json::Error),
35}
36
37/// Alias for a [Result][std::result::Result] where the error variant is an [Error].
38pub type Result<T> = std::result::Result<T, Error>;
39
40/// An API client backed by [reqwest::Client] that can make requests to the Apollo platform API.
41pub struct PlatformClient {
42    pub(crate) inner: reqwest::Client,
43    pub(crate) url: String,
44    pub(crate) api_key: String,
45    pub(crate) sudo: bool,
46}
47
48// Custom Debug impl to prevent us dumping the api key
49impl fmt::Debug for PlatformClient {
50    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
51        f.debug_struct("PlatformClient")
52            .field("url", &self.url)
53            .finish()
54    }
55}
