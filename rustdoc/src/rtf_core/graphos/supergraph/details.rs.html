11/// An error encountered while attempting to fetch details for a supergraph from the platform API.
12#[derive(Debug, thiserror::Error)]
13#[error("unable to fetch details for {graph_id}@{variant}: {cause}")]
14pub struct FetchError {
15    /// The graph ID being fetched
16    pub graph_id: String,
17    /// The graph variant being fetched
18    pub variant: String,
19    /// The error encountered
20    pub cause: FetchErrorCause,
21}
22
23/// Causes of errors when fetching supergraph details from studio.
24/// Should always be wrapped in a [FetchError] providing the graph ID and variant that the error is
25/// associated with.
26#[derive(Debug, thiserror::Error)]
27pub enum FetchErrorCause {
28    /// An empty string was returned for the supergraph SDL
29    #[error("empty supergraph schema returned")]
30    EmptySchema,
31
32    /// A graphQL error was encountered while attempting to pull supergraph details
33    #[error(transparent)]
34    Graphql(#[from] platform_query::Error),
35
36    /// There was no build information for the latest launch
37    #[error("no build found as part of the latest launch found")]
38    NoBuild,
39
40    /// There was no latest launch
41    #[error("no latest launch found")]
42    NoLaunch,
43
44    /// No subgraphs were found
45    #[error("no subgraphs found")]
46    NoSubgraphs,
47
48    /// Offline licenses are not enabled for this organisation
49    #[error("offline licenses are not enabled for this organisation")]
50    OfflineLicenseNotEnabled,
51
52    /// The requested operation could not be found in studio
53    #[error("not a known operation in studio")]
54    UnknownOperation,
55
56    /// The requested organisation could not be found in studio
57    #[error("not a known organisation in studio")]
58    UnknownOrganisation,
59
60    /// The requested supergraph could not be found in Studio
61    #[error("not a known supergraph in studio")]
62    UnknownSupergraph,
63
64    /// The requested supergraph variant could not be found in Studio
65    #[error("not a known variant in studio")]
66    UnknownVariant,
67}

78/// Supergraph schema details
79///
80/// To pull supergraph details from studio see [SupergraphDetails::fetch].
81#[derive(Debug, Clone, PartialEq, Eq)]
82pub struct SupergraphDetails {
83    /// The GraphOS graphID for this supergraph
84    pub graph_id: String,
85    /// The GraphOS variant for this supergraph
86    pub variant: String,
87    /// The full SDL format schema for this supergraph
88    pub supergraph_sdl: String,
89    /// The list of subgraphs associated with this supergraph
90    pub subgraphs: Vec<Subgraph>,
91}
92
93impl SupergraphDetails {
94    /// Attempt to fetch the data we need for writing out the supergraph and subgraph schemas
95    /// associated with a particular graph ID and variant.
96    ///
97    /// # Errors
98    /// This method will error if the underlying graphQL request fails or if the data returned is
99    /// insufficient for us to construct [SupergraphDetails]. Failure modes are enumerated and
100    /// documented as part of [FetchErrorCause].
101    pub async fn fetch(
102        graph_id: impl Into<String>,
103        variant: impl Into<String>,
104        client: &impl platform_query::Client,
105    ) -> Result<Self, FetchError> {
106        let graph_id = graph_id.into();
107        let variant = variant.into();
108
109        info!(%graph_id, %variant, "pulling supergraph details");
110        let graph_ref = raw_supergraph_details::Variables {
111            graph_id: graph_id.clone(),
112            variant: variant.clone(),
113        };
114
115        RawSupergraphDetails::fetch(graph_ref, client)
116            .await
117            .map_err(|cause| FetchError {
118                graph_id,
119                variant,
120                cause,
121            })
122    }
123
124    /// Attempt to rewrite the subgraph url directives in this schema to use the provided urls
125    /// instead.
126    pub fn rewrite_subgraph_urls(
127        &mut self,
128        subgraph_urls: &HashMap<String, String>,
129    ) -> Result<(), &'static str> {
130        match rewrite_subgraph_urls(&self.supergraph_sdl, subgraph_urls) {
131            Some(new_sdl) => {
132                self.supergraph_sdl = new_sdl;
133                Ok(())
134            }
135
136            None => {
137                error!("Unable to rewrite subgraph URLs");
138                Err("Unable to rewrite subgraph URLs")
139            }
140        }
141    }
142
143    /// Write out only the schemas held in this [SupergraphDetails].
144    pub fn write_schemas(&self, out_dir: &Path) -> graphos::Result<()> {
145        debug!("writing supergraph SDL");
146        fs::write(out_dir.join("supergraph.graphql"), &self.supergraph_sdl)?;
147
148        debug!("writing supergraph SDLs");
149        let sg_dir = out_dir.join("subgraphs");
150        if sg_dir.exists() && !sg_dir.is_dir() {
151            return Err(graphos::Error::Io(io::Error::new(
152                io::ErrorKind::NotADirectory,
153                format!("{} is not a directory", sg_dir.display()),
154            )));
155        } else if !sg_dir.exists() {
156            debug!("  creating subgraph directory: {}", sg_dir.display());
157            fs::create_dir(&sg_dir)?;
158        }
159
160        for sg in self.subgraphs.iter() {
161            debug!("  writing subgraph: {}", sg.name);
162            fs::write(sg_dir.join(format!("{}.graphql", sg.name)), &sg.sdl)?;
163        }
164
165        Ok(())
166    }
167}

182impl PlatformQuery for RawSupergraphDetails {
183    type Output = SupergraphDetails;
184    type Error = FetchErrorCause;
185
186    fn try_parse(
187        data: Self::ResponseData,
188        raw_supergraph_details::Variables { graph_id, variant }: raw_supergraph_details::Variables,
189    ) -> Result<SupergraphDetails, FetchErrorCause> {
190        use raw_supergraph_details::RawSupergraphDetailsServiceVariantLatestApprovedLaunchBuildResult as BuildRes;
191
192        let graph_variant = data
193            .service
194            .ok_or(FetchErrorCause::UnknownSupergraph)?
195            .variant
196            .ok_or(FetchErrorCause::UnknownVariant)?;
197
198        let build_result = graph_variant
199            .latest_approved_launch
200            .ok_or(FetchErrorCause::NoLaunch)?
201            .build
202            .and_then(|build| build.result)
203            .ok_or(FetchErrorCause::NoBuild)?;
204
205        let supergraph_sdl = match build_result {
206            BuildRes::BuildFailure => return Err(FetchErrorCause::NoBuild),
207            BuildRes::BuildSuccess(res) => res.core_schema.core_document,
208        };
209
210        if supergraph_sdl.is_empty() {
211            return Err(FetchErrorCause::EmptySchema);
212        }
213
214        let subgraphs: Vec<_> = match graph_variant.source_variant {
215            Some(v) => {
216                info!("variant is a contract; fetching subgraph schemas from source variant");
217                v.subgraphs
218                    .ok_or(FetchErrorCause::NoSubgraphs)?
219                    .into_iter()
220                    .map(|raw| Subgraph {
221                        name: raw.name,
222                        sdl: raw.active_partial_schema.sdl,
223                    })
224                    .collect()
225            }
226            None => graph_variant
227                .subgraphs
228                .ok_or(FetchErrorCause::NoSubgraphs)?
229                .into_iter()
230                .map(|raw| Subgraph {
231                    name: raw.name,
232                    sdl: raw.active_partial_schema.sdl,
233                })
234                .collect(),
235        };
236
237        if subgraphs.is_empty() {
238            return Err(FetchErrorCause::NoSubgraphs);
239        }
240
241        Ok(SupergraphDetails {
242            graph_id: graph_id.to_string(),
243            variant: variant.to_string(),
244            supergraph_sdl,
245            subgraphs,
246        })
247    }
248}

250/// Rewrite the given supergraph SDL to set the provided subgraph URLs in place of what is
251/// currently there.
252fn rewrite_subgraph_urls(sdl: &str, subgraph_urls: &HashMap<String, String>) -> Option<String> {
253    let mut schema = Schema::parse(sdl, "supergraph.graphql").ok()?;
254
255    let join_graph_enum = match schema.types.get_mut("join__Graph") {
256        Some(ExtendedType::Enum(e)) => e,
257        Some(_) => {
258            warn!("join__Graph exists but is not an enum type");
259            return None;
260        }
261        None => {
262            warn!("supergraph SDL missing join__Graph enum");
263            return None;
264        }
265    };
266
267    let mut rewritten_count = 0;
268
269    for value_def in join_graph_enum.get_mut()?.values.values_mut() {
270        for directive in value_def.get_mut()?.directives.0.iter_mut() {
271            if directive.name.as_str() != "join__graph" {
272                continue;
273            }
274
275            let sg_name = directive.specified_argument_by_name("name")?;
276            let sg_name_str = sg_name.as_str()?;
277
278            let url = match subgraph_urls.get(sg_name_str) {
279                Some(u) => u,
280                None => {
281                    warn!("URL map missing entry for a subgraph in the schema");
282                    return None;
283                }
284            };
285
286            *directive.get_mut()?.specified_argument_by_name_mut("url")? =
287                Node::new(Value::String(url.clone()));
288
289            rewritten_count += 1;
290        }
291    }
292
293    debug!(
294        rewritten = rewritten_count,
295        provided = subgraph_urls.len(),
296        "rewrote subgraph URLs in supergraph SDL"
297    );
298
299    Some(schema.to_string())
300}
