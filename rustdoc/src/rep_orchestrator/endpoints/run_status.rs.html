1//! Fetch the status of a given [TestRun] by its UUID.
2use crate::{Error, Result, conn, db::TestRun, response_types::TestRunSummary};
3use axum::{Json, extract::Path};
4use uuid::Uuid;
5
6pub async fn handler(Path(id): Path<Uuid>) -> Result<Json<TestRunSummary>> {
7    let conn = conn!();
8
9    match TestRun::get_by_uuid(&id, conn).await? {
10        Some(ex) => Ok(Json(ex.try_into_summary(conn).await?)),
11        None => Err(Error::UnknownTestRun { id }),
12    }
13}
