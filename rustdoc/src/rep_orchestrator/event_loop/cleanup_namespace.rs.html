1use crate::{
2    db::TestExecution,
3    event_loop::{Error, EventData, Result},
4    k8s,
5};
6use tracing::info;
7
8pub(super) async fn try_run<K>(
9    test_execution: TestExecution,
10    clients: K,
11) -> Result<Option<EventData>>
12where
13    K: k8s::Client,
14{
15    let execution_id = test_execution.uuid();
16    info!(%execution_id, "deleting namespace");
17    clients
18        .delete_workload_namespace(&execution_id.to_string())
19        .await
20        .map_err(|error| Error::DeleteNamespace { error })?;
21
22    Ok(None)
23}
