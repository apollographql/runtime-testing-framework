23const OUTPUT_COLLECTOR_SCRIPT: &str = r#"
24echo "Waiting for scenario to complete..."
25while [ ! -f /shared/scenario-exited ]; do
26  sleep 5
27done

29echo "Scenario finished. Showing output..."

34const RESOLVE_SCRIPT: &str = r#"
35set -e
36
37echo "Resolving scenario..."
38rtf resolve scenario /scenario/scenario.yaml --outdir /shared/providers
39
40echo "Writing out user scenario script..."
41cat << 'EOF' > /shared/scenario.sh
42#!/bin/sh
43set -ex
44
45echo "Sourcing RTF env vars"
46. /shared/providers/scenario.env
47
48export OUTDIR=/shared/output
49export RTF_OUTPUT="$OUTDIR/RTF_OUTPUT"
50
51echo "Running user specified scenario..."
52__SCENARIO_COMMAND__
53EOF
54
55echo "Writing out run script..."
56cat << 'EOF' > /shared/run.sh
57#!/bin/sh
58set -ex
59
60trap 'touch /shared/scenario-exited' EXIT
61echo "Exit trap installed"
62
63mkdir -p /shared/output
64
65echo "Running test scenario"
66{ /shared/scenario.sh 2>&1; echo $? > /shared/scenario_exit_status; } |
67  tee /shared/output/output.log
68EOF
69
70chmod -R 777 /shared/scenario.sh
71chmod -R 777 /shared/run.sh
72
73echo "Contents of scenario script"
74cat /shared/scenario.sh
75
76ls -laR /shared/providers/
77"#;

79/// Create a new [JobSpec] for the given [DockerScenario].
80pub fn scenario_job(execution_id: &Uuid, scenario: &DockerScenario) -> JobSpec {
81    JobSpec {
82        backoff_limit: Some(0), // don't retry failed scenarios
83        ttl_seconds_after_finished: Some(TTL_SECONDS_AFTER_FINISHED),
84        template: PodTemplateSpec {
85            metadata: Some(ObjectMeta {
86                labels: Some(BTreeMap::from([(
87                    EXECUTION_ID_LABEL.to_owned(),
88                    execution_id.to_string(),
89                )])),
90                ..Default::default()
91            }),
92            spec: Some(PodSpec {
93                restart_policy: Some("Never".to_owned()),
94                init_containers: Some(vec![init_container_spec(scenario)]),
95                containers: vec![
96                    scenario_run_container_spec(scenario),
97                    output_collector_container_spec(),
98                ],
99                volumes: Some(scenario_volumes()),
100                ..Default::default()
101            }),
102        },
103        ..Default::default()
104    }
105}
106
107fn init_container_spec(scenario: &DockerScenario) -> Container {
108    Container {
109        name: "rtf-resolve".to_owned(),
110        image: Some(TOOLBOX_IMAGE.to_owned()),
111        command: Some(vec!["/bin/sh".to_owned(), "-c".to_owned()]),
112        args: Some(vec![
113            RESOLVE_SCRIPT.replace("__SCENARIO_COMMAND__", &scenario.command()),
114        ]),
115        volume_mounts: Some(vec![
116            VolumeMount {
117                name: VOLUME_MOUNT_NAME_CONFIG.to_owned(),
118                mount_path: "/scenario".to_owned(),
119                read_only: Some(true),
120                ..Default::default()
121            },
122            VolumeMount {
123                name: VOLUME_MOUNT_NAME_SHARED.to_owned(),
124                mount_path: SHARED_DIR_PATH.to_owned(),
125                read_only: Some(false),
126                ..Default::default()
127            },
128        ]),
129        ..Default::default()
130    }
131}
132
133fn scenario_run_container_spec(scenario: &DockerScenario) -> Container {
134    Container {
135        name: "scenario-runner".to_owned(),
136        image: Some(scenario.docker_image()),
137        command: Some(vec!["/bin/sh".to_owned(), "/shared/run.sh".to_owned()]),
138        working_dir: Some(SHARED_DIR_PATH.to_owned()),
139        volume_mounts: Some(vec![VolumeMount {
140            name: VOLUME_MOUNT_NAME_SHARED.to_owned(),
141            mount_path: SHARED_DIR_PATH.to_owned(),
142            read_only: Some(false),
143            ..Default::default()
144        }]),
145        lifecycle: Some(Lifecycle {
146            pre_stop: Some(LifecycleHandler {
147                exec: Some(ExecAction {
148                    command: Some(vec![
149                        "/bin/sh".to_owned(),
150                        "-c".to_owned(),
151                        "sync".to_owned(),
152                    ]),
153                }),
154                ..Default::default()
155            }),
156            ..Default::default()
157        }),
158        ..Default::default()
159    }
160}
