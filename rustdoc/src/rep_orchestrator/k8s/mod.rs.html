40/// Kubernetes API actions that only interact with the management cluster.
41pub trait ManagementClient: Clone + Send + Sync + 'static {
42    /// Create a new argo [Workflow] in the management cluster for provisioning an ephemeral
43    /// namespace in the workload cluster and deploying services into it as defined by an RTF
44    /// Test Plan.
45    fn create_argo_workflow(
46        &self,
47        execution_id: &Uuid,
48        spec: WorkflowSpec,
49    ) -> impl Future<Output = Result<Workflow>> + Send;
50}
51
52/// Kubernetes API actions that only interact with the workload cluster.
53pub trait WorkloadClient: Clone + Send + Sync + 'static {
54    /// Create a new k8s [Job] for running an RTF Scenario in an ephemeral namespace within the
55    /// workload cluster.
56    fn create_job(
57        &self,
58        ns: &str,
59        name: &str,
60        execution_id: &Uuid,
61        spec: JobSpec,
62    ) -> impl Future<Output = Result<Job>> + Send;
63
64    /// Wait for a k8s [Job] running within the workload cluster to reach a terminal state,
65    /// selecting the job by its execution ID label.
66    fn wait_for_job(
67        &self,
68        ns: &str,
69        execution_id: &Uuid,
70    ) -> impl Future<Output = WatchOutcome> + Send;
71
72    /// Delete an ephemeral namespace within the workload cluster.
73    fn delete_workload_namespace(&self, ns: &str) -> impl Future<Output = Result<()>> + Send;
74}
75
76/// Kubernetes API actions that interact with both the management and workload clusters.
77pub trait FullClient: ManagementClient + WorkloadClient {
78    /// Wait for a [Workflow] running within the management cluster to reach a terminal state,
79    /// selecting the workflow by its execution ID label. The workload cluster is also watched
80    /// so that pods stuck in an unrunnable state can short-circuit the wait.
81    fn wait_for_workflow(&self, execution_id: &Uuid) -> impl Future<Output = WatchOutcome> + Send;
82}

84/// Terminal states for argo [Workflow]s and k8s [Job]s.
85#[derive(Debug, Clone)]
86pub enum WatchOutcome {
87    Succeeded,
88    Failed(String),
89    ContainerUnrunnable(String),
90    WatcherError(String),
91    StreamClosed,
92}
93
94impl fmt::Display for WatchOutcome {
95    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
96        match self {
97            Self::Succeeded => write!(f, "Succeeded"),
98            Self::Failed(msg) => write!(f, "Failed ({msg})"),
99            Self::ContainerUnrunnable(reason) => {
100                write!(f, "pod stuck in unrunnable waiting state: {reason}")
101            }
102            Self::WatcherError(msg) => write!(f, "Watcher error ({msg})"),
103            Self::StreamClosed => write!(f, "Watcher stream closed unexpectedly"),
104        }
105    }
106}
