51    pub entrypoint: String,
52    pub on_exit: String,

54    pub volumes: Vec<Volume>,
55    pub ttl_strategy: Option<TtlStrategy>,

59impl WorkflowSpec {
60    pub fn for_execution(ex: &TestExecution, orchestrator_url: &str) -> Self {
61        let execution_id = ex.uuid();
62        let configmap_name = env_configmap_name(&execution_id);
63        let env_vars = ex.toolbox_env_vars(orchestrator_url);
64        let namespace = execution_id.to_string();
65
66        Self {
67            service_account_name: "argo-workflow".to_owned(),
68            entrypoint: "main".to_owned(),
69            on_exit: "cleanup".to_owned(),
70            templates: vec![
71                TemplateDef::Main(MainTemplate::new()),
72                TemplateDef::Task(create_namespace(&namespace, env_vars.clone())),
73                TemplateDef::Task(create_pull_secret(&namespace, env_vars.clone())),
74                TemplateDef::Task(create_service_account(&namespace, env_vars.clone())),
75                TemplateDef::Task(deploy_environment(
76                    &configmap_name,
77                    &namespace,
78                    env_vars.clone(),
79                )),
80                TemplateDef::Task(cleanup(&configmap_name, env_vars)),
81            ],
82            volumes: vec![Volume {
83                name: "kubeconfig".into(),
84                secret: Some(SecretVolumeSource {
85                    secret_name: Some("workload-kubeconfig".into()),
86                    ..Default::default()
87                }),
88                ..Default::default()
89            }],
90            ttl_strategy: Some(TtlStrategy {
91                seconds_after_completion: Some(TTL_SECONDS_AFTER_FINISHED),
92            }),
93            // Delete the workflow pods if the workflow succeeds
94            // Retains failed pods for debugging
95            pod_g_c: Some(PodGC {
96                strategy: "OnWorkflowSuccess".to_string(),
97            }),
98        }
99    }
100}

159#[derive(Debug, Clone, Deserialize, Serialize, JsonSchema)]
160pub struct TaskTemplate {
161    pub name: String,
162    pub container: Container,
163    pub volumes: Option<Vec<Volume>>,
164}
165
166impl TaskTemplate {
167    /// Build a task container that runs `rep-orchestrator-cli <args...>` inside the toolbox image.
168    fn new(
169        name: &str,
170        cli_args: Vec<String>,
171        volume_mounts: Vec<VolumeMount>,
172        volumes: Option<Vec<Volume>>,
173        env: Vec<EnvVar>,
174    ) -> Self {
175        Self {
176            name: name.into(),
177            container: Container {
178                name: name.into(),
179                image: Some(TOOLBOX_IMAGE.into()),
180                command: Some(vec![CLI_BINARY.to_owned()]),
181                args: Some(cli_args),
182                volume_mounts: Some(volume_mounts),
183                env: Some(env),
184                ..Default::default()
185            },
186            volumes,
187        }
188    }
189}

200fn create_namespace(namespace: &str, env: Vec<EnvVar>) -> TaskTemplate {
201    TaskTemplate::new(
202        "create-namespace",
203        vec![
204            "create-namespace".into(),
205            "--namespace".into(),
206            namespace.into(),
207            "--kubeconfig".into(),
208            KUBECONFIG_PATH.into(),
209        ],
210        vec![kubeconfig_volume_mount()],
211        None,
212        env,
213    )
214}

216fn create_pull_secret(namespace: &str, env: Vec<EnvVar>) -> TaskTemplate {
217    TaskTemplate::new(
218        "create-pull-secret",
219        vec![
220            "create-pull-secret".into(),
221            "--namespace".into(),
222            namespace.into(),
223            "--kubeconfig".into(),
224            KUBECONFIG_PATH.into(),
225            "--docker-config".into(),
226            DOCKER_CONFIG_PATH.into(),
227        ],
228        vec![
229            kubeconfig_volume_mount(),
230            VolumeMount {
231                name: "gcr-secret".into(),
232                mount_path: "/gcr-secret".into(),
233                read_only: Some(true),
234                ..Default::default()
235            },
236        ],
237        Some(vec![Volume {
238            name: "gcr-secret".into(),
239            secret: Some(SecretVolumeSource {
240                secret_name: Some("gcr-secret".into()),
241                items: Some(vec![KeyToPath {
242                    key: ".dockerconfigjson".into(),
243                    path: "config.json".into(),
244                    ..Default::default()
245                }]),
246                ..Default::default()
247            }),
248            ..Default::default()
249        }]),
250        env,
251    )
252}
