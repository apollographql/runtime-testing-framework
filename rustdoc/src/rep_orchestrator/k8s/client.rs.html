22// Container waiting reasons that indicate a pod is permanently stuck and will never produce an
23// exit code. These surface as `containerStatuses[].state.waiting.reason` in the pod status.
24//
25// Unlike transient failures (OOMKilled, CrashLoopBackOff, Error), which produce an exit code and
26// propagate to the parent Job/Workflow as Failed, these states leave the container in Waiting
27// indefinitely — the Job/Workflow never transitions, and the execution hangs.
28//
29// The reason strings are kubelet implementation details, not part of the Kubernetes API contract.
30// Image-related reasons originate in pkg/kubelet/images/types.go [1]; container creation reasons
31// in pkg/kubelet/kuberuntime/kuberuntime_container.go [2]. The official pod lifecycle docs
32// describe the Waiting state but do not enumerate possible reason values [3].
33//
34// This list covers known cases but is not exhaustive — other Waiting reasons may exist that
35// would also cause executions to hang.
36//
37// [1] https://github.com/kubernetes/kubernetes/blob/cec8f06d2ce283d1563d37849619887aa2b11c9f/pkg/kubelet/images/types.go
38// [2] https://github.com/kubernetes/kubernetes/blob/cec8f06d2ce283d1563d37849619887aa2b11c9f/pkg/kubelet/kuberuntime/kuberuntime_container.go
39// [3] https://kubernetes.io/docs/concepts/workloads/pods/pod-lifecycle/#container-states
40const UNRUNNABLE_REASONS: &[&str] = &[
41    "ImagePullBackOff",
42    "ErrImagePull",
43    "ErrImageNeverPull",
44    "InvalidImageName",
45    "CreateContainerConfigError",
46    "CreateContainerError",
47];

58    /// Construct clients where the management client uses the pod's own in-cluster credentials
59    /// and the workload client uses an explicit kubeconfig file. Used in prod where the
60    /// orchestrator runs inside the mgmt cluster.
61    pub async fn try_new(workload_path: &str, workload_context: &str) -> Result<Self> {
62        let management = Client::try_from(Config::incluster_env()?)?;
63        let kfg = Kubeconfig::read_from(workload_path)?;
64
65        Ok(Self {
66            management,
67            workload: client_for_context(kfg, workload_context).await?,
68        })
69    }

71    /// Helper for obtaining an [Api] client associated with the appropriate cluster namespace.
72    pub fn namespaced_api<K>(&self, cluster: Cluster, ns: &str) -> Api<K>
73    where
74        K: Resource<Scope = NamespaceResourceScope>,
75        <K as Resource>::DynamicType: Default,
76    {
77        let client = match cluster {
78            Cluster::Management => self.management.clone(),
79            Cluster::Workload => self.workload.clone(),
80        };
81
82        Api::namespaced(client, ns)
83    }

216async fn client_for_context(kfg: Kubeconfig, context: &str) -> Result<Client> {
217    let client = Client::try_from(
218        Config::from_custom_kubeconfig(
219            kfg,
220            &KubeConfigOptions {

222                ..Default::default()
223            },
224        )
225        .await?,
226    )?;

292    async fn wait_for_workflow(&self, execution_id: &Uuid) -> WatchOutcome {
293        // Watch for the workflow to complete
294        let wf_api: Api<Workflow> = self.namespaced_api(Cluster::Management, CLUSTER_API_NAMESPACE);
295        let wf_labels = format!("{EXECUTION_ID_LABEL}={execution_id}");
296        let wf_config = watcher::Config::default().labels(&wf_labels);
297        let mut wf_stream = pin!(watcher(wf_api, wf_config).applied_objects());
298
299        // Watch the argo workflow pods to make sure they do not get into an unrunnable state
300        // This is not handled particularly well by Argo natively
301        let mgmt_pod_api: Api<Pod> =
302            self.namespaced_api(Cluster::Management, CLUSTER_API_NAMESPACE);
303        let wf_pod_labels = format!("workflows.argoproj.io/workflow=provision-env-{execution_id}");
304        let wf_pod_config = watcher::Config::default().labels(&wf_pod_labels);
305        let mut mgmt_pod_stream = pin!(watcher(mgmt_pod_api, wf_pod_config).applied_objects());
306
307        // Watch the pods being deployed to the workload cluster to make sure they do not get into an unrunnable state
308
309        // TODO This should move to the orchestrator cli once we switch to using that and not the inline
310        // shell logic
311        let workload_ns = execution_id.to_string();
312        let workload_pod_api: Api<Pod> = self.namespaced_api(Cluster::Workload, &workload_ns);
313        let workload_pod_config = watcher::Config::default();
314        let mut workload_pod_stream =
315            pin!(watcher(workload_pod_api, workload_pod_config).applied_objects());
316
317        loop {
318            tokio::select! {
319                Some(res) = wf_stream.next() => {
320                    if let Some(outcome) = handle_workflow_event(res) { return outcome; }
321                }
322                Some(res) = mgmt_pod_stream.next() => {
323                    if let Some(outcome) = handle_pod_watch_event(res) { return outcome; }
324                }
325                Some(res) = workload_pod_stream.next() => {
326                    if let Some(outcome) = handle_pod_watch_event(res) { return outcome; }
327                }
328                else => return WatchOutcome::StreamClosed
329            }
330        }
331    }

333    async fn wait_for_job(&self, ns: &str, execution_id: &Uuid) -> WatchOutcome {
334        // Watch for the job to complete
335        let job_api: Api<Job> = self.namespaced_api(Cluster::Workload, ns);
336        let job_labels = format!("{EXECUTION_ID_LABEL}={execution_id}");
337        let job_config = watcher::Config::default().labels(&job_labels);
338        let mut job_stream = pin!(watcher(job_api, job_config).applied_objects());
339
340        // Watch the pods being deployed to the workload cluster to make sure they do not get into an unrunnable state
341        let pod_api: Api<Pod> = self.namespaced_api(Cluster::Workload, ns);
342        let pod_labels = format!("{EXECUTION_ID_LABEL}={execution_id}");
343        let pod_config = watcher::Config::default().labels(&pod_labels);
344        let mut pod_stream = pin!(watcher(pod_api, pod_config).applied_objects());
345
346        loop {
347            tokio::select! {
348                Some(res) = job_stream.next() => {
349                    if let Some(outcome) = handle_job_event(res) { return outcome; }
350                }
351                Some(res) = pod_stream.next() => {
352                    if let Some(outcome) = handle_pod_watch_event(res) { return outcome; }
353                }
354                else => return WatchOutcome::StreamClosed
355            }
356        }
357    }

359    async fn delete_workload_namespace(&self, ns: &str) -> Result<()> {
360        let crb_api: Api<ClusterRoleBinding> = Api::all(self.workload.clone());
361        if let Err(e) = crb_api
362            .delete(&format!("{OUTPUT_COLLECTOR}-{ns}"), &Default::default())
363            .await
364        {
365            warn!("failed to delete ClusterRoleBinding {OUTPUT_COLLECTOR}-{ns}: {e}");
366        }
367
368        let api: Api<Namespace> = Api::all(self.workload.clone());
369        api.delete(ns, &Default::default()).await?;
370
371        Ok(())
372    }

375fn check_pod_for_unrunnable(pod: &Pod) -> Option<WatchOutcome> {
376    let statuses = pod.status.as_ref()?.container_statuses.as_deref()?;
377    for status in statuses {
378        if let Some(waiting) = status.state.as_ref()?.waiting.as_ref()
379            && let Some(reason) = &waiting.reason
380            && UNRUNNABLE_REASONS.contains(&reason.as_str())
381        {
382            return Some(WatchOutcome::ContainerUnrunnable(reason.clone()));
383        }
384    }
385
386    None
387}
388
389fn handle_pod_watch_event(res: result::Result<Pod, watcher::Error>) -> Option<WatchOutcome> {
390    match res {
391        Ok(pod) => check_pod_for_unrunnable(&pod),
392        Err(e) => Some(WatchOutcome::WatcherError(e.to_string())),
393    }
394}

396fn handle_workflow_event(res: result::Result<Workflow, watcher::Error>) -> Option<WatchOutcome> {
397    let wf = match res {
398        Ok(wf) => wf,
399        Err(e) => return Some(WatchOutcome::WatcherError(e.to_string())),
400    };
401
402    let status = wf.status.as_ref()?;
403    match status.phase.as_deref() {
404        Some("Succeeded") => Some(WatchOutcome::Succeeded),
405
406        Some("Failed") | Some("Error") => {
407            let msg = status
408                .message
409                .clone()
410                .unwrap_or_else(|| "argo workflow failed".into());
411
412            Some(WatchOutcome::Failed(msg))
413        }
414
415        Some("Running") => None,
416
417        _ => {
418            error!(?status, "unexpected workflow status");
419            None
420        }
421    }
422}
423
424fn handle_job_event(res: result::Result<Job, watcher::Error>) -> Option<WatchOutcome> {
425    let job = match res {
426        Ok(job) => job,
427        Err(e) => return Some(WatchOutcome::WatcherError(e.to_string())),
428    };
429
430    let status = job.status.as_ref()?;
431    let conditions = status.conditions.as_deref().unwrap_or_default();
432    if conditions
433        .iter()
434        .any(|c| c.type_ == "Complete" && c.status == "True")
435    {
436        return Some(WatchOutcome::Succeeded);
437    }
438
439    if conditions
440        .iter()
441        .any(|c| c.type_ == "Failed" && c.status == "True")
442    {
443        return Some(WatchOutcome::Failed("job failed".into()));
444    }
445
446    None
