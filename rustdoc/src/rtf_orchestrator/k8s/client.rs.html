23// Container waiting reasons that indicate a pod is permanently stuck and will never produce an
24// exit code. These surface as `containerStatuses[].state.waiting.reason` in the pod status.
25//
26// Unlike transient failures (OOMKilled, CrashLoopBackOff, Error), which produce an exit code and
27// propagate to the parent Job/Workflow as Failed, these states leave the container in Waiting
28// indefinitely — the Job/Workflow never transitions, and the execution hangs.
29//
30// The reason strings are kubelet implementation details, not part of the Kubernetes API contract.
31// Image-related reasons originate in pkg/kubelet/images/types.go [1]; container creation reasons
32// in pkg/kubelet/kuberuntime/kuberuntime_container.go [2]. The official pod lifecycle docs
33// describe the Waiting state but do not enumerate possible reason values [3].
34//
35// This list covers known cases but is not exhaustive — other Waiting reasons may exist that
36// would also cause executions to hang.
37//
38// [1] https://github.com/kubernetes/kubernetes/blob/cec8f06d2ce283d1563d37849619887aa2b11c9f/pkg/kubelet/images/types.go
39// [2] https://github.com/kubernetes/kubernetes/blob/cec8f06d2ce283d1563d37849619887aa2b11c9f/pkg/kubelet/kuberuntime/kuberuntime_container.go
40// [3] https://kubernetes.io/docs/concepts/workloads/pods/pod-lifecycle/#container-states
41const UNRUNNABLE_REASONS: &[&str] = &[
42    "ImagePullBackOff",
43    "ErrImagePull",
44    "ErrImageNeverPull",
45    "InvalidImageName",
46    "CreateContainerConfigError",
47    "CreateContainerError",
48];

55/// Typestate marker for a [ClusterClients] slot that holds an initialised [Client]
56/// for the management cluster.
57/// The underlying [Client] does not need to be rebuilt for the management cluster
58/// as the `kubeconfig` tokens are refreshable through kube client

374    async fn wait_for_job(
375        &mut self,
376        ns: &str,
377        execution_id: &Uuid,
378        poll_interval_secs: u64,
379        retry_window_secs: u64,
380    ) -> WatchOutcome {
381        let labels = format!("{EXECUTION_ID_LABEL}={execution_id}");
382        let params = ListParams::default().labels(&labels);
383        let poll_interval = time::Duration::from_secs(poll_interval_secs);
384        let retry_window = Duration::seconds(retry_window_secs as i64);
385        // Used to keep track of when the first client failure was detected so a maximum
386        // retry window is not exceeded
387        let mut first_failure: Option<DateTime<Utc>> = None;
388
389        loop {
390            sleep(poll_interval).await;
391
392            let result = async {
393                let job_api: Api<Job> = self.workload_api(ns);
394                let pod_api: Api<Pod> = self.workload_api(ns);
395                poll_job_and_pods(&job_api, &pod_api, &params).await
396            }
397            .await;
398
399            match classify_poll_result(result, &mut first_failure, retry_window) {
400                PollDecision::Terminal(outcome) => return outcome,
401                PollDecision::KeepWaiting => {}
402                PollDecision::RetryAfterRefresh => {
403                    let _ = self.refresh_workload_client().await;
404                }
405            }
406        }
407    }
408
409    async fn delete_workload_namespace(&mut self, ns: &str) -> Result<()> {

425impl FullClient for ClusterClients<AvailableManagement, AvailableWorkload> {
426    async fn wait_for_workflow(
427        &mut self,
428        execution_id: &Uuid,
429        poll_interval_secs: u64,
430        retry_window_secs: u64,
431    ) -> WatchOutcome {
432        let wf_labels = format!("{EXECUTION_ID_LABEL}={execution_id}");
433        let wf_params = ListParams::default().labels(&wf_labels);
434
435        // Argo workflow pods aren't handled particularly well by Argo natively, so we watch them
436        // directly to catch a stuck pod the workflow status itself won't report.
437        let wf_pod_labels = format!("workflows.argoproj.io/workflow=provision-env-{execution_id}");
438        let wf_pod_params = ListParams::default().labels(&wf_pod_labels);
439
440        let workload_ns = execution_id.to_string();
441        let workload_pod_params = ListParams::default();
442        let poll_interval = time::Duration::from_secs(poll_interval_secs);
443        let retry_window = Duration::seconds(retry_window_secs as i64);
444        // Used to keep track of when the first client failure was detected so a maximum
445        // retry window is not exceeded
446        let mut first_failure: Option<DateTime<Utc>> = None;
447
448        loop {
449            sleep(poll_interval).await;
450
451            let wf_api: Api<Workflow> = self.management_api(ORCHESTRATOR_NAMESPACE);
452            let mgmt_pod_api: Api<Pod> = self.management_api(ORCHESTRATOR_NAMESPACE);
453
454            let result = async {
455                let workload_pod_api: Api<Pod> = self.workload_api(&workload_ns);
456                poll_workflow_and_pods(
457                    &wf_api,
458                    &wf_params,
459                    &mgmt_pod_api,
460                    &wf_pod_params,
461                    &workload_pod_api,
462                    &workload_pod_params,
463                )
464                .await
465            }
466            .await;
467
468            match classify_poll_result(result, &mut first_failure, retry_window) {
469                PollDecision::Terminal(outcome) => return outcome,
470                PollDecision::KeepWaiting => {}
471                PollDecision::RetryAfterRefresh => {
472                    let _ = self.refresh_workload_client().await;
473                }
474            }
475        }
476    }

479fn check_pod_for_unrunnable(pod: &Pod) -> Option<WatchOutcome> {
480    let pod_status = pod.status.as_ref()?;
481    let containers = pod_status.container_statuses.as_deref().unwrap_or_default();
482    let init_containers = pod_status
483        .init_container_statuses
484        .as_deref()
485        .unwrap_or_default();
486
487    for status in containers.iter().chain(init_containers) {
488        if let Some(reason) = status
489            .state
490            .as_ref()
491            .and_then(|s| s.waiting.as_ref())
492            .and_then(|w| w.reason.as_deref())
493            && UNRUNNABLE_REASONS.contains(&reason)
494        {
495            return Some(WatchOutcome::ContainerUnrunnable(reason.to_owned()));
496        }
497    }
498
499    None
500}
501
502/// Detect a `scenario-runner` container that was killed by the OOM killer. Unlike the Waiting
503/// reasons in `check_pod_for_unrunnable`, this container *did* terminate with a real exit code —
504/// but its sibling `output-collector` sidecar waits on a sentinel file that a SIGKILLed process
505/// never gets to write, so the pod hangs indefinitely without this check.
506///
507/// Scoped to the named `scenario-runner` container specifically: this must only ever run against
508/// the scenario Job's pod, never against environment pods, where an OOMKilled container is
509/// expected to self-heal via the Deployment's normal restart behaviour.
510fn check_scenario_runner_oomkilled(pod: &Pod) -> Option<WatchOutcome> {
511    let statuses = pod.status.as_ref()?.container_statuses.as_deref()?;
512    let status = statuses
513        .iter()
514        .find(|s| s.name == SCENARIO_RUNNER_CONTAINER)?;
515    let terminated = status.state.as_ref()?.terminated.as_ref()?;
516
517    if terminated.reason.as_deref() == Some("OOMKilled") {
518        Some(WatchOutcome::ContainerUnrunnable("OOMKilled".to_owned()))
519    } else {
520        None
521    }
522}
523
524fn check_workflow_status(wf: &Workflow) -> Option<WatchOutcome> {
525    let status = wf.status.as_ref()?;
526    match status.phase.as_deref() {
527        Some("Succeeded") => Some(WatchOutcome::Succeeded),
528
529        Some("Failed") | Some("Error") => {
530            let msg = status
531                .message
532                .clone()
533                .unwrap_or_else(|| "argo workflow failed".into());
534
535            Some(WatchOutcome::Failed(msg))
536        }
537
538        Some("Running") => None,
539
540        _ => {
541            error!(?status, "unexpected workflow status");
542            None
543        }
544    }
545}
546
547fn check_job_status(job: &Job) -> Option<WatchOutcome> {
548    let status = job.status.as_ref()?;
549    let conditions = status.conditions.as_deref().unwrap_or_default();
550    if conditions
551        .iter()
552        .any(|c| c.type_ == "Complete" && c.status == "True")
553    {
554        return Some(WatchOutcome::Succeeded);
555    }
556
557    if conditions
558        .iter()
559        .any(|c| c.type_ == "Failed" && c.status == "True")
560    {
561        return Some(WatchOutcome::Failed("job failed".into()));
562    }
563
564    None
565}

567/// List `api` and apply `check` to the first matching item, if any.
568async fn poll_first<K>(
569    api: &Api<K>,
570    params: &ListParams,
571    check: impl Fn(&K) -> Option<WatchOutcome>,
572) -> Result<Option<WatchOutcome>>
573where
574    K: Clone + std::fmt::Debug + serde::de::DeserializeOwned,
575{
576    let list = api.list(params).await?;
577
578    Ok(list.items.first().and_then(check))
579}
580
581/// List pods matching `params` and check each for a permanently-stuck waiting state.
582async fn poll_pods(api: &Api<Pod>, params: &ListParams) -> Result<Option<WatchOutcome>> {
583    let pods = api.list(params).await?;
584
585    Ok(pods.items.iter().find_map(check_pod_for_unrunnable))
586}
587
588/// List pods matching `params` and check each for a permanently-stuck waiting state or an
589/// OOMKilled `scenario-runner` container. Only used for the scenario Job's own pod — see
590/// `check_scenario_runner_oomkilled`.
591async fn poll_scenario_pods(api: &Api<Pod>, params: &ListParams) -> Result<Option<WatchOutcome>> {
592    let pods = api.list(params).await?;
593
594    Ok(pods.items.iter().find_map(|pod| {
595        check_pod_for_unrunnable(pod).or_else(|| check_scenario_runner_oomkilled(pod))
596    }))
597}
598
599async fn poll_job_and_pods(
600    job_api: &Api<Job>,
601    pod_api: &Api<Pod>,
602    params: &ListParams,
603) -> Result<Option<WatchOutcome>> {
604    if let Some(outcome) = poll_first(job_api, params, check_job_status).await? {
605        return Ok(Some(outcome));
606    }
607
608    poll_scenario_pods(pod_api, params).await
609}

611async fn poll_workflow_and_pods(
612    wf_api: &Api<Workflow>,
613    wf_params: &ListParams,
614    mgmt_pod_api: &Api<Pod>,
615    mgmt_pod_params: &ListParams,
616    workload_pod_api: &Api<Pod>,
617    workload_pod_params: &ListParams,
618) -> Result<Option<WatchOutcome>> {
619    if let Some(outcome) = poll_first(wf_api, wf_params, check_workflow_status).await? {
620        return Ok(Some(outcome));
621    }
622
623    if let Some(outcome) = poll_pods(mgmt_pod_api, mgmt_pod_params).await? {
624        return Ok(Some(outcome));
625    }
626
627    poll_pods(workload_pod_api, workload_pod_params).await
628}

630/// What a `wait_for_job`/`wait_for_workflow` loop should do next after one poll attempt.
631enum PollDecision {
632    /// The poll succeeded but no terminal state was reached yet; sleep and poll again.
633    KeepWaiting,
634    /// A retryable error occurred and the retry window has not been exceeded; rebuild the
635    /// workload client before polling again, on the chance the error was caused by an expired
636    /// token.
637    RetryAfterRefresh,
638    /// The wait loop is done: either a terminal state was reached, a non-retryable error
639    /// occurred, or the retry window was exceeded.
640    Terminal(WatchOutcome),
641}
642
643/// Shared bookkeeping between `wait_for_job` and `wait_for_workflow`: tracks how long a run of
644/// retryable errors has been ongoing via `first_failure`, resetting it on any successful poll,
645/// and gives up once `retry_window` is exceeded.
646fn classify_poll_result(
647    result: Result<Option<WatchOutcome>>,
648    first_failure: &mut Option<DateTime<Utc>>,
649    retry_window: Duration,
650) -> PollDecision {
651    match result {
652        Ok(Some(outcome)) => PollDecision::Terminal(outcome),
653        // Client api is working so clear first failure
654        Ok(None) => {
655            *first_failure = None;
656            PollDecision::KeepWaiting
657        }
658        Err(e) if e.is_retryable() => {
659            let first_failure_at = *first_failure.get_or_insert_with(Utc::now);
660
661            if Utc::now() - first_failure_at > retry_window {
662                PollDecision::Terminal(WatchOutcome::WatchErrors(
663                    Error::MaxRetriesExceeded.to_string(),
664                ))
665            } else {
666                PollDecision::RetryAfterRefresh
667            }
668        }
669        Err(e) => PollDecision::Terminal(WatchOutcome::WatchErrors(e.to_string())),
670    }
671}
