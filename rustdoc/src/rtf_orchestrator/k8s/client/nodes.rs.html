6/// A struct to represent an allocation of node labels to nodes
7#[derive(Debug)]
8pub(crate) struct NodeAllocationPlan {
9    // Map of node ID to the label (key, value) to apply to it
10    inner: HashMap<String, (String, String)>,
11}
12
13impl NodeAllocationPlan {
14    /// Create an allocation, testing whether it's valid throughout.
15    pub(crate) fn try_new(
16        mut nodes: Vec<Node>,
17        allocation: Vec<(String, u32)>,
18    ) -> Result<Self, NodeAllocationError> {
19        use NodeAllocationError::{WeightOverflow, ZeroWeights};
20
21        let node_total: u32 = nodes.len() as u32;
22
23        let allocation_total: u32 = allocation.iter().map(|(_, weight)| weight).sum();
24
25        // If somehow we try to allocate 0 weight then just stop because it will make other
26        // calculations nonsensical.
27        if allocation_total == 0 {
28            return Err(ZeroWeights);
29        }
30
31        // Use checked multiply here so we don't fall victim to overflows (or at least we can
32        // be alerted when they happen).
33        let mut node_counts: Vec<(String, u32)> = allocation
34            .into_iter()
35            .map(|(label, weight)| {
36                weight
37                    .checked_mul(node_total)
38                    .map(|scaled| (label, scaled / allocation_total))
39                    .ok_or(WeightOverflow)
40            })
41            .collect::<Result<_, _>>()?;
42
43        // Flooring each proportion can leave a shortfall versus `node_total`; hand those nodes
44        // out one at a time to whichever group is currently smallest.
45        let shortfall = node_total - node_counts.iter().map(|(_, count)| count).sum::<u32>();
46        for _ in 0..shortfall {
47            let (_, count) = node_counts
48                .iter_mut()
49                .min_by_key(|(_, count)| *count)
50                .expect("allocation is non-empty as weights summed to more than zero");
51            *count += 1;
52        }
53
54        // Transform the allocation into the inner data structure
55        let mut inner = HashMap::new();
56        for (label, amount) in node_counts {
57            for _ in 0..amount {
58                let node = nodes.pop().unwrap();
59                inner.insert(
60                    node.name_any(),
61                    (NODE_ALLOCATION_LABEL.to_string(), label.clone()),
62                );
63            }
64        }
65
66        Ok(NodeAllocationPlan { inner })
67    }
68}

70impl IntoIterator for NodeAllocationPlan {
71    type Item = (String, (String, String));
72    type IntoIter = std::collections::hash_map::IntoIter<String, (String, String)>;
73
74    fn into_iter(self) -> Self::IntoIter {
75        self.inner.into_iter()
76    }
77}

79impl From<HashMap<String, (String, String)>> for NodeAllocationPlan {
80    fn from(value: HashMap<String, (String, String)>) -> Self {
81        Self { inner: value }
82    }
83}
84
85impl PartialEq for NodeAllocationPlan {
86    fn eq(&self, other: &Self) -> bool {
87        self.inner.eq(&other.inner)
88    }
89}

91#[derive(Debug, PartialEq, thiserror::Error)]
92pub enum NodeAllocationError {
93    #[error("Weights supplied equalled zero")]
94    ZeroWeights,
95    #[error("Allocation weight too large to compute a node split")]
96    WeightOverflow,
97}

105    fn summary(plan: &NodeAllocationPlan) -> HashMap<String, u32> {
106        let mut summary = HashMap::new();
107        for (_, label) in plan.inner.values() {
108            summary
109                .entry(label.clone())
110                .and_modify(|counter| *counter += 1)
111                .or_insert(1);
112        }
113
114        summary
115    }

117    fn generate_nodes(nodes: Vec<&str>) -> Vec<Node> {
118        nodes
119            .into_iter()
120            .map(|name| Node {
121                metadata: ObjectMeta {
122                    name: Some(name.to_string()),
123                    ..Default::default()
124                },
125                spec: None,
126                status: None,
127            })
128            .collect()
129    }
130
131    fn allocation(weights: &[(&str, u32)]) -> Vec<(String, u32)> {
132        weights.iter().map(|(k, v)| (k.to_string(), *v)).collect()
133    }
134
135    fn counts(weights: &[(&str, u32)]) -> HashMap<String, u32> {
136        weights.iter().map(|(k, v)| (k.to_string(), *v)).collect()
137    }
