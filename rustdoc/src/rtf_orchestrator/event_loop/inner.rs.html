13#[derive(Debug)]
14pub(super) struct EventQueueInner {
15    /// Pending events for in-progress executions, independent of cluster
16    pending_non_provisions: VecDeque<Event>,
17    /// Pending provisioning events for new executions, queued per pool
18    pending_provisions: HashMap<PoolId, VecDeque<PendingProvision>>,
19    /// Ordering for obtaining the next queueable provisioning event, visited round-robin
20    pool_provision_order: VecDeque<PoolId>,
21    /// The clusters that make up each pool
22    pool_clusters: HashMap<PoolId, Vec<ClusterId>>,
23    /// Pools that runs can claim dedicated clusters from
24    dedicated_pools: HashSet<PoolId>,
25    /// Uuids for the set of executions whose namespaces are live, per cluster
26    running_executions: HashMap<ClusterId, HashSet<Uuid>>,
27    /// Maximum number of live namespaces, per cluster
28    max_concurrent_executions: HashMap<ClusterId, usize>,
29    /// Clusters that are reserved for, or owned by, a run that requires a dedicated cluster
30    claims: HashMap<ClusterId, ClusterClaim>,
31}

48        self.running_executions.get(cluster).map_or(0, |s| s.len())

51    pub(super) fn is_empty(&self) -> bool {
52        self.pending_provisions.values().all(|q| q.is_empty())
53            && self.pending_non_provisions.is_empty()

56    pub(super) fn push_event(&mut self, evt: QueueEvent) {
57        match evt {
58            QueueEvent::Provision(pool, evt) => self
59                .pending_provisions
60                .entry(pool)
61                .or_default()
62                .push_back(evt),
63            QueueEvent::Other(evt) => self.pending_non_provisions.push_back(evt),
64        }
65    }
66
67    pub(super) fn next_event(&mut self) -> Option<Event> {
68        if let Some(evt) = self.pending_non_provisions.pop_front() {
69            return Some(evt);
70        } else if let Some(evt) = self.runnable_provisioning_event() {
71            self.insert_running_execution(evt.test_execution.uuid(), evt.cluster.clone());
72            return Some(evt);
73        }
74
75        None
76    }

78    /// Find the next pool in round-robin order with a queued provisioning event that can be
79    /// assigned to a cluster, and remove that event from the queue.
80    ///
81    /// Events within a pool are considered in order, but one that is waiting on a dedicated
82    /// cluster does not block the events behind it.
83    pub(super) fn runnable_provisioning_event(&mut self) -> Option<Event> {
84        for _ in 0..self.pool_provision_order.len() {
85            let pool = self.pool_provision_order.front()?.clone();
86            self.pool_provision_order.rotate_left(1);
87
88            let n_pending = self.pending_provisions.get(&pool).map_or(0, VecDeque::len);
89            for i in 0..n_pending {
90                let (run_uuid, requires_dedicated) = {
91                    let pending = &self.pending_provisions[&pool][i];
92                    (pending.run_uuid, pending.requires_dedicated)
93                };
94
95                let (cluster, data) = match self.try_assign(&pool, run_uuid, requires_dedicated) {
96                    Some(details) => details,
97                    None => continue,
98                };
99
100                let provision = self.pending_provisions.get_mut(&pool)?.remove(i)?;
101
102                return Some(Event::new(provision.test_execution, cluster, data));
103            }
104        }
105
106        None
107    }

109    /// Attempt to find a cluster that is able to accepts a new execution from the given test run.
110    pub(super) fn try_assign(
111        &mut self,
112        pool: &PoolId,
113        run_uuid: Uuid,
114        requires_dedicated: bool,
115    ) -> Option<(ClusterId, EventData)> {
116        use ClusterClaim::{Acquiring, Owned, Reserved};
117
118        let clusters = self.pool_clusters.get(pool)?;
119        let cluster_with_claim = |claim| {
120            clusters
121                .iter()
122                .find(|c| self.claims.get(*c) == Some(&claim))
123        };
124
125        // If we don't need a dedicated cluster and the pool doesn't support dedicated,
126        // assign to the least loaded cluster (provided we have capacity)
127        if !(requires_dedicated && self.dedicated_pools.contains(pool)) {
128            let cluster = clusters
129                .iter()
130                .filter(|c| !self.claims.contains_key(*c) && self.has_capacity(c))
131                .min_by_key(|c| self.running_on(c))?;
132
133            return Some((cluster.clone(), EventData::ResolveConfig));
134        }
135
136        // All assignments past this point are for runs requiring dedicated clusters
137
138        // If this run already owns a cluster, assign to that cluster (provided we have capacity)
139        if let Some(cluster) = cluster_with_claim(Owned(run_uuid)) {
140            return self
141                .has_capacity(cluster)
142                .then(|| (cluster.clone(), EventData::ResolveConfig));
143        }
144
145        // If this run is in the process of acquiring its cluster then all other executions block
146        // until that process has finished
147        if cluster_with_claim(Acquiring(run_uuid)).is_some() {
148            return None;
149        }
150
151        // If this run has an outstanding reservation assign to that cluster if it is now free and
152        // update the claim to Acquiring (otherwise we continue waiting)
153        if let Some(cluster) = cluster_with_claim(Reserved(run_uuid)) {
154            if self.running_on(cluster) > 0 || !self.has_capacity(cluster) {
155                return None;
156            }
157            self.claims.insert(cluster.clone(), Acquiring(run_uuid));
158            return Some((cluster.clone(), EventData::AcquireCluster));
159        }
160
161        let unclaimed: Vec<&ClusterId> = clusters
162            .iter()
163            .filter(|c| !self.claims.contains_key(*c))
164            .collect();
165
166        // If we have an empty unclaimed cluster, claim it and assign to it
167        let first_empty = unclaimed.iter().find(|c| self.running_on(c) == 0);
168        if let Some(cluster) = first_empty {
169            self.claims.insert((*cluster).clone(), Acquiring(run_uuid));
170
171            return Some(((*cluster).clone(), EventData::AcquireCluster));
172        }
173
174        // Finally, if we have at least one unclaimed cluster then reserve the one with the fewest
175        // ongoing executions before returning None
176        if let Some(cluster) = unclaimed.iter().min_by_key(|c| self.running_on(c)) {
177            self.claims.insert((*cluster).clone(), Reserved(run_uuid));
178        }
179
180        None
181    }

183    pub(super) fn has_capacity(&self, cluster: &ClusterId) -> bool {
184        let max = self
185            .max_concurrent_executions
186            .get(cluster)
187            .copied()
188            .unwrap_or(0);
189
190        self.running_on(cluster) < max
191    }

193    /// Immediately remove any outstanding reservations and schedule a ReleaseCluster event for any
194    /// clusters currently claimed by this run.
195    pub(super) fn schedule_cluster_release(&mut self, run_uuid: Uuid, ex: &TestExecution) {
196        let mut released = Vec::new();
197
198        self.claims.retain(|cluster, claim| match claim {
199            ClusterClaim::Reserved(run) if *run == run_uuid => false,
200            ClusterClaim::Acquiring(run) | ClusterClaim::Owned(run) if *run == run_uuid => {
201                released.push(cluster.clone());
202                true
203            }
204            _ => true,
205        });
206
207        for cluster in released.into_iter() {
208            self.pending_non_provisions.push_front(Event::new(
209                ex.clone(),
210                cluster,
211                EventData::ReleaseCluster,
212            ));
213        }
214    }

239    pub(super) fn remove_running_execution(&mut self, ex_uuid: Uuid) -> bool {
240        for running in self.running_executions.values_mut() {
241            if running.remove(&ex_uuid) {
242                return true;
243            }
244        }
245
246        false
247    }

249    pub(super) fn cluster_for_execution(&mut self, ex_uuid: Uuid) -> Option<ClusterId> {
250        for (cid, running) in self.running_executions.iter() {
251            if running.contains(&ex_uuid) {
252                return Some(cid.clone());
253            }
254        }
255
256        None
257    }

259    pub(super) fn abort_pending_provisions_for(&mut self, run_uuid: Uuid) -> Vec<TestExecution> {
260        let mut aborted = Vec::new();
261
262        for evts in self.pending_provisions.values_mut() {
263            evts.retain(|evt| {
264                if evt.run_uuid == run_uuid {
265                    aborted.push(evt.test_execution.clone());
266                    false
267                } else {
268                    true
269                }
270            });
271        }
272
273        aborted
274    }

304    pub(super) fn purge_execution_state(&mut self, ex: TestExecution) -> Option<()> {
305        let ex_id = ex.uuid();
306
307        // drop provision events first to ensure that we don't create any _new_ namespaces
308        for events in self.pending_provisions.values_mut() {
309            events.retain(|evt| evt.test_execution.uuid() != ex_id)
310        }
311
312        if let Some(cluster) = self.cluster_for_execution(ex_id) {
313            // The execution made it as far as being provisioned so we need to nuke the namespace
314            self.running_executions.get_mut(&cluster)?.remove(&ex_id);
315            self.pending_non_provisions
316                .retain(|evt| evt.test_execution.uuid() != ex_id);
317            self.pending_non_provisions.push_front(Event::new(
318                ex,
319                cluster,
320                EventData::PurgeNamespace,
321            ));
322        };
323
324        None
325    }

327    pub(super) fn snapshot_state(&self) -> InnerSnapshotState {
328        let mut summary = SnapshotSummary::default();
329        let mut pools: BTreeMap<String, PoolQueueState> = BTreeMap::new();
330        let mut clusters: BTreeMap<String, ClusterQueueState> = BTreeMap::new();
331
332        for (pool_id, events) in self.pending_provisions.iter() {
333            let state = pools.entry(pool_id.to_string()).or_default();
334            for evt in events.iter() {
335                state.pending_provisions.push(PendingProvisionSummary {
336                    execution_id: evt.test_execution.uuid(),
337                    run_id: evt.run_uuid,
338                    requires_dedicated: evt.requires_dedicated,
339                });
340            }
341        }
342
343        for (cluster_id, running) in self.running_executions.iter() {
344            let state = clusters.entry(cluster_id.to_string()).or_default();
345            state.running_executions.extend(running.iter().copied());
346            state.running_executions.sort_unstable();
347            summary.running += running.len();
348        }
349
350        for evt in self.pending_non_provisions.iter() {
351            let state = clusters.entry(evt.cluster.to_string()).or_default();
352            state.pending_non_provisions.push(event_summary(evt));
353            summary.pending_non_provisions += 1;
354        }
355
356        // Ensure that we have default summaries for pools/clusters that are currently empty
357        for (pool_id, cluster_ids) in self.pool_clusters.iter() {
358            pools.entry(pool_id.to_string()).or_default();
359
360            for cluster_id in cluster_ids.iter() {
361                clusters.entry(cluster_id.to_string()).or_default();
362            }
363        }
364
365        let claims: Vec<_> = self
366            .claims
367            .iter()
368            .map(|(cid, claim)| (cid.to_string(), *claim, self.running_on(cid)))
369            .collect();
370
371        InnerSnapshotState {
372            summary,
373            claims,
374            pools,
375            clusters,
376        }
377    }

415#[derive(Debug, Clone, Copy, PartialEq, Eq)]
416pub(super) enum ClusterClaim {
417    /// The run is waiting for the executions currently running on the cluster to finish
418    Reserved(Uuid),
419    /// The run has the cluster to itself and is in the process of running acquisition actions
420    Acquiring(Uuid),
421    /// The run has the cluster to itself and it is ready for its executions to start
422    Owned(Uuid),
423}

547        inner.remove_running_execution(ex);
548        assert_eq!(inner.try_assign_insert_dedicated(run), Some(acquire("a")));
