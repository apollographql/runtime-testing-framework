20pub(super) struct EventQueueInner {
21    /// Pending events for in-progress executions, independent of cluster
22    pending_non_provisions: VecDeque<Event>,
23    /// Pending provisioning events for new executions, queued per pool
24    pending_provisions: HashMap<PoolId, VecDeque<PendingProvision>>,
25    /// Ordering for obtaining the next queueable provisioning event, visited round-robin
26    pool_provision_order: VecDeque<PoolId>,
27    /// The clusters that make up each pool
28    pool_clusters: HashMap<PoolId, Vec<ClusterId>>,
29    /// Pools that runs can claim dedicated clusters from
30    dedicated_pools: HashSet<PoolId>,
31    /// Uuids for the set of executions whose namespaces are live, per cluster
32    running_executions: HashMap<ClusterId, HashSet<Uuid>>,
33    /// Maximum number of live namespaces, per cluster
34    max_concurrent_executions: HashMap<ClusterId, usize>,
35    /// Clusters that are reserved for, or owned by, a run that requires a dedicated cluster
36    claims: HashMap<ClusterId, ClusterClaim>,
37    /// Per-user execution limits per pool
38    per_user_limits: HashMap<PoolId, PerUserExecutionConfig>,
39    /// State shared with the rest of the event loop, used to look up which run and user an
40    /// execution belongs to.
41    ///
42    /// Lock order: always take the lock on [EventQueueInner] before this one.
43    shared: Arc<Mutex<Shared>>,
44}

95    pub(super) fn push_event(&mut self, evt: QueueEvent) {
96        match evt {
97            QueueEvent::Provision(pool, evt) => self
98                .pending_provisions
99                .entry(pool)
100                .or_default()
101                .push_back(evt),
102            QueueEvent::Other(evt) => self.pending_non_provisions.push_back(evt),
103        }
104    }

106    pub(super) async fn next_event(&mut self) -> Option<Event> {
107        if let Some(evt) = self.pending_non_provisions.pop_front() {
108            return Some(evt);
109        } else if let Some(evt) = self.runnable_provisioning_event().await {
110            self.insert_running_execution(evt.test_execution.uuid(), evt.cluster.clone());
111            return Some(evt);
112        }
113
114        None
115    }

117    /// Find the next pool in round-robin order with a queued provisioning event that can be
118    /// assigned to a cluster, and remove that event from the queue.
119    ///
120    /// Events within a pool are considered in order, but one that is waiting on a dedicated
121    /// cluster does not block the events behind it.
122    pub(super) async fn runnable_provisioning_event(&mut self) -> Option<Event> {
123        for _ in 0..self.pool_provision_order.len() {
124            let pool = self.pool_provision_order.front()?.clone();
125            self.pool_provision_order.rotate_left(1);
126
127            let n_pending = self.pending_provisions.get(&pool).map_or(0, VecDeque::len);
128            for i in 0..n_pending {
129                let pending = &self.pending_provisions[&pool][i];
130                let (run_uuid, requires_dedicated) = (pending.run_uuid, pending.requires_dedicated);
131
132                if self
133                    .user_at_max_concurrent_runs(&pool, &pending.initiated_by_user, run_uuid)
134                    .await
135                {
136                    continue;
137                }
138
139                let (cluster, data) = match self.try_assign(&pool, run_uuid, requires_dedicated) {
140                    Some(details) => details,
141                    None => continue,
142                };
143
144                let provision = self.pending_provisions.get_mut(&pool)?.remove(i)?;
145
146                return Some(Event::new(provision.test_execution, cluster, data));
147            }
148        }
149
150        None
151    }

153    /// Returns whether starting an execution of `run_uuid` would put `user` over
154    /// their concurrent runs limit for `pool`.
155    async fn user_at_max_concurrent_runs(
156        &self,
157        pool: &PoolId,
158        user: &UserType,
159        run_uuid: Uuid,
160    ) -> bool {
161        let Some(per_user) = self.per_user_limits.get(pool) else {
162            return false;
163        };
164
165        let running = self
166            .pool_clusters
167            .get(pool)
168            .into_iter()
169            .flatten()
170            .flat_map(|cluster| self.execution_ids_for_cluster(cluster));
171        // Only hold the shared lock for as long as it takes to count the user's ongoing runs
172        let ongoing_runs = self
173            .shared
174            .lock()
175            .await
176            .runs_initiated_by(running, user.user_identity().as_deref());
177
178        rate_limit::user_at_max_concurrent_runs(per_user, user, &ongoing_runs, run_uuid)
179    }

181    /// Attempt to find a cluster that is able to accepts a new execution from the given test run.
182    pub(super) fn try_assign(
183        &mut self,
184        pool: &PoolId,
185        run_uuid: Uuid,
186        requires_dedicated: bool,
187    ) -> Option<(ClusterId, EventData)> {
188        use ClusterClaim::{Acquiring, Owned, Reserved};
189
190        let clusters = self.pool_clusters.get(pool)?;
191        let cluster_with_claim = |claim| {
192            clusters
193                .iter()
194                .find(|c| self.claims.get(*c) == Some(&claim))
195        };
196
197        // If we don't need a dedicated cluster and the pool doesn't support dedicated,
198        // assign to the least loaded cluster (provided we have capacity)
199        if !(requires_dedicated && self.dedicated_pools.contains(pool)) {
200            let cluster = clusters
201                .iter()
202                .filter(|c| !self.claims.contains_key(*c) && self.has_capacity(c))
203                .min_by_key(|c| self.running_on(c))?;
204
205            return Some((cluster.clone(), EventData::ResolveConfig));
206        }
207
208        // All assignments past this point are for runs requiring dedicated clusters
209
210        // If this run already owns a cluster, assign to that cluster (provided we have capacity)
211        if let Some(cluster) = cluster_with_claim(Owned(run_uuid)) {
212            return self
213                .has_capacity(cluster)
214                .then(|| (cluster.clone(), EventData::ResolveConfig));
215        }
216
217        // If this run is in the process of acquiring its cluster then all other executions block
218        // until that process has finished
219        if cluster_with_claim(Acquiring(run_uuid)).is_some() {
220            return None;
221        }
222
223        // If this run has an outstanding reservation assign to that cluster if it is now free and
224        // update the claim to Acquiring (otherwise we continue waiting)
225        if let Some(cluster) = cluster_with_claim(Reserved(run_uuid)) {
226            if self.running_on(cluster) > 0 || !self.has_capacity(cluster) {
227                return None;
228            }
229            self.claims.insert(cluster.clone(), Acquiring(run_uuid));
230            return Some((cluster.clone(), EventData::AcquireCluster));
231        }
232
233        let unclaimed: Vec<&ClusterId> = clusters
234            .iter()
235            .filter(|c| !self.claims.contains_key(*c))
236            .collect();
237
238        // If we have an empty unclaimed cluster, claim it and assign to it
239        let first_empty = unclaimed.iter().find(|c| self.running_on(c) == 0);
240        if let Some(cluster) = first_empty {
241            self.claims.insert((*cluster).clone(), Acquiring(run_uuid));
242
243            return Some(((*cluster).clone(), EventData::AcquireCluster));
244        }
245
246        // Finally, if we have at least one unclaimed cluster then reserve the one with the fewest
247        // ongoing executions before returning None
248        if let Some(cluster) = unclaimed.iter().min_by_key(|c| self.running_on(c)) {
249            self.claims.insert((*cluster).clone(), Reserved(run_uuid));
250        }
251
252        None
253    }

255    pub(super) fn has_capacity(&self, cluster: &ClusterId) -> bool {
256        let max = self
257            .max_concurrent_executions
258            .get(cluster)
259            .copied()
260            .unwrap_or(0);
261
262        self.running_on(cluster) < max
263    }

265    /// Immediately remove any outstanding reservations and schedule a ReleaseCluster event for any
266    /// clusters currently claimed by this run.
267    pub(super) fn schedule_cluster_release(&mut self, run_uuid: Uuid, ex: &TestExecution) {
268        let mut released = Vec::new();
269
270        self.claims.retain(|cluster, claim| match claim {
271            ClusterClaim::Reserved(run) if *run == run_uuid => false,
272            ClusterClaim::Acquiring(run) | ClusterClaim::Owned(run) if *run == run_uuid => {
273                released.push(cluster.clone());
274                true
275            }
276            _ => true,
277        });
278
279        for cluster in released.into_iter() {
280            self.pending_non_provisions.push_front(Event::new(
281                ex.clone(),
282                cluster,
283                EventData::ReleaseCluster,
284            ));
285        }
286    }

380    pub(super) fn purge_execution_state(&mut self, ex: TestExecution) -> Option<()> {
381        let ex_id = ex.uuid();
382
383        // drop provision events first to ensure that we don't create any _new_ namespaces
384        for events in self.pending_provisions.values_mut() {
385            events.retain(|evt| evt.test_execution.uuid() != ex_id)
386        }
387
388        if let Some(cluster) = self.cluster_for_execution(ex_id) {
389            // The execution made it as far as being provisioned so we need to nuke the namespace
390            self.running_executions.get_mut(&cluster)?.remove(&ex_id);
391            self.pending_non_provisions
392                .retain(|evt| evt.test_execution.uuid() != ex_id);
393            self.pending_non_provisions.push_front(Event::new(
394                ex,
395                cluster,
396                EventData::PurgeNamespace,
397            ));
398        };
399
400        None
401    }

403    pub(super) fn snapshot_state(&self) -> InnerSnapshotState {
404        let mut summary = SnapshotSummary::default();
405        let mut pools: BTreeMap<String, PoolQueueState> = BTreeMap::new();
406        let mut clusters: BTreeMap<String, ClusterQueueState> = BTreeMap::new();
407
408        for (pool_id, events) in self.pending_provisions.iter() {
409            let state = pools.entry(pool_id.to_string()).or_default();
410            for evt in events.iter() {
411                state.pending_provisions.push(PendingProvisionSummary {
412                    execution_id: evt.test_execution.uuid(),
413                    run_id: evt.run_uuid,
414                    requires_dedicated: evt.requires_dedicated,
415                });
416            }
417            summary.queued += events.len();
418        }
419
420        for (cluster_id, running) in self.running_executions.iter() {
421            let state = clusters.entry(cluster_id.to_string()).or_default();
422            state.running_executions.extend(running.iter().copied());
423            state.running_executions.sort_unstable();
424            summary.running += running.len();
425        }
426
427        for evt in self.pending_non_provisions.iter() {
428            let state = clusters.entry(evt.cluster.to_string()).or_default();
429            state.pending_non_provisions.push(event_summary(evt));
430            summary.pending_non_provisions += 1;
431        }
432
433        // Ensure that we have default summaries for pools/clusters that are currently empty
434        for (pool_id, cluster_ids) in self.pool_clusters.iter() {
435            pools.entry(pool_id.to_string()).or_default();
436
437            for cluster_id in cluster_ids.iter() {
438                clusters.entry(cluster_id.to_string()).or_default();
439            }
440        }
441
442        let claims: Vec<_> = self
443            .claims
444            .iter()
445            .map(|(cid, claim)| (cid.to_string(), *claim, self.running_on(cid)))
446            .collect();
447
448        InnerSnapshotState {
449            summary,
450            claims,
451            pools,
452            clusters,
453        }
454    }
