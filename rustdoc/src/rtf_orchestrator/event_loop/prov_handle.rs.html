25/// A handle for submitting provisioning requests to the `EventQueue`.
26#[derive(Debug, Clone)]
27pub struct ProvisioningHandle {
28    /// Shared state with the parent event queue
29    shared: Arc<Mutex<Shared>>,
30    /// Sender for submitting provisioning events to the event queue
31    tx: UnboundedSender<QueueEvent>,
32}
33
34impl ProvisioningHandle {
35    pub(super) fn new(shared: Arc<Mutex<Shared>>, tx: UnboundedSender<QueueEvent>) -> Self {
36        Self { shared, tx }
37    }

39    /// Run a closure with access to the shared event queue state.
40    ///
41    /// This method holds the mutex lock on the shared state for the duration of the closure. As
42    /// such, you _must_ ensure that closures are quick to execute. If multiple long running
43    /// operations are required, prefer extracting the state you need from [Shared] and
44    /// manipulating it after releasing the lock before re-acquiring.
45    async fn with_shared<F, T>(&self, f: F) -> T
46    where
47        F: FnOnce(&mut Shared) -> T,
48    {
49        f(&mut *self.shared.lock().await)
50    }

52    pub(crate) async fn cache_for_test_run(
53        &self,
54        run_uuid: Uuid,
55        initiated_by: Option<String>,
56        workload_config: WorkloadConfig,
57        requires_dedicated: bool,
58        ctx: OrchestratorContext,
59        test_plan: OrchestratorTestPlan,
60    ) {
61        self.with_shared(|shared| {
62            shared.runs.insert(
63                run_uuid,
64                RunState {
65                    ctx: Arc::new(ctx),
66                    test_plan,
67                    executions: HashSet::new(),
68                    initiated_by,
69                    workload_config,
70                    requires_dedicated,
71                },
72            )
73        })
74        .await;
75    }
76
77    /// Drop the in-memory run entry for the given run uuid without touching the ref-count
78    /// bookkeeping. Used by the resolver when it has cached a payload but every `init_execution`
79    /// for the run failed — there will be no `mark_execution_complete` for this run, so eviction
80    /// has to happen here instead.
81    pub(crate) async fn evict_cached_run_state(&self, run_uuid: Uuid) {
82        self.with_shared(|shared| {
83            shared.runs.remove(&run_uuid);
84        })
85        .await;
86    }

88    /// Create the DB row for a new execution and submit a provisioning event to the event loop,
89    /// decrementing `n_queued` in the process.
90    ///
91    /// The event waits in `EventQueue::pending_provisions` until namespace capacity is
92    /// available — the gating happens inside `EventQueue::next_event`, not here.
93    pub(crate) async fn request_provisioning<H>(
94        &self,
95        tr: &TestRun,
96        name: &str,
97        index: usize,
98        pool: PoolId,
99        update_handle: &mut H,
100    ) -> crate::Result<bool>
101    where
102        H: UpdateHandle,
103    {
104        let mut shared = self.shared.lock().await;
105
106        let requires_dedicated = match shared.runs.get(&tr.uuid()) {
107            Some(r) => r.requires_dedicated,
108            None => {
109                // run was cancelled while we were resolving executions
110                return Ok(false);
111            }
112        };
113
114        let ex = update_handle.init_execution(tr, name, index).await?;
115
116        assert!(
117            shared.n_queued > 0,
118            "request_provisioning called with n_queued == 0"
119        );
120        shared.n_queued -= 1;
121        shared.register_execution(ex.uuid(), tr.uuid());
122        drop(shared);
123
124        if let Err(e) = self.tx.send(QueueEvent::Provision(
125            pool,
126            PendingProvision::new(ex, tr.uuid(), requires_dedicated),
127        )) {
128            error!(%e, "event loop channel closed");
129            return Err(ResolverError::EventChannelClosed.into());
130        }
131
132        Ok(true)
133    }

135    /// Register an already-created execution against its parent run's in-memory state, without
136    /// creating a DB row or submitting a provisioning event.
137    #[cfg(test)]
138    pub(crate) async fn register_existing_execution(&self, ex_uuid: Uuid, run_uuid: Uuid) {
139        self.with_shared(|shared| shared.register_execution(ex_uuid, run_uuid))
140            .await;
141    }

143    async fn templated_and_checked_version(
144        &self,
145        ex: &TestExecution,
146    ) -> resolver::Result<(OrchestratorTestPlan, Arc<OrchestratorContext>)> {
147        let (mut test_plan, ctx) = self
148            .with_shared(|shared| shared.variant_with_context(ex))
149            .await?;
150
151        let template_ctx = ctx.new_template_context(test_plan.variables.clone());
152
153        test_plan
154            .try_template(&mut Vec::new(), &StableSource::TestPlan, &template_ctx)
155            .map_err(ResolverError::VariantTemplating)?;
156
157        test_plan.try_check(&mut Vec::new(), ctx.as_ref())?;
158
159        Ok((test_plan, ctx))
160    }

162    /// Resolve the environment and scenario config for `ex` and store it in
163    /// `resolved_execution_cache` keyed by execution UUID.
164    pub(crate) async fn resolve_and_cache_config(
165        &self,
166        ex: &TestExecution,
167    ) -> resolver::Result<()> {
168        let (mut test_plan, ctx) = self.templated_and_checked_version(ex).await?;
169
170        // Make sure that we run the run inlining without holding the lock on shared
171        let inline_cache = ctx.inline_cache();
172        test_plan
173            .environment
174            .execution
175            .try_inline(
176                InlineMode::All,
177                ctx.as_ref(),
178                &mut *inline_cache.lock().await,
179            )
180            .await?;
181
182        test_plan
183            .scenario
184            .execution
185            .try_inline(
186                InlineMode::All,
187                ctx.as_ref(),
188                &mut *inline_cache.lock().await,
189            )
190            .await?;
191
192        // Clear custom provider details to allow `rtf resolve` to run
193        test_plan.environment.custom_providers = Vec::new();
194        test_plan.scenario.custom_providers = Vec::new();
195
196        let image = test_plan.scenario.execution.docker_image();
197        let command = test_plan.scenario.execution.command();
198        let environment = test_plan.environment.execution.clone();
199
200        let env_yaml = serde_yaml::to_string(&test_plan.environment)
201            .map_err(|e| ResolverError::Serialisation(e.to_string()))?;
202        let scenario_yaml = serde_yaml::to_string(&test_plan.scenario)
203            .map_err(|e| ResolverError::Serialisation(e.to_string()))?;
204        let execution_variables = serde_json::to_string_pretty(&test_plan.variables)
205            .map_err(|e| ResolverError::Serialisation(e.to_string()))?;
206
207        let env_prom_queries = test_plan
208            .environment
209            .execution
210            .prometheus_queries()
211            .iter()
212            .map(|q| {
213                q.with_namespace_label_filter(&ex.uuid().to_string())
214                    .expect("promql query should have been validated by try_check")
215            })
216            .collect();
217        let scenario_prom_queries = test_plan
218            .scenario
219            .execution
220            .output_collection
221            .prometheus
222            .iter()
223            .map(|q| {
224                q.with_namespace_label_filter(&ex.uuid().to_string())
225                    .expect("promql query should have been validated by try_check")
226            })
227            .collect();
228        let output_collection = OutputCollectionResponse {
229            execution_variables,
230            prometheus: PrometheusQueries {
231                environment: env_prom_queries,
232                scenario: scenario_prom_queries,
233            },
234        };
235
236        self.with_shared(|shared| {
237            let exec = shared
238                .executions
239                .get_mut(&ex.uuid())
240                .ok_or(ResolverError::UnknownExecution(ex.uuid()))?;
241
242            exec.resolved_config = Some(ResolvedExecutionConfig {
243                env_yaml,
244                scenario_yaml,
245                output_collection,
246                docker_image: image,
247                docker_command: command,
248                environment,
249            });
250
251            Ok(())
252        })
253        .await
254    }

256    /// Send an event to the event loop.
257    pub(crate) fn send_event(
258        &self,
259        ex: TestExecution,
260        cluster: ClusterId,
261        data: EventData,
262    ) -> resolver::Result<()> {
263        self.tx
264            .send(QueueEvent::Other(Event::new(ex, cluster, data)))
265            .map_err(|_| ResolverError::EventChannelClosed)
266    }
267
268    /// Return `n` queued execution claims back to the shared state.
269    ///
270    /// # Safety
271    /// The caller must pass an `n` that matches the number of unused claims they previously
272    /// obtained from `try_reserve_pending_executions`.
273    pub async unsafe fn release_pending_execution_claim(&self, n: usize) {
274        let mut shared = self.shared.lock().await;
275        shared.n_queued -= n;
276    }

331    async fn request_provisioning_happy_path() {
332        let (h, mut rx) = test_prov_handle();
333        let tr = TestRun::create_stub(1, "test");
334        let mut mock = MockUpdateHandle::with_run(tr.clone());
335        cache_stub_run(&h, tr.uuid()).await;
336        h.shared.lock().await.n_queued = 1;
337
338        let res = h
339            .request_provisioning(&tr, "test", 0, DEFAULT_POOL, &mut mock)
340            .await;
341
342        assert_matches!(
343            res,
344            Ok(true),
345            "should have submitted the execution: {res:?}"
346        );
347
348        h.with_shared(|shared| {
349            assert_eq!(shared.n_queued, 0, "n_queued should have been decremented")
350        })
351        .await;
352
353        let event = rx.try_recv().expect("event should have been sent");
354        let ex_uuid = mock.test_executions[0].uuid();
355
356        assert_matches!(
357            event,
358            QueueEvent::Provision(pool, PendingProvision { test_execution, run_uuid, requires_dedicated: false })
359                if pool == DEFAULT_POOL
360                    && test_execution.uuid() == ex_uuid
361                    && run_uuid == tr.uuid()
362        );
363    }
