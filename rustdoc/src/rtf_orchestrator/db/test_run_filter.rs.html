6#[derive(Debug, Clone, Default, PartialEq, Eq)]
7pub struct TestRunFilter {
8    pub name: Option<String>,
9    pub initiated_by: Option<String>,
10    pub started_after: Option<DateTime<Utc>>,
11    pub started_before: Option<DateTime<Utc>>,
12    pub known_test_plan_uuid: Option<Uuid>,
13    pub known_test_plan_name: Option<String>,
14}
15
16impl TestRunFilter {
17    fn is_empty(&self) -> bool {
18        self.name.is_none()
19            && self.initiated_by.is_none()
20            && self.started_after.is_none()
21            && self.started_before.is_none()
22            && self.known_test_plan_uuid.is_none()
23            && self.known_test_plan_name.is_none()
24    }
25
26    fn needs_known_test_plan_join(&self) -> bool {
27        self.known_test_plan_uuid.is_some() || self.known_test_plan_name.is_some()
28    }
29
30    fn push_joins(&self, qb: &mut QueryBuilder<'_, Postgres>) {
31        if self.needs_known_test_plan_join() {
32            qb.push(
33                r#"
34                JOIN
35                  known_test_plan_run
36                ON
37                  known_test_plan_run.test_run_id = test_run.id
38                JOIN
39                  known_test_plan
40                ON
41                  known_test_plan.id = known_test_plan_run.known_test_plan_id
42                "#,
43            );
44        }
45    }
46
47    fn push_where_clause(&self, qb: &mut QueryBuilder<'_, Postgres>) {
48        if self.is_empty() {
49            return;
50        }
51
52        qb.push(" WHERE TRUE");
53
54        if let Some(name) = &self.name {
55            qb.push(" AND name = ").push_bind(name.clone());
56        }
57        if let Some(user) = &self.initiated_by {
58            // `NULL ILIKE anything` is `NULL` (never matches), so a run with no recorded
59            // initiator would be unfindable no matter what's searched — even searching for
60            // "unknown", which is exactly what such a run displays as. Coalescing to the same
61            // placeholder the UI shows keeps search consistent with what's on screen.
62            qb.push(" AND COALESCE(initiated_by, ")
63                .push_bind(UNKNOWN_INITIATOR)
64                .push(") ILIKE ")
65                .push_bind(substring_pattern(user));
66        }
67        if let Some(dt) = self.started_after {
68            qb.push(" AND started_at >= ").push_bind(dt);
69        }
70        if let Some(dt) = self.started_before {
71            qb.push(" AND started_at <= ").push_bind(dt);
72        }
73        if let Some(uuid) = &self.known_test_plan_uuid {
74            qb.push(" AND known_test_plan.uuid = ").push_bind(*uuid);
75        }
76        if let Some(name) = &self.known_test_plan_name {
77            qb.push(" AND known_test_plan.name = ")
78                .push_bind(name.clone());
79        }
80    }
81
82    pub async fn runs_matching(
83        &self,
84        limit: i64,
85        offset: i64,
86        conn: &mut PgConnection,
87    ) -> Result<Vec<TestRun>> {
88        let mut qb = QueryBuilder::new("SELECT test_run.* FROM test_run");
89        self.push_joins(&mut qb);
90        self.push_where_clause(&mut qb);
91
92        qb.push(" ORDER BY started_at DESC LIMIT ")
93            .push_bind(limit)
94            .push(" OFFSET ")
95            .push_bind(offset);
96
97        Ok(qb.build_query_as::<TestRun>().fetch_all(conn).await?)
98    }
99
100    pub async fn n_matching(&self, conn: &mut PgConnection) -> Result<i64> {
101        let mut qb = QueryBuilder::new("SELECT COUNT(*) FROM test_run");
102        self.push_joins(&mut qb);
103        self.push_where_clause(&mut qb);
104
105        Ok(qb.build_query_scalar().fetch_one(conn).await?)
106    }
107}

109/// Builds an `ILIKE` pattern matching `term` as a case-insensitive substring anywhere in the
110/// column, escaping `%`, `_`, and `\` in `term` first so they match literally rather than as `LIKE`
111/// wildcards (Postgres's default `LIKE`/`ILIKE` escape character is `\`, so no `ESCAPE` clause is
112/// needed on the query side).
113fn substring_pattern(term: &str) -> String {
114    let escaped = term
115        .replace('\\', "\\\\")
116        .replace('%', "\\%")
117        .replace('_', "\\_");
118
119    format!("%{escaped}%")
120}
