6/// Insert a resolved workload config, returning the id of the row that holds it.
7pub async fn upsert_workload_config(
8    config: &WorkloadConfig,
9    conn: &mut PgConnection,
10) -> Result<i32> {
11    let data = serde_json::to_value(config).expect("workload config to serialize");
12    let (id,): (i32,) = sqlx::query_as(
13        "INSERT INTO workload_config (data)
14         VALUES ($1)
15         ON CONFLICT (data) DO UPDATE SET data = EXCLUDED.data
16         RETURNING id;
17        ",
18    )
19    .bind(data)
20    .fetch_one(conn)
21    .await?;
22
23    Ok(id)
24}
25
26/// Fetch the resolved workload config stored under the given id.
27pub async fn get_workload_config(id: i32, conn: &mut PgConnection) -> Result<WorkloadConfig> {
28    let (data,): (Value,) = sqlx::query_as("SELECT data FROM workload_config WHERE id = $1;")
29        .bind(id)
30        .fetch_one(conn)
31        .await?;
32
33    serde_json::from_value(data).map_err(Error::MalformedWorkloadConfig)
34}
