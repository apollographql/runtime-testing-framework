28/// Axum extractor that only admits callers [ServerState::identify_user] resolves to
29/// [UserType::Admin], rejecting everyone else with [Error::Unauthorized].
30pub struct AdminUser(pub String);
31
32impl FromRequestParts<ServerState> for AdminUser {
33    type Rejection = Error;
34
35    async fn from_request_parts(
36        parts: &mut Parts,
37        state: &ServerState,
38    ) -> Result<Self, Self::Rejection> {
39        match state.identify_user(&parts.headers).await? {
40            UserType::Admin(email) => Ok(Self(email)),
41            UserType::User(_) | UserType::Unknown => Err(Error::Unauthorized),
42        }
43    }
44}

46/// Axum extractor that parses the `Authorization: Bearer <token>` header.
47///
48/// The extracted token should be validated against the per-execution token stored in the
49/// database using [BearerToken::verify].
50pub struct BearerToken(String);
51
52impl BearerToken {
53    pub fn verify(&self, expected: Uuid) -> crate::Result<()> {
54        let provided: Uuid = self.0.parse().map_err(|_| {
55            warn!("Provided bearer token is not a valid UUID, returning 403");
56            Error::Unauthorized
57        })?;
58        if provided == expected {
59            Ok(())
60        } else {
61            Err(Error::Unauthorized)
62        }
63    }
64}
65
66impl<S> FromRequestParts<S> for BearerToken
67where
68    S: Send + Sync,
69{
70    type Rejection = Error;
71
72    async fn from_request_parts(parts: &mut Parts, state: &S) -> Result<Self, Self::Rejection> {
73        let TypedHeader(Authorization(bearer)) =
74            TypedHeader::<Authorization<Bearer>>::from_request_parts(parts, state)
75                .await
76                .map_err(|err| {
77                    warn!(%err, "Failed to parse bearer token, returning 403");
78                    Error::Unauthorized
79                })?;
80
81        Ok(Self(bearer.token().to_owned()))
82    }
83}
