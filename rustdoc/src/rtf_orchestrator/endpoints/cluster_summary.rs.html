17pub async fn handler() -> Result<Json<ClusterSummaryResponse>> {
18    let clusters = &Config::get().workload_clusters;
19    let hourly = cached_hourly_execution_counts(clusters.available_clusters()).await?;
20
21    Ok(Json(cluster_summary(clusters, hourly)))
22}
23
24#[cached(ttl_secs = 600)]
25async fn cached_hourly_execution_counts(
26    clusters: Vec<ClusterId>,
27) -> Result<HashMap<ClusterId, Vec<HourlyCount>>> {
28    Ok(hourly_execution_counts(&clusters, Utc::now(), conn!()).await?)
29}

31fn cluster_summary(
32    clusters: &WorkloadClusters,
33    mut hourly: HashMap<ClusterId, Vec<HourlyCount>>,
34) -> ClusterSummaryResponse {
35    let pools = clusters
36        .cluster_pools
37        .iter()
38        .map(|(name, pool)| WorkloadPoolSummary {
39            name: name.to_string(),
40            clusters: pool
41                .available_clusters
42                .iter()
43                .filter_map(|cluster| {
44                    clusters
45                        .available_clusters
46                        .iter()
47                        .find(|c| c.name == *cluster)
48                })
49                .map(|cfg| WorkloadClusterSummary {
50                    name: cfg.name.clone(),
51                    execution: execution_summary(&cfg.execution, &pool.per_user_limits),
52                    hourly_executions: hourly
53                        .remove(&ClusterId::new(&cfg.name))
54                        .unwrap_or_default(),
55                })
56                .collect(),
57        })
58        .collect();
59
60    ClusterSummaryResponse {
61        max_queued_executions: clusters.max_queued_executions,
62        pools,
63    }
64}
65
66fn execution_summary(
67    cfg: &ClusterExecutionConfig,
68    per_user: &PerUserExecutionConfig,
69) -> ClusterExecutionSummary {
70    ClusterExecutionSummary {
71        max_concurrent: cfg.max_concurrent,
72        failed_execution_ttl_secs: cfg.failed_execution_ttl_secs,
73        retry_window_secs: cfg.retry_window_secs,
74        poll_interval_secs: cfg.poll_interval_secs,
75        exclusive_nodes: cfg.exclusive_nodes,
76        scenario_node_selector: cfg.scenario_node_selector.clone(),
77        namespace_cleanup_timeout_secs: cfg.namespace_cleanup_timeout_secs,
78        per_user: PerUserExecutionSummary {
79            max_concurrent_runs: per_user.max_concurrent_runs,
80            max_queued_runs: per_user.max_queued_runs,
81            max_queued_executions: per_user.max_queued_executions,
82            max_runs_per_hour: per_user.max_runs_per_hour,
83        },
84    }
85}

87#[cfg(test)]
88mod tests {
89    use super::*;
90    use crate::config::{ClusterPools, NamedPoolConfig, PoolConfig};
91    use simple_test_case::test_case;
92
93    fn pool_layout(resp: &ClusterSummaryResponse) -> Vec<(&str, Vec<&str>)> {
94        resp.pools
95            .iter()
96            .map(|p| {
97                let names = p.clusters.iter().map(|c| c.name.as_str()).collect();
98                (p.name.as_str(), names)
99            })
100            .collect()
101    }
102
103    fn pool(clusters: &[&str]) -> PoolConfig {
104        PoolConfig {
105            supports_dedicated: false,
106            available_clusters: clusters.iter().map(|c| c.to_string()).collect(),
107            per_user_limits: Default::default(),
108        }
109    }
110
111    #[test_case(pool(&["a"]), vec![], &[("default", &["a"])]; "default only")]
112    #[test_case(
113        pool(&["a"]),
114        vec![NamedPoolConfig { name: "perf".into(), config: pool(&["b", "c"]) }],
115        &[("default", &["a"]), ("perf", &["b", "c"])];
116        "default and an additional pool with several clusters"
117    )]
118    #[test]
119    fn cluster_summary_follows_the_configured_pools(
120        default: PoolConfig,
121        additional: Vec<NamedPoolConfig>,
122        expected: &[(&str, &[&str])],
123    ) {
124        let mut clusters =
125            WorkloadClusters::for_test_with_available_clusters(10, "a", &["a", "b", "c"]);
126        clusters.cluster_pools = ClusterPools {
127            default,
128            additional,
129        };
130
131        let resp = cluster_summary(&clusters, HashMap::new());
132        let expected: Vec<_> = expected.iter().map(|(p, cs)| (*p, cs.to_vec())).collect();
133
134        assert_eq!(pool_layout(&resp), expected);
135    }
