23pub async fn handler() -> Result<Json<ClusterSummaryResponse>> {
24    let clusters = &Config::get().workload_clusters;
25    let hourly = cached_hourly_execution_counts(clusters.available_clusters()).await?;
26    let nodes = cached_node_summaries(clusters.available_clusters()).await;
27
28    Ok(Json(cluster_summary(clusters, hourly, nodes)))
29}

31#[cached(ttl_secs = 600)]
32async fn cached_hourly_execution_counts(
33    clusters: Vec<ClusterId>,
34) -> Result<HashMap<ClusterId, Vec<HourlyCount>>> {
35    Ok(hourly_execution_counts(&clusters, Utc::now(), conn!()).await?)
36}
37
38#[cached(ttl_secs = 600)]
39async fn cached_node_summaries(clusters: Vec<ClusterId>) -> HashMap<ClusterId, NodesSummary> {
40    let per_cluster = Config::get().workload_clusters.per_cluster_config();
41
42    let fetches = clusters.iter().filter_map(|id| {
43        per_cluster.get(id).map(|cfg| async move {
44            cluster_node_summary(cfg)
45                .await
46                .map(|summary| (id.clone(), summary))
47        })
48    });
49
50    join_all(fetches).await.into_iter().flatten().collect()
51}

53/// `None` when the cluster's nodes could not be fetched (e.g. it is unreachable) - a single
54/// broken cluster should not take down the whole summary.
55async fn cluster_node_summary(cfg: &WorkloadClusterConfig) -> Option<NodesSummary> {
56    let mut clients =
57        k8s::ClusterClients::try_new_workload(&cfg.kubeconfig_path(), &cfg.workload_context)
58            .await
59            .inspect_err(|e| warn!(cluster = %cfg.name, %e, "failed to build workload client"))
60            .ok()?;
61
62    clients
63        .nodes_summary()
64        .await
65        .inspect_err(|e| warn!(cluster = %cfg.name, %e, "failed to fetch node summary"))
66        .ok()
67}

69fn cluster_summary(
70    clusters: &WorkloadClusters,
71    mut hourly: HashMap<ClusterId, Vec<HourlyCount>>,
72    mut nodes: HashMap<ClusterId, NodesSummary>,
73) -> ClusterSummaryResponse {
74    let pools = clusters
75        .cluster_pools
76        .iter()
77        .map(|(name, pool)| WorkloadPoolSummary {
78            name: name.to_string(),
79            supports_dedicated: pool.supports_dedicated,
80            per_user: per_user_summary(&pool.per_user_limits),
81            clusters: pool
82                .available_clusters
83                .iter()
84                .filter_map(|cluster| {
85                    clusters
86                        .available_clusters
87                        .iter()
88                        .find(|c| c.name == *cluster)
89                })
90                .map(|cfg| WorkloadClusterSummary {
91                    name: cfg.name.clone(),
92                    execution: execution_summary(&cfg.execution),
93                    hourly_executions: hourly
94                        .remove(&ClusterId::new(&cfg.name))
95                        .unwrap_or_default(),
96                    nodes: nodes.remove(&ClusterId::new(&cfg.name)),
97                })
98                .collect(),
99        })
100        .collect();
101
102    ClusterSummaryResponse {
103        max_queued_executions: clusters.max_queued_executions,
104        pools,
105    }
106}

108fn per_user_summary(per_user: &PerUserExecutionConfig) -> PerUserExecutionSummary {
109    PerUserExecutionSummary {
110        max_concurrent_runs: per_user.max_concurrent_runs,
111        max_queued_runs: per_user.max_queued_runs,
112        max_queued_executions: per_user.max_queued_executions,
113        max_runs_per_hour: per_user.max_runs_per_hour,
114    }
115}

117fn execution_summary(cfg: &ClusterExecutionConfig) -> ClusterExecutionSummary {
118    ClusterExecutionSummary {
119        max_concurrent: cfg.max_concurrent,
120        failed_execution_ttl_secs: cfg.failed_execution_ttl_secs,
121        retry_window_secs: cfg.retry_window_secs,
122        poll_interval_secs: cfg.poll_interval_secs,
123        exclusive_nodes: cfg.exclusive_nodes,
124        scenario_node_selector: cfg.scenario_node_selector.clone(),
125        namespace_cleanup_timeout_secs: cfg.namespace_cleanup_timeout_secs,
126    }
127}

129#[cfg(test)]
130mod tests {
131    use super::*;
132    use crate::config::{ClusterPools, NamedPoolConfig, PoolConfig};
133    use simple_test_case::test_case;
134
135    fn pool_layout(resp: &ClusterSummaryResponse) -> Vec<(&str, Vec<&str>)> {
136        resp.pools
137            .iter()
138            .map(|p| {
139                let names = p.clusters.iter().map(|c| c.name.as_str()).collect();
140                (p.name.as_str(), names)
141            })
142            .collect()
143    }
144
145    fn pool(clusters: &[&str]) -> PoolConfig {
146        PoolConfig {
147            supports_dedicated: false,
148            available_clusters: clusters.iter().map(|c| c.to_string()).collect(),
149            per_user_limits: Default::default(),
150            workload_config: Default::default(),
151        }
152    }
153
154    #[test_case(pool(&["a"]), vec![], &[("default", &["a"])]; "default only")]
155    #[test_case(
156        pool(&["a"]),
157        vec![NamedPoolConfig { name: "perf".into(), config: pool(&["b", "c"]) }],
158        &[("default", &["a"]), ("perf", &["b", "c"])];
159        "default and an additional pool with several clusters"
160    )]
161    #[test]
162    fn cluster_summary_follows_the_configured_pools(
163        default: PoolConfig,
164        additional: Vec<NamedPoolConfig>,
165        expected: &[(&str, &[&str])],
166    ) {
167        let mut clusters =
168            WorkloadClusters::for_test_with_available_clusters(10, "a", &["a", "b", "c"]);
169        clusters.cluster_pools = ClusterPools {
170            default,
171            additional,
172        };
173
174        let resp = cluster_summary(&clusters, HashMap::new(), HashMap::new());
175        let expected: Vec<_> = expected.iter().map(|(p, cs)| (*p, cs.to_vec())).collect();
176
177        assert_eq!(pool_layout(&resp), expected);
178    }
179
180    #[test]
181    fn cluster_summary_reports_dedicated_support_and_user_limits_per_pool() {
182        let mut dedicated = pool(&["a"]);
183        dedicated.supports_dedicated = true;
184        dedicated.per_user_limits.max_queued_runs = 7;
185        let mut perf = pool(&["b"]);
186        perf.per_user_limits.max_queued_runs = 3;
187        let mut clusters = WorkloadClusters::for_test_with_available_clusters(10, "a", &["a", "b"]);
188        clusters.cluster_pools = ClusterPools {
189            default: dedicated,
190            additional: vec![NamedPoolConfig {
191                name: "perf".into(),
192                config: perf,
193            }],
194        };
195
196        let resp = cluster_summary(&clusters, HashMap::new(), HashMap::new());
197        let pools: Vec<_> = resp
198            .pools
199            .iter()
200            .map(|p| {
201                (
202                    p.name.as_str(),
203                    p.supports_dedicated,
204                    p.per_user.max_queued_runs,
205                )
206            })
207            .collect();
208
209        assert_eq!(pools, vec![("default", true, 7), ("perf", false, 3)]);
210    }
