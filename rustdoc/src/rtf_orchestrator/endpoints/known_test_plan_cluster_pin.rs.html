1//! Pin or unpin the workload cluster a known test plan's runs execute in.
2//!
3//! Gated to admins only (see [AdminUser]).
4use crate::{
5    Error, Result, conn,
6    db::{ClusterId, KnownTestPlan},
7    endpoints::AdminUser,
8    state::ServerState,
9};
10use axum::{
11    Json,
12    extract::{Path, State},
13};
14use rtf_orchestrator_shared::known_test_plan::{
15    KnownTestPlanSummary, SetPinnedWorkloadClusterRequest,
16};
17use uuid::Uuid;
18
19pub async fn set_handler(
20    _admin: AdminUser,
21    Path(uuid): Path<Uuid>,
22    State(ServerState { eq_state, .. }): State<ServerState>,
23    Json(req): Json<SetPinnedWorkloadClusterRequest>,
24) -> Result<Json<KnownTestPlanSummary>> {
25    let conn = conn!();
26
27    let requested = ClusterId::new(&req.cluster);
28    if !eq_state.available_clusters().await.contains(&requested) {
29        return Err(Error::UnknownWorkloadCluster {
30            cluster: req.cluster,
31        });
32    }
33
34    let mut known = KnownTestPlan::get_by_uuid(&uuid, conn)
35        .await?
36        .ok_or_else(|| Error::UnknownTestPlan {
37            identifier: uuid.to_string(),
38        })?;
39
40    known
41        .set_pinned_workload_cluster(Some(&req.cluster), conn)
42        .await?;
43
44    Ok(Json(known.into_summary()))
45}
46
47pub async fn clear_handler(
48    _admin: AdminUser,
49    Path(uuid): Path<Uuid>,
50) -> Result<Json<KnownTestPlanSummary>> {
51    let conn = conn!();
52
53    let mut known = KnownTestPlan::get_by_uuid(&uuid, conn)
54        .await?
55        .ok_or_else(|| Error::UnknownTestPlan {
56            identifier: uuid.to_string(),
57        })?;
58
59    known.set_pinned_workload_cluster(None, conn).await?;
60
61    Ok(Json(known.into_summary()))
62}

77    fn admin_header_value() -> String {
78        format!("accounts.google.com:{ADMIN_EMAIL}")

81    async fn register_plan(conn: &mut PgConnection) -> KnownTestPlan {
82        KnownTestPlan::register(&unique("plan"), None, "org", "repo", &unique("path"), conn)
83            .await
84            .unwrap()
85    }

87    #[cfg_attr(not(feature = "db_tests"), ignore)]
88    #[tokio::test]
89    async fn admin_can_pin_and_unpin_a_configured_cluster() -> anyhow::Result<()> {
90        let tss = TestServerState::new_with_admins_and_clusters(
91            &[ADMIN_EMAIL],
92            vec![ClusterId::new("alpha"), ClusterId::new("beta")],
93        );
94
95        let plan = register_plan(conn!()).await;
96        assert_eq!(plan.pinned_workload_cluster(), None);
97
98        let resp = tss
99            .test_server
100            .post(&format!("/test-plan/{}/pinned-cluster", plan.uuid()))
101            .add_header(IAP_USER_EMAIL_HEADER, admin_header_value())
102            .json(&SetPinnedWorkloadClusterRequest {
103                cluster: "beta".to_owned(),
104            })
105            .await;
106        assert_eq!(resp.status_code(), StatusCode::OK, "{:?}", resp.text());
107
108        let pinned: KnownTestPlanSummary = resp.json();
109        assert_eq!(pinned.pinned_workload_cluster.as_deref(), Some("beta"));
110
111        let resp = tss
112            .test_server
113            .delete(&format!("/test-plan/{}/pinned-cluster", plan.uuid()))
114            .add_header(IAP_USER_EMAIL_HEADER, admin_header_value())
115            .await;
116        assert_eq!(resp.status_code(), StatusCode::OK);
117
118        let unpinned: KnownTestPlanSummary = resp.json();
119        assert_eq!(unpinned.pinned_workload_cluster, None);
120
121        Ok(())
122    }
