1//! List Test Plans registered with the orchestrator, matching an optional name filter, or fetch a
2//! single one by UUID.
3use crate::{
4    Error, Result, conn,
5    db::{KnownTestPlan, KnownTestPlanFilter},
6};
7use axum::{
8    Json,
9    extract::{Path, Query},
10};
11use rtf_orchestrator_shared::known_test_plan::{
12    KnownTestPlanListParams, KnownTestPlanListResponse, KnownTestPlanSummary,
13};
14use uuid::Uuid;
15
16const DEFAULT_LIMIT: i64 = 20;
17const MAX_LIMIT: i64 = 100;
18
19pub async fn list_handler(
20    Query(params): Query<KnownTestPlanListParams>,
21) -> Result<Json<KnownTestPlanListResponse>> {
22    let conn = conn!();
23    let filter = KnownTestPlanFilter { name: params.name };
24
25    let limit = params.limit.unwrap_or(DEFAULT_LIMIT).clamp(1, MAX_LIMIT);
26    let offset = params.offset.unwrap_or(0).max(0);
27
28    let test_plans = filter.matching(limit, offset, conn).await?;
29    let total = filter.n_matching(conn).await?;
30
31    Ok(Json(KnownTestPlanListResponse {
32        test_plans: test_plans.into_iter().map(|tp| tp.into_summary()).collect(),
33        total,
34    }))
35}
36
37pub async fn by_uuid_handler(Path(uuid): Path<Uuid>) -> Result<Json<KnownTestPlanSummary>> {
38    match KnownTestPlan::get_by_uuid(&uuid, conn!()).await? {
39        Some(plan) => Ok(Json(plan.into_summary())),
40        None => Err(Error::UnknownTestPlan {
41            identifier: uuid.to_string(),
42        }),
43    }
44}

57    #[cfg_attr(not(feature = "db_tests"), ignore)]

60        let tss = TestServerState::new();
61        let conn = conn!();
62        let name = unique("match-me");
63        KnownTestPlan::register(&name, None, "org", "repo", &unique("path"), conn).await?;
64        KnownTestPlan::register(
65            &unique("not-this-one"),
66            None,
67            "org",
68            "repo",
69            &unique("path"),
70            conn,
71        )
72        .await?;
73
74        let resp = tss
75            .test_server
76            .get("/test-plan")
77            .add_query_param("name", &name)
78            .await;
79
80        assert_eq!(resp.status_code(), StatusCode::OK);
81
82        let body: KnownTestPlanListResponse = resp.json();
83
84        assert_eq!(body.total, 1, "{body:?}");
