1//! Reads the caller's identity off the `X-Goog-Authenticated-User-Email` header that Google IAP
2//! attaches to every request once it authenticates the caller.
3use axum::http::HeaderMap;
4use rtf_integrations::orchestrator::TRIGGER_REPO_HEADER;
5use tracing::warn;
6
7pub const IAP_USER_EMAIL_HEADER: &str = "x-goog-authenticated-user-email";
8
9/// Reads the `X-Goog-Authenticated-User-Email` header (format `prefix:email`, e.g.
10/// `accounts.google.com:someone@apollographql.com`) and returns the email.
11pub fn extract_authenticated_user_email(headers: &HeaderMap) -> Option<String> {
12    headers
13        .get(IAP_USER_EMAIL_HEADER)
14        .or_else(|| {
15            warn!("{IAP_USER_EMAIL_HEADER} header missing from request");
16
17            None
18        })
19        .and_then(|value| {
20            value
21                .to_str()
22                .inspect_err(|err| warn!(%err, "{IAP_USER_EMAIL_HEADER} header is not valid UTF-8"))
23                .ok()
24        })
25        .and_then(|value| {
26            value.split_once(':').or_else(|| {
27                warn!("{IAP_USER_EMAIL_HEADER} header has no colon separator");
28
29                None
30            })
31        })
32        .map(|(_, email)| email)
33        .filter(|email| {
34            let is_empty = email.is_empty();
35            if is_empty {
36                warn!("{IAP_USER_EMAIL_HEADER} header has an empty email portion");
37            }
38
39            !is_empty
40        })
41        .map(|email| email.to_owned())
42}
43
44/// Parse the `X-Rtf-Trigger-Repo` header (format `org/repo`) and split it into its org and repo
45/// components.
46/// We only pay attention to this for known automation users which are already verified via the IAP
47/// header so we simply trust the value that has been set.
48pub fn try_extract_trigger_repo(headers: &HeaderMap) -> Option<(String, String)> {
49    headers
50        .get(TRIGGER_REPO_HEADER)
51        .and_then(|value| value.to_str().ok())
52        .and_then(|value| value.split_once('/'))
53        .filter(|(org, repo)| !org.is_empty() && !repo.is_empty() && !repo.contains('/'))
54        .map(|(org, repo)| (org.to_owned(), repo.to_owned()))
55}

57#[cfg(test)]
58mod tests {
59    use super::*;
60    use axum::http::HeaderValue;
61    use simple_test_case::test_case;
62
63    #[test]
64    fn extract_initiator_returns_none_when_header_is_missing() {
65        assert_eq!(extract_authenticated_user_email(&HeaderMap::new()), None);
66    }
67
68    #[test_case(b"accounts.google.com:someone@apollographql.com", Some("someone@apollographql.com"); "valid")]
69    #[test_case(b"someone@apollographql.com", None; "missing prefix before the colon")]
70    #[test_case(b"accounts.google.com:", None; "missing the user email")]
71    #[test_case(&[0xff, 0xfe], None; "invalid utf8")]
72    #[test]
73    fn extract_initiator_returns_expected_data(header_value_bytes: &[u8], expected: Option<&str>) {
74        let mut headers = HeaderMap::new();
75        headers.insert(
76            IAP_USER_EMAIL_HEADER,
77            HeaderValue::from_bytes(header_value_bytes).unwrap(),
78        );
79
80        assert_eq!(
81            extract_authenticated_user_email(&headers),
82            expected.map(|s| s.to_string())
83        );
84    }
85
86    #[test]
87    fn extract_trigger_repo_returns_none_when_header_is_missing() {
88        assert_eq!(try_extract_trigger_repo(&HeaderMap::new()), None);
89    }
90
91    #[test_case(b"foo-org/bar-repo", Some(("foo-org", "bar-repo")); "valid")]
92    #[test_case(b"foo-org", None; "no slash")]
93    #[test_case(b"/bar-repo", None; "no org before slash")]
94    #[test_case(b"foo-org/bar-repo/baz-path", None; "more than one slash")]
95    #[test_case(&[0xff, 0xfe], None; "invalid utf8")]
96    #[test]
97    fn extract_trigger_repo_returns_expected_data(
98        header_value_bytes: &[u8],
99        expected: Option<(&str, &str)>,
100    ) {
101        let mut headers = HeaderMap::new();
102        headers.insert(
103            TRIGGER_REPO_HEADER,
104            HeaderValue::from_bytes(header_value_bytes).unwrap(),
105        );
106
107        assert_eq!(
108            try_extract_trigger_repo(&headers),
109            expected.map(|(org, repo)| (org.to_string(), repo.to_string()))
110        );
111    }
112}
