99impl IntoResponse for Error {
100    fn into_response(self) -> Response {
101        let msg = self.to_string();
102
103        let raw = match self {
104            Self::FileUploadAlreadyRequested
105            | Self::InvalidFileProviderUsage { .. }
106            | Self::UnknownWorkloadPool { .. }
107            | Self::DedicatedClusterNotSupported { .. }
108            | Self::InvalidWorkloadConfig(_)
109            | Self::NodeLabelsRequireDedicatedCluster
110            | Self::NodeLabelsRequirePinnedPool
111            | Self::Prepare(_)
112            | Self::Db(db::Error::MissingExitCode)
113            | Self::Db(db::Error::InvalidFailedExitCode)
114            | Self::Db(db::Error::InvalidExitCode { .. })
115            | Self::Db(db::Error::InvalidExecutionStatus { .. }) => (
116                StatusCode::BAD_REQUEST,
117                Json(json!({ "error": "BAD_REQUEST", "message": msg })),
118            ),
119
120            Self::FileUploadNotAvailable => (
121                StatusCode::NOT_FOUND,
122                Json(json!({ "error": "NOT_FOUND", "message": msg })),
123            ),
124
125            Self::FileUploadNotReady | Self::Db(db::Error::KnownTestPlanAlreadyExists) => (
126                StatusCode::CONFLICT,
127                Json(json!({ "error": "CONFLICT", "message": msg })),
128            ),
129
130            Self::InsufficientCapacity => (
131                StatusCode::SERVICE_UNAVAILABLE,
132                Json(json!({ "error": "SERVICE_UNAVAILABLE", "message": msg })),
133            ),
134
135            Self::RateLimited { pool, reason } => (
136                StatusCode::TOO_MANY_REQUESTS,
137                Json(
138                    json!({ "error": "TOO_MANY_REQUESTS", "message": msg, "pool": pool,  "reason": reason }),
139                ),
140            ),
141
142            Self::Unauthorized => (
143                StatusCode::FORBIDDEN,
144                Json(json!({ "error": "FORBIDDEN", "message": msg })),
145            ),
146
147            Self::UnknownTestExecution { id } => (
148                StatusCode::NOT_FOUND,
149                Json(json!({ "error": "NOT_FOUND", "id": id, "message": msg })),
150            ),
151
152            Self::UnknownTestRun { id } => (
153                StatusCode::NOT_FOUND,
154                Json(json!({ "error": "NOT_FOUND", "id": id, "message": msg })),
155            ),
156
157            Self::UnknownTestPlan { identifier } => (
158                StatusCode::NOT_FOUND,
159                Json(json!({ "error": "NOT_FOUND", "identifier": identifier, "message": msg })),
160            ),
161
162            _ => (
163                StatusCode::INTERNAL_SERVER_ERROR,
164                Json(json!({ "error": "INTERNAL", "message": msg })),
165            ),
166        };
167
168        raw.into_response()
169    }
170}

172impl From<EnvironmentSummaryError> for Error {
173    fn from(err: EnvironmentSummaryError) -> Self {
174        match err {
175            EnvironmentSummaryError::Formats(e) => e.into(),
176            EnvironmentSummaryError::Inlining(e) => ResolverError::Inlining(e).into(),
177            EnvironmentSummaryError::Templating(e) => ResolverError::TemplatingCheck(e).into(),
178        }
179    }
