33#[serde(rename_all = "camelCase")]
34pub struct WorkflowSpec {
35    pub service_account_name: String,
36    pub entrypoint: String,
37    pub on_exit: String,
38    pub templates: Vec<TemplateDef>,
39    pub volumes: Vec<Volume>,
40}
41
42impl WorkflowSpec {
43    pub fn for_execution_id(execution_id: &Uuid) -> Self {
44        let configmap_name = env_configmap_name(execution_id);
45        let namespace = execution_id.to_string();
46
47        Self {
48            service_account_name: "argo-workflow".to_owned(),
49            entrypoint: "main".to_owned(),
50            on_exit: "cleanup".to_owned(),
51            templates: vec![
52                TemplateDef::Main(MainTemplate::new()),
53                TemplateDef::Task(create_namespace(&namespace)),
54                TemplateDef::Task(create_pull_secret(&namespace)),
55                TemplateDef::Task(deploy_environment(&configmap_name, &namespace)),
56                TemplateDef::Task(cleanup(&configmap_name)),
57            ],
58            volumes: vec![Volume {
59                name: "kubeconfig".into(),
60                secret: Some(SecretVolumeSource {
61                    secret_name: Some("workload-kubeconfig".into()),
62                    ..Default::default()
63                }),
64                ..Default::default()
65            }],
66        }
67    }
68}

122#[derive(Debug, Clone, Deserialize, Serialize, JsonSchema)]
123pub struct TaskTemplate {
124    name: String,
125    container: Container,
126    volumes: Option<Vec<Volume>>,
127}
128
129impl TaskTemplate {
130    fn new(
131        name: &str,
132        arg: String,
133        volume_mounts: Vec<VolumeMount>,
134        volumes: Option<Vec<Volume>>,
135    ) -> Self {
136        Self {
137            name: name.into(),
138            container: Container {
139                name: name.into(),
140                image: Some(TOOLBOX_IMAGE.into()),
141                command: Some(vec!["/bin/sh".to_owned(), "-c".to_owned()]),
142                args: Some(vec![arg]),
143                volume_mounts: Some(volume_mounts),
144                ..Default::default()
145            },
146            volumes,
147        }
148    }
149}

151const CREATE_NAMESPACE_SCRIPT: &str = r#"
152set -e
153echo "Creating namespace '__NAMESPACE__' in workload cluster..."
154kubectl \
155  --kubeconfig=/kubeconfig/value \
156  create namespace __NAMESPACE__ \
157  --dry-run=client \
158  -o yaml |
159    kubectl --kubeconfig=/kubeconfig/value apply -f -
160
161echo "Namespace created successfully."
162"#;
163
164fn create_namespace(namespace: &str) -> TaskTemplate {
165    TaskTemplate::new(
166        "create-namespace",
167        CREATE_NAMESPACE_SCRIPT.replace("__NAMESPACE__", namespace),
168        vec![VolumeMount {
169            name: "kubeconfig".into(),
170            mount_path: "/kubeconfig".into(),
171            read_only: Some(true),
172            ..Default::default()
173        }],
174        None,
175    )
176}

178const CREATE_PULL_SECRET_SCRIPT: &str = r#"
179set -e
180echo "Creating image pull secret in namespace '__NAMESPACE__'..."
181kubectl --kubeconfig=/kubeconfig/value \
182  create secret docker-registry gcr-secret \
183  --namespace=__NAMESPACE__ \
184  --from-file=.dockerconfigjson=/gcr-secret/config.json \
185  --dry-run=client -o yaml |
186    kubectl --kubeconfig=/kubeconfig/value apply -f -
187
188echo "Patching default service account..."
189kubectl --kubeconfig=/kubeconfig/value \
190  patch serviceaccount default \
191  --namespace=__NAMESPACE__ \
192  -p '{"imagePullSecrets": [{"name": "gcr-secret"}]}'
193
194echo "Pull secret created successfully."
195"#;
196
197fn create_pull_secret(namespace: &str) -> TaskTemplate {
198    TaskTemplate::new(
199        "create-pull-secret",
200        CREATE_PULL_SECRET_SCRIPT.replace("__NAMESPACE__", namespace),
201        vec![
202            VolumeMount {
203                name: "kubeconfig".into(),
204                mount_path: "/kubeconfig".into(),
205                read_only: Some(true),
206                ..Default::default()
207            },
208            VolumeMount {
209                name: "gcr-secret".into(),
210                mount_path: "/gcr-secret".into(),
211                read_only: Some(true),
212                ..Default::default()
213            },
214        ],
215        Some(vec![Volume {
216            name: "gcr-secret".into(),
217            secret: Some(SecretVolumeSource {
218                secret_name: Some("gcr-secret".into()),
219                items: Some(vec![KeyToPath {
220                    key: ".dockerconfigjson".into(),
221                    path: "config.json".into(),
222                    ..Default::default()
223                }]),
224                ..Default::default()
225            }),
226            ..Default::default()
227        }]),
228    )
229}

231const DEPLOY_ENV_SCRIPT: &str = r#"
232set -e
233WORKDIR=/tmp/rtf-work
234mkdir -p $WORKDIR
235mkdir -p $WORKDIR/k8s
236
237echo "Resolving environment docker-compose files..."
238rtf resolve environment /environment/environment.yaml --outdir $WORKDIR/output
239
240# Source RTF env vars
241. $WORKDIR/output/setup/setup.env
242echo "Converting to kubernetes manifests..."
243KOMPOSE_ARGS=""
244while IFS= read -r f || [ -n "$f" ]; do
245  [ -n "$f" ] && KOMPOSE_ARGS="$KOMPOSE_ARGS -f $f"
246done < "$COMPOSE_FILES"
247kompose convert $KOMPOSE_ARGS -o $WORKDIR/k8s/
248
249echo "Applying manifests to namespace '__NAMESPACE__'..."
250kubectl --kubeconfig=/kubeconfig/value apply \
251  -n __NAMESPACE__ \
252  -f $WORKDIR/k8s/
253
254echo "Waiting for deployments to become available..."
255kubectl --kubeconfig=/kubeconfig/value wait \
256  --for=condition=available \
257  deployment \
258  --all \
259  -n __NAMESPACE__ \
260  --timeout=300s
261
262echo "Environment deployed successfully."
263"#;

265fn deploy_environment(configmap_name: &str, namespace: &str) -> TaskTemplate {
266    TaskTemplate::new(
267        "deploy-environment",
268        DEPLOY_ENV_SCRIPT.replace("__NAMESPACE__", namespace),
269        vec![
270            VolumeMount {
271                name: "kubeconfig".into(),
272                mount_path: "/kubeconfig".into(),
273                read_only: Some(true),
274                ..Default::default()
275            },
276            VolumeMount {
277                name: "environment".into(),
278                mount_path: "/environment".into(),
279                read_only: Some(true),
280                ..Default::default()
281            },
282        ],
283        Some(vec![Volume {
284            name: "environment".to_owned(),
285            config_map: Some(ConfigMapVolumeSource {
286                name: configmap_name.to_owned(),
287                ..Default::default()
288            }),
289            ..Default::default()
290        }]),
291    )
292}
