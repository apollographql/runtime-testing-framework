12#[derive(Debug)]
13pub struct Event {
14    pub execution_id: Uuid,
15    pub ty: EventType,
16}
17
18#[derive(Debug)]
19pub enum EventType {
20    ProvisionEnvironment(DockerComposeEnvironment, DockerScenario),
21    RunScenario(DockerScenario),
22    CleanupNamespace,
23}
24
25pub async fn event_loop_task(tx: UnboundedSender<Event>, mut rx: UnboundedReceiver<Event>) {
26    // FIXME: handle getting the kubeconfig path correctly here
27    let clients = match ClusterClients::try_new(
28        Path::new(&env::var("KUBECONFIG_PATH").expect("no kubeconfig path")),
29        "kind-rtf-mgmt",
30        "kind-rtf-workload",
31    )
32    .await
33    {
34        Ok(clients) => clients,
35        Err(error) => {
36            error!(%error, "unable to create k8s clients");
37            return;
38        }
39    };
40
41    loop {
42        let evt = match rx.recv().await {
43            Some(evt) => evt,
44            None => {
45                info!("Event loop channel closed. Exiting event loop task");
46                return;
47            }
48        };
49
50        if let Err(e) = handle_event(evt, &tx, &clients).await {
51            error!(%e, "error processing event");
52        }
53    }
54}
55
56async fn handle_event(
57    evt: Event,
58    tx: &UnboundedSender<Event>,
59    clients: &ClusterClients,
60) -> anyhow::Result<()> {
61    match evt.ty {
62        EventType::ProvisionEnvironment(environment, scenario) => {
63            provision_environment::run(evt.execution_id, environment, scenario, tx, clients).await
64        }
65        EventType::RunScenario(scenario) => {
66            run_scenario::run(evt.execution_id, scenario, tx, clients).await
67        }
68        EventType::CleanupNamespace => cleanup_environment::run(evt.execution_id, clients).await,
69    }
70}
